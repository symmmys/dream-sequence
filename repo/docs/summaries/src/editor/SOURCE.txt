        y: position.y - building.position.y,
        z: -dx * sin + dz * cos

class BuildingPlacement {
    constructor(system, editorCamera) {
        this.system = system;
        this.editorCamera = editorCamera;

        this.activeBuilding = null;
    }

    select(building) {
        this.activeBuilding = building;

        if (building) {
            this.editorCamera.focus(
                building.position
            );
        }
    }

    update(patch = {}) {
        const building = this.activeBuilding;

        if (!building) {
            throw new Error("No building selected.");
        }

        const position = building.position;

        if (Number.isFinite(patch.x)) {
            position.x = patch.x;
        }

        if (Number.isFinite(patch.z)) {
            position.z = patch.z;
        }

        if (Number.isFinite(patch.y)) {
            position.y = patch.y;
        }

        if (Number.isFinite(patch.rotation)) {
            building.rotation = patch.rotation;
        }

        this.system.rebuild(building);

        this.editorCamera.focus(position);
    }

    setHorizontal(x, z) {
        this.update({ x, z });
    }

    setElevation(y) {
        this.update({ y });
    }

    setRotation(rotation) {
        this.update({ rotation });
    }

    getState() {
        if (!this.activeBuilding) return null;

        return {
            position: { ...this.activeBuilding.position },
            rotation: this.activeBuilding.rotation
        };
    }
}

export { BuildingPlacement };

        );
    }

    createComponentLines(component, building) {
        const p = component.position;
        const d = component.dimensions;

        if (
            !p || !d ||
            ![
                p.x, p.y, p.z,
                d.x, d.y, d.z,
                building.position.x,
                building.position.y,
                building.position.z
            ].every(Number.isFinite) ||
            d.x <= 0 || d.y <= 0 || d.z <= 0
        ) {
            return null;
        }

        const angle = building.rotation ?? 0;
        const cos = Math.cos(angle);
        const sin = Math.sin(angle);

        const worldPosition = new BABYLON.Vector3(
            building.position.x + p.x * cos - p.z * sin,
            building.position.y + p.y,
            building.position.z + p.x * sin + p.z * cos
        );

        const rotation = BABYLON.Quaternion.RotationAxis(
            BABYLON.Axis.Y,
            angle + (component.rotation ?? 0)
        );

        const matrix = BABYLON.Matrix.Compose(
            new BABYLON.Vector3(d.x, d.y, d.z),
            rotation,
            worldPosition
        );

        const projected =
            this.geometryRenderer.transformInstanceMatrix(matrix);

        if (!projected) return null;

        const corners = [];

        for (let i = 0; i < 8; i++) {
            const local = new BABYLON.Vector3(
                (i & 1) ? 0.5 : -0.5,
                (i & 2) ? 0.5 : -0.5,
                (i & 4) ? 0.5 : -0.5
            );

            corners.push(
                BABYLON.Vector3.TransformCoordinates(
                    local,
                    projected
                )
            );
        }

        const edges = [
            [0, 1], [0, 2], [0, 4],
            [1, 3], [1, 5],
            [2, 3], [2, 6],
            [3, 7], [4, 5],
            [4, 6], [5, 7], [6, 7]
        ];

        return edges.map(([a, b]) => [
            corners[a],
            corners[b]
        ]);
    }

    show(building) {
        this.clear();

        if (!building?.generatedGeometry) {
            throw new Error(
                "Building has no generated geometry to preview."
            );
        }

        this.buildingId = building.id;

        for (const component of building.generatedGeometry) {
            const edges = this.createComponentLines(
                component,
                building
            );

            if (!edges) continue;

            const mesh = BABYLON.MeshBuilder.CreateLineSystem(
                "building-preview-outline",
                { lines: edges },
                this.scene
            );

            mesh.color = this.color;
            mesh.alpha = 1;
            mesh.isPickable = false;
            mesh.renderingGroupId = 2;

            mesh.metadata = {
                type: "building-preview-outline",
                buildingId: building.id
            };

            this.lines.push(mesh);
        }

        return this.lines.length;
    }

    clear() {
        for (const mesh of this.lines) {
            mesh.dispose();
        }

        this.lines.length = 0;
        this.buildingId = null;
    }

    commit(building) {
        if (
            building &&
            building.id !== this.buildingId
        ) {
            throw new Error(
                "Cannot commit a different preview building."
            );
        }

        this.clear();
    }

    dispose() {
        this.clear();
    }
}

export { BuildingPreviewRenderer };

class EditorCamera {

    }

        this.camera.detachControl();
    }

        this.detach();
        this.camera.dispose();
    }
}

class FloorPlanBuilder {
    constructor(options = {}) {
        this.wallHeight = options.wallHeight ?? 4;
        this.wallThickness = options.wallThickness ?? 0.35;
        this.doorHeight = options.doorHeight ?? 2.1;
        this.windowHeight = options.windowHeight ?? 1.2;
        this.windowSill = options.windowSill ?? 0.9;
        this.minimumSegmentPixels =
            options.minimumSegmentPixels ?? 5;
    }

    setWallHeight(value) {
        if (
            !Number.isFinite(value) ||
            value < 2.4 ||
            value > 1000
        ) {
            throw new Error(
                "Wall height must be between 2.4 and 1000."
            );
        }

        this.wallHeight = value;
        return value;
    }

    setWallThickness(value) {
        if (
            !Number.isFinite(value) ||
            value < 0.1 ||
            value > 1000
        ) {
            throw new Error(
                "Wall thickness must be between 0.1 and 1000."
            );
        }

        this.wallThickness = value;
        return value;
    }

    classify(r, g, b, a) {
        if (a < 128) return null;

        if (r < 75 && g < 75 && b < 75) {
            return "wall";
        }

        if (r > 150 && g < 110 && b < 110) {
            return "door";
        }

        if (g > 130 && r < 120 && b < 130) {
            return "window";
        }

        return null;
    }

    extractSegments(imageData) {
        const { data, width, height } = imageData;

        if (
            !data ||
            !Number.isInteger(width) ||
            !Number.isInteger(height) ||
            width <= 0 ||
            height <= 0 ||
            data.length < width * height * 4
        ) {
            throw new Error("Invalid floor-plan image data.");
        }

        const active = new Map();
        const completed = [];

        for (let y = 0; y < height; y++) {
            const current = new Set();
            let x = 0;

            while (x < width) {
                const index = (y * width + x) * 4;

                const kind = this.classify(
                    data[index],
                    data[index + 1],
                    data[index + 2],
                    data[index + 3]
                );

                if (!kind) {
                    x++;
                    continue;
                }

                const start = x;

                while (x < width) {
                    const i = (y * width + x) * 4;

                    if (
                        this.classify(
                            data[i],
                            data[i + 1],
                            data[i + 2],
                            data[i + 3]
                        ) !== kind
                    ) {
                        break;
                    }

                    x++;
                }

                const end = x - 1;
                const key = `${kind}:${start}:${end}`;

                current.add(key);

                const previous = active.get(key);

                if (previous) {
                    previous.maxY = y;
                } else {
                    active.set(key, {
                        kind,
                        minX: start,
                        maxX: end,
                        minY: y,
                        maxY: y
                    });
                }
            }

            for (const [key, segment] of active) {
                if (!current.has(key)) {
                    completed.push(segment);
                    active.delete(key);
                }
            }
        }

        completed.push(...active.values());

        return completed.filter(segment => {
            const w = segment.maxX - segment.minX + 1;
            const h = segment.maxY - segment.minY + 1;

            return Math.max(w, h) >= this.minimumSegmentPixels;
        });
    }

    build(building, imageData, options = {}) {
        if (
            !building ||
            typeof building.addComponent !== "function" ||
            !Array.isArray(building.components)
        ) {
            throw new Error("Invalid building model.");
        }

        const width = options.width ?? 20;
        const depth = options.depth ?? 20;
        const floorIndex = options.floor ?? 0;

        if (
            !Number.isFinite(width) ||
            !Number.isFinite(depth) ||
            width <= 0 ||
            depth <= 0 ||
            !Number.isInteger(floorIndex) ||
            floorIndex < 0
        ) {
            throw new Error("Invalid floor dimensions.");
        }

        const segments = this.extractSegments(imageData);

        if (!segments.length) {
            throw new Error(
                "No architectural lines detected."
            );
        }

        const scaleX = width / imageData.width;
        const scaleZ = depth / imageData.height;

        const floorY = floorIndex * this.wallHeight;
        const prefix = `floor-plan-${floorIndex}-`;

        const generated = [];

        const counts = {
            walls: 0,
            doors: 0,
            windows: 0,
            floors: 0
        };

        const add = (
            kind,
            type,
            position,
            dimensions,
            material
        ) => {
            generated.push({
                kind,
                type: prefix + type,
                position,
                dimensions,
                material
            });
        };

        add(
            "floors",
            "floor",
            { x: 0, y: floorY - 0.075, z: 0 },
            { x: width, y: 0.15, z: depth },
            "floor"
        );

        counts.floors++;

        for (const segment of segments) {
            const pixelWidth =
                segment.maxX - segment.minX + 1;

            const pixelHeight =
                segment.maxY - segment.minY + 1;

            const horizontal = pixelWidth >= pixelHeight;

            const centerPixelX =
                (segment.minX + segment.maxX + 1) * 0.5;

            const centerPixelY =
                (segment.minY + segment.maxY + 1) * 0.5;

            const x =
                (centerPixelX / imageData.width - 0.5) * width;

            const z =
                (centerPixelY / imageData.height - 0.5) * depth;

            const length = horizontal
                ? pixelWidth * scaleX
                : pixelHeight * scaleZ;

            const detectedThickness = Math.max(
                0.08,
                Math.min(
                    0.35,
                    horizontal
                        ? pixelHeight * scaleZ
                        : pixelWidth * scaleX
                )
            );

            const thickness = segment.kind === "wall"
                ? this.wallThickness
                : detectedThickness;

            const dimensions = horizontal
                ? { x: length, z: thickness }
                : { x: thickness, z: length };

            if (segment.kind === "wall") {
                add(
                    "walls",
                    "wall",
                    {
                        x,
                        y: floorY + this.wallHeight * 0.5,
                        z
                    },
                    {
                        ...dimensions,
                        y: this.wallHeight
                    },
                    "wall"
                );

                counts.walls++;
                continue;
            }

            if (segment.kind === "door") {
                add(
                    "doors",
                    "door",
                    {
                        x,
                        y: floorY + this.doorHeight * 0.5,
                        z
                    },
                    {
                        ...dimensions,
                        y: this.doorHeight
                    },
                    "wood"
                );

                counts.doors++;
                continue;
            }

            if (segment.kind === "window") {
                const windowDepth = Math.max(
                    0.035,
                    detectedThickness * 0.25
                );

                add(
                    "glazing",
                    "window",
                    {
                        x,
                        y: floorY + this.windowSill +
                            this.windowHeight * 0.5,
                        z
                    },
                    horizontal
                        ? {
                            x: length,
                            y: this.windowHeight,
                            z: windowDepth
                        }
                        : {
                            x: windowDepth,
                            y: this.windowHeight,
                            z: length
                        },
                    "glass"
                );

                counts.windows++;
            }
        }

        const legacyTypes = new Set([
            "floor-plan-floor",
            "floor-plan-wall",
            "floor-plan-door",
            "floor-plan-window"
        ]);

        building.components = building.components.filter(
            component =>
                !legacyTypes.has(component.type) &&
                !component.type?.startsWith(prefix)
        );

        for (const component of generated) {
            building.addComponent(component);
        }

        return {
            ...counts,
            floor: floorIndex,
            totalSegments: segments.length,
            wallHeight: this.wallHeight,
            wallThickness: this.wallThickness,
            generatedComponents: generated.length
        };
    }
}

export { FloorPlanBuilder };

/**
 * FloorPlanReference
 *
 * Manages an independently positioned architectural
 * reference image in the Somnium building editor.
 *
 * Authoritative coordinates remain in simulation space.
 * SpatialTransform handles rendering coordinates.
 *
 * Babylon CreateGround uses local X-Z dimensions.
 */
class FloorPlanReference {
    constructor(scene, spatialTransform) {
        this.scene = scene;
        this.transform = spatialTransform;

        this.mesh = null;
        this.material = null;
        this.texture = null;
        this.objectURL = null;
        this.image = null;

        this.width = 20;
        this.depth = 20;

        this.opacity = 0.65;
        this.visible = true;

        this.position = {
            x: 0,
            y: 0,
            z: 0
        };

        this.altitudeOffset = 0;
        this.rotation = 0;
        this.loaded = false;
    }

    validatePosition(position) {
        if (
            !position ||
            !Number.isFinite(position.x) ||
            !Number.isFinite(position.y) ||
            !Number.isFinite(position.z)
        ) {
            throw new Error(
                "Invalid floor-plan position."
            );
        }
    }

    async load(file) {
        if (
            !file ||
            typeof file.type !== "string" ||
            !file.type.startsWith("image/")
        ) {
            throw new Error(
                "Please select a valid image file."
            );
        }

        this.remove();

        try {
            this.objectURL = URL.createObjectURL(file);
            this.image = new Image();

            await new Promise((resolve, reject) => {
                this.image.onload = resolve;
                this.image.onerror = () => reject(
                    new Error(
                        "Could not load floor-plan image."
                    )
                );

                this.image.src = this.objectURL;
            });

            if (
                !this.image.naturalWidth ||
                !this.image.naturalHeight
            ) {
                throw new Error(
                    "Floor-plan image has invalid dimensions."
                );
            }

            this.texture = new BABYLON.Texture(
                this.objectURL,
                this.scene,
                false,
                false,
                BABYLON.Texture.TRILINEAR_SAMPLINGMODE
            );

            await new Promise((resolve, reject) => {
                if (this.texture.isReady()) {
                    resolve();
                    return;
                }

                this.texture.onLoadObservable.addOnce(resolve);

                this.texture.onError = message => {
                    reject(
                        new Error(
                            message ||
                            "Floor-plan texture failed."
                        )
                    );
                };
            });

            this.material = new BABYLON.StandardMaterial(
                "building-floor-plan-material",
                this.scene
            );

            this.material.diffuseTexture = this.texture;
            this.material.emissiveTexture = this.texture;
            this.material.disableLighting = true;
            this.material.backFaceCulling = false;
            this.material.alpha = this.opacity;
            this.material.transparencyMode =
                BABYLON.Material.MATERIAL_ALPHABLEND;

            this.mesh = BABYLON.MeshBuilder.CreateGround(
                "building-floor-plan",
                {
                    width: 1,
                    height: 1,
                    subdivisions: 1
                },
                this.scene
            );

            this.mesh.material = this.material;
            this.mesh.isPickable = false;
            this.mesh.receiveShadows = false;
            this.mesh.renderingGroupId = 1;

            this.mesh.metadata = {
                type: "floor-plan-reference",
                reference: true
            };

            this.loaded = true;

            this.setVisible(this.visible);
            this.updateTransform();

            return this;

        } catch (error) {
            this.remove();
            throw error;
        }
    }

    async readPixels() {
        if (!this.image || !this.loaded) {
            throw new Error(
                "Load a floor-plan image first."
            );
        }

        const canvas = document.createElement("canvas");

        canvas.width = this.image.naturalWidth;
        canvas.height = this.image.naturalHeight;

        const context = canvas.getContext("2d", {
            willReadFrequently: true
        });

        if (!context) {
            throw new Error(
                "Image processing canvas unavailable."
            );
        }

        context.drawImage(this.image, 0, 0);

        return context.getImageData(
            0,
            0,
            canvas.width,
            canvas.height
        );
    }

    setPosition(position) {
        this.validatePosition(position);

        this.position = {
            x: position.x,
            y: position.y,
            z: position.z
        };

        this.updateTransform();
    }

    setAltitudeOffset(offset) {
        if (!Number.isFinite(offset)) {
            throw new Error(
                "Invalid altitude offset."
            );
        }

        this.altitudeOffset = offset;
        this.updateTransform();
    }

    setDimensions(width, depth) {
        if (
            !Number.isFinite(width) ||
            !Number.isFinite(depth) ||
            width <= 0 ||
            depth <= 0
        ) {
            throw new Error(
                "Invalid floor-plan dimensions."
            );
        }

        this.width = width;
        this.depth = depth;

        this.updateTransform();
    }

    setRotation(rotation) {
        if (!Number.isFinite(rotation)) {
            throw new Error(
                "Invalid floor-plan rotation."
            );
        }

        this.rotation = rotation;
        this.updateTransform();
    }

    updateTransform(
        position = this.position,
        dimensions = {},
        rotation = this.rotation
    ) {
        if (!this.mesh) return;

        if (!this.transform) {
            throw new Error(
                "SpatialTransform is unavailable."
            );
        }

        if (position) {
            this.validatePosition(position);

            this.position = {
                x: position.x,
                y: position.y,
                z: position.z
            };
        }

        const width = dimensions.width ?? this.width;
        const depth = dimensions.depth ?? this.depth;

        if (
            !Number.isFinite(width) ||
            !Number.isFinite(depth) ||
            width <= 0 ||
            depth <= 0 ||
            !Number.isFinite(rotation)
        ) {
            throw new Error(
                "Invalid floor-plan transformation."
            );
        }

        this.width = width;
        this.depth = depth;
        this.rotation = rotation;

        const worldPosition = {
            x: this.position.x,
            y: this.position.y + this.altitudeOffset,
            z: this.position.z
        };

        const rendered = this.transform.worldToRender(
            worldPosition
        );

        const cos = Math.cos(rotation);
        const sin = Math.sin(rotation);

        const xAxis = this.transform.transformDirection(
            worldPosition,
            {
                x: cos,
                y: 0,
                z: -sin
            }
        );

        const zAxis = this.transform.transformDirection(
            worldPosition,
            {
                x: sin,
                y: 0,
                z: cos
            }
        );

        const xScale = Math.hypot(
            xAxis.x,
            xAxis.y,
            xAxis.z
        );

        const zScale = Math.hypot(
            zAxis.x,
            zAxis.y,
            zAxis.z
        );

        if (
            !Number.isFinite(xScale) ||
            !Number.isFinite(zScale) ||
            xScale <= 0 ||
            zScale <= 0
        ) {
            throw new Error(
                "Invalid projected floor-plan dimensions."
            );
        }

        this.mesh.position.set(
            rendered.x,
            rendered.y + 0.08,
            rendered.z
        );

        // CreateGround occupies local X-Z, not X-Y.
        this.mesh.scaling.set(
            width * xScale,
            1,
            depth * zScale
        );

        // Align the ground plane to its projected X axis.
        const projectedYaw = Math.atan2(
            -xAxis.z,
            xAxis.x
        );

        this.mesh.rotation.set(
            0,
            projectedYaw,
            0
        );

        this.mesh.isVisible = this.visible;
        this.mesh.computeWorldMatrix(true);
    }

    setOpacity(value) {
        if (!Number.isFinite(value)) {
            throw new Error(
                "Invalid floor-plan opacity."
            );
        }

        this.opacity = Math.max(
            0,
            Math.min(1, value)
        );

        if (this.material) {
            this.material.alpha = this.opacity;
        }
    }

    setVisible(value) {
        this.visible = Boolean(value);

        if (this.mesh) {
            this.mesh.setEnabled(this.visible);
        }
    }

    toggleVisibility() {
        this.setVisible(!this.visible);
        return this.visible;
    }

    remove() {
        this.mesh?.dispose();
        this.material?.dispose();
        this.texture?.dispose();

        this.mesh = null;
        this.material = null;
        this.texture = null;

        if (this.objectURL) {
            URL.revokeObjectURL(this.objectURL);
            this.objectURL = null;
        }

        this.image = null;
        this.loaded = false;
    }

    dispose() {
        this.remove();
    }
}
