const LOG_LEVELS = Object.freeze({
    SILENT: 0,
    ERROR: 1,
    WARN: 2,
    NORMAL: 3,
    DETAIL: 4,
    DEBUG: 5
});

class SomniumLogger {
    static levels = LOG_LEVELS;
    static level = LOG_LEVELS.WARN;

    static setLevel(level) {
        if (
            !Number.isInteger(level) ||
            level < LOG_LEVELS.SILENT ||
            level > LOG_LEVELS.DEBUG
        ) {
            throw new RangeError(
                "Invalid Somnium logging level."
            );
        }

        this.level = level;
    }

    static getLevel() {
        return this.level;
    }

    static enabled(level) {
        return this.level >= level;
    }

    static error(scope, message, details = null) {
        if (!this.enabled(LOG_LEVELS.ERROR)) return;
        console.error(`[${scope}] ${message}`, details ?? "");
    }

    static warn(scope, message, details = null) {
        if (!this.enabled(LOG_LEVELS.WARN)) return;
        console.warn(`[${scope}] ${message}`, details ?? "");
    }

    static info(scope, message, details = null) {
        if (!this.enabled(LOG_LEVELS.NORMAL)) return;
        console.info(`[${scope}] ${message}`, details ?? "");
    }

    static detail(scope, message, details = null) {
        if (!this.enabled(LOG_LEVELS.DETAIL)) return;
        console.info(`[${scope}] ${message}`, details ?? "");
    }

    static debug(scope, message, details = null) {
        if (!this.enabled(LOG_LEVELS.DEBUG)) return;
        console.debug(`[${scope}] ${message}`, details ?? "");
    }
}

export { SomniumLogger, LOG_LEVELS };

function hashNoise2D(x, z, seed = 1) {

    let n =
        Math.imul(x | 0, 374761393) +
        Math.imul(z | 0, 668265263) +
        Math.imul(seed | 0, 1442695041);

    n = Math.imul(
        n ^ (n >>> 13),
        1274126177
    );

    n ^= n >>> 16;

    return (n >>> 0) / 4294967295;

}


function smoothNoise2D(x, z, seed = 1) {

    const ix = Math.floor(x);
    const iz = Math.floor(z);

    const fx = x - ix;
    const fz = z - iz;

    const u = fx * fx * (3 - 2 * fx);
    const v = fz * fz * (3 - 2 * fz);

    const a = hashNoise2D(ix, iz, seed);
    const b = hashNoise2D(ix + 1, iz, seed);
    const c = hashNoise2D(ix, iz + 1, seed);
    const d = hashNoise2D(ix + 1, iz + 1, seed);

    const ab = BABYLON.Scalar.Lerp(a, b, u);
    const cd = BABYLON.Scalar.Lerp(c, d, u);

    return BABYLON.Scalar.Lerp(ab, cd, v);

}


function fractalNoise2D(
    x,
    z,
    seed,
    octaves = 4
) {

    let value = 0;
    let amplitude = 1;
    let frequency = 1;
    let totalAmplitude = 0;

    for (let i = 0; i < octaves; i++) {

        value += smoothNoise2D(
            x * frequency,
            z * frequency,
            seed + i * 1013
        ) * amplitude;

        totalAmplitude += amplitude;

        amplitude *= 0.5;
        frequency *= 2;

    }

    return value / totalAmplitude;

}
export { hashNoise2D, smoothNoise2D, fractalNoise2D };

function CreateSeededRandom(seed) {

    let state = seed >>> 0;

    return function() {

        state += 0x6D2B79F5;

        let t = state;

        t = Math.imul(
            t ^ (t >>> 15),
            t | 1
        );

        t ^= t + Math.imul(
            t ^ (t >>> 7),
            t | 61
        );

        return (
            (t ^ (t >>> 14)) >>> 0
        ) / 4294967296;

    };

}
export { CreateSeededRandom }
