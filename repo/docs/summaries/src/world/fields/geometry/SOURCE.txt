/** Shared continuous W-freedom field and generalized chunk reachability math. */

import { norm4, subtract4 } from './spatial-4d.js?v=somnium-20261108-phase16c-anomaly-optics-2';

const AXES = ['x', 'y', 'z', 'w'];
const EPSILON = 1e-9;

/**
 * Resolves an anomaly's ambient-space field from its actual slice endpoints.
 * The default horizon extends beyond the complete axis-aligned anomaly region.
 */
function resolveAnomalyInfluence(anomaly, sourceAmbient, destinationAmbient, sourceNormal = null) {
    validatePosition4(sourceAmbient, 'Source ambient position');
    validatePosition4(destinationAmbient, 'Destination ambient position');
    const influence = anomaly.influence ?? {};
    const centerMode = influence.centerMode ?? (influence.center ? 'fixed' : 'endpoints');
    const center = centerMode === 'fixed' && influence.center
        ? { ...validatePosition4(influence.center, 'Influence center') } : {
        x: (sourceAmbient.x + destinationAmbient.x) / 2,
        y: (sourceAmbient.y + destinationAmbient.y) / 2,
        z: (sourceAmbient.z + destinationAmbient.z) / 2,
        w: (sourceAmbient.w + destinationAmbient.w) / 2
    };
    const endpointDelta = subtract4(destinationAmbient, sourceAmbient);
    const span = norm4(endpointDelta);
    // The W reach budget measures separation normal to the source slice, not
    // incidental XYZ displacement between the two portal mouths.
    const wSpan = sourceNormal
        ? Math.abs(sourceNormal.x * endpointDelta.x + sourceNormal.y * endpointDelta.y +
            sourceNormal.z * endpointDelta.z + sourceNormal.w * endpointDelta.w)
        : Math.abs(endpointDelta.w);
    const regionRadius = getRegionBoundingRadius(anomaly.region, center);
    const innerRadius = influence.innerRadius ?? clamp(span * 0.06, 1.5, 4);
    const horizonRadius = influence.horizonRadius ?? Math.max(
        regionRadius + Math.max(12, Math.min(24, span * 0.25)),
        innerRadius + 1
    );
    const defaultWReach = sourceNormal ? wSpan : (wSpan > EPSILON ? wSpan : span);
    const maxWReach = influence.maxWReach ?? defaultWReach;
    const maxWMovementScale = influence.maxWMovementScale ?? 1;

    validateRadius(innerRadius, 'Anomaly inner influence radius');
    validateRadius(horizonRadius, 'Anomaly horizon radius');
    if (horizonRadius <= innerRadius) {
        throw new RangeError('Anomaly horizon radius must exceed its inner influence radius.');
    }
    if (!Number.isFinite(maxWReach) || maxWReach <= 0) {
        throw new RangeError('Anomaly maximum W reach must be a positive finite number.');
    }
    if (!Number.isFinite(maxWMovementScale) || maxWMovementScale < 0 || maxWMovementScale > 1) {
        throw new RangeError('Anomaly W movement scale must be between zero and one.');
    }

    return {
        ...anomaly,
        influence: {
            center,
            horizonRadius,
            innerRadius,
            maxWReach,
            maxWMovementScale,
            freedomMultiplier: influence.freedomMultiplier ?? 1,
            reachMultiplier: influence.reachMultiplier ?? 1,
            centerMode
        }
    };
}

/** Evaluates the ambient 4D smootherstep field at one point. */
function evaluateAnomalyInfluence(anomaly, position4D, options = {}) {
    validatePosition4(position4D, 'Influence evaluation position');
    const influence = anomaly?.influence;
    if (!influence?.center || !Number.isFinite(influence.horizonRadius) ||
        !Number.isFinite(influence.innerRadius) || !Number.isFinite(influence.maxWReach)) {
        return {
            distance: Infinity,
            normalized: 0,
            freedom: 0,
            wReach: 0,
            wMovementScale: 0,
            insideHorizon: false,
            insideInnerRegion: false
        };
    }

    const distance = norm4(subtract4(position4D, influence.center));
    const normalized = clamp(
        (influence.horizonRadius - distance) /
            (influence.horizonRadius - influence.innerRadius),
        0,
        1
    );
    const configuredFreedom = clamp(influence.freedomMultiplier ?? 1, 0, 1);
    const configuredReach = clamp(influence.reachMultiplier ?? 1, 0, 1);
    const freedom = smootherstep(normalized) * configuredFreedom *
        clamp(options.freedomMultiplier ?? 1, 0, 1);
    const reachMultiplier = configuredReach * clamp(options.reachMultiplier ?? 1, 0, 1);
    const wReach = influence.maxWReach * freedom * reachMultiplier;
    const wMovementScale = influence.maxWMovementScale * freedom;

    return {
        distance,
        normalized,
        freedom,
        wReach,
        wMovementScale,
        insideHorizon: distance < influence.horizonRadius,
        insideInnerRegion: distance <= influence.innerRadius
    };
}

/**
 * Generic reachability predicate. Spatial and W dimensions remain independent
 * inputs so additional metric dimensions can be added without rewriting callers.
 */
function evaluateChunkReachability(parameters = {}) {
    const spatialDistance = requireNonNegative(parameters.spatialDistance, 'Spatial distance');
    const spatialReach = requireNonNegative(parameters.spatialReach, 'Spatial reach');
    const wDistance = requireNonNegative(parameters.wDistance ?? 0, 'W distance');
    const wReach = requireNonNegative(parameters.wReach ?? 0, 'W reach');
    const spatialReachable = spatialDistance <= spatialReach + EPSILON;
    const wReachable = wDistance <= wReach + EPSILON;
    return {
        reachable: spatialReachable && wReachable,
        spatialReachable,
        wReachable,
        spatialDistance,
        spatialReach,
        wDistance,
        wReach
    };
}

function getRegionBoundingRadius(region, center) {
    if (!region?.min || !region?.max) return 0;
    let maximum = 0;
    for (let mask = 0; mask < 16; mask++) {
        const corner = {};
        for (let index = 0; index < AXES.length; index++) {
            const axis = AXES[index];
            corner[axis] = mask & (1 << index) ? region.max[axis] : region.min[axis];
        }
        maximum = Math.max(maximum, norm4(subtract4(corner, center)));
    }
    return maximum;
}

function smootherstep(value) {
    const t = clamp(value, 0, 1);
    return t * t * t * (t * (t * 6 - 15) + 10);
}

function validatePosition4(value, label) {
    if (!value || !AXES.every((axis) => Number.isFinite(value[axis]))) {
        throw new TypeError(`${label} must contain finite x/y/z/w coordinates.`);
    }
    return value;
}

function validateRadius(value, label) {
    if (!Number.isFinite(value) || value <= 0) throw new RangeError(`${label} must be positive and finite.`);
}

function requireNonNegative(value, label) {
    if (!Number.isFinite(value) || value < 0) throw new RangeError(`${label} must be non-negative and finite.`);
    return value;
}

function clamp(value, minimum, maximum) {
    return Math.min(maximum, Math.max(minimum, value));
}

import { GeometryRenderer } from './geometry-renderer.js';

class BabylonGeometryAdapter {
    constructor(scene, options = {}) {
        if (!scene) throw new TypeError('BabylonGeometryAdapter requires a scene.');
        this.scene = scene;
        this.renderers = new Map();
        this.useGpu = options.useGpu ?? false;
        this.opticalScene = options.opticalScene ?? null;
        this.opticalBindings = new Map();
        this.statistics = { applied: 0, skipped: 0, missingSlice: 0 };
    }

    registerSlice(slice, spatialTransform) {
        if (!slice?.id || !spatialTransform) throw new TypeError('A slice and transform are required.');
        if (this.renderers.has(slice.id)) throw new Error(`Geometry slice already registered: ${slice.id}`);
        const renderer = new GeometryRenderer(this.scene, spatialTransform);
        this.renderers.set(slice.id, { slice, spatialTransform, renderer });
        return renderer;
    }

    applyMesh(mesh, sliceId = mesh?.metadata?.sliceId) {
        if (!mesh || !sliceId) return this.skipMissingSlice();
        const binding = this.renderers.get(sliceId);
        if (!binding) return this.skipMissingSlice();
        if (mesh.parent) {
            console.warn(`[GeometryAdapter] Parent transforms are unsupported for ${mesh.name}; mesh kept unchanged.`);
            this.statistics.skipped++;
            return false;
        }
        bakeMeshTransform(mesh);
        mesh.metadata = { ...(mesh.metadata || {}), sliceId, geometryRenderPath: 'cpu' };
        this.registerOpticalMesh(mesh, sliceId);
        const applied = binding.renderer.applyMesh(mesh);
        if (applied) this.statistics.applied++;
        else this.statistics.skipped++;
        return applied;
    }

    skipMissingSlice() {
        this.statistics.missingSlice++;
        return false;
    }

    restoreMesh(mesh) {
        const sliceId = mesh?.metadata?.sliceId;
        const binding = this.renderers.get(sliceId);
        this.unregisterOpticalMesh(mesh);
        return binding ? binding.renderer.restoreMesh(mesh) : false;
    }

    registerOpticalMesh(mesh, sliceId) {
        if (!this.opticalScene || !mesh || shouldIgnoreOpticalMesh(mesh)) return false;
        const positions = mesh.getVerticesData?.(BABYLON.VertexBuffer.PositionKind);
        const indices = mesh.getIndices?.();
        if (!positions?.length || !indices?.length) return false;
        const previousRevision = this.opticalBindings.get(mesh)?.revision ?? 0;
        this.unregisterOpticalMesh(mesh);
        const id = `babylon-mesh:${mesh.uniqueId ?? mesh.id ?? mesh.name}`;
        const materialId = mesh.material?.id ?? mesh.material?.name ?? mesh.name ?? id;
        const opticalMaterial = mesh.metadata?.opticalMaterial ?? { id: `material:${materialId}` };
        const registration = this.opticalScene.registerTriangleMesh({
            id,
            sliceId,
            positions,
            indices,
            chunkAddress: mesh.metadata?.chunkAddress ?? (
                Number.isSafeInteger(mesh.metadata?.chunkX) && Number.isSafeInteger(mesh.metadata?.chunkZ)
                    ? { sliceId, chunkX: mesh.metadata.chunkX, chunkZ: mesh.metadata.chunkZ }
                    : null
            ),
            material: opticalMaterial,
            revision: previousRevision + 1,
            enabled: mesh.isEnabled?.() !== false
        });
        this.opticalBindings.set(mesh, { id, revision: registration.revision });
        return true;
    }

    unregisterOpticalMesh(mesh) {
        const binding = this.opticalBindings.get(mesh);
        if (!binding) return false;
        this.opticalScene?.unregister(binding.id);
        this.opticalBindings.delete(mesh);
        return true;
    }

    getStats() {
        return {
            registeredSlices: this.renderers.size,
            renderers: Array.from(this.renderers, rendererStats),
            ...this.statistics,
            opticalRegistrations: this.opticalBindings.size,
            gpuEnabled: this.useGpu
        };
    }

    dispose() {
        for (const mesh of this.opticalBindings.keys()) this.unregisterOpticalMesh(mesh);
        for (const binding of this.renderers.values()) binding.renderer.dispose();
        this.renderers.clear();
    }
}

function shouldIgnoreOpticalMesh(mesh) {
    const metadata = mesh.metadata ?? {};
    return metadata.opticalIgnore === true || metadata.renderOnly === true ||
        metadata.geometryProxy === true || metadata.geometryDebugGrid === true ||
        metadata.demoLabel === true || metadata.hypercube4D === true ||
        metadata.hypercube4DSectionVertex !== undefined || mesh.isPickable === false && metadata.opticalSurface !== true;
}

function bakeMeshTransform(mesh) {
    mesh.computeWorldMatrix(true);
    if (mesh.bakeCurrentTransformIntoVertices) mesh.bakeCurrentTransformIntoVertices();
}

function rendererStats(entry) {
    return { sliceId: entry[0], ...entry[1].renderer.getStats() };
}

export { BabylonGeometryAdapter };

/** Builds a local 3D view of an authoritative four-dimensional corridor. */

import { normalize4, transportPerpendicularFrame } from './spatial-4d.js';
import { sampleSpatialCorridorFrames } from './spatial-corridor.js';

class CorridorRenderer {
    constructor(scene, corridor, spatialTransform, options = {}) {
        this.scene = scene;
        this.corridor = corridor;
        this.spatialTransform = spatialTransform;
        this.options = {
            segments: options.segments ?? 48,
            radialSegments: options.radialSegments ?? 12,
            wToZScale: options.wToZScale ?? 0.28,
            displayScale: options.displayScale ?? 0.7,
            opacity: options.opacity ?? 0.22,
            anchorT: options.anchorT ?? 0
        };
        this.validateOptions();
        this.frameSamples = sampleSpatialCorridorFrames(corridor, this.options.segments);
        const anchor = this.getFrameSample(this.options.anchorT);
        this.startPosition = this.spatialTransform.worldToRender(
            this.options.anchorT < 0.5 ? corridor.source.position : corridor.destination.position
        );
        this.firstMapped = map4To3(anchor.position, this.options.wToZScale);
        this.mesh = this.createMesh();
        this.material = createCorridorMaterial(
            scene,
            `geometryCorridorMaterial_${corridor.id}_${this.options.anchorT < 0.5 ? 'source' : 'destination'}`,
            this.options.opacity
        );
        this.mesh.material = this.material;
        this.mesh.isPickable = false;
        this.mesh.metadata = {
            ...(this.mesh.metadata || {}),
            corridorId: corridor.id,
            geometryProxy: true,
            renderOnly: true
        };
    }

    validateOptions() {
        const { segments, radialSegments, wToZScale, displayScale, opacity, anchorT } = this.options;
        if (!Number.isSafeInteger(segments) || segments < 2 || segments > 4096 ||
            !Number.isSafeInteger(radialSegments) || radialSegments < 3 ||
            !Number.isFinite(wToZScale) || !Number.isFinite(displayScale) || displayScale <= 0 ||
            !Number.isFinite(opacity) || opacity <= 0 || opacity > 1 ||
            !Number.isFinite(anchorT) || anchorT < 0 || anchorT > 1) {
            throw new RangeError('Invalid corridor rendering configuration.');
        }
    }

    createMesh() {
        const samples = this.frameSamples.map(this.mapSample.bind(this));
        const buffers = createTubeBuffers(samples, this.options.radialSegments);
        const mesh = new BABYLON.Mesh(
            `corridorProxy_${this.corridor.id}_${this.options.anchorT < 0.5 ? 'source' : 'destination'}`,
            this.scene
        );
        const data = new BABYLON.VertexData();
        data.positions = buffers.positions;
        data.normals = buffers.normals;
        data.colors = buffers.colors;
        data.indices = buffers.indices;
        data.applyToMesh(mesh, true);
        return mesh;
    }

    mapSample(sample) {
        const mapped = map4To3(sample.position, this.options.wToZScale);
        const scale = this.options.displayScale;
        const axes = sample.axes.map(mapAxis.bind(null, this.options.wToZScale));
        return {
            center: new BABYLON.Vector3(
                this.startPosition.x + (mapped.x - this.firstMapped.x) * scale,
                this.startPosition.y + (mapped.y - this.firstMapped.y) * scale,
                this.startPosition.z + (mapped.z - this.firstMapped.z) * scale
            ),
            tangent: mapAxis(this.options.wToZScale, sample.tangent),
            axes,
            axes4D: sample.axes,
            radius: sample.radius * scale
        };
    }

    getFrameSample(t) {
        const scaled = t * this.options.segments;
        const lower = Math.min(this.options.segments, Math.floor(scaled));
        const upper = Math.min(this.options.segments, lower + 1);
        const blend = scaled - lower;
        const left = this.frameSamples[lower];
        const right = this.frameSamples[upper];
        const tangent = normalize4(interpolate4(left.tangent, right.tangent, blend));
        return {
            position: interpolate4(left.position, right.position, blend),
            tangent,
            axes: transportPerpendicularFrame(left.axes, tangent),
            radius: left.radius + (right.radius - left.radius) * blend
        };
    }

    getDisplaySample(t) {
        if (!Number.isFinite(t) || t < 0 || t > 1) throw new RangeError('Corridor sample must be in [0, 1].');
        const sample = this.getFrameSample(t);
        const mapped = this.mapSample(sample);
        return {
            position: mapped.center,
            tangent: mapped.tangent,
            axes: createDisplayAxes(mapped),
            crossSectionAxes: mapped.axes.map(function toPlainAxis(axis) {
                return { x: axis.x, y: axis.y, z: axis.z };
            }),
            tangent4D: sample.tangent,
            axes4D: sample.axes,
            spatialPosition4D: sample.position,
            radius: mapped.radius
        };
    }

    dispose() {
        this.mesh.dispose();
        this.material.dispose();
    }
}

function createTubeBuffers(samples, radialSegments) {
    const positions = [];
    const normals = [];
    const colors = [];
    const indices = [];
    for (let row = 0; row < samples.length; row++) {
        const frame = createDisplayFrame(samples[row]);
        writeTubeRing(samples[row], frame, radialSegments, row, samples.length, positions, normals, colors);
        if (row > 0) writeTubeIndices(row, radialSegments, indices);
    }
    return { positions, normals, colors, indices };
}

function createDisplayAxes(sample) {
    const tangent = normalize3(sample.tangent);
    let first = orthogonalize3(sample.axes[0], tangent);
    if (length3(first) < 1e-5) first = orthogonalize3(leastAlignedAxis(tangent), tangent);
    first.normalize();
    let second = orthogonalize3(sample.axes[1], tangent);
    second = second.subtract(first.scale(BABYLON.Vector3.Dot(second, first)));
    if (length3(second) < 1e-5) second = BABYLON.Vector3.Cross(tangent, first);
    second.normalize();
    return [first, second];
}

function leastAlignedAxis(tangent) {
    const axes = [
        new BABYLON.Vector3(1, 0, 0),
        new BABYLON.Vector3(0, 1, 0),
        new BABYLON.Vector3(0, 0, 1)
    ];
    axes.sort(function compareAlignment(left, right) {
        return Math.abs(BABYLON.Vector3.Dot(left, tangent)) -
            Math.abs(BABYLON.Vector3.Dot(right, tangent));
    });
    return axes[0];
}

function createDisplayFrame(sample) {
    const [first, second] = createDisplayAxes(sample);
    return { first, second };
}

function writeTubeRing(sample, frame, radialSegments, row, rowCount, positions, normals, colors) {
    for (let column = 0; column <= radialSegments; column++) {
        const angle = column / radialSegments * Math.PI * 2;
        const normal = frame.first.scale(Math.cos(angle)).add(frame.second.scale(Math.sin(angle))).normalize();
        const point = sample.center.add(normal.scale(sample.radius));
        positions.push(point.x, point.y, point.z);
        normals.push(normal.x, normal.y, normal.z);
        const blend = row / (rowCount - 1);
        colors.push(0.22 + 0.18 * blend, 0.5 + 0.22 * (1 - blend), 0.96, 0.82);
    }
}

function writeTubeIndices(row, radialSegments, indices) {
    const previousStart = (row - 1) * (radialSegments + 1);
    const currentStart = row * (radialSegments + 1);
    for (let column = 0; column < radialSegments; column++) {
        const a = previousStart + column;
        const b = currentStart + column;
        indices.push(a, b, a + 1, a + 1, b, b + 1);
    }
}

function interpolate4(left, right, amount) {
    return {
        x: left.x + (right.x - left.x) * amount,
        y: left.y + (right.y - left.y) * amount,
        z: left.z + (right.z - left.z) * amount,
        w: left.w + (right.w - left.w) * amount
    };
}

function mapAxis(wToZScale, vector) { return map4To3(vector, wToZScale); }

function map4To3(vector, wToZScale) {
    return new BABYLON.Vector3(vector.x, vector.y, vector.z + vector.w * wToZScale);
}

function orthogonalize3(vector, tangent) {
    return vector.subtract(tangent.scale(BABYLON.Vector3.Dot(vector, tangent)));
}

function normalize3(vector) {
    const result = new BABYLON.Vector3(vector.x, vector.y, vector.z);
    return result.lengthSquared() > 1e-12 ? result.normalize() : new BABYLON.Vector3(0, 0, 1);
}

function length3(vector) { return Math.hypot(vector.x, vector.y, vector.z); }

function createCorridorMaterial(scene, name, opacity) {
    const material = new BABYLON.StandardMaterial(name, scene);
    material.useVertexColors = true;
    material.diffuseColor = BABYLON.Color3.White();
    material.emissiveColor = new BABYLON.Color3(0.08, 0.12, 0.3);
    material.alpha = opacity;
    material.disableDepthWrite = true;
    material.backFaceCulling = false;
    material.disableLighting = true;
    return material;
}

        this.entryPosition = this.player.getRenderPosition();
        this.entryYaw = this.player.yaw;
        this.activeRenderer = this.renderers.get(this.currentSliceId) ?? this.renderer;
        this.state = 'transit';
        this.movement.setMovementSuspended(true);
        this.player.setSpatialContext?.(
            this.startWorld.transform,
            this.startWorld.slice.id,
            this.startWorld.layerMask
        );
        this.camera.setSpatialContext(this.startWorld.transform, this.startWorld.layerMask);
        this.corridorChunk?.enter(this.player, { direction: this.direction, progress: 0 });
        this.updatePlayerOnCorridor();
        this.onSliceChanged(this.currentSliceId, {
            transitioning: true,
            corridorId: this.corridor.id,
            renderSliceId: this.currentSliceId,
            direction: this.direction
        });
        return true;
    }

    tryBeginEntityTransit(entity, fromSliceId = entity?.sliceId) {
        this.assertActive();
        if (!entity?.worldPosition || !Number.isFinite(entity.yaw)) return false;
        const forwardX = Math.sin(entity.yaw);
        const forwardZ = Math.cos(entity.yaw);
        for (const neighbor of this.topology.getTraversableNeighbors(fromSliceId)) {
            if (neighbor.corridorId !== this.corridor.id) continue;
            const fromSource = this.corridor.source.sliceId === fromSliceId;
            const endpoint = fromSource ? this.corridor.source : this.corridor.destination;
            const dx = endpoint.position.x - entity.worldPosition.x;
            const dz = endpoint.position.z - entity.worldPosition.z;
            const distance = Math.hypot(dx, dz);
            if (distance < 1e-6 || distance > this.options.triggerDistance) continue;
            const heading = (forwardX * dx + forwardZ * dz) / distance;
            if (heading >= this.options.headingThreshold) return this.beginEntityTransit(entity, fromSliceId);
        }
        return false;
    }

    beginEntityTransit(entity, fromSliceId = entity?.sliceId) {
        this.assertActive();
        if (!entity || entity === this.player) throw new TypeError('Use player traversal for the active player.');
        if (this.entityTransits.has(entity)) return false;
        const link = this.topology.getTraversableNeighbors(fromSliceId)
            .find(function findCorridor(neighbor) { return neighbor.corridorId === this.corridor.id; }, this);
        if (!link) return false;
        const position = entity.worldPosition;
        if (!position || !this.hasChunkConnection(fromSliceId, position)) return false;
        const fromSource = fromSliceId === this.corridor.source.sliceId;
        const target = this.worlds.get(link.sliceId);
        if (!target) throw new Error(`Missing corridor destination slice: ${link.sliceId}`);
        const speed = entity.corridorSpeed ?? this.options.entitySpeed;
        if (!Number.isFinite(speed) || speed <= 0) throw new RangeError('Entity corridorSpeed must be positive.');
        const state = {
            entity, fromSliceId, target, direction: fromSource ? 1 : -1,
            progress: 0, speed,
            entryPosition: getEntityRenderPosition(entity),
            entryYaw: entity.yaw ?? 0,
            renderer: this.renderers.get(fromSliceId) ?? this.renderer
        };
        this.entityTransits.set(entity, state);
        this.corridorChunk?.enter(entity, { direction: state.direction, progress: 0 });
        return true;
    }

    hasChunkConnection(sliceId, position) {
        if (!this.chunkGraph) return true;
        const address = this.chunkGraph.getAddress(sliceId, position);
        for (let offsetX = -1; offsetX <= 1; offsetX++) {
            for (let offsetZ = -1; offsetZ <= 1; offsetZ++) {
                const nearby = {
                    sliceId,
                    chunkX: address.chunkX + offsetX,
                    chunkZ: address.chunkZ + offsetZ
                };
                const connected = this.chunkGraph.getNeighbors(nearby).some(function isThisCorridor(node) {
                    return node.kind === 'corridor' && node.id === this.corridor.id;
                }, this);
                if (connected) return true;
            }
        }
        return false;
    }

    advanceTransit(deltaSeconds) {
        const input = this.getMovementInput();
        const forward = clamp(Number(input.forward) || 0, -1, 1);
        const strafe = clamp(Number(input.strafe) || 0, -1, 1);
        const vertical = clamp(Number(input.vertical) || 0, -1, 1);
        const crossSectionW = clamp(Number(input.crossSectionW) || 0, -1, 1);
        const speed = input.sprint ? this.options.sprintSpeed : this.options.speed;
        const next = this.progress + forward * speed * deltaSeconds / this.corridorLength;
        if (next >= 1) {
            this.progress = 1;
            this.updatePlayerOnCorridor();
            this.completeTransit(this.target);
            return;
        }
        if (next < 0 && this.progress > 0) {
            this.progress = 0;
            this.updatePlayerOnCorridor();
            this.completeTransit(this.startWorld);
            return;
        }
        this.progress = clamp(next, 0, 1);
        const radius = this.getCurrentRadius();
        const transverseLimit = radius * 0.62;
        this.offsetU = clamp(
            this.offsetU + strafe * this.options.transverseSpeed * deltaSeconds,
            -transverseLimit,
            transverseLimit
        );
        this.offsetV = clamp(
            this.offsetV + vertical * this.options.transverseSpeed * deltaSeconds,
            -transverseLimit,
            transverseLimit
        );
        this.offsetW = clamp(
            this.offsetW + crossSectionW * this.options.transverseSpeed * deltaSeconds,
            -transverseLimit,
            transverseLimit
        );
        this.constrainCrossSection(transverseLimit);
        this.updatePlayerOnCorridor();
    }

    updatePlayerOnCorridor() {
        const sample = this.getTransitSample(
            this.activeRenderer,
            this.progress,
            this.direction,
            this.offsetU,
            this.offsetV,
            this.offsetW,
            this.entryPosition,
            this.entryYaw
        );
        this.spatialPosition4D = sample.spatialPosition4D;
        this.player.spatialPosition4D = { ...sample.spatialPosition4D };
        this.player.setRenderPosition(sample.position, sample.yaw);
        this.corridorChunk?.updateOccupant(this.player, this.progress, sample.spatialPosition4D);
        this.camera.setRenderAnchor(sample.position, sample.yaw, this.getCameraLookTarget());
    }

    getCameraLookTarget() {
        const ahead = Math.min(0.985, Math.max(0.76, this.progress + 0.18));
        const t = this.direction > 0 ? ahead : 1 - ahead;
        const sample = this.activeRenderer.getDisplaySample(t);
        const axes = sample.axes ?? [];
        const cross = sample.crossSectionAxes ?? [];
        const scale = this.activeRenderer.options?.displayScale ?? 1;
        const axisU = normalize3(axes[0] ?? cross[0]);
        const axisV = orthogonalize3(normalize3(axes[1] ?? cross[1]), axisU);
        const axisW = normalizeOrZero3(cross[2]);
        return add3(
            add3(
                add3(sample.position, scale3(axisU, this.offsetU * scale)),
                scale3(axisV, this.offsetV * scale)
            ),
            scale3(axisW, this.offsetW * scale)
        );
    }

    advanceOtherEntities(deltaSeconds) {
        for (const state of this.entityTransits.values()) {
            state.progress = Math.min(1, state.progress + state.speed * deltaSeconds / this.corridorLength);
            const sample = this.getTransitSample(
                state.renderer, state.progress, state.direction, 0, 0,
                0, state.entryPosition, state.entryYaw
            );
            state.entity.spatialPosition4D = { ...sample.spatialPosition4D };
            setEntityRenderPosition(state.entity, sample.position, sample.yaw);
            this.corridorChunk?.updateOccupant(state.entity, state.progress, sample.spatialPosition4D);
            if (state.progress >= 1) this.completeEntityTransit(state);
        }
    }

    constrainCrossSection(limit) {
        const length = Math.hypot(this.offsetU, this.offsetV, this.offsetW);
        if (length <= limit || length <= 1e-9) return;
        const scale = limit / length;
        this.offsetU *= scale;
        this.offsetV *= scale;
        this.offsetW *= scale;
    }

    getTransitSample(renderer, progress, direction, offsetU, offsetV, offsetW, entryPosition, entryYaw) {
        const t = direction > 0 ? progress : 1 - progress;
        const sample = renderer.getDisplaySample(t);
        const axes = sample.axes ?? [];
        const axisU = normalize3(axes[0]);
        const axisV = orthogonalize3(normalize3(axes[1]), axisU);
        const axisW = normalizeOrZero3(sample.crossSectionAxes?.[2]);
        const scale = renderer.options?.displayScale ?? 1;
        const offset = add3(
            add3(scale3(axisU, offsetU * scale), scale3(axisV, offsetV * scale)),
            scale3(axisW, offsetW * scale)
        );
        const target = add3(sample.position, offset);
        const travelled = progress * this.corridorLength;
        const blend = smoothstep(travelled / this.options.entryBlendDistance);
        const position = lerpPoint(entryPosition, target, blend);
        const yaw = lerpAngle(entryYaw, getYaw(sample.tangent, direction), blend);
        const base = sample.spatialPosition4D ?? sample.position4D ?? { x: 0, y: 0, z: 0, w: 0 };
        const axes4D = sample.axes4D ?? [];
        const spatialPosition4D = addScaled4(base, axes4D[0], offsetU);
        const secondPosition4D = addScaled4(spatialPosition4D, axes4D[1], offsetV);
        const finalPosition4D = addScaled4(secondPosition4D, axes4D[2], offsetW);
        return { position, yaw, progress, spatialPosition4D: finalPosition4D };
    }

    getCurrentRadius() {
        const t = this.direction > 0 ? this.progress : 1 - this.progress;
        const radius = this.corridor.radiusProfile;
        for (let index = 1; index < radius.length; index++) {
            if (t > radius[index].t) continue;
            const left = radius[index - 1];
            const right = radius[index];
            const blend = (t - left.t) / (right.t - left.t);
            return left.radius + (right.radius - left.radius) * blend;
        }
        return radius[radius.length - 1].radius;
    }

    completeTransit(world) {
        const arrival = this.createArrivalPosition(world, this.spatialPosition4D);
        this.player.setSpatialContext(world.transform, world.slice.id, world.layerMask);
        this.player.yaw = world.slice.id === this.corridor.destination.sliceId ? Math.PI : 0;
        this.player.spatialPosition4D = null;
        this.player.setWorldPosition(arrival);
        this.movement.setTerrain(world.terrain);
        this.movement.setMovementSuspended(false);
        this.camera.clearRenderAnchor();
        this.camera.setSpatialContext(world.transform, world.layerMask);
        this.currentSliceId = world.slice.id;
        this.corridorChunk?.leave(this.player);
        this.state = 'idle';
        this.progress = 0;
        this.direction = 0;
        this.startWorld = null;
        this.target = null;
        this.activeRenderer = null;
        this.offsetU = 0;
        this.offsetV = 0;
        this.offsetW = 0;
        this.spatialPosition4D = null;
        this.onSliceChanged(this.currentSliceId, { transitioning: false, corridorId: this.corridor.id });
        return arrival;
    }

    completeEntityTransit(state) {
        const { entity, target } = state;
        const arrival = this.createArrivalPosition(target, entity.spatialPosition4D);
        setEntitySpatialContext(entity, target);
        setEntityWorldPosition(entity, arrival, target);
        entity.spatialPosition4D = null;
        this.corridorChunk?.leave(entity);
        this.entityTransits.delete(entity);
        return arrival;
    }

    createArrivalPosition(world, spatialPosition4D) {
        const atDestination = world.slice.id === this.corridor.destination.sliceId;
        const endpoint = atDestination ? this.corridor.destination : this.corridor.source;
        const local = spatialPosition4D && world.slice.frame
            ? ambientToSlice(world.slice.frame, spatialPosition4D).position
            : endpoint.position;
        const x = clamp(local.x, world.terrain.min.x, world.terrain.max.x);
        const z = clamp(local.z + (atDestination ? 5 : -5), world.terrain.min.z, world.terrain.max.z);
        const sample = world.terrain.sampleAt(x, z);
        if (!sample) throw new RangeError('Corridor arrival lies outside destination terrain.');
        return { x, y: sample.y, z };
    }

    getState() {
        return {
            state: this.state,
            currentSliceId: this.currentSliceId,
            startSliceId: this.startWorld?.slice.id ?? null,
            destinationSliceId: this.target?.slice.id ?? null,
            progress: this.progress,
            direction: this.direction,
            corridorLength: this.corridorLength,
            speed: this.options.speed,
            position4D: this.spatialPosition4D && { ...this.spatialPosition4D },
            corridorOccupants: this.corridorChunk?.occupants.size ?? this.entityTransits.size
        };
    }

    getEntityTransitState(entity) {
        const state = this.entityTransits.get(entity);
        return state ? {
            fromSliceId: state.fromSliceId,
            toSliceId: state.target.slice.id,
            progress: state.progress,
            direction: state.direction
        } : null;
    }

    validateOptions() {
        const { triggerDistance, speed, sprintSpeed, transverseSpeed, entitySpeed,
            entryBlendDistance, lengthSegments, headingThreshold } = this.options;
        if (![triggerDistance, speed, sprintSpeed, transverseSpeed, entitySpeed, entryBlendDistance]
            .every(Number.isFinite) || triggerDistance <= 0 || speed <= 0 || sprintSpeed < speed ||
            transverseSpeed <= 0 || entitySpeed <= 0 || entryBlendDistance <= 0 ||
            !Number.isSafeInteger(lengthSegments) || lengthSegments < 8 || lengthSegments > 4096) {
            throw new RangeError('Corridor movement speeds and sampling options must be positive and valid.');
        }
        if (!Number.isFinite(headingThreshold) || headingThreshold < -1 || headingThreshold > 1) {
            throw new RangeError('Corridor traversal headingThreshold must be between -1 and 1.');
        }
    }

    assertActive() {
        if (this.disposed || !this.active) throw new Error('Corridor traversal is not active.');
    }

    dispose() {
        if (this.disposed) return;
        this.disposed = true;
        this.active = false;
        if (this.state === 'transit') {
            this.movement.setMovementSuspended(false);
            this.corridorChunk?.leave(this.player);
        }
        for (const entity of this.entityTransits.keys()) this.corridorChunk?.leave(entity);
        this.entityTransits.clear();
        this.state = 'disposed';
        this.target = null;
        this.startWorld = null;
    }
}

function measureCorridor(corridor, segments) {
    let length = 0;
    let previous = sampleSpatialCorridor(corridor, 0).position;
    for (let index = 1; index <= segments; index++) {
        const current = sampleSpatialCorridor(corridor, index / segments).position;
        length += Math.hypot(
            current.x - previous.x, current.y - previous.y,
            current.z - previous.z, current.w - previous.w
        );
        previous = current;
    }
    if (!Number.isFinite(length) || length <= 1e-9) throw new RangeError('Corridor length must be positive.');
    return length * (corridor.intrinsicLengthScale ?? 1);
}

function getYaw(tangent, direction) {
    return Math.atan2(tangent.x * direction, tangent.z * direction);
}

function normalize3(vector) {
    const length = Math.hypot(vector?.x ?? 0, vector?.y ?? 0, vector?.z ?? 0);
    return length > 1e-9
        ? { x: vector.x / length, y: vector.y / length, z: vector.z / length }
        : { x: 1, y: 0, z: 0 };
}

function normalizeOrZero3(vector) {
    const length = Math.hypot(vector?.x ?? 0, vector?.y ?? 0, vector?.z ?? 0);
    return length > 1e-9
        ? { x: vector.x / length, y: vector.y / length, z: vector.z / length }
        : { x: 0, y: 0, z: 0 };
}

function orthogonalize3(vector, axis) {
    const dot = vector.x * axis.x + vector.y * axis.y + vector.z * axis.z;
    return normalize3({
        x: vector.x - axis.x * dot,
        y: vector.y - axis.y * dot,
        z: vector.z - axis.z * dot
    });
}

function add3(left, right) {
    return { x: left.x + right.x, y: left.y + right.y, z: left.z + right.z };
}

function scale3(vector, amount) {
    return { x: vector.x * amount, y: vector.y * amount, z: vector.z * amount };
}

function addScaled4(position, axis, amount) {
    if (!axis || !amount) return { ...position };
    return {
        x: position.x + axis.x * amount,
        y: position.y + axis.y * amount,
        z: position.z + axis.z * amount,
        w: position.w + axis.w * amount
    };
}

function lerpPoint(left, right, amount) {
    return {
        x: left.x + (right.x - left.x) * amount,
        y: left.y + (right.y - left.y) * amount,
        z: left.z + (right.z - left.z) * amount
    };
}

function lerpAngle(left, right, amount) {
    const delta = Math.atan2(Math.sin(right - left), Math.cos(right - left));
    return left + delta * amount;
}

function smoothstep(value) {
    const amount = clamp(value, 0, 1);
    return amount * amount * (3 - 2 * amount);
}

function clamp(value, minimum, maximum) {
    return Math.min(maximum, Math.max(minimum, value));
}

function getEntityRenderPosition(entity) {
    if (typeof entity.getRenderPosition === 'function') return entity.getRenderPosition();
    const position = entity.root?.position ?? entity.mesh?.position ?? entity.position;
    if (!position) throw new TypeError('Spatial entities require getRenderPosition() or a render position.');
    return { x: position.x, y: position.y, z: position.z };
}

function setEntityRenderPosition(entity, position, yaw) {
    if (typeof entity.setRenderPosition === 'function') {
        entity.setRenderPosition(position, yaw);
        return;
    }
    const target = entity.root ?? entity.mesh ?? entity;
    if (typeof target.position?.set !== 'function') throw new TypeError('Spatial entities require setRenderPosition() or a Babylon position.');
    target.position.set(position.x, position.y, position.z);
    if (Number.isFinite(yaw) && target.rotation) target.rotation.y = yaw;
}

function setEntitySpatialContext(entity, world) {
    if (typeof entity.setSpatialContext === 'function') {
        entity.setSpatialContext(world.transform, world.slice.id, world.layerMask);
        return;
    }
    entity.transform = world.transform;
    entity.sliceId = world.slice.id;
    entity.layerMask = world.layerMask;
    const target = entity.root ?? entity.mesh ?? entity;
    if (target.layerMask !== undefined) target.layerMask = world.layerMask;
    if (target.metadata) target.metadata.sliceId = world.slice.id;
}

function setEntityWorldPosition(entity, position, world) {
    if (typeof entity.setWorldPosition === 'function') {
        entity.setWorldPosition(position);
        return;
    }
    entity.worldPosition = { ...position };
    const render = world.transform.worldToRender(position);
    setEntityRenderPosition(entity, render, entity.yaw ?? 0);
}

function requireObject(value, name) {
    if (!value || typeof value !== 'object') throw new TypeError(`CorridorTraversalController requires ${name}.`);
    return value;
}

export { CorridorTraversalController, measureCorridor };

class GeodesicSolver {

    constructor(geometry, options = {}) {

        this.geometry = geometry;

        this.iterations =
            options.iterations ?? 80;

        this.relaxation =
            options.relaxation ?? 0.15;

        this.segments =
            options.segments ?? 24;

    }

    solve(start, end) {

        const points = [];

        for (
            let i = 0;
            i <= this.segments;
            i++
        ) {

            const t = i / this.segments;

            points.push({

                x: start.x + (end.x - start.x) * t,

                y: start.y + (end.y - start.y) * t,

                z: start.z + (end.z - start.z) * t

            });

        }

        const step = this.relaxation;

        for (
            let iteration = 0;
            iteration < this.iterations;
            iteration++
        ) {

            for (
                let i = 1;
                i < points.length - 1;
                i++
            ) {

                const previous = points[i - 1];

                const current = points[i];

                const next = points[i + 1];

                const midpoint = {

                    x: (previous.x + next.x) * 0.5,

                    y: (previous.y + next.y) * 0.5,

                    z: (previous.z + next.z) * 0.5

                };

                const candidate = {

                    x: current.x +
                        (midpoint.x - current.x) * step,

                    y: current.y +
                        (midpoint.y - current.y) * step,

                    z: current.z +
                        (midpoint.z - current.z) * step

                };

                // Reject candidates outside bounded geometries.
                if (
                    typeof this.geometry.isInside === "function" &&
                    !this.geometry.isInside(candidate)
                ) {
                    continue;
                }

                points[i] = candidate;

            }

        }

        return {

            points,

            length: this.calculateLength(points)

        };

    }

    calculateLength(points) {

        let length = 0;

        for (
            let i = 0;
            i < points.length - 1;
            i++
        ) {

            length += this.geometry.distance(
                points[i],
                points[i + 1]
            );

        }

        return length;

    }

}

export { GeodesicSolver };

/** Toggleable per-slice metric grid, optionally conformed to local terrain. */

class GeometryDebugGrid {
    constructor(scene, adapter, options = {}) {
        this.scene = scene;
        this.adapter = adapter;
        this.sliceId = options.sliceId;
        this.size = options.size ?? 24;
        this.spacing = options.spacing ?? 2;
        this.center = {
            x: options.center?.x ?? 0,
            z: options.center?.z ?? 0
        };
        this.heightAt = typeof options.heightAt === 'function' ? options.heightAt : null;
        this.sampleStep = options.sampleStep ?? Math.min(this.spacing, 2);
        this.visible = options.visible ?? true;
        this.active = true;
        this.mesh = this.createMesh(options.y ?? 0.04);
        this.mesh.isVisible = this.active && this.visible;
    }

    createMesh(height) {
        const mesh = BABYLON.MeshBuilder.CreateLineSystem(
            `geometryMetricDebugGrid-${this.sliceId ?? 'unbound'}`,
            {
                lines: createGridLines(
                    this.size, this.spacing, height, this.center,
                    this.heightAt, this.sampleStep
                )
            },
            this.scene
        );
        mesh.color = new BABYLON.Color3(0.18, 0.78, 0.94);
        mesh.alpha = 0.72;
        mesh.isPickable = false;
        mesh.metadata = { sliceId: this.sliceId, geometryDebugGrid: true };
        this.adapter.applyMesh(mesh, this.sliceId);
        return mesh;
    }

    toggle() {
        this.setVisible(!this.visible);
        return this.visible;
    }

    setVisible(visible) {
        this.visible = Boolean(visible);
        this.mesh.isVisible = this.active && this.visible;
    }

    setActive(active) {
        this.active = Boolean(active);
        this.mesh.isVisible = this.active && this.visible;
    }

    getInfo() {
        return {
            visible: this.visible,
            active: this.active,
            sliceId: this.sliceId,
            size: this.size,
            spacing: this.spacing
        };
    }

    dispose() {
        this.mesh.dispose();
    }
}

function createGridLines(size, spacing, height, center, heightAt, sampleStep) {
    const lines = [];
    const half = size / 2;
    const segments = Math.max(1, Math.ceil(size / sampleStep));
    const step = size / segments;

    for (let offset = -half; offset <= half + 1e-6; offset += spacing) {
        if (!heightAt) {
            lines.push([
                new BABYLON.Vector3(center.x + offset, height, center.z - half),
                new BABYLON.Vector3(center.x + offset, height, center.z + half)
            ]);
            lines.push([
                new BABYLON.Vector3(center.x - half, height, center.z + offset),
                new BABYLON.Vector3(center.x + half, height, center.z + offset)
            ]);
            continue;
        }

        let longitudinal = [];
        let transverse = [];
        for (let index = 0; index <= segments; index++) {
            const distance = -half + index * step;
            const xForLongitudinal = center.x + offset;
            const zForLongitudinal = center.z + distance;
            const longY = sampleHeight(heightAt, xForLongitudinal, zForLongitudinal);
            longitudinal = appendTerrainSample(
                lines, longitudinal, xForLongitudinal, longY, zForLongitudinal
            );

            const xForTransverse = center.x + distance;
            const zForTransverse = center.z + offset;
            const transverseY = sampleHeight(heightAt, xForTransverse, zForTransverse);
            transverse = appendTerrainSample(
                lines, transverse, xForTransverse, transverseY, zForTransverse
            );
        }
        flushTerrainLine(lines, longitudinal);
        flushTerrainLine(lines, transverse);
    }
    return lines;
}

function sampleHeight(heightAt, x, z) {
    try {
        const value = heightAt(x, z);
        return Number.isFinite(value) ? value : null;
    } catch (error) {
        return null;
    }
}

function appendTerrainSample(lines, line, x, y, z) {
    if (Number.isFinite(y)) {
        line.push(new BABYLON.Vector3(x, y, z));
        return line;
    }
    flushTerrainLine(lines, line);
    return [];
}

function flushTerrainLine(lines, line) {
    if (line.length > 1) lines.push(line);
}

/**
 * GeometryRenderer
 *
 * Applies spatial transformations to Babylon.js
 * mesh geometry.
 *
 * Initial target:
 * - Streamed terrain meshes.
 *
 * Preserves:
 * - Original vertex positions.
 * - Original vertex normals.
 * - Mesh materials.
 * - Vertex colors.
 * - Mesh indices.
 *
 * Simulation coordinates remain unchanged.
 */

class GeometryRenderer {

    constructor(scene, spatialTransform, options = {}) {

        if (!scene || !spatialTransform) {
            throw new Error(
                "GeometryRenderer requires a scene and SpatialTransform."
            );
        }

        this.scene = scene;
        this.transform = spatialTransform;

        this.enabled = options.enabled ?? true;

        this.meshCache = new WeakMap();

        this.statistics = {
            transformedMeshes: 0,
            transformedVertices: 0,
            restoredMeshes: 0
        };

        console.log(
            "[GeometryRenderer] Initialized:",
            this.transform.getInfo()
        );
    }

    // -----------------------------------------
    // MESH TRANSFORMATION
    // -----------------------------------------

    applyMesh(mesh) {

        if (!this.enabled || !mesh) {
            return false;
        }

        if (this.transform.geometryName === "euclidean") {
            return false;
        }

        const cached = this.meshCache.get(mesh);

        if (cached) {
            return false;
        }

        const positions =
            mesh.getVerticesData(
                BABYLON.VertexBuffer.PositionKind
            );

        const indices = mesh.getIndices();

        if (!positions || positions.length === 0) {
            return false;
        }

        // Preserve authoritative mesh geometry.

        const originalPositions =
            new Float32Array(positions);

        const originalNormals =
            mesh.getVerticesData(
                BABYLON.VertexBuffer.NormalKind
            );

        const cache = {
            positions: originalPositions,

            normals: originalNormals
                ? new Float32Array(originalNormals)
                : null
        };

        this.meshCache.set(mesh, cache);

        // Project around a stable local anchor so mesh neighbourhoods expand.
        const anchor = getPositionAnchor(positions);
        const projectedAnchor = this.transform.worldToRender(anchor);
        const visualScale = this.transform.getVisualScale(anchor);
        const transformedPositions = new Float32Array(positions.length);

        for (let index = 0; index < positions.length; index += 3) {
            const projected = this.transform.worldToRender({
                x: positions[index],
                y: positions[index + 1],
                z: positions[index + 2]
            });
            transformedPositions[index] = projectedAnchor.x + (projected.x - projectedAnchor.x) * visualScale;
            transformedPositions[index + 1] = projectedAnchor.y + (projected.y - projectedAnchor.y) * visualScale;
            transformedPositions[index + 2] = projectedAnchor.z + (projected.z - projectedAnchor.z) * visualScale;
        }

        const transformedNormals = originalNormals && indices
            ? new Float32Array(positions.length)
            : null;
        if (transformedNormals) {
            BABYLON.VertexData.ComputeNormals(transformedPositions, indices, transformedNormals);
        }

        mesh.setVerticesData(
            BABYLON.VertexBuffer.PositionKind,
            transformedPositions,
            true,
            3
        );

        if (transformedNormals) {
            mesh.setVerticesData(
                BABYLON.VertexBuffer.NormalKind,
                transformedNormals,
                true,
                3
            );
        }

        mesh.refreshBoundingInfo();

        mesh.computeWorldMatrix(true);

        mesh.metadata = {
            ...(mesh.metadata || {}),
            spatialGeometry: this.transform.geometryName,
            spatiallyTransformed: true
        };

        this.statistics.transformedMeshes++;

        this.statistics.transformedVertices += positions.length / 3;

        return true;
    }

    // -----------------------------------------
    // RESTORE ORIGINAL GEOMETRY
    // -----------------------------------------

    restoreMesh(mesh) {

        if (!mesh) return false;

        const cache = this.meshCache.get(mesh);

        if (!cache) return false;

        mesh.setVerticesData(
            BABYLON.VertexBuffer.PositionKind,
            cache.positions,
            true,
            3
        );

        if (cache.normals) {

            mesh.setVerticesData(
                BABYLON.VertexBuffer.NormalKind,
                cache.normals,
                true,
                3
            );
        }

        mesh.refreshBoundingInfo();

        mesh.computeWorldMatrix(true);

        if (mesh.metadata) {

            mesh.metadata.spatiallyTransformed = false;

            delete mesh.metadata.spatialGeometry;
        }

        this.meshCache.delete(mesh);

        this.statistics.restoredMeshes++;

        return true;
    }

    // -----------------------------------------
    // RESTORE ALL KNOWN CHUNKS
    // -----------------------------------------

    restoreWorld(worldManager) {

        if (!worldManager) return;

        for (const chunk of worldManager.chunks.values()) {

            if (chunk.mesh) {
                this.restoreMesh(chunk.mesh);
            }
        }
    }

    // -----------------------------------------
    // REPROJECT ALL KNOWN CHUNKS
    // -----------------------------------------

    reprojectWorld(worldManager) {

        if (!worldManager) return;

        for (const chunk of worldManager.chunks.values()) {

            if (!chunk.mesh) continue;

            this.restoreMesh(chunk.mesh);

            this.applyMesh(chunk.mesh);
        }
    }

    // -----------------------------------------
    // CONFIGURATION
    // -----------------------------------------

    setEnabled(enabled, worldManager = null) {

        this.enabled = !!enabled;

        if (!worldManager) return;

        if (this.enabled) {
            this.reprojectWorld(worldManager);
        } else {
            this.restoreWorld(worldManager);
        }
    }

    // -----------------------------------------
    // DIAGNOSTICS
    // -----------------------------------------

    getStats() {

        return {
            enabled: this.enabled,

            geometry: this.transform.geometryName,
            ...this.statistics
        };
    }

    // -----------------------------------------
    // DISPOSAL
    // -----------------------------------------

    dispose(worldManager = null) {

        if (worldManager) {
            this.restoreWorld(worldManager);
        }

        this.enabled = false;

        console.log(
            "[GeometryRenderer] Disposed:",
            this.getStats()
        );
    }
	

    // -----------------------------------------
    // THIN INSTANCE TRANSFORMATION
    // -----------------------------------------

    transformInstanceMatrix(matrix) {

        if (
            !this.enabled ||
            this.transform.geometryName === "euclidean"
        ) {
            return matrix;
        }

        if (
            !matrix ||
            !matrix.m ||
            !Array.from(matrix.m).every(Number.isFinite)
        ) {
            return null;
        }

        // Extract translation directly from the matrix.

        const translation = matrix.getTranslation();

        if (
            !translation ||
            !Number.isFinite(translation.x) ||
            !Number.isFinite(translation.y) ||
            !Number.isFinite(translation.z)
        ) {
            return null;
        }

        // Reject positions outside the authoritative
        // projection domain.

        let projected;

        try {

            projected =
                this.transform.worldToRender({
                    x: translation.x,
                    y: translation.y,
                    z: translation.z
                });

        } catch (error) {

            return null;
        }

        const scaling = new BABYLON.Vector3();
        const rotation = new BABYLON.Quaternion();
        const decomposedTranslation =
            new BABYLON.Vector3();

        if (
            !matrix.decompose(
                scaling,
                rotation,
                decomposedTranslation
            )
        ) {
            return null;
        }

        // Validate the local transformation Jacobian.

        let J;

        try {

            J = this.transform.getJacobian({
                x: translation.x,
                y: translation.y,
                z: translation.z
            });

        } catch (error) {

            return null;
        }

        if (!J.every(Number.isFinite)) {
            return null;
        }

        // Estimate volume scaling.

        const determinant =
            J[0] * (J[4] * J[8] - J[5] * J[7]) -
            J[1] * (J[3] * J[8] - J[5] * J[6]) +
            J[2] * (J[3] * J[7] - J[4] * J[6]);

        if (!Number.isFinite(determinant)) {
            return null;
        }

        const scaleFactor = Math.cbrt(
            Math.max(Math.abs(determinant), 1e-18)
        );

        const adjustedScale =
            scaling.scale(scaleFactor);

        return BABYLON.Matrix.Compose(
            adjustedScale,
            rotation,
            new BABYLON.Vector3(
                projected.x,
                projected.y,
                projected.z
            )
        );
    }

	
    // -----------------------------------------
    // CAMERA BOUNDARY PROTECTION
    // -----------------------------------------

    constrainCamera(camera) {

        if (
            !camera ||
            this.transform.geometryName !== "poincare"
        ) {
            return;
        }

        const scale = this.transform.renderScale;
        const radius = this.transform.projection.radius;

        const x = camera.position.x / scale;
        const y = camera.position.y / scale;
        const z = camera.position.z / scale;

        const distance = Math.hypot(x, y, z);

        const maximumRadius =
            radius * (1 - 1e-9);

        if (distance >= maximumRadius) {

            const factor =
                maximumRadius / Math.max(distance, 1e-12);

            camera.position.scaleInPlace(factor);
        }
    }

    getWorldCameraPosition(camera) {

        this.constrainCamera(camera);

        return this.transform.renderToWorld({
            x: camera.position.x,
            y: camera.position.y,
            z: camera.position.z
        });
    }




}

function getPositionAnchor(positions) {
    const count = positions.length / 3;
    const anchor = { x: 0, y: 0, z: 0 };
    for (let index = 0; index < positions.length; index += 3) {
        anchor.x += positions[index];
        anchor.y += positions[index + 1];
        anchor.z += positions[index + 2];
    }
    anchor.x /= count;
    anchor.y /= count;
    anchor.z /= count;
    return anchor;
}

export { GeometryRenderer };

/**
 * GeometryShaderManager
 *
 * GPU-side spatial projection for Babylon.js.
 *
 * Initial implementation:
 * - Three-dimensional Poincare projection.
 * - Configurable curvature.
 * - Configurable world bounds.
 * - Basic diffuse lighting.
 * - Independent shader material management.
 *
 * Initial limitation:
 * - Ordinary meshes only.
 * - Thin instances require a separate shader path.
 */

class GeometryShaderManager {

    constructor(scene, spatialTransform, options = {}) {

        if (!scene || !spatialTransform) {
            throw new Error(
                "GeometryShaderManager requires a scene and SpatialTransform."
            );
        }

        this.scene = scene;
        this.transform = spatialTransform;

        this.enabled = options.enabled ?? true;

        this.materials = new Set();
        this.originalMaterials = new WeakMap();

        this.renderScale =
            options.renderScale ??
            spatialTransform.renderScale;

        this.initializeShaders();

        console.log(
            "[GeometryShaderManager] Initialized."
        );
    }

    // -----------------------------------------
    // SHADER REGISTRATION
    // -----------------------------------------

    initializeShaders() {

        BABYLON.Effect.ShadersStore[
            "somniumPoincareVertexShader"
        ] = this.getVertexShader();

        BABYLON.Effect.ShadersStore[
            "somniumPoincareFragmentShader"
        ] = this.getFragmentShader();
    }

    // -----------------------------------------
    // VERTEX SHADER
    // -----------------------------------------

    getVertexShader() {

        return `
            precision highp float;

            attribute vec3 position;
            attribute vec3 normal;

            uniform mat4 world;
	    uniform mat4 viewProjection;

            uniform vec3 worldCenter;
            uniform vec3 worldHalfBounds;

            uniform float ballRadius;
            uniform float characteristicLength;
            uniform float renderScale;
            uniform float boundaryEpsilon;

            varying vec3 vNormal;
            varying vec3 vWorldPosition;

            float safeTanh(float value) {

                float x = clamp(value, 0.0, 20.0);

                float e = exp(2.0 * x);

                return (e - 1.0) / (e + 1.0);
            }

            vec3 poincareProject(vec3 worldPosition) {

                vec3 delta =
                    worldPosition - worldCenter;

                float distanceFromCenter =
                    length(delta);

                if (distanceFromCenter < 0.000001) {
                    return vec3(0.0);
                }

                vec3 direction =
                    delta / distanceFromCenter;

                vec3 safeDirection =
                    max(abs(direction), vec3(0.000001));

                vec3 boundaryDistances =
                    worldHalfBounds / safeDirection;

                float boundaryDistance =
                    min(
                        boundaryDistances.x,
                        min(
                            boundaryDistances.y,
                            boundaryDistances.z
                        )
                    );

                float u =
                    distanceFromCenter /
                    boundaryDistance;

                u = clamp(
                    u,
                    0.0,
                    1.0 - boundaryEpsilon
                );

                float hyperbolicDistance =
                    characteristicLength *
                    u / (1.0 - u);

                float projectedRadius =
                    ballRadius *
                    safeTanh(
                        hyperbolicDistance /
                        (2.0 * ballRadius)
                    );

                return direction * projectedRadius;
            }

	void main(void) {

    vec4 worldPosition =
        world * vec4(position, 1.0);

    vec3 projectedPosition =
        poincareProject(worldPosition.xyz);

    vWorldPosition =
        projectedPosition * renderScale;

    vNormal =
        normalize(mat3(world) * normal);

    gl_Position =
        viewProjection *
        vec4(vWorldPosition, 1.0);
}

        `;
    }

    // -----------------------------------------
    // FRAGMENT SHADER
    // -----------------------------------------

    getFragmentShader() {

        return `
            precision highp float;

            uniform vec4 baseColor;
            uniform vec3 lightDirection;
            uniform float ambientStrength;

            varying vec3 vNormal;
            varying vec3 vWorldPosition;

            void main(void) {

                vec3 normal =
                    normalize(vNormal);

                float diffuse =
                    max(
                        dot(normal, normalize(-lightDirection)),
                        0.0
                    );

                float illumination =
                    ambientStrength +
                    (1.0 - ambientStrength) * diffuse;

                vec3 color =
                    baseColor.rgb * illumination;

                gl_FragColor =
                    vec4(color, baseColor.a);
            }
        `;
    }

    // -----------------------------------------
    // MATERIAL CREATION
    // -----------------------------------------

    createMaterial(name, options = {}) {

        const material =
            new BABYLON.ShaderMaterial(
                name,
                this.scene,
                {
                    vertex: "somniumPoincare",
                    fragment: "somniumPoincare"
                },
                {
                    attributes: [
                        "position",
                        "normal"
                    ],

                    uniforms: [
                        "world",
                        "worldViewProjection",
                        "worldCenter",
                        "worldHalfBounds",
                        "ballRadius",
                        "characteristicLength",
                        "renderScale",
                        "boundaryEpsilon",
                        "baseColor",
                        "lightDirection",
                        "ambientStrength"
                    ]
                }
            );

        const projection =
            this.transform.projection;

        const bounds = projection.bounds;

        material.setVector3(
            "worldCenter",
            new BABYLON.Vector3(
                projection.center.x,
                projection.center.y,
                projection.center.z
            )
        );

        material.setVector3(
            "worldHalfBounds",
            new BABYLON.Vector3(
                (bounds.maxX - bounds.minX) / 2,
                (bounds.maxY - bounds.minY) / 2,
                (bounds.maxZ - bounds.minZ) / 2
            )
        );

        material.setFloat(
            "ballRadius",
            projection.radius
        );

        material.setFloat(
            "characteristicLength",
            projection.characteristicLength
        );

        material.setFloat(
            "renderScale",
            this.renderScale
        );

        material.setFloat(
            "boundaryEpsilon",
            projection.boundaryEpsilon
        );

        material.setColor4(
            "baseColor",
            options.color ??
                new BABYLON.Color3(0.35, 0.55, 0.3),
            options.alpha ?? 1
        );

        material.setVector3(
            "lightDirection",
            options.lightDirection ??
                new BABYLON.Vector3(-0.5, -1, -0.3).normalize()
        );

        material.setFloat(
            "ambientStrength",
            options.ambientStrength ?? 0.3
        );

        material.backFaceCulling =
            options.backFaceCulling ?? true;

material.onBindObservable.add(mesh => {

    const camera = this.scene.activeCamera;

    if (!camera) return;

    const world = mesh.getWorldMatrix();

    const viewProjection =
        camera.getViewMatrix().multiply(
            camera.getProjectionMatrix()
        );

    material.setMatrix("world", world);

    material.setMatrix(
        "viewProjection",
        viewProjection
    );
});
        this.materials.add(material);

        return material;
    }

    // -----------------------------------------
    // APPLY TO MESH
    // -----------------------------------------

    apply(mesh, options = {}) {

        if (!this.enabled || !mesh) {
            return false;
        }

        if (
            mesh.metadata?.spatiallyTransformed
        ) {
            throw new Error(
                "Cannot apply GPU projection to a CPU-projected mesh."
            );
        }

        if (
            mesh.metadata?.geometryShaderApplied
        ) {
            return false;
        }

        if (!this.originalMaterials.has(mesh)) {

            this.originalMaterials.set(
                mesh,
                mesh.material
            );
        }

        const material =
            this.createMaterial(
                options.name ??
                    `poincare-${mesh.name}`,
                options
            );

        mesh.material = material;

        mesh.metadata = {
            ...(mesh.metadata || {}),
            geometryShaderApplied: true
        };

        return true;
    }

    // -----------------------------------------
    // RESTORE ORIGINAL MATERIAL
    // -----------------------------------------

    restore(mesh) {

        if (!mesh) return false;

        if (!this.originalMaterials.has(mesh)) {
            return false;
        }

        mesh.material =
            this.originalMaterials.get(mesh);

        this.originalMaterials.delete(mesh);

        if (mesh.metadata) {
            delete mesh.metadata.geometryShaderApplied;
        }

        return true;
    }

    // -----------------------------------------
    // CONFIGURATION
    // -----------------------------------------

    setEnabled(enabled) {

        this.enabled = !!enabled;
    }

    // -----------------------------------------
    // DIAGNOSTICS
    // -----------------------------------------

    getStats() {

        return {
            enabled: this.enabled,
            geometry: this.transform.geometryName,
            activeMaterials: this.materials.size,
            renderScale: this.renderScale
        };
    }

    // -----------------------------------------
    // DISPOSAL
    // -----------------------------------------

    dispose() {

        for (const material of this.materials) {
            material.dispose();
        }

        this.materials.clear();

        console.log(
            "[GeometryShaderManager] Disposed."
        );
    }
}

export { GeometryShaderManager };

/**
 * HyperbolicCameraController
 *
 * Converts ordinary Babylon.js camera input into
 * hyperbolically distance-limited geodesic movement.
 *
 * Authoritative world coordinates remain unchanged.
 */

class HyperbolicCameraController {
    constructor(camera, spatialTransform, options = {}) {
        if (!camera || !spatialTransform) {
            throw new Error(
                "HyperbolicCameraController requires a camera and SpatialTransform."
            );
        }

        this.camera = camera;
        this.transform = spatialTransform;
        this.projection = spatialTransform.projection;
        this.metric = spatialTransform.metric;

        this.enabled = options.enabled ?? true;

        this.maxHyperbolicSpeed =
            options.maxHyperbolicSpeed ?? 2.5;

        this.boundaryMargin =
            options.boundaryMargin ?? 1e-7;

        this.maxDeltaSeconds =
            options.maxDeltaSeconds ?? 0.05;

        this.statistics = {
            updates: 0,
            geodesicMoves: 0,
            boundaryCorrections: 0,
            limitedMoves: 0
        };

        this.previousPosition = this.getBallPosition();

        this.constrainCamera();

        this.previousPosition = this.getBallPosition();

        console.log(
            "[HyperbolicCameraController] Initialized:",
            this.getInfo()
        );
    }

    // -----------------------------------------
    // POSITION CONVERSION
    // -----------------------------------------

    getBallPosition() {
        const scale = this.transform.renderScale;

        return {
            x: this.camera.position.x / scale,
            y: this.camera.position.y / scale,
            z: this.camera.position.z / scale
        };
    }

    setBallPosition(position) {
        const scale = this.transform.renderScale;

        this.camera.position.set(
            position.x * scale,
            position.y * scale,
            position.z * scale
        );
    }

    // -----------------------------------------
    // OPEN BALL PROTECTION
    // -----------------------------------------

    constrainPosition(position) {
        const radius = this.projection.radius;

        const maximumRadius =
            radius * (1 - this.boundaryMargin);

        const distance = Math.hypot(
            position.x,
            position.y,
            position.z
        );

        if (distance < maximumRadius) {
            return {
                x: position.x,
                y: position.y,
                z: position.z
            };
        }

        const factor =
            maximumRadius / Math.max(distance, 1e-15);

        this.statistics.boundaryCorrections++;

        return {
            x: position.x * factor,
            y: position.y * factor,
            z: position.z * factor
        };
    }

    constrainCamera() {
        const position = this.constrainPosition(
            this.getBallPosition()
        );

        this.setBallPosition(position);

        return position;
    }

    // -----------------------------------------
    // GEODESIC CAMERA MOVEMENT
    // -----------------------------------------

    update(deltaSeconds) {
        if (!this.enabled) {
            return;
        }

        if (!Number.isFinite(deltaSeconds) ||
            deltaSeconds <= 0) {
            return;
        }

        const elapsed = Math.min(
            deltaSeconds,
            this.maxDeltaSeconds
        );

        const start = this.previousPosition;

        // Babylon.js has already processed ordinary
        // keyboard and mouse camera input.
        const target = this.constrainPosition(
            this.getBallPosition()
        );

        let distance;

        try {
            distance = this.metric.distance(
                start,
                target
            );
        } catch (error) {
            this.setBallPosition(start);
            return;
        }

        if (!Number.isFinite(distance)) {
            this.setBallPosition(start);
            return;
        }

        if (distance < 1e-10) {
            this.setBallPosition(target);
            this.previousPosition = target;
            return;
        }

        // Limit movement by hyperbolic distance.
        const maximumStep =
            this.maxHyperbolicSpeed * elapsed;

        const fraction = Math.min(
            1,
            maximumStep / distance
        );

        if (fraction < 1) {
            this.statistics.limitedMoves++;
        }

        // Analytic Poincare geodesic interpolation.
        const nextPosition =
            this.metric.interpolateGeodesic(
                start,
                target,
                fraction
            );

        const safePosition =
            this.constrainPosition(nextPosition);

        this.setBallPosition(safePosition);

        this.previousPosition = safePosition;

        this.statistics.updates++;
        this.statistics.geodesicMoves++;
    }

    // -----------------------------------------
    // AUTHORITATIVE WORLD POSITION
    // -----------------------------------------

    getWorldPosition() {
        this.constrainCamera();

        return this.transform.renderToWorld({
            x: this.camera.position.x,
            y: this.camera.position.y,
            z: this.camera.position.z
        });
    }

    // -----------------------------------------
    // TELEPORT / POSITION SYNCHRONIZATION
    // -----------------------------------------

    synchronize() {
        this.constrainCamera();
        this.previousPosition = this.getBallPosition();
    }

    // -----------------------------------------
    // CONFIGURATION
    // -----------------------------------------

    setEnabled(enabled) {
        this.enabled = !!enabled;

        this.synchronize();
    }

    getInfo() {
        return {
            enabled: this.enabled,
            geometry: "poincare",
            movement: "hyperbolic-geodesic",
            maxHyperbolicSpeed: this.maxHyperbolicSpeed,
            boundaryMargin: this.boundaryMargin,
            boundaryReachable: false,
            statistics: { ...this.statistics }
        };
    }

    getStats() {
        return this.getInfo();
    }

    // -----------------------------------------
    // DISPOSAL
    // -----------------------------------------

    dispose() {
        this.enabled = false;
        this.camera = null;
        this.previousPosition = null;
    }
}

/** Provides tesseract topology and true 3D hyperplane sections. */

import { rotate4 } from './spatial-4d.js?v=somnium-20261108-phase16c-anomaly-optics-2';

const AXES = ['x', 'y', 'z', 'w'];
const FACET_BITS = [1, 2, 4, 8];
const ROTATION_PLANES = ['xw', 'yw', 'zw'];

function createHypercubeVertices(halfExtent = 2) {
    assertPositive(halfExtent, 'Hypercube half-extent');
    return Array.from({ length: 16 }, function vertexAt(unused, mask) {
        return {
            x: (mask & 1) ? halfExtent : -halfExtent,
            y: (mask & 2) ? halfExtent : -halfExtent,
            z: (mask & 4) ? halfExtent : -halfExtent,
            w: (mask & 8) ? halfExtent : -halfExtent
        };
    });
}

function createHypercubeEdges(vertices) {
    if (!Array.isArray(vertices) || vertices.length !== 16) {
        throw new TypeError('A tesseract requires exactly 16 vertices.');
    }
    const edges = [];
    for (let index = 0; index < vertices.length; index++) {
        for (const bit of FACET_BITS) {
            const next = index ^ bit;
            if (next > index) edges.push([vertices[index], vertices[next]]);
        }
    }
    return edges;
}

function rotateHypercubePoint(point, angles = {}) {
    assertVector4(point);
    let rotated = { ...point };
    for (const plane of ROTATION_PLANES) {
        const angle = angles[plane] ?? 0;
        if (!Number.isFinite(angle)) throw new TypeError(`The ${plane} rotation must be finite.`);
        rotated = rotate4(rotated, plane, angle);
    }
    return rotated;
}

/** Intersects the rotated tesseract with the 3D hyperplane W = viewerW. */
function createHypercubeSlice(vertices, angles, viewerW, options = {}) {
    if (!Array.isArray(vertices) || vertices.length !== 16) {
        throw new TypeError('A tesseract slice requires exactly 16 vertices.');
    }
    if (!Number.isFinite(viewerW)) throw new TypeError('The slice W coordinate must be finite.');
    const tolerance = options.tolerance ?? 1e-7;
    assertPositive(tolerance, 'Slice tolerance');
    const rotated = vertices.map(function rotateVertex(vertex) {
        return rotateHypercubePoint(vertex, angles);
    });
    const polygons = [];

    for (const bit of FACET_BITS) {
        for (const positive of [false, true]) {
            const facet = rotated.filter(function belongsToFacet(vertex, index) {
                return Boolean(index & bit) === positive;
            });
            if (facet.every(function liesInSlice(vertex) { return Math.abs(vertex.w - viewerW) <= tolerance; })) {
                continue;
            }
            const points = collectFacetIntersections(rotated, bit, positive, viewerW, tolerance);
            const ordered = orderPolygon(points, tolerance);
            if (ordered.length >= 3) polygons.push(ordered);
        }
    }

    return buildSliceGeometry(polygons, tolerance, viewerW);
}

function collectFacetIntersections(vertices, bit, positive, viewerW, tolerance) {
    const points = new Map();
    for (let left = 0; left < vertices.length; left++) {
        if (Boolean(left & bit) !== positive) continue;
        for (const edgeBit of FACET_BITS) {
            const right = left ^ edgeBit;
            if (edgeBit === bit || right <= left || Boolean(right & bit) !== positive) continue;
            for (const point of intersectSegment(vertices[left], vertices[right], viewerW, tolerance)) {
                points.set(pointKey(point, tolerance), point);
            }
        }
    }
    return [...points.values()];
}

function intersectSegment(start, end, viewerW, tolerance) {
    const left = start.w - viewerW;
    const right = end.w - viewerW;
    if (Math.abs(left) <= tolerance && Math.abs(right) <= tolerance) return [toVector3(start), toVector3(end)];
    if (Math.abs(left) <= tolerance) return [toVector3(start)];
    if (Math.abs(right) <= tolerance) return [toVector3(end)];
    if ((left < 0) === (right < 0)) return [];
    const t = left / (left - right);
    return [{
        x: start.x + (end.x - start.x) * t,
        y: start.y + (end.y - start.y) * t,
        z: start.z + (end.z - start.z) * t
    }];
}

function orderPolygon(points, tolerance) {
    if (points.length < 3) return [];
    const center = average3(points);
    const first = normalize3(subtract3(points[0], center));
    let normal = null;
    for (let index = 1; index < points.length; index++) {
        const candidate = cross3(first, subtract3(points[index], center));
        if (length3(candidate) > tolerance) {
            normal = normalize3(candidate);
            break;
        }
    }
    if (!normal) return [];
    const second = normalize3(cross3(normal, first));
    return points.slice().sort(function sortAroundCenter(left, right) {
        const a = subtract3(left, center);
        const b = subtract3(right, center);
        return Math.atan2(dot3(a, second), dot3(a, first)) -
            Math.atan2(dot3(b, second), dot3(b, first));
    });
}

function buildSliceGeometry(polygons, tolerance, viewerW) {
    const vertices = [];
    const vertexMap = new Map();
    const faces = [];
    const edgeMap = new Map();
    for (const polygon of polygons) {
        const face = polygon.map(function indexVertex(point) {
            const key = pointKey(point, tolerance);
            if (!vertexMap.has(key)) {
                vertexMap.set(key, vertices.length);
                vertices.push(point);
            }
            return vertexMap.get(key);
        });
        if (new Set(face).size < 3) continue;
        faces.push(face);
        for (let index = 0; index < face.length; index++) {
            const a = face[index];
            const b = face[(index + 1) % face.length];
            edgeMap.set(a < b ? `${a}:${b}` : `${b}:${a}`, a < b ? [a, b] : [b, a]);
        }
    }
    return {
        vertices,
        faces,
        edges: [...edgeMap.values()],
        viewerW,
        empty: vertices.length === 0
    };
}


function assertVector4(point) {
    if (!point || !AXES.every(function finiteAxis(axis) { return Number.isFinite(point[axis]); })) {
        throw new TypeError('A tesseract vertex must have four finite coordinates.');
    }
}

function assertPositive(value, label) {
    if (!Number.isFinite(value) || value <= 0) throw new RangeError(`${label} must be positive.`);
}

function toVector3(point) { return { x: point.x, y: point.y, z: point.z }; }
function pointKey(point, tolerance) {
    return [point.x, point.y, point.z].map(function quantize(value) {
        return Math.round(value / tolerance);
    }).join(':');
}
function average3(points) {
    const sum = points.reduce(function addPoint(total, point) { return add3(total, point); }, { x: 0, y: 0, z: 0 });
    return { x: sum.x / points.length, y: sum.y / points.length, z: sum.z / points.length };
}
function subtract3(a, b) { return { x: a.x - b.x, y: a.y - b.y, z: a.z - b.z }; }
function add3(a, b) { return { x: a.x + b.x, y: a.y + b.y, z: a.z + b.z }; }
function cross3(a, b) {
    return { x: a.y * b.z - a.z * b.y, y: a.z * b.x - a.x * b.z, z: a.x * b.y - a.y * b.x };
}
function dot3(a, b) { return a.x * b.x + a.y * b.y + a.z * b.z; }
function length3(value) { return Math.hypot(value.x, value.y, value.z); }
function normalize3(value) {
    const length = length3(value);
    return length > 1e-12 ? { x: value.x / length, y: value.y / length, z: value.z / length } : { x: 1, y: 0, z: 0 };
}

export {
    createHypercubeEdges, createHypercubeSlice, createHypercubeVertices,
    rotateHypercubePoint
};

/**
 * Cross-slice optical transport through explicitly optically-enabled
 * slice-transition anomalies. This module is CPU-only and keeps all ray state
 * in authoritative ambient R4 / slice-local coordinates.
 *
 * A slice aperture is represented by a spherical capture boundary in that
 * slice. At entry, the aperture's metric-normal covector is transferred to
 * the anomaly's slice-normal covector. At exit the transfer is reversed.
 * Canonical momentum tangent to the aperture boundary is preserved and the
 * normal coefficient is solved from the Hamiltonian (frequency) constraint.
 */

    subtract4

const AXES4 = ['x', 'y', 'z', 'w'];
const EPSILON = 1e-8;
const DEFAULT_MAX_TRANSITIONS = 8;

/**
 * Builds a positive-definite four-dimensional optical field for one anomaly.
 * The tangential 3D metric smoothly blends from the source slice to the
 * destination slice along the common slice normal. W coupling reuses the
 * anomaly's radial smootherstep envelope but is independent of player stats.
 */
function createAnomalyOpticalField(anomaly, sourceBinding, destinationBinding, options = {}) {
    validateAnomaly(anomaly);
    validateBinding(sourceBinding, anomaly.source.sliceId, 'source');
    validateBinding(destinationBinding, anomaly.destination.sliceId, 'destination');

    const sourceFrame = sourceBinding.slice.frame;
    const destinationFrame = destinationBinding.slice.frame;
    const normalAlignment = Math.abs(dot4(sourceFrame.normal, destinationFrame.normal));
    if (normalAlignment < 1 - 1e-7) {
        throw new RangeError('Optical anomaly fields require parallel or anti-parallel slice normals.');
    }

    const sourceAmbient = sliceToAmbient(sourceFrame, anomaly.source.position);
    const destinationAmbient = sliceToAmbient(destinationFrame, anomaly.destination.position);
    const signedSpan = dot4(subtract4(destinationAmbient, sourceAmbient), sourceFrame.normal);
    if (Math.abs(signedSpan) <= EPSILON) {
        throw new RangeError('Optical anomaly endpoints must have non-zero separation along the slice normal.');
    }

    const sourceAxes = sourceFrame.axes;
    const destinationAxes = destinationFrame.axes;
    const sourceTransform = sourceBinding.spatialTransform;
    const destinationTransform = destinationBinding.spatialTransform;
    if (typeof sourceTransform?.metricAtWorld !== 'function' ||
        typeof destinationTransform?.metricAtWorld !== 'function') {
        throw new TypeError('Both optical anomaly endpoint bindings require SpatialTransform.metricAtWorld().');
    }

    const refractiveIndexAt = options.refractiveIndexAt ?? anomaly.optics.refractiveIndex;
    const gradientStep = options.gradientStep;
    const metricSampling = options.metricSampling ?? anomaly.optics.metricSampling ?? 'endpoint';
    if (!['endpoint', 'pointwise', 'ambient'].includes(metricSampling)) {
        throw new RangeError(`Unsupported anomaly metric sampling mode: ${metricSampling}`);
    }
    // A 4D bridge connects two different intrinsic 3D metrics. Their tangent
    // coordinates are not globally identified, so the default bridge uses
    // aperture metrics and interpolates them only along the bridge normal.
    // The entry-side endpoint is sampled at the ray's actual aperture hit,
    // ensuring the 3D/4D interface is metric-continuous even when a curved
    // Poincare chart varies appreciably across the aperture. The opposite
    // endpoint remains sampled at its configured centre; pointwise mode is an
    // explicit opt-in for anomalies whose two charts share a broad domain.
    const entryOverride = options.endpointMetricOverride;
    const sourceEndpointMetric = entryOverride?.side === 'source'
        ? readMetric(entryOverride.metric)
        : readMetric(sourceTransform.metricAtWorld(anomaly.source.position));
    const destinationEndpointMetric = entryOverride?.side === 'destination'
        ? rebaseMetric3(readMetric(entryOverride.metric), destinationAxes, sourceAxes)
        : rebaseMetric3(
            readMetric(destinationTransform.metricAtWorld(anomaly.destination.position)),
            destinationAxes,
            sourceAxes
        );
    const sliceMetricAt = (_sourceLocalPosition, context) => {
        if (metricSampling === 'ambient') return [1, 0, 0, 0, 1, 0, 0, 0, 1];
        const ambientPosition = context.ambientPosition;
        const normalOffset = dot4(subtract4(ambientPosition, sourceFrame.origin), sourceFrame.normal);
        const normalizedOffset = normalOffset / signedSpan;
        if (!Number.isFinite(signedSpan) || !Number.isFinite(normalizedOffset)) {
            throw new RangeError(`Anomaly optical metric received a non-finite bridge coordinate (offset=${normalOffset}, span=${signedSpan}, w=${ambientPosition.w}).`);
        }
        const t = smootherstep(clamp(normalizedOffset, 0, 1));
        if (metricSampling === 'endpoint') {
            return blendScaledPositiveDefiniteMetrics3(
                [sourceEndpointMetric, destinationEndpointMetric], [1 - t, t]
            );
        }

        const destinationWeight = t;
        const sourceWeight = 1 - destinationWeight;
        const sourceLocal = ambientToSlice(sourceFrame, ambientPosition).position;
        const destinationLocal = ambientToSlice(destinationFrame, ambientPosition).position;
        if (destinationWeight <= 1e-10) return readMetric(sourceTransform.metricAtWorld(sourceLocal));
        if (sourceWeight <= 1e-10) {
            return rebaseMetric3(
                readMetric(destinationTransform.metricAtWorld(destinationLocal)),
                destinationAxes,
                sourceAxes
            );
        }
        const sourceMetric = readMetric(sourceTransform.metricAtWorld(sourceLocal));
        const destinationMetric = rebaseMetric3(
            readMetric(destinationTransform.metricAtWorld(destinationLocal)),
            destinationAxes,
            sourceAxes
        );
        return blendScaledPositiveDefiniteMetrics3(
            [sourceMetric, destinationMetric], [sourceWeight, destinationWeight]
        );
    };

    const field = createOpticalMetricField({
        sliceFrame: sourceFrame,
        sliceMetricAt,
        refractiveIndexAt,
        wCouplingAt(position) {
            return evaluateOpticalAnomalyCoupling(anomaly, position).coupling;
        },
        normalMetricScaleAt(_position, _context, baseMetric) {
            return geometricMeanMetricScale3(baseMetric);
        },
        containsPosition(position) {
            if (metricSampling === 'endpoint' || metricSampling === 'ambient') return true;
            const sourceLocal = ambientToSlice(sourceFrame, position).position;
            const destinationLocal = ambientToSlice(destinationFrame, position).position;
            const sourceValid = typeof sourceTransform.containsMetricDomainPosition !== 'function' ||
                sourceTransform.containsMetricDomainPosition(sourceLocal);
            const destinationValid = typeof destinationTransform.containsMetricDomainPosition !== 'function' ||
                destinationTransform.containsMetricDomainPosition(destinationLocal);
            return sourceValid && destinationValid;
        },
        gradientStep
    });

    return Object.freeze({
        ...field,
        anomalyId: anomaly.id,
        anomalyRevision: anomaly.revision,
        sourceAmbient,
        destinationAmbient,
        signedSpan,
        metricSampling,
        sourceBinding,
        destinationBinding
    });
}

/**
 * Creates a topology-aware optical tracer. Traversal permission is never read:
 * only WorldTopology's explicit optical-neighbor API can authorize a crossing.
 */
function createAnomalyOpticalTracer({
    topology,
    sliceRegistry,
    sceneRegistry,
    chunkGraph = null,
    debugRenderer = null
} = {}) {
    if (!topology || typeof topology.getCorridor !== 'function' ||
        typeof topology.getOpticallyConnectedNeighbors !== 'function') {
        throw new TypeError('Anomaly optical tracing requires a WorldTopology with optical connectivity.');
    }
    if (!sliceRegistry || typeof sliceRegistry.require !== 'function') {
        throw new TypeError('Anomaly optical tracing requires a SliceWorldRegistry.');
    }
    if (!sceneRegistry || typeof sceneRegistry.intersectSegment !== 'function') {
        throw new TypeError('Anomaly optical tracing requires an OpticalSceneRegistry.');
    }

    const sliceFieldCache = new Map();
    const anomalyFieldCache = new Map();

    function getSliceField(sliceId, metricOptions = {}) {
        const binding = sliceRegistry.require(sliceId);
        const cacheKey = `${sliceId}:${binding.slice.revision ?? 1}:${metricOptions.cacheKey ?? 'default'}`;
        if (!sliceFieldCache.has(cacheKey)) {
            sliceFieldCache.set(cacheKey, createSliceOpticalField(
                binding.slice, binding.spatialTransform, metricOptions
            ));
        }
        return sliceFieldCache.get(cacheKey);
    }

    function getAnomalyField(anomaly, metricOptions = {}) {
        if (metricOptions.endpointMetricOverride) {
            return createAnomalyOpticalField(
                anomaly,
                sliceRegistry.require(anomaly.source.sliceId),
                sliceRegistry.require(anomaly.destination.sliceId),
                metricOptions
            );
        }
        const cacheKey = `${anomaly.id}:${anomaly.revision}:${metricOptions.cacheKey ?? 'default'}`;
        if (!anomalyFieldCache.has(cacheKey)) {
            anomalyFieldCache.set(cacheKey, createAnomalyOpticalField(
                anomaly,
                sliceRegistry.require(anomaly.source.sliceId),
                sliceRegistry.require(anomaly.destination.sliceId),
                metricOptions
            ));
        }
        return anomalyFieldCache.get(cacheKey);
    }

    function evaluateReachability(anomalyId, position4, targetSliceId, reachabilityOptions = {}) {
        const anomaly = topology.getCorridor(anomalyId);
        validateAnomaly(anomaly, { requireEnabled: false });
        if (!anomaly.optics.enabled) {
            return { reachable: false, reason: 'optics-disabled', anomalyId, targetSliceId };
        }

        const endpoint = resolveEndpoint(anomaly, targetSliceId);
        const fromSliceId = targetSliceId === anomaly.source.sliceId
            ? anomaly.destination.sliceId : anomaly.source.sliceId;
        if (!topology.hasOpticalConnection(fromSliceId, targetSliceId)) {
            return {
                reachable: false,
                reason: 'no-optical-topology-edge',
                anomalyId,
                fromSliceId,
                targetSliceId
            };
        }

        const targetBinding = sliceRegistry.require(targetSliceId);
        const localTarget = ambientToSlice(targetBinding.slice.frame, position4);
        const endpointDistance = Math.hypot(
            localTarget.position.x - endpoint.position.x,
            localTarget.position.y - endpoint.position.y,
            localTarget.position.z - endpoint.position.z
        );
        // Reachability normally uses the local influence field. During initial
        // aperture entry, callers may supply the anomaly centre as the
        // coupling reference: this answers whether the configured anomaly can
        // bridge the complete slice separation at full coupling, while keeping
        // aperture and chunk-spatial tests anchored to the actual ray position.
        const couplingPosition = reachabilityOptions.couplingPosition4D ?? position4;
        const coupling = evaluateOpticalAnomalyCoupling(anomaly, couplingPosition);
        const targetPlaneDistance = Math.abs(localTarget.normalOffset);
        const wReachable = targetPlaneDistance <= coupling.opticalWReach + EPSILON;
        const apertureReachable = endpointDistance <= (anomaly.optics.apertureRadius ?? anomaly.entryRadius) + EPSILON;

        let chunkReachability = null;
        if (wReachable && apertureReachable && chunkGraph) {
            if (!chunkGraph.hasSlice(fromSliceId) || !chunkGraph.hasSlice(targetSliceId)) {
                chunkReachability = {
                    reachable: false,
                    reason: 'chunk-graph-slice-unregistered'
                };
            } else {
                const fromBinding = sliceRegistry.require(fromSliceId);
                const fromLocal = ambientToSlice(fromBinding.slice.frame, position4).position;
                const sourceAddress = chunkGraph.getAddress(fromSliceId, fromLocal);
                const targetAddress = chunkGraph.getAddress(targetSliceId, endpoint.position);
                chunkReachability = chunkGraph.getOpticalReachability(sourceAddress, targetAddress, {
                    position4D: position4,
                    couplingPosition4D: couplingPosition,
                    corridorId: anomaly.id,
                    // A spherical aperture can straddle one X and one Z chunk
                    // boundary at once. Include the diagonal Manhattan-2 case
                    // so the physical aperture and graph test agree.
                    maxChunkDistance: reachabilityOptions.maxChunkDistance ?? 2
                });
            }
        }

        const reachable = wReachable && apertureReachable &&
            (chunkReachability === null || chunkReachability.reachable);
        return {
            reachable,
            reason: !wReachable ? 'insufficient-optical-w-reach' :
                !apertureReachable ? 'outside-optical-aperture' :
                    chunkReachability && !chunkReachability.reachable
                        ? chunkReachability.reason ??
                            (!chunkReachability.spatialReachable
                                ? 'spatial-distance-exceeded'
                                : !chunkReachability.wReachable
                                    ? 'insufficient-optical-w-reach'
                                    : 'optical-chunk-unreachable')
                        : null,
            anomalyId,
            fromSliceId,
            targetSliceId,
            wDistance: targetPlaneDistance,
            opticalWReach: coupling.opticalWReach,
            coupling: coupling.coupling,
            apertureDistance: endpointDistance,
            chunkReachability
        };
    }

    function trace(initialRay, options = {}) {
        const startSliceId = options.sliceId ?? initialRay?.sliceId;
        if (typeof startSliceId !== 'string' || !startSliceId.trim()) {
            throw new TypeError('An initial sliceId is required for topology-aware optical tracing.');
        }
        const maxTransitions = options.maxTransitions ?? DEFAULT_MAX_TRANSITIONS;
        if (!Number.isSafeInteger(maxTransitions) || maxTransitions < 0 || maxTransitions > 128) {
            throw new RangeError('maxTransitions must be an integer from zero through 128.');
        }

        let state = createOpticalRayState(initialRay);
        let context = { kind: 'slice', sliceId: startSliceId };
        const pathSegments = [];
        const transitions = [];
        let status = 'max-path-length';
        let terminalEvent = null;
        let segmentCount = 0;
        let transitionCount = 0;
        // Preserve the exact field instance used to solve the entry interface.
        // In endpoint mode it may contain an aperture-local metric override;
        // recreating it for the anomaly segment would introduce an artificial
        // metric jump immediately after the boundary.
        let activeAnomalyField = null;

        while (segmentCount < (maxTransitions * 2 + 1)) {
            if (context.kind === 'slice') {
                const binding = sliceRegistry.require(context.sliceId);
                const field = getSliceField(context.sliceId, options.sliceMetricOptions ?? {});
                const neighbors = topology.getOpticallyConnectedNeighbors(context.sliceId);
                const candidates = neighbors.map((neighbor) => ({
                    neighbor,
                    anomaly: topology.getCorridor(neighbor.corridorId)
                })).filter(({ anomaly }) => anomaly.type === SLICE_TRANSITION_ANOMALY);
                const callerEvent = options.sliceTraceOptions?.segmentEvent;
                const sliceOptions = { ...(options.sliceTraceOptions ?? {}) };
                delete sliceOptions.segmentEvent;
                sliceOptions.segmentEvent = (segment) => {
                    const entry = findApertureEntry(segment, candidates, binding);
                    if (entry) return entry;
                    return typeof callerEvent === 'function' ? callerEvent(segment) : null;
                };
                const result = traceOpticalRayInSlice(
                    field, sceneRegistry, context.sliceId, state, sliceOptions
                );
                pathSegments.push({ kind: 'slice', sliceId: context.sliceId, result });
                segmentCount++;
                if (result.event?.kind !== 'anomaly-entry') {
                    status = result.status;
                    terminalEvent = result.event;
                    state = result.finalState;
                    break;
                }

                if (transitionCount >= maxTransitions) {
                    result.status = 'max-transitions';
                    result.event = { kind: 'max-transitions', maxTransitions };
                    status = 'max-transitions';
                    terminalEvent = result.event;
                    state = result.finalState;
                    break;
                }
                const anomaly = topology.getCorridor(result.event.anomalyId);
                const incidentMetric = field.sample(result.event.ambientPosition).tangentMetric;
                const anomalyField = getAnomalyField(anomaly, {
                    ...(options.anomalyMetricOptions ?? {}),
                    endpointMetricOverride: { side: result.event.side, metric: incidentMetric }
                });
                if (Math.abs(anomalyField.signedSpan) > anomaly.optics.maxWOpticalReach + EPSILON) {
                    result.status = 'optical-range-limit';
                    result.event = {
                        kind: 'optical-range-limit', anomalyId: anomaly.id,
                        requiredWReach: Math.abs(anomalyField.signedSpan),
                        maximumWReach: anomaly.optics.maxWOpticalReach
                    };
                    status = result.status;
                    terminalEvent = result.event;
                    state = result.finalState;
                    break;
                }
                const entryTargetSliceId = result.event.side === 'source'
                    ? anomaly.destination.sliceId : anomaly.source.sliceId;
                const entryReachability = evaluateReachability(
                    anomaly.id,
                    result.event.ambientPosition,
                    entryTargetSliceId,
                    {
                        ...(options.chunkReachabilityOptions ?? {}),
                        // Validate the anomaly's full configured W bridge at
                        // its maximum-coupling centre. Local coupling at the
                        // aperture still controls whether momentum can enter.
                        couplingPosition4D: anomaly.influence.center
                    }
                );
                if (!entryReachability.reachable) {
                    result.status = 'optical-entry-blocked';
                    result.event = {
                        kind: 'optical-entry-blocked',
                        anomalyId: anomaly.id,
                        reachability: entryReachability
                    };
                    status = result.status;
                    terminalEvent = result.event;
                    state = result.finalState;
                    break;
                }
                const entryResult = transferSliceMomentumIntoAnomaly({
                    anomaly,
                    anomalyField,
                    sliceField: field,
                    slice: binding.slice,
                    state: result.finalState,
                    event: result.event,
                    side: result.event.side
                });
                result.status = entryResult.transmitted ? 'anomaly-entry' : 'interface-blocked';
                result.interface = entryResult.diagnostics;
                if (!entryResult.transmitted) {
                    status = 'interface-blocked';
                    terminalEvent = { kind: 'interface-blocked', ...entryResult.diagnostics };
                    state = result.finalState;
                    break;
                }
                state = createOpticalRayState({
                    ...result.finalState,
                    momentum: entryResult.momentum
                });
                activeAnomalyField = anomalyField;
                context = {
                    kind: 'anomaly',
                    anomalyId: anomaly.id,
                    entrySide: result.event.side
                };
                transitions.push({
                    anomalyId: anomaly.id,
                    fromSliceId: context.entrySide === 'source' ? anomaly.source.sliceId : anomaly.destination.sliceId,
                    toSliceId: context.entrySide === 'source' ? anomaly.destination.sliceId : anomaly.source.sliceId,
                    direction: 'into-anomaly',
                    interface: entryResult.diagnostics
                });
                continue;
            }

            const anomaly = topology.getCorridor(context.anomalyId);
            const anomalyField = activeAnomalyField ??
                getAnomalyField(anomaly, options.anomalyMetricOptions ?? {});
            const entryOpticalPath = state.opticalPathLength;
            const result = traceOpticalRay(anomalyField, state, {
                ...(options.anomalyTraceOptions ?? {}),
                segmentEvent: (segment) => findAnomalyEvent(
                    segment,
                    anomaly,
                    context.entrySide,
                    anomalyField,
                    (event) => evaluateReachability(
                        anomaly.id,
                        event.ambientPosition,
                        event.sliceId,
                        options.chunkReachabilityOptions ?? {}
                    )
                )
            });
            const attenuationDistance = Math.max(0,
                result.finalState.opticalPathLength - entryOpticalPath
            );
            const attenuation = Math.exp(-anomaly.optics.attenuationPerUnit * attenuationDistance);
            result.finalState.intensity *= attenuation;
            pathSegments.push({ kind: 'anomaly', anomalyId: anomaly.id, result });
            segmentCount++;

            if (result.event?.kind !== 'anomaly-exit') {
                status = result.event?.kind ?? result.status;
                terminalEvent = result.event;
                state = result.finalState;
                break;
            }

            const exit = result.event;
            const destinationBinding = sliceRegistry.require(exit.sliceId);
            const destinationField = getSliceField(exit.sliceId, options.sliceMetricOptions ?? {});
            const exitResult = transferAnomalyMomentumIntoSlice({
                anomaly,
                anomalyField,
                sliceField: destinationField,
                slice: destinationBinding.slice,
                state: result.finalState,
                event: exit
            });
            result.status = exitResult.transmitted ? 'anomaly-exit' : 'interface-blocked';
            result.interface = exitResult.diagnostics;
            if (!exitResult.transmitted) {
                status = 'interface-blocked';
                terminalEvent = { kind: 'interface-blocked', ...exitResult.diagnostics };
                state = result.finalState;
                break;
            }
            state = createOpticalRayState({
                ...result.finalState,
                momentum: exitResult.momentum,
                intensity: result.finalState.intensity * anomaly.optics.transmission
            });
            transitions.push({
                anomalyId: anomaly.id,
                fromSliceId: anomaly.source.sliceId === exit.sliceId
                    ? anomaly.destination.sliceId : anomaly.source.sliceId,
                toSliceId: exit.sliceId,
                direction: 'out-of-anomaly',
                interface: exitResult.diagnostics
            });
            activeAnomalyField = null;
            context = { kind: 'slice', sliceId: exit.sliceId };
            transitionCount++;
            if (transitionCount > maxTransitions) {
                status = 'max-transitions';
                terminalEvent = { kind: 'max-transitions', maxTransitions };
                break;
            }
        }

        const diagnostics = {
            segmentCount,
            transitionCount,
            sliceSegments: pathSegments.filter((segment) => segment.kind === 'slice').length,
            anomalySegments: pathSegments.filter((segment) => segment.kind === 'anomaly').length,
            acceptedSteps: pathSegments.reduce((sum, segment) => sum + (segment.result.diagnostics?.acceptedSteps ?? 0), 0),
            rejectedSteps: pathSegments.reduce((sum, segment) => sum + (segment.result.diagnostics?.rejectedSteps ?? 0), 0),
            pathLength: state.opticalPathLength,
            travelTime: state.travelTime,
            intensity: state.intensity
        };
        const combined = {
            status,
            event: terminalEvent,
            finalState: state,
            context,
            pathSegments,
            transitions,
            diagnostics
        };

        if (options.drawDebug && debugRenderer) {
            renderTopologyDebug(combined, sliceRegistry, debugRenderer, options.debugStyle ?? {});
        }
        return combined;
    }

    return Object.freeze({
        trace,
        createAnomalyField(anomalyId, metricOptions = {}) {
            return getAnomalyField(topology.getCorridor(anomalyId), metricOptions);
        },
        evaluateReachability,
        getOpticallyConnectedNeighbors(sliceId) {
            return topology.getOpticallyConnectedNeighbors(sliceId);
        },
        getOpticalChunkReachability(sourceAddress, targetAddress, reachabilityOptions = {}) {
            if (!chunkGraph || typeof chunkGraph.getOpticalReachability !== 'function') {
                throw new Error('Optical chunk reachability requires a SpatialChunkGraph.');
            }
            return chunkGraph.getOpticalReachability(sourceAddress, targetAddress, reachabilityOptions);
        },
        getOpticallyReachableChunkCandidates(sourceSliceId, position4, reachabilityOptions = {}) {
            if (!chunkGraph || typeof chunkGraph.getOpticallyReachableChunkCandidates !== 'function') {
                throw new Error('Optically reachable chunk candidates require a SpatialChunkGraph.');
            }
            return chunkGraph.getOpticallyReachableChunkCandidates(sourceSliceId, position4, reachabilityOptions);
        },
        clearCache() {
            sliceFieldCache.clear();
            anomalyFieldCache.clear();
        }
    });
}

function findApertureEntry(segment, candidates, binding) {
    const start = ambientToSlice(binding.slice.frame, segment.startPosition).position;
    const end = ambientToSlice(binding.slice.frame, segment.endPosition).position;
    let best = null;
    for (const { neighbor, anomaly } of candidates) {
        const side = neighbor.entrySide;
        if (side === 'destination' && !anomaly.optics.bidirectional) continue;
        const endpoint = anomaly[side];
        const radius = anomaly.optics.apertureRadius ?? anomaly.entryRadius;
        // Slice coordinates are records, while the small vector helpers below
        // intentionally use dense arrays. Convert explicitly at this boundary.
        const hit = intersectSphereEntry(
            [start.x, start.y, start.z],
            [end.x, end.y, end.z],
            [endpoint.position.x, endpoint.position.y, endpoint.position.z],
            radius
        );
        if (!hit || (best && hit.parameter >= best.parameter)) continue;
        const localPosition = { x: hit.point[0], y: hit.point[1], z: hit.point[2] };
        const ambientPosition = sliceToAmbient(binding.slice.frame, localPosition);
        if (!containsAmbientPoint(anomaly.region, ambientPosition, 1e-6)) continue;
        best = {
            kind: 'anomaly-entry',
            parameter: hit.parameter,
            anomalyId: anomaly.id,
            side,
            apertureNormal: hit.normal,
            ambientPosition,
            localPosition
        };
    }
    return best;
}

function findAnomalyEvent(segment, anomaly, entrySide, field, evaluateExitReachability = null) {
    const events = [];
    const sourceBinding = field.sourceBinding;
    const destinationBinding = field.destinationBinding;
    const sourceAmbient = field.sourceAmbient;
    const destinationAmbient = field.destinationAmbient;
    const normalSign = Math.sign(field.signedSpan);
    const orientedNormal = scale4(sourceBinding.slice.frame.normal, normalSign);
    const apertureRadius = anomaly.optics.apertureRadius ?? anomaly.entryRadius;

    for (const side of ['source', 'destination']) {
        if (!anomaly.optics.bidirectional && side === entrySide) continue;
        const binding = side === 'source' ? sourceBinding : destinationBinding;
        const endpoint = side === 'source' ? anomaly.source : anomaly.destination;
        const endpointAmbient = side === 'source' ? sourceAmbient : destinationAmbient;
        const inwardNormal = side === 'source' ? orientedNormal : scale4(orientedNormal, -1);
        const d0 = dot4(subtract4(segment.startPosition, endpointAmbient), inwardNormal);
        const d1 = dot4(subtract4(segment.endPosition, endpointAmbient), inwardNormal);
        if (!(d0 > EPSILON && d1 <= EPSILON)) continue;
        const parameter = d0 / (d0 - d1);
        const point = interpolate4(segment.startPosition, segment.endPosition, parameter);
        const local = ambientToSlice(binding.slice.frame, point).position;
        const normal = normalize3OrFallback([
            local.x - endpoint.position.x,
            local.y - endpoint.position.y,
            local.z - endpoint.position.z
        ], projectAmbientVectorToFrame(segment.endPosition, segment.startPosition, binding.slice.frame.axes));
        const apertureDistance = Math.hypot(
            local.x - endpoint.position.x,
            local.y - endpoint.position.y,
            local.z - endpoint.position.z
        );
        if (apertureDistance <= apertureRadius + 1e-6) {
            const exitEvent = {
                kind: 'anomaly-exit',
                anomalyId: anomaly.id,
                side,
                sliceId: endpoint.sliceId,
                localPosition: local,
                apertureNormal: normal,
                ambientPosition: point
            };
            const reachability = typeof evaluateExitReachability === 'function'
                ? evaluateExitReachability(exitEvent) : null;
            events.push({
                parameter,
                priority: 0,
                event: reachability && !reachability.reachable
                    ? {
                        kind: 'optical-exit-blocked',
                        anomalyId: anomaly.id,
                        side,
                        sliceId: endpoint.sliceId,
                        ambientPosition: point,
                        reachability
                    }
                    : exitEvent
            });
        }
    }

    const regionParameter = segmentAabbExitParameter(
        segment.startPosition, segment.endPosition, anomaly.region
    );
    if (regionParameter !== null) {
        events.push({
            parameter: regionParameter,
            priority: 1,
            event: { kind: 'anomaly-boundary', anomalyId: anomaly.id }
        });
    }

    const horizonParameter = segmentSphereExitParameter(
        segment.startPosition,
        segment.endPosition,
        anomaly.influence.center,
        anomaly.influence.horizonRadius
    );
    if (horizonParameter !== null) {
        events.push({
            parameter: horizonParameter,
            priority: 2,
            event: { kind: 'optical-horizon', anomalyId: anomaly.id }
        });
    }

    if (!events.length) return null;
    events.sort((left, right) => left.parameter - right.parameter || left.priority - right.priority);
    return { ...events[0].event, parameter: events[0].parameter };
}

function transferSliceMomentumIntoAnomaly({ anomaly, anomalyField, sliceField, slice, state, event, side }) {
    const point = event.ambientPosition ?? state.position;
    const sliceMomentum = projectCovectorToFrame(state.momentum, slice.frame.axes);
    const sliceSample = sliceField.sample(point);
    const aperture = event.apertureNormal;
    const decomposition = decomposeCovector3(sliceMomentum, aperture, sliceSample.tangentCometric);
    const incomingHamiltonian = evaluateOpticalHamiltonian(sliceField, point, state.momentum);
    const tangentAmbient = liftCovectorFromFrame(slice.frame.axes, decomposition.tangential);
    const anomalyMomentumTangent = projectCovectorToFrame(tangentAmbient, anomalyField.frame.axes);
    const anomalySample = anomalyField.sample(point);
    const tangentQuadratic = quadratic3(anomalySample.tangentCometric, anomalyMomentumTangent);
    const availableTwiceHamiltonian = 2 * incomingHamiltonian - tangentQuadratic;
    const coupling = anomalySample.wCoupling;
    if (coupling <= 1e-8 || availableTwiceHamiltonian < -1e-8) {
        return {
            transmitted: false,
            momentum: null,
            diagnostics: {
                rule: 'aperture-normal-to-anomaly-normal',
                reason: coupling <= 1e-8 ? 'zero-w-optical-coupling' : 'insufficient-interface-energy',
                incomingHamiltonian,
                tangentHamiltonian: tangentQuadratic / 2,
                wCoupling: coupling
            }
        };
    }

    // A 3D aperture boundary has a two-dimensional normal space in ambient R4:
    // its radial conormal plus the slice normal. Aim the unconstrained launch
    // component toward the anomaly influence centre. The radial component is
    // deliberately biasable: the radial W-coupling gradient can refract a ray
    // away from the core, so a modest inward bias helps it enter the full-field
    // region before the ray is redirected toward the opposite aperture.
    const radialCovector = liftCovectorFromFrame(slice.frame.axes, aperture);
    const wCovector = anomalyField.frame.normal;
    const targetDirection = subtract4(anomaly.influence.center, point);
    const cometric = anomalySample.cometric;
    const radialImage = multiply4(cometric, radialCovector);
    const wImage = multiply4(cometric, wCovector);
    const g00 = dot4(radialCovector, radialImage);
    const g01 = dot4(radialCovector, wImage);
    const g11 = dot4(wCovector, wImage);
    const determinant = g00 * g11 - g01 * g01;
    if (!Number.isFinite(determinant) || determinant <= 1e-14) {
        return {
            transmitted: false,
            momentum: null,
            diagnostics: {
                rule: 'aperture-normal-to-anomaly-normal',
                reason: 'degenerate-anomaly-interface-normal-plane',
                incomingHamiltonian,
                wCoupling: coupling
            }
        };
    }
    const rhs0 = dot4(radialCovector, targetDirection);
    const rhs1 = dot4(wCovector, targetDirection);
    let coefficient0 = anomaly.optics.entryRadialBias *
        (rhs0 * g11 - rhs1 * g01) / determinant;
    let coefficient1 = (rhs1 * g00 - rhs0 * g01) / determinant;
    let preferredNormalCovector = add4(
        scale4(radialCovector, coefficient0),
        scale4(wCovector, coefficient1)
    );
    let normalQuadratic = dot4(
        preferredNormalCovector,
        multiply4(cometric, preferredNormalCovector)
    );
    if (!Number.isFinite(normalQuadratic) || normalQuadratic <= 1e-14) {
        // A fixed endpoint-centred field can put its centre exactly on the
        // aperture. In that degenerate case, preserve the old axial W transfer.
        const expectedWSign = Math.sign(dot4(
            subtract4(anomalyField.destinationAmbient, anomalyField.sourceAmbient),
            anomalyField.frame.normal
        )) * (side === 'source' ? 1 : -1);
        coefficient0 = 0;
        coefficient1 = expectedWSign || 1;
        preferredNormalCovector = scale4(wCovector, coefficient1);
        normalQuadratic = dot4(preferredNormalCovector, multiply4(cometric, preferredNormalCovector));
    }

    const normalScale = Math.sqrt(Math.max(0, availableTwiceHamiltonian) / normalQuadratic);
    const momentum = add4(tangentAmbient, scale4(preferredNormalCovector, normalScale));
    const outgoingHamiltonian = evaluateOpticalHamiltonian(anomalyField, point, momentum);
    const drift = Math.abs(outgoingHamiltonian - incomingHamiltonian);
    return {
        transmitted: drift <= 1e-5 * Math.max(1, incomingHamiltonian),
        momentum,
        diagnostics: {
            rule: 'aperture-normal-to-anomaly-normal',
            entryRadialBias: anomaly.optics.entryRadialBias,
            reason: drift <= 1e-5 * Math.max(1, incomingHamiltonian) ? null : 'interface-hamiltonian-mismatch',
            incomingHamiltonian,
            outgoingHamiltonian,
            hamiltonianDrift: drift,
            wCoupling: coupling,
            normalMomentum: normalScale,
            preferredAnomalyDirection: {
                x: radialImage.x * coefficient0 + wImage.x * coefficient1,
                y: radialImage.y * coefficient0 + wImage.y * coefficient1,
                z: radialImage.z * coefficient0 + wImage.z * coefficient1,
                w: radialImage.w * coefficient0 + wImage.w * coefficient1
            },
            preservedApertureTangentMomentum: true
        }
    };
}

function transferAnomalyMomentumIntoSlice({ anomaly, anomalyField, sliceField, slice, state, event }) {
    const point = event.ambientPosition ?? state.position;
    const sliceSample = sliceField.sample(point);
    const sliceMomentum = projectCovectorToFrame(state.momentum, slice.frame.axes);
    const decomposition = decomposeCovector3(
        sliceMomentum, event.apertureNormal, sliceSample.tangentCometric
    );
    const incomingHamiltonian = evaluateOpticalHamiltonian(anomalyField, point, state.momentum);
    const tangentQuadratic = quadratic3(sliceSample.tangentCometric, decomposition.tangential);
    const availableTwiceHamiltonian = 2 * incomingHamiltonian - tangentQuadratic;
    const normalNormSquared = decomposition.normalNormSquared;
    if (availableTwiceHamiltonian < -1e-8 || normalNormSquared <= 1e-12) {
        return {
            transmitted: false,
            momentum: null,
            diagnostics: {
                rule: 'anomaly-normal-to-aperture-normal',
                reason: 'insufficient-interface-energy',
                incomingHamiltonian,
                tangentHamiltonian: tangentQuadratic / 2
            }
        };
    }
    const radialMomentum = Math.sqrt(Math.max(0, availableTwiceHamiltonian) / normalNormSquared);
    const outgoingLocal = add3(
        decomposition.tangential,
        scale3(event.apertureNormal, radialMomentum)
    );
    const momentum = liftCovectorFromFrame(slice.frame.axes, outgoingLocal);
    const outgoingHamiltonian = evaluateOpticalHamiltonian(sliceField, point, momentum);
    const drift = Math.abs(outgoingHamiltonian - incomingHamiltonian);
    return {
        transmitted: drift <= 1e-5 * Math.max(1, incomingHamiltonian),
        momentum,
        diagnostics: {
            rule: 'anomaly-normal-to-aperture-normal',
            reason: drift <= 1e-5 * Math.max(1, incomingHamiltonian) ? null : 'interface-hamiltonian-mismatch',
            incomingHamiltonian,
            outgoingHamiltonian,
            hamiltonianDrift: drift,
            normalMomentum: radialMomentum,
            preservedApertureTangentMomentum: true
        }
    };
}

function decomposeCovector3(momentum, apertureConormal, cometric) {
    const transformed = multiply3(cometric, apertureConormal);
    const normalNormSquared = dot3(apertureConormal, transformed);
    if (!Number.isFinite(normalNormSquared) || normalNormSquared <= 1e-12) {
        throw new RangeError('Aperture conormal is degenerate in the slice optical metric.');
    }
    const metricMomentum = multiply3(cometric, momentum);
    const coefficient = dot3(apertureConormal, metricMomentum) / normalNormSquared;
    const tangential = subtract3(momentum, scale3(apertureConormal, coefficient));
    return { coefficient, tangential, normalNormSquared };
}

function resolveEndpoint(anomaly, sliceId) {
    if (anomaly.source.sliceId === sliceId) return anomaly.source;
    if (anomaly.destination.sliceId === sliceId) return anomaly.destination;
    throw new Error(`Slice ${sliceId} is not an endpoint of anomaly ${anomaly.id}.`);
}

function validateAnomaly(anomaly, { requireEnabled = true } = {}) {
    if (!anomaly || anomaly.type !== SLICE_TRANSITION_ANOMALY) {
        throw new TypeError('Phase 16C optical transport currently requires a slice-transition anomaly.');
    }
    if (requireEnabled && !anomaly.optics?.enabled) {
        throw new RangeError(`Anomaly ${anomaly.id} is not optically enabled.`);
    }
    if (!anomaly.influence?.center || !Number.isFinite(anomaly.influence.horizonRadius) ||
        !Number.isFinite(anomaly.influence.innerRadius)) {
        throw new TypeError(`Anomaly ${anomaly.id} has no resolved optical influence field.`);
    }
}

function validateBinding(binding, expectedSliceId, label) {
    if (!binding?.slice?.id || binding.slice.id !== expectedSliceId || !binding.slice.frame ||
        typeof binding.spatialTransform?.metricAtWorld !== 'function') {
        throw new TypeError(`Anomaly optical ${label} binding is incomplete or belongs to another slice.`);
    }
}

function intersectSphereEntry(from, to, center, radius) {
    const direction = subtract3(to, from);
    const offset = subtract3(from, center);
    const a = dot3(direction, direction);
    if (a <= 1e-16) return null;
    const b = 2 * dot3(offset, direction);
    const c = dot3(offset, offset) - radius * radius;
    const discriminant = b * b - 4 * a * c;
    if (discriminant < 0) return null;
    const root = Math.sqrt(Math.max(0, discriminant));
    const roots = [(-b - root) / (2 * a), (-b + root) / (2 * a)].sort((x, y) => x - y);
    for (const parameter of roots) {
        if (parameter < 0 || parameter > 1) continue;
        const point = add3(from, scale3(direction, parameter));
        const radial = normalize3OrFallback(subtract3(point, center), [1, 0, 0]);
        if (dot3(direction, radial) < -1e-10) return { parameter, point, normal: radial };
    }
    return null;
}

function segmentAabbExitParameter(from, to, bounds) {
    const direction = subtract4(to, from);
    let best = Infinity;
    for (const axis of AXES4) {
        if (to[axis] > bounds.max[axis] && direction[axis] > 0) {
            best = Math.min(best, (bounds.max[axis] - from[axis]) / direction[axis]);
        } else if (to[axis] < bounds.min[axis] && direction[axis] < 0) {
            best = Math.min(best, (bounds.min[axis] - from[axis]) / direction[axis]);
        }
    }
    return Number.isFinite(best) && best >= 0 && best <= 1 ? best : null;
}

function segmentSphereExitParameter(from, to, center, radius) {
    const direction = subtract4(to, from);
    const offset = subtract4(from, center);
    const startDistance = norm4(offset);
    const endDistance = norm4(subtract4(to, center));
    if (startDistance >= radius - 1e-8 || endDistance < radius) return null;
    const a = dot4(direction, direction);
    if (a <= 1e-16) return null;
    const b = 2 * dot4(offset, direction);
    const c = dot4(offset, offset) - radius * radius;
    const discriminant = b * b - 4 * a * c;
    if (discriminant < 0) return null;
    const root = Math.sqrt(discriminant);
    const parameter = (-b + root) / (2 * a);
    return parameter >= 0 && parameter <= 1 ? parameter : null;
}

function readMetric(metric) {
    const values = Array.isArray(metric) || ArrayBuffer.isView(metric) ? metric : metric?.values;
    if ((!Array.isArray(values) && !ArrayBuffer.isView(values)) || values.length !== 9) {
        throw new TypeError('SpatialTransform.metricAtWorld() must provide a 3x3 metric.');
    }
    const result = Array.from(values, Number);
    if (!result.every(Number.isFinite)) throw new TypeError('Slice optical metric contains non-finite values.');
    return result;
}

function rebaseMetric3(metric, fromAxes, toAxes) {
    const rotation = Array.from({ length: 3 }, (_, row) =>
        Array.from({ length: 3 }, (_, column) => dot4(toAxes[row], fromAxes[column]))
    );
    const result = new Array(9).fill(0);
    for (let row = 0; row < 3; row++) {
        for (let column = 0; column < 3; column++) {
            for (let a = 0; a < 3; a++) {
                for (let b = 0; b < 3; b++) {
                    result[row * 3 + column] += rotation[row][a] * metric[a * 3 + b] * rotation[column][b];
                }
            }
        }
    }
    return result;
}

function projectCovectorToFrame(vector, axes) {
    return axes.map((axis) => dot4(vector, axis));
}

function projectAmbientVectorToFrame(to, from, axes) {
    const delta = subtract4(to, from);
    return axes.map((axis) => dot4(delta, axis));
}

function liftCovectorFromFrame(axes, vector) {
    let result = { x: 0, y: 0, z: 0, w: 0 };
    for (let index = 0; index < 3; index++) {
        result = {
            x: result.x + axes[index].x * vector[index],
            y: result.y + axes[index].y * vector[index],
            z: result.z + axes[index].z * vector[index],
            w: result.w + axes[index].w * vector[index]
        };
    }
    return result;
}

function multiply4(matrix, vector) {
    const axes = ['x', 'y', 'z', 'w'];
    const values = new Array(4);
    for (let row = 0; row < 4; row++) {
        values[row] = 0;
        for (let column = 0; column < 4; column++) {
            values[row] += matrix[row * 4 + column] * vector[axes[column]];
        }
    }
    return { x: values[0], y: values[1], z: values[2], w: values[3] };
}

function multiply3(matrix, vector) {
    return [
        matrix[0] * vector[0] + matrix[1] * vector[1] + matrix[2] * vector[2],
        matrix[3] * vector[0] + matrix[4] * vector[1] + matrix[5] * vector[2],
        matrix[6] * vector[0] + matrix[7] * vector[1] + matrix[8] * vector[2]
    ];
}

function quadratic3(matrix, vector) {
    return dot3(vector, multiply3(matrix, vector));
}

function add3(a, b) {
    return [a[0] + b[0], a[1] + b[1], a[2] + b[2]];
}

function subtract3(a, b) {
    return [a[0] - b[0], a[1] - b[1], a[2] - b[2]];
}

function scale3(vector, factor) {
    return [vector[0] * factor, vector[1] * factor, vector[2] * factor];
}

function dot3(a, b) {
    return a[0] * b[0] + a[1] * b[1] + a[2] * b[2];
}

function normalize3OrFallback(vector, fallback) {
    const length = Math.hypot(vector[0], vector[1], vector[2]);
    if (length > 1e-12) return vector.map((value) => value / length);
    const fallbackLength = Math.hypot(fallback[0], fallback[1], fallback[2]);
    if (fallbackLength <= 1e-12) return [1, 0, 0];
    return fallback.map((value) => value / fallbackLength);
}

function interpolate4(a, b, t) {
    return {
        x: a.x + (b.x - a.x) * t,
        y: a.y + (b.y - a.y) * t,
        z: a.z + (b.z - a.z) * t,
        w: a.w + (b.w - a.w) * t
    };
}

function smootherstep(value) {
    const t = clamp(value, 0, 1);
    // The polynomial can overshoot by a few ulps at t≈1. Keep the convex
    // interpolation weights exactly inside [0, 1] before blending SPD metrics.
    if (t === 0 || t === 1) return t;
    return clamp(t * t * t * (t * (t * 6 - 15) + 10), 0, 1);
}

function clamp(value, minimum, maximum) {
    return Math.min(maximum, Math.max(minimum, value));
}

import { smootherstep } from './anomaly-influence-field.js?v=somnium-20261108-phase16c-anomaly-optics-2';

const OPTICAL_MATERIAL_SCHEMA_VERSION = 1;
const ANOMALY_OPTICS_SCHEMA_VERSION = 1;
const EPSILON = 1e-12;

/**
 * Creates the initial optical material contract.
 * Reflection/refraction behavior is consumed in a later optics phase; this
 * phase only validates and preserves the coefficients for future consumers.
 */
function createOpticalMaterial(options = {}) {
    const schemaVersion = options.schemaVersion ?? OPTICAL_MATERIAL_SCHEMA_VERSION;
    if (schemaVersion !== OPTICAL_MATERIAL_SCHEMA_VERSION) {
        throw new RangeError(`Unsupported optical material schema version: ${schemaVersion}`);
    }

    const material = {
        schemaVersion,
        id: requireId(options.id, 'Optical material id'),
        enabled: options.enabled ?? true,
        refractiveIndex: options.refractiveIndex ?? 1,
        attenuationPerUnit: options.attenuationPerUnit ?? 0,
        transmission: options.transmission ?? 1,
        reflection: options.reflection ?? 0
    };

    if (typeof material.enabled !== 'boolean') {
        throw new TypeError('Optical material enabled must be boolean.');
    }
    requirePositive(material.refractiveIndex, 'Optical material refractiveIndex');
    requireNonNegative(material.attenuationPerUnit, 'Optical material attenuationPerUnit');
    requireUnitInterval(material.transmission, 'Optical material transmission');

        throw new RangeError('Optical material transmission and reflection cannot sum above one.');
    }

    return material;

/**
 * Creates an optical permission/configuration independent of player traversal.
 * Disabled by default so adding the contract cannot silently make old
 * traversable anomalies transmit light.
 */
function createAnomalyOptics(options = {}) {
    const schemaVersion = options.schemaVersion ?? ANOMALY_OPTICS_SCHEMA_VERSION;
    if (schemaVersion !== ANOMALY_OPTICS_SCHEMA_VERSION) {
        throw new RangeError(`Unsupported anomaly optics schema version: ${schemaVersion}`);
    }

    const optics = {
        schemaVersion,
        enabled: options.enabled ?? false,
        bidirectional: options.bidirectional ?? true,
        wCouplingScale: options.wCouplingScale ?? 1,
        // Remaps the shared smootherstep envelope for optics without changing
        // its centre, radii, endpoint values, or smooth zero-slope horizon.
        // Values below one expose optical W freedom earlier in the shell.
        wCouplingProfileBias: options.wCouplingProfileBias ?? 1,
        maxWOpticalReach: options.maxWOpticalReach ?? null,
        apertureRadius: options.apertureRadius ?? null,
        interfaceRule: options.interfaceRule ?? 'aperture-normal-transfer',
        // Endpoint sampling makes the bridge metric a stable interpolation of
        // the two aperture tangent metrics. Pointwise sampling is available
        // for future anomalies that explicitly want ambient chart gradients
        // to continue through the full 4D interior.
        metricSampling: options.metricSampling ?? 'endpoint',
        // 'ambient' gives the anomaly interior its own isotropic Euclidean
        // tangent metric, decoupled from endpoint chart scale.
        // Controls the share of normal-plane launch momentum aimed radially
        // into the anomaly versus along its fourth-dimensional normal.
        entryRadialBias: options.entryRadialBias ?? 1,
        refractiveIndex: options.refractiveIndex ?? 1,
        attenuationPerUnit: options.attenuationPerUnit ?? 0,
        transmission: options.transmission ?? 1
    };

    if (typeof optics.enabled !== 'boolean' || typeof optics.bidirectional !== 'boolean') {
        throw new TypeError('Anomaly optics enabled and bidirectional flags must be boolean.');
    }
    requireUnitInterval(optics.wCouplingScale, 'Anomaly optics wCouplingScale');
    requirePositive(optics.wCouplingProfileBias, 'Anomaly optics wCouplingProfileBias');
    if (optics.wCouplingProfileBias < 0.001 || optics.wCouplingProfileBias > 100) {
        throw new RangeError('Anomaly optics wCouplingProfileBias must be between 0.001 and 100.');
    }
    if (optics.maxWOpticalReach !== null) {
        requirePositive(optics.maxWOpticalReach, 'Anomaly optics maxWOpticalReach');
    }
    if (optics.apertureRadius !== null) {
        requirePositive(optics.apertureRadius, 'Anomaly optics apertureRadius');
    }
    if (optics.interfaceRule !== 'aperture-normal-transfer') {
        throw new RangeError(`Unsupported anomaly optical interface rule: ${optics.interfaceRule}`);
    }
    if (!['endpoint', 'pointwise', 'ambient'].includes(optics.metricSampling)) {
        throw new RangeError(`Unsupported anomaly optical metric sampling mode: ${optics.metricSampling}`);
    }
    requirePositive(optics.entryRadialBias, 'Anomaly optics entryRadialBias');
    if (optics.entryRadialBias > 16) {
        throw new RangeError('Anomaly optics entryRadialBias cannot exceed 16.');
    }
    if (optics.enabled && optics.maxWOpticalReach === null) {
        throw new TypeError('Enabled anomaly optics requires an explicit maxWOpticalReach.');
    }
    requirePositive(optics.refractiveIndex, 'Anomaly optics refractiveIndex');
    requireNonNegative(optics.attenuationPerUnit, 'Anomaly optics attenuationPerUnit');
    requireUnitInterval(optics.transmission, 'Anomaly optics transmission');

    return optics;
}

function evaluateOpticalAnomalyCoupling(anomaly, position4) {
    const optics = createAnomalyOptics(anomaly?.optics ?? {});
    if (!optics.enabled) {
        return { enabled: false, coupling: 0, opticalWReach: 0, distance: Infinity };
    }

    const influence = anomaly.influence;
    if (!influence?.center || !Number.isFinite(influence.horizonRadius) ||
        !Number.isFinite(influence.innerRadius) || influence.horizonRadius <= influence.innerRadius) {
        throw new TypeError('Enabled anomaly optics requires a resolved influence center and valid radii.');
    }
    validatePosition4(position4, 'Optical anomaly evaluation position');

    const distance = Math.hypot(
        position4.x - influence.center.x,
        position4.y - influence.center.y,
        position4.z - influence.center.z,
        position4.w - influence.center.w
    );
    const t = clamp(
        (influence.horizonRadius - distance) /
            (influence.horizonRadius - influence.innerRadius),
        0,
        1
    );
    const envelope = smootherstep(t);
    const biasDenominator = envelope + optics.wCouplingProfileBias * (1 - envelope);
    const opticalEnvelope = envelope === 0 ? 0 :
        envelope === 1 ? 1 : envelope / biasDenominator;
    const coupling = clamp(opticalEnvelope * optics.wCouplingScale, 0, 1);

    return {
        enabled: true,
        distance,
        normalized: t,
        envelope,
        opticalEnvelope,
        coupling,
        opticalWReach: optics.maxWOpticalReach * coupling
    };
}

function validatePosition4(value, label) {
    if (!value || !['x', 'y', 'z', 'w'].every((axis) => Number.isFinite(value[axis]))) {
        throw new TypeError(`${label} must contain finite x/y/z/w coordinates.`);
    }
}

function requireId(value, label) {
    if (typeof value !== 'string' || value.trim() === '') {
        throw new TypeError(`${label} must be a non-empty string.`);
    }
    return value.trim();
}

function requirePositive(value, label) {
    if (!Number.isFinite(value) || value <= 0) {
        throw new RangeError(`${label} must be positive and finite.`);
    }
}

function requireNonNegative(value, label) {
    if (!Number.isFinite(value) || value < 0) {
        throw new RangeError(`${label} must be non-negative and finite.`);
    }
}

function requireUnitInterval(value, label) {
    if (!Number.isFinite(value) || value < 0 || value > 1) {
        throw new RangeError(`${label} must be between zero and one.`);
    }
}

function clamp(value, minimum, maximum) {
    return Math.min(maximum, Math.max(minimum, value));
}

/** Render-space-only visualization for authoritative slice optical paths. */

import { ambientToSlice } from './spatial-4d.js?v=somnium-20261108-phase16c-anomaly-optics-2';

class OpticalDebugRenderer {
    constructor(scene, options = {}) {
        if (!scene) throw new TypeError('OpticalDebugRenderer requires a Babylon scene.');
        this.scene = scene;
        this.name = options.name ?? 'somniumOpticalRayDebug';
        this.meshes = [];
        this.visible = options.visible ?? true;
        this.disposed = false;
    }

    renderTrace(result, options = {}) {
        this.assertActive();
        const slice = options.slice;
        const transform = options.spatialTransform;
        if (!slice?.id || !slice.frame || typeof transform?.worldToRender !== 'function') {
            throw new TypeError('Optical debug rendering requires a slice and its SpatialTransform.');
        }
        const tolerance = options.sliceTolerance ?? 1e-6;
        if (!Number.isFinite(tolerance) || tolerance <= 0) throw new RangeError('Optical debug slice tolerance must be positive.');
        const color = resolveColor(options.color);
        const groups = [];
        let active = [];
        for (const sample of result?.samples ?? []) {
            const local = ambientToSlice(slice.frame, sample.position);
            if (Math.abs(local.normalOffset) > tolerance) {
                flushGroup(groups, active);
                active = [];
                continue;
            }
            const render = transform.worldToRender(local.position);
            if (![render.x, render.y, render.z].every(Number.isFinite)) {
                flushGroup(groups, active);
                active = [];
                continue;
            }
            active.push(new BABYLON.Vector3(render.x, render.y, render.z));
        }
        flushGroup(groups, active);
        this.clear();
        let index = 0;
        for (const points of groups) {
            if (points.length < 2) continue;
            const mesh = BABYLON.MeshBuilder.CreateLines(`${this.name}_${slice.id}_${index++}`, { points }, this.scene);
            mesh.color = new BABYLON.Color3(color.r, color.g, color.b);
            mesh.alpha = options.alpha ?? 1;
            mesh.isPickable = false;
            mesh.alwaysSelectAsActiveMesh = true;
            mesh.metadata = {
                ...(mesh.metadata || {}),
                opticalDebugRay: true,
                sliceId: slice.id,
                opticalStatus: result.status,
                opticalEvent: result.event?.primitiveType ?? null
            };
            mesh.isVisible = this.visible;
            this.meshes.push(mesh);
        }
        return this.meshes.slice();
    }

    setVisible(visible) {
        this.assertActive();
        this.visible = Boolean(visible);
        for (const mesh of this.meshes) mesh.isVisible = this.visible;
    }

    clear() {
        if (this.disposed) return;
        for (const mesh of this.meshes) mesh.dispose(false, false);
        this.meshes.length = 0;
    }

    getStats() {
        return { visible: this.visible, meshCount: this.meshes.length, disposed: this.disposed };
    }

    assertActive() {
        if (this.disposed) throw new Error('OpticalDebugRenderer has been disposed.');
    }

    dispose() {
        if (this.disposed) return;
        this.clear();
        this.disposed = true;
    }
}

function flushGroup(groups, points) {
    if (points.length > 1) groups.push(points);
}

function resolveColor(value) {
    if (!value) return { r: 0.22, g: 0.92, b: 1 };
    const color = { r: value.r, g: value.g, b: value.b };
    if (![color.r, color.g, color.b].every((component) => Number.isFinite(component) && component >= 0 && component <= 1)) {
        throw new RangeError('Optical debug color components must be between zero and one.');
    }
    return color;
}

/**
 * CPU-side optical metric construction in authoritative ambient R4.
 * This module has no Babylon.js/render-space dependencies.
 */

import {
    ambientToSlice,
    assertVector4,
    createCanonicalSliceFrame,
    createSliceFrame
} from './spatial-4d.js?v=somnium-20261108-phase16c-anomaly-optics-2';

const AXES = ['x', 'y', 'z', 'w'];
const MATRIX3_SIZE = 9;
const MATRIX4_SIZE = 16;
const SPD_EPSILON = 1e-14;
const DEFAULT_GRADIENT_STEP = 1e-4;

/**
 * Creates an optical metric field over ambient four-space.
 *
 * sliceMetricAt(localPosition, context) returns a 3x3 intrinsic slice metric,
 * either as a nine-value row-major array or an object with a `values` array.
 * refractiveIndexAt and wCouplingAt may be constants or callbacks.
 *
 * At exact zero W coupling, `metric` is null and `cometric` remains finite and
 * positive-semidefinite. This represents a genuinely constrained normal
 * direction without inserting Infinity into a matrix.
 */
function createOpticalMetricField(options = {}) {
    const frame = options.sliceFrame
        ? createSliceFrame(options.sliceFrame)
        : createCanonicalSliceFrame(options.sliceW ?? 0);
    const sliceMetricAt = options.sliceMetricAt ?? identityMetric3;
    const refractiveIndexAt = options.refractiveIndexAt ?? options.refractiveIndex ?? 1;
    const wCouplingAt = options.wCouplingAt ?? options.wCoupling ?? 0;
    const normalMetricScaleAt = options.normalMetricScaleAt ?? options.normalMetricScale ?? 1;
    const gradientStep = options.gradientStep ?? DEFAULT_GRADIENT_STEP;
    const containsPosition = options.containsPosition ?? (() => true);

    if (typeof sliceMetricAt !== 'function') {
        throw new TypeError('sliceMetricAt must be a function.');
    }
    if (typeof containsPosition !== 'function') {
        throw new TypeError('containsPosition must be a function.');
    }
    validatePositive(gradientStep, 'Optical metric gradient step');

    function sample(position) {
        assertVector4(position);
        const local = ambientToSlice(frame, position);
        const context = {
            ambientPosition: copy4(position),
            localPosition: { ...local.position },
            normalOffset: local.normalOffset,
            frame
        };
        const baseMetric = readMetric3(sliceMetricAt(local.position, context));
        const refractiveIndex = evaluateScalar(refractiveIndexAt, position, context, 'Refractive index');
        const wCoupling = evaluateScalar(wCouplingAt, position, context, 'W optical coupling');
        const baseNormalMetricScale = typeof normalMetricScaleAt === 'function'
            ? normalMetricScaleAt(copy4(position), context, baseMetric.slice())
            : normalMetricScaleAt;
        validatePositive(refractiveIndex, 'Refractive index');
        validateUnitInterval(wCoupling, 'W optical coupling');
        validatePositive(baseNormalMetricScale, 'Normal optical metric scale');
        // The fourth coordinate shares the same ambient unit as the slice
        // coordinates. Its reference metric scale therefore follows the local
        // intrinsic slice metric instead of assuming an unrelated unit scale.
        // This is essential when a compactified Poincare slice has metric
        // coefficients many orders of magnitude smaller than Euclidean space.
        const opticalNormalMetricScale = baseNormalMetricScale * refractiveIndex * refractiveIndex;
        if (!Number.isFinite(opticalNormalMetricScale) || opticalNormalMetricScale <= 0) {
            throw new RangeError('Normal optical metric scale must remain positive and finite.');
        }

        const opticalMetric3 = baseMetric.map((value) => value * refractiveIndex * refractiveIndex);
        const inverseBaseMetric = invertSpd3(baseMetric);
        const inverseOpticalMetric3 = inverseBaseMetric.map((value) => value / (refractiveIndex * refractiveIndex));
        const tangentCometric = lift3To4(frame.axes, inverseOpticalMetric3);
        const normal = frame.normal;
        const cometric = tangentCometric.map((value, index) => {
            const row = Math.floor(index / 4);
            const column = index % 4;
            return value + (wCoupling * wCoupling / opticalNormalMetricScale) *
                normal[AXES[row]] * normal[AXES[column]];
        });
        validateFiniteMatrix(cometric, MATRIX4_SIZE, 'Optical cometric');
        validateSymmetricMatrix(cometric, 4, 'Optical cometric');

        const isConstrained = wCoupling === 0;
        let metric = null;
        if (!isConstrained) {
            const tangentMetric = lift3To4(frame.axes, opticalMetric3);
            const inverseNormalScale = opticalNormalMetricScale / (wCoupling * wCoupling);
            if (!Number.isFinite(inverseNormalScale)) {
                throw new RangeError('W optical coupling is too small to construct a finite metric tensor.');
            }
            metric = tangentMetric.map((value, index) => {
                const row = Math.floor(index / 4);
                const column = index % 4;
                return value + inverseNormalScale * normal[AXES[row]] * normal[AXES[column]];
            });
            validateSpdMatrix(metric, 4, 'Optical metric');
        }

        return {
            position: copy4(position),
            localPosition: { ...local.position },
            normalOffset: local.normalOffset,
            refractiveIndex,
            wCoupling,
            normalMetricScale: opticalNormalMetricScale,
            isConstrained,
            metric,
            cometric,
            tangentMetric: opticalMetric3,
            tangentCometric: inverseOpticalMetric3,
            frame
        };
    }

    function cometricAt(position) {
        return sample(position).cometric;
    }

    function cometricGradientAt(position, gradientOptions = {}) {
        assertVector4(position);
        const step = gradientOptions.step ?? gradientStep;
        validatePositive(step, 'Cometric gradient step');
        const gradient = [];
        for (const axis of AXES) {
            const forward = copy4(position);
            const backward = copy4(position);
            forward[axis] += step;
            backward[axis] -= step;
            const positive = sample(forward).cometric;
            const negative = sample(backward).cometric;
            gradient.push(positive.map((value, index) => (value - negative[index]) / (2 * step)));
        }
        return gradient;
    }

    return Object.freeze({
        frame,
        sample,
        metricAt: (position) => sample(position).metric,
        cometricAt,
        cometricGradientAt,
        containsPosition(position) {
            assertVector4(position);
            return containsPosition(copy4(position)) === true;
        }
    });
}

/** Convex blend of 3x3 SPD slice metric tensors in the same local basis. */
function blendPositiveDefiniteMetrics3(metrics, weights) {
    if (!Array.isArray(metrics) || metrics.length === 0 || !Array.isArray(weights) || weights.length !== metrics.length) {
        throw new TypeError('Metric blending requires matching non-empty metric and weight arrays.');
    }
    const normalizedWeights = normalizeWeights(weights);
    const result = new Array(MATRIX3_SIZE).fill(0);
    metrics.forEach((metric, metricIndex) => {
        const values = readMetric3(metric);
        for (let index = 0; index < MATRIX3_SIZE; index++) {
            result[index] += values[index] * normalizedWeights[metricIndex];
        }
    });
    validateSpdMatrix(result, 3, 'Blended optical slice metric');
    return result;
}


/**
 * Scale-balanced SPD interpolation for slice metrics expressed in different
 * world-coordinate units. Each tensor is factored into its determinant scale
 * and unit-determinant shape; scales interpolate geometrically while shapes
 * interpolate convexly. This avoids a numerically singular linear blend when
 * two valid charts differ by many orders of magnitude in optical units.
 */
function blendScaledPositiveDefiniteMetrics3(metrics, weights) {
    if (!Array.isArray(metrics) || metrics.length === 0 ||
        !Array.isArray(weights) || weights.length !== metrics.length) {
        throw new TypeError('Scaled metric blending requires matching non-empty metric and weight arrays.');
    }
    const normalizedWeights = normalizeWeights(weights);
    const shape = new Array(MATRIX3_SIZE).fill(0);
    let logScale = 0;
    metrics.forEach((metric, index) => {
        const values = readMetric3(metric);
        const scale = geometricMeanMetricScale3(values);
        const weight = normalizedWeights[index];
        logScale += weight * Math.log(scale);
        for (let component = 0; component < MATRIX3_SIZE; component++) {
            shape[component] += weight * values[component] / scale;
        }
    });
    validateSpdMatrix(shape, 3, 'Scale-balanced blended optical slice metric shape');
    const blendedScale = Math.exp(logScale);
    validatePositive(blendedScale, 'Scale-balanced blended optical slice metric scale');
    return shape.map((value) => value * blendedScale);
}

/** Maps a slice-coordinate SpatialTransform metric callback into this API. */
function spatialTransformMetricProvider(spatialTransform) {
    if (!spatialTransform || typeof spatialTransform.metricAtWorld !== 'function') {
        throw new TypeError('A SpatialTransform with metricAtWorld(position) is required.');
    }
    return (localPosition) => spatialTransform.metricAtWorld(localPosition);
}

function lift3To4(axes, matrix3) {
    const result = new Array(MATRIX4_SIZE).fill(0);
    for (let row = 0; row < 4; row++) {
        for (let column = 0; column < 4; column++) {
            for (let a = 0; a < 3; a++) {
                for (let b = 0; b < 3; b++) {
                    result[row * 4 + column] +=
                        axes[a][AXES[row]] * matrix3[a * 3 + b] * axes[b][AXES[column]];
                }
            }
        }
    }
    return result;
}

function readMetric3(metric) {
    const values = Array.isArray(metric) || ArrayBuffer.isView(metric) ? metric : metric?.values;
    if (!values || values.length !== MATRIX3_SIZE) {
        throw new TypeError('A slice metric must contain nine row-major values.');
    }
    const result = Array.from(values);
    validateFiniteMatrix(result, MATRIX3_SIZE, 'Slice metric');
    validateSymmetricMatrix(result, 3, 'Slice metric');
    validateSpdMatrix(result, 3, 'Slice metric');
    return result;
}

/**
 * Coordinate-invariant characteristic scale (det(g)^(1/3)) of a 3D SPD metric.
 * Computation is normalized first so very small or very large metric units are
 * handled without determinant underflow/overflow.
 */
function geometricMeanMetricScale3(matrix) {
    const values = readMetric3(matrix);
    const scale = matrixScale(values);
    if (!(scale > 0) || !Number.isFinite(scale)) {
        throw new RangeError('Cannot derive a normal metric scale from a degenerate slice metric.');
    }
    const normalized = values.map((value) => value / scale);
    const a = normalized[0], b = normalized[1], c = normalized[2];
    const d = normalized[4], e = normalized[5], f = normalized[8];
    const determinant = a * (d * f - e * e) +
        b * (c * e - b * f) + c * (b * e - c * d);
    if (!Number.isFinite(determinant) || determinant <= 0) {
        throw new RangeError('Cannot derive a normal metric scale from a non-positive or non-finite slice determinant.');
    }
    const result = scale * Math.cbrt(determinant);
    validatePositive(result, 'Geometric-mean slice metric scale');
    return result;
}

function invertSpd3(matrix) {
    validateSpdMatrix(matrix, 3, 'Slice metric');
    // Determinants scale cubically. Testing an absolute determinant threshold
    // incorrectly rejects perfectly valid metrics expressed in small world
    // units (the live Main World Poincare metric is around 1e-7 per entry).
    // Normalize first so the conditioning test is scale invariant.
    const scale = matrixScale(matrix);
    const normalized = matrix.map((value) => value / scale);
    const a = normalized[0], b = normalized[1], c = normalized[2];
    const d = normalized[4], e = normalized[5], f = normalized[8];
    const cofactor00 = d * f - e * e;
    const cofactor01 = c * e - b * f;
    const cofactor02 = b * e - c * d;
    const cofactor11 = a * f - c * c;
    const cofactor12 = b * c - a * e;
    const cofactor22 = a * d - b * b;
    const determinant = a * cofactor00 + b * cofactor01 + c * cofactor02;
    if (!Number.isFinite(determinant) || determinant <= SPD_EPSILON) {
        throw new RangeError('Slice metric is singular or ill-conditioned.');
    }
    const inverse = [
        cofactor00 / determinant / scale,
        cofactor01 / determinant / scale,
        cofactor02 / determinant / scale,
        cofactor01 / determinant / scale,
        cofactor11 / determinant / scale,
        cofactor12 / determinant / scale,
        cofactor02 / determinant / scale,
        cofactor12 / determinant / scale,
        cofactor22 / determinant / scale
    ];
    validateFiniteMatrix(inverse, MATRIX3_SIZE, 'Inverse slice metric');
    return inverse;
}

function validateSpdMatrix(matrix, size, label) {
    validateFiniteMatrix(matrix, size * size, label);
    validateSymmetricMatrix(matrix, size, label);
    // Cholesky pivots are also scale dependent. Normalize the tensor before
    // testing positive-definiteness, retaining a relative conditioning bound.
    const scale = matrixScale(matrix);
    if (!(scale > 0) || !Number.isFinite(scale)) {
        throw new RangeError(`${label} must be positive definite.`);
    }
    const normalized = matrix.map((value) => value / scale);
    const lower = new Array(size * size).fill(0);
    for (let row = 0; row < size; row++) {
        for (let column = 0; column <= row; column++) {
            let value = normalized[row * size + column];
            for (let k = 0; k < column; k++) {
                value -= lower[row * size + k] * lower[column * size + k];
            }
            if (row === column) {
                if (value <= SPD_EPSILON || !Number.isFinite(value)) {
                    throw new RangeError(`${label} must be positive definite.`);
                }
                lower[row * size + column] = Math.sqrt(value);
            } else {
                lower[row * size + column] = value / lower[column * size + column];
            }
        }
    }
    return lower;
}

function matrixScale(matrix) {
    return matrix.reduce((maximum, value) => Math.max(maximum, Math.abs(value)), 0);
}

function validateSymmetricMatrix(matrix, size, label) {
    const scale = matrixScale(matrix);
    const tolerance = Math.max(Number.MIN_VALUE, scale * 1e-9);
    for (let row = 0; row < size; row++) {
        for (let column = row + 1; column < size; column++) {
            if (Math.abs(matrix[row * size + column] - matrix[column * size + row]) > tolerance) {
                throw new TypeError(`${label} must be symmetric.`);
            }
        }
    }
}

function validateFiniteMatrix(matrix, expectedLength, label) {
    if (!Array.isArray(matrix) || matrix.length !== expectedLength || !matrix.every(Number.isFinite)) {
        throw new TypeError(`${label} must contain ${expectedLength} finite values.`);
    }
}

function normalizeWeights(weights) {
    let sum = 0;
    const result = weights.map((weight) => {
        if (!Number.isFinite(weight) || weight < 0) throw new RangeError('Metric blend weights must be non-negative and finite.');
        sum += weight;
        return weight;
    });
    if (!Number.isFinite(sum) || sum <= 0) throw new RangeError('Metric blend weights must have a positive sum.');
    return result.map((weight) => weight / sum);
}

function evaluateScalar(source, position, context, label) {
    const value = typeof source === 'function' ? source(copy4(position), context) : source;
    if (!Number.isFinite(value)) throw new TypeError(`${label} must evaluate to a finite number.`);
    return value;
}

function validatePositive(value, label) {
    if (!Number.isFinite(value) || value <= 0) throw new RangeError(`${label} must be positive and finite.`);
}

function validateUnitInterval(value, label) {
    if (!Number.isFinite(value) || value < 0 || value > 1) throw new RangeError(`${label} must be between zero and one.`);
}

function identityMetric3() {
    return [1, 0, 0, 0, 1, 0, 0, 0, 1];
}

    return { x: position.x, y: position.y, z: position.z, w: position.w };

/** Deterministic CPU reference integrator for Hamiltonian optical rays in R4. */

import { assertVector4 } from './spatial-4d.js?v=somnium-20261108-phase16c-anomaly-optics-2';

const AXES = ['x', 'y', 'z', 'w'];
const STATE_SIZE = 8;
const DEFAULTS = Object.freeze({
    maxPathLength: 10,
    initialStepSize: 0.025,
    minStepSize: 1e-7,
    maxStepSize: 0.1,
    absoluteTolerance: 1e-9,
    relativeTolerance: 1e-7,
    gradientStep: 1e-4,
    maxSteps: 4096,
    speedOfLight: 1,
    includeSamples: true,
    segmentEvent: null
});

function createOpticalRayState(options = {}) {
    assertVector4(options.position);
    assertVector4(options.momentum);
    const opticalPathLength = options.opticalPathLength ?? 0;
    const travelTime = options.travelTime ?? 0;
    const intensity = options.intensity ?? 1;
    validateNonNegative(opticalPathLength, 'Optical path length');
    validateNonNegative(travelTime, 'Ray travel time');
    if (!Number.isFinite(intensity) || intensity < 0 || intensity > 1) {
        throw new RangeError('Ray intensity must be between zero and one.');
    }
    return {
        position: copy4(options.position),
        momentum: copy4(options.momentum),
        opticalPathLength,
        travelTime,
        intensity
    };
}

/** Scales a canonical covector so its initial Hamiltonian equals targetH. */
function normalizeOpticalMomentum(field, position, momentum, targetH = 0.5) {
    requireField(field);
    assertVector4(position);
    assertVector4(momentum);
    validatePositive(targetH, 'Target optical Hamiltonian');
    const sample = field.sample(position);
    const currentH = hamiltonianFromCometric(sample.cometric, momentum);
    if (!Number.isFinite(currentH) || currentH <= 1e-15) {
        throw new RangeError('Optical ray momentum has no component in an available propagation direction.');
    }
    const scale = Math.sqrt(targetH / currentH);
    return {
        x: momentum.x * scale,
        y: momentum.y * scale,
        z: momentum.z * scale,
        w: momentum.w * scale
    };
}

function evaluateOpticalHamiltonian(field, position, momentum) {
    requireField(field);
    assertVector4(position);
    assertVector4(momentum);
    return hamiltonianFromCometric(field.cometricAt(position), momentum);
}

/**
 * Traces one ray with adaptive RK4 step-doubling.
 * The path parameter is normalized optical arclength when H = 1/2. The solver
 * also accumulates the measured optical path and travel time independently.
 *
 * Declared metric-domain boundaries are handled as explicit terminal events: the
 * adaptive integrator reduces a step that crosses the boundary and returns the
 * last valid state if no safe step remains. All other metric/geometry errors
 * propagate unchanged so invalid configurations are not silently hidden.
 */
function traceOpticalRay(field, initialRay, options = {}) {
    requireField(field);
    const config = { ...DEFAULTS, ...options };
    validateTraceOptions(config);

    const ray = createOpticalRayState(initialRay);
    let vector = stateToArray(ray);
    let lambda = 0;
    let stepSize = config.initialStepSize;
    let opticalPathLength = ray.opticalPathLength;
    let travelTime = ray.travelTime;
    let acceptedSteps = 0;
    let rejectedSteps = 0;
    let maxRelativeStepError = 0;
    let event = null;
    const samples = [];
    if (typeof field.containsPosition === 'function' && !field.containsPosition(ray.position)) {
        throw new RangeError('Initial optical ray lies outside the metric domain.');
    }
    let initialHamiltonian = evaluateOpticalHamiltonian(field, ray.position, ray.momentum);
    if (!Number.isFinite(initialHamiltonian) || initialHamiltonian <= 1e-15) {
        throw new RangeError('Initial optical ray must have positive finite Hamiltonian.');
    }
    const startingHamiltonian = initialHamiltonian;
    if (config.includeSamples) samples.push(makeSample(vector, lambda, opticalPathLength, travelTime, initialHamiltonian));

    let status = 'max-path-length';
    while (opticalPathLength - ray.opticalPathLength < config.maxPathLength && acceptedSteps + rejectedSteps < config.maxSteps) {
        const currentPosition = arrayToPosition(vector);
        const currentMomentum = arrayToMomentum(vector);
        const currentHamiltonian = evaluateOpticalHamiltonian(field, currentPosition, currentMomentum);
        const opticalSpeed = Math.sqrt(Math.max(0, 2 * currentHamiltonian));
        if (!Number.isFinite(opticalSpeed) || opticalSpeed <= 1e-14) {
            status = 'stalled';
            break;
        }

        const remainingOpticalPath = config.maxPathLength - (opticalPathLength - ray.opticalPathLength);
        if (remainingOpticalPath <= 1e-12) break;
        const h = Math.min(stepSize, config.maxStepSize, remainingOpticalPath / opticalSpeed);
        if (h < config.minStepSize) {
            status = 'step-underflow';
            break;
        }

        // Step doubling estimates the fourth-order RK error and provides a
        // Richardson-corrected candidate without external numerical libraries.
        // If an intermediate RK stage crosses a finite chart boundary, reject
        // and halve the step until the last valid state is safely retained.
        let full;
        let halfA;
        let halfB;
        let error;
        let corrected = null;
        let nextPosition = null;
        let nextMomentum = null;
        let nextHamiltonian = null;
        try {
            full = rk4Step(field, vector, h, config.gradientStep);
            halfA = rk4Step(field, vector, h / 2, config.gradientStep);
            halfB = rk4Step(field, halfA, h / 2, config.gradientStep);
            error = estimateStepError(full, halfB, config.absoluteTolerance, config.relativeTolerance);
            maxRelativeStepError = Math.max(maxRelativeStepError, error);

            if (error <= 1) {
                corrected = halfB.map((value, index) => value + (value - full[index]) / 15);
                nextPosition = arrayToPosition(corrected);
                assertInsideMetricDomain(field, nextPosition);
                nextMomentum = arrayToMomentum(corrected);
                nextHamiltonian = evaluateOpticalHamiltonian(field, nextPosition, nextMomentum);
            }
        } catch (error) {
            if (!isMetricDomainBoundaryError(error)) throw error;
            rejectedSteps++;
            const reducedStep = h * 0.5;
            if (reducedStep < config.minStepSize) {
                status = 'domain-boundary';
                event = { kind: 'chart-domain-boundary', position: currentPosition };
                break;
            }
            stepSize = reducedStep;
            continue;
        }

        if (error <= 1) {
            if (!Number.isFinite(nextHamiltonian) || nextHamiltonian <= 0) {
                throw new RangeError('Optical ray integration produced an invalid Hamiltonian state.');
            }

            const nextOpticalSpeed = Math.sqrt(2 * nextHamiltonian);
            const opticalIncrement = h * (opticalSpeed + nextOpticalSpeed) / 2;
            const boundedIncrement = Math.min(remainingOpticalPath, opticalIncrement);
            const segment = {
                startPosition: currentPosition,
                endPosition: nextPosition,
                startMomentum: currentMomentum,
                endMomentum: nextMomentum,
                startOpticalPathLength: opticalPathLength,
                endOpticalPathLength: opticalPathLength + boundedIncrement,
                startTravelTime: travelTime,
                endTravelTime: travelTime + boundedIncrement / config.speedOfLight
            };
            const detectedEvent = typeof config.segmentEvent === 'function'
                ? config.segmentEvent(segment)
                : null;
            if (detectedEvent) {
                const parameter = detectedEvent.parameter;
                if (!Number.isFinite(parameter) || parameter < 0 || parameter > 1) {
                    throw new RangeError('Optical segment event parameter must be between zero and one.');
                }
                vector = corrected.map((value, index) => vector[index] + (value - vector[index]) * parameter);
                const hitPosition = arrayToPosition(vector);
                if (typeof field.containsPosition === 'function' && !field.containsPosition(hitPosition)) {
                    vector = stateToArray({ position: currentPosition, momentum: currentMomentum });
                    status = 'domain-boundary';
                    event = { kind: 'chart-domain-boundary', position: currentPosition };
                    break;
                }
                const hitMomentum = arrayToMomentum(vector);
                const hitHamiltonian = evaluateOpticalHamiltonian(field, hitPosition, hitMomentum);
                if (!Number.isFinite(hitHamiltonian) || hitHamiltonian <= 0) {
                    throw new RangeError('Optical segment event produced an invalid Hamiltonian state.');
                }
                const hitOpticalIncrement = boundedIncrement * parameter;
                opticalPathLength += hitOpticalIncrement;
                travelTime += hitOpticalIncrement / config.speedOfLight;
                lambda += h * parameter;
                acceptedSteps++;
                initialHamiltonian = hitHamiltonian;
                event = { ...detectedEvent, parameter };
                if (config.includeSamples) samples.push(makeSample(
                    vector, lambda, opticalPathLength, travelTime, hitHamiltonian
                ));
                status = 'event';
                break;
            }

            opticalPathLength += boundedIncrement;
            travelTime += boundedIncrement / config.speedOfLight;
            lambda += h;
            vector = corrected;
            acceptedSteps++;
            initialHamiltonian = nextHamiltonian;
            if (config.includeSamples) samples.push(makeSample(
                vector, lambda, opticalPathLength, travelTime, nextHamiltonian
            ));

            const factor = error <= 1e-12 ? 2.5 : clamp(0.9 * Math.pow(1 / error, 0.2), 0.2, 2.5);
            stepSize = clamp(h * factor, config.minStepSize, config.maxStepSize);
        } else {
            rejectedSteps++;
            const factor = clamp(0.9 * Math.pow(1 / error, 0.2), 0.1, 0.5);
            stepSize = h * factor;
            if (stepSize < config.minStepSize) {
                status = 'step-underflow';
                break;
            }
        }
    }

    if (acceptedSteps + rejectedSteps >= config.maxSteps &&
        opticalPathLength - ray.opticalPathLength < config.maxPathLength) {
        status = 'max-steps';
    }

    const finalPosition = arrayToPosition(vector);
    const finalMomentum = arrayToMomentum(vector);
    const finalHamiltonian = evaluateOpticalHamiltonian(field, finalPosition, finalMomentum);
    const absoluteHamiltonianDrift = Math.abs(finalHamiltonian - startingHamiltonian);
    const relativeHamiltonianDrift = absoluteHamiltonianDrift / Math.max(Math.abs(startingHamiltonian), 1e-15);
    const finalState = {
        position: finalPosition,
        momentum: finalMomentum,
        opticalPathLength,
        travelTime,
        intensity: ray.intensity
    };

    return {
        status,
        event,
        finalState,
        samples,
        diagnostics: {
            initialHamiltonian: startingHamiltonian,
            finalHamiltonian,
            absoluteHamiltonianDrift,
            relativeHamiltonianDrift,
            acceptedSteps,
            rejectedSteps,
            maxRelativeStepError,
            pathParameter: lambda
        }
    };
}

/** Stable, timestamp-free JSON representation for deterministic ray debugging. */
function serializeOpticalRayResult(result, precision = 12) {
    if (!result || !result.finalState || !Array.isArray(result.samples) || !result.diagnostics) {
        throw new TypeError('A complete optical ray result is required for serialization.');
    }
    if (!Number.isSafeInteger(precision) || precision < 0 || precision > 15) {
        throw new RangeError('Optical ray serialization precision must be an integer from zero through fifteen.');
    }
    const round = (value) => Number(value.toFixed(precision));
    const point = (value) => AXES.map((axis) => round(value[axis]));
    const output = {
        schemaVersion: 1,
        status: result.status,
        finalState: {
            position: point(result.finalState.position),
            momentum: point(result.finalState.momentum),
            opticalPathLength: round(result.finalState.opticalPathLength),
            travelTime: round(result.finalState.travelTime),
            intensity: round(result.finalState.intensity)
        },
        samples: result.samples.map((sample) => ({
            pathParameter: round(sample.pathParameter),
            opticalPathLength: round(sample.opticalPathLength),
            travelTime: round(sample.travelTime),
            position: point(sample.position),
            momentum: point(sample.momentum),
            hamiltonian: round(sample.hamiltonian)
        })),
        diagnostics: Object.fromEntries(Object.entries(result.diagnostics).map(([key, value]) => [key, round(value)]))
    };
    return JSON.stringify(output);
}

function rk4Step(field, state, h, gradientStep) {
    const k1 = derivative(field, state, gradientStep);
    const k2 = derivative(field, addScaled(state, k1, h / 2), gradientStep);
    const k3 = derivative(field, addScaled(state, k2, h / 2), gradientStep);
    const k4 = derivative(field, addScaled(state, k3, h), gradientStep);
    return state.map((value, index) => value + h * (
        k1[index] + 2 * k2[index] + 2 * k3[index] + k4[index]
    ) / 6);
}

function derivative(field, state, gradientStep) {
    const position = arrayToPosition(state);
    assertInsideMetricDomain(field, position);
    const momentum = arrayToMomentum(state);
    const sample = field.sample(position);
    const velocity = multiply4(sample.cometric, momentum);
    const gradient = field.cometricGradientAt(position, { step: gradientStep });
    const momentumDerivative = new Array(4).fill(0);
    for (let coordinate = 0; coordinate < 4; coordinate++) {
        let contraction = 0;
        for (let row = 0; row < 4; row++) {
            for (let column = 0; column < 4; column++) {
                contraction += gradient[coordinate][row * 4 + column] *
                    momentum[AXES[row]] * momentum[AXES[column]];
            }
        }
        momentumDerivative[coordinate] = -0.5 * contraction;
    }
    return [
        velocity.x, velocity.y, velocity.z, velocity.w,
        ...momentumDerivative
    ];
}


function assertInsideMetricDomain(field, position) {
    if (typeof field.containsPosition === 'function' && !field.containsPosition(position)) {
        const error = new RangeError('Optical ray reached a finite geometry chart boundary.');
        error.code = 'OPTICAL_METRIC_DOMAIN_BOUNDARY';
        throw error;
    }
}

function isMetricDomainBoundaryError(error) {
    return error?.code === 'OPTICAL_METRIC_DOMAIN_BOUNDARY';
}

function estimateStepError(full, half, absoluteTolerance, relativeTolerance) {
    let maximum = 0;
    for (let index = 0; index < STATE_SIZE; index++) {
        const scale = absoluteTolerance + relativeTolerance * Math.max(Math.abs(full[index]), Math.abs(half[index]));
        maximum = Math.max(maximum, Math.abs(half[index] - full[index]) / (15 * scale));
    }
    return maximum;
}

function hamiltonianFromCometric(cometric, momentum) {
    const values = AXES.map((axis) => momentum[axis]);
    let result = 0;
    for (let row = 0; row < 4; row++) {
        for (let column = 0; column < 4; column++) {
            result += values[row] * cometric[row * 4 + column] * values[column];
        }
    }
    return 0.5 * result;
}

function multiply4(matrix, vector) {
    const values = AXES.map((axis) => vector[axis]);
    const result = new Array(4).fill(0);
    for (let row = 0; row < 4; row++) {
        for (let column = 0; column < 4; column++) {
            result[row] += matrix[row * 4 + column] * values[column];
        }
    }
    return { x: result[0], y: result[1], z: result[2], w: result[3] };
}

    return {
        pathParameter,
        opticalPathLength,
        travelTime,
        position: arrayToPosition(state),
        momentum: arrayToMomentum(state),
        hamiltonian
    };
}

    return { x: state[0], y: state[1], z: state[2], w: state[3] };
}

function arrayToMomentum(state) {
    return { x: state[4], y: state[5], z: state[6], w: state[7] };
}

function addScaled(state, derivativeValues, scale) {
    return state.map((value, index) => value + derivativeValues[index] * scale);
}

/**
 * Slice-aware CPU optical scene registry.
 *
 * Geometry is retained in authoritative slice-local coordinates, before any
 * Babylon/render projection. Static triangle surfaces receive per-registration
 * BVHs; streamed heightfields use analytic height sampling.
 */

import { createOpticalMaterial } from './optical-contracts.js?v=somnium-20261108-phase16c-anomaly-optics-2';
import { createSliceAddress, getSliceAddressKey } from './world-slice.js?v=somnium-20261108-phase16c-anomaly-optics-2';

const AXES = ['x', 'y', 'z'];
const DEFAULT_LEAF_SIZE = 8;
const EPSILON = 1e-10;

class OpticalSceneRegistry {
    constructor(options = {}) {
        this.leafSize = options.leafSize ?? DEFAULT_LEAF_SIZE;
        if (!Number.isSafeInteger(this.leafSize) || this.leafSize < 1 || this.leafSize > 128) {
            throw new RangeError('Optical BVH leaf size must be an integer from 1 through 128.');
        }
        this.triangleMeshes = new Map();
        this.heightfields = new Map();
        this.disposed = false;
        this.statistics = {
            registeredTriangleMeshes: 0,
            registeredHeightfields: 0,
            triangleCount: 0,
            queries: 0,
            bvhNodesVisited: 0,
            trianglesTested: 0,
            heightfieldSamples: 0,
            hits: 0
        };
    }

    registerTriangleMesh(options = {}) {
        this.assertActive();
        const id = requireId(options.id, 'Optical mesh id');
        const sliceId = requireId(options.sliceId, 'Optical mesh sliceId');
        const geometry = validateTriangleGeometry(options.positions, options.indices);
        const chunkAddress = normalizeChunkAddress(options.chunkAddress, sliceId);
        const material = normalizeMaterial(options.material, id);
        const revision = requireRevision(options.revision);
        const enabled = options.enabled ?? true;
        if (typeof enabled !== 'boolean') throw new TypeError('Optical mesh enabled must be boolean.');

        this.unregister(id);
        const triangles = buildTriangles(geometry.positions, geometry.indices);
        const root = triangles.length ? buildBvh(triangles, this.leafSize) : null;
        const entry = {
            id, sliceId, chunkAddress, chunkKey: chunkAddress ? getSliceAddressKey(chunkAddress) : null,
            revision, enabled, material, root, triangleCount: triangles.length,
            sourceVertexCount: geometry.positions.length / 3
        };
        this.triangleMeshes.set(id, entry);
        this.statistics.registeredTriangleMeshes = this.triangleMeshes.size;
        this.statistics.triangleCount = countTriangles(this.triangleMeshes);
        return describeEntry(entry);
    }

    registerHeightfield(options = {}) {
        this.assertActive();
        const id = requireId(options.id, 'Optical heightfield id');
        const sliceId = requireId(options.sliceId, 'Optical heightfield sliceId');
        const terrain = options.terrain;
        if (!terrain || typeof terrain.sampleHeight !== 'function' ||
            !Number.isSafeInteger(terrain.width) || terrain.width < 2 ||
            !Number.isSafeInteger(terrain.height) || terrain.height < 2) {
            throw new TypeError('Optical heightfield requires a sampled terrain with width and height.');
        }
        const bounds = normalizeHeightfieldBounds(options.bounds);
        const chunkAddress = normalizeChunkAddress(options.chunkAddress, sliceId);
        const material = normalizeMaterial(options.material, id);
        const revision = requireRevision(options.revision);
        const enabled = options.enabled ?? true;
        if (typeof enabled !== 'boolean') throw new TypeError('Optical heightfield enabled must be boolean.');

        this.unregister(id);
        const entry = {
            id, sliceId, terrain, bounds, chunkAddress,
            chunkKey: chunkAddress ? getSliceAddressKey(chunkAddress) : null,
            material, revision, enabled,
            kind: 'heightfield'
        };
        this.heightfields.set(id, entry);
        this.statistics.registeredHeightfields = this.heightfields.size;
        return describeEntry(entry);
    }

    unregister(id) {
        this.assertActive();
        const removedMesh = this.triangleMeshes.delete(id);
        const removedHeightfield = this.heightfields.delete(id);
        this.statistics.registeredTriangleMeshes = this.triangleMeshes.size;
        this.statistics.registeredHeightfields = this.heightfields.size;
        this.statistics.triangleCount = countTriangles(this.triangleMeshes);
        return removedMesh || removedHeightfield;
    }

    intersectSegment(sliceId, from, to, options = {}) {
        this.assertActive();
        const normalizedSliceId = requireId(sliceId, 'Optical query sliceId');
        validatePosition3(from, 'Optical segment start');
        validatePosition3(to, 'Optical segment end');
        const segment = createSegment(from, to);
        if (segment.length <= EPSILON) return null;

        const allowedChunks = options.allowedChunkAddresses
            ? new Set(options.allowedChunkAddresses.map((address) => getSliceAddressKey(address)))
            : null;
        const minimumParameter = options.minimumParameter ?? 1e-7;
        if (!Number.isFinite(minimumParameter) || minimumParameter < 0 || minimumParameter >= 1) {
            throw new RangeError('Optical segment minimumParameter must be in [0, 1).');
        }

        this.statistics.queries++;
        let best = null;
        const counters = { bvhNodesVisited: 0, trianglesTested: 0, heightfieldSamples: 0 };

        for (const entry of this.triangleMeshes.values()) {
            if (!entry.enabled || !entry.material.enabled || entry.sliceId !== normalizedSliceId ||
                !entry.root || !isAllowedChunk(entry.chunkKey, allowedChunks)) continue;
            const candidate = intersectBvh(entry.root, segment, minimumParameter, best?.parameter ?? 1, counters);
            if (candidate && isBetterHit(candidate, best, entry.id)) {
                best = makeHit(entry, candidate, segment, 'triangle');
            }
        }

        for (const entry of this.heightfields.values()) {
            if (!entry.enabled || !entry.material.enabled || entry.sliceId !== normalizedSliceId ||
                !isAllowedChunk(entry.chunkKey, allowedChunks)) continue;
            const candidate = intersectHeightfield(entry, segment, minimumParameter, best?.parameter ?? 1, counters);
            if (candidate && isBetterHit(candidate, best, entry.id)) {
                best = makeHit(entry, candidate, segment, 'heightfield');
            }
        }

        this.statistics.bvhNodesVisited += counters.bvhNodesVisited;
        this.statistics.trianglesTested += counters.trianglesTested;
        this.statistics.heightfieldSamples += counters.heightfieldSamples;
        if (best) this.statistics.hits++;
        return best;
    }

    getStats() {
        this.assertActive();
        return {
            ...this.statistics,
            registeredTriangleMeshes: this.triangleMeshes.size,
            registeredHeightfields: this.heightfields.size,
            triangleCount: countTriangles(this.triangleMeshes),
            slices: [...new Set([
                ...[...this.triangleMeshes.values()].map((entry) => entry.sliceId),
                ...[...this.heightfields.values()].map((entry) => entry.sliceId)
            ])].sort()
        };
    }

    assertActive() {
        if (this.disposed) throw new Error('OpticalSceneRegistry has been disposed.');
    }

    dispose() {
        if (this.disposed) return;
        this.triangleMeshes.clear();
        this.heightfields.clear();
        this.disposed = true;
        this.statistics.registeredTriangleMeshes = 0;
        this.statistics.registeredHeightfields = 0;
        this.statistics.triangleCount = 0;
    }
}

function validateTriangleGeometry(positions, indices) {
    if (!ArrayBuffer.isView(positions) && !Array.isArray(positions)) {
        throw new TypeError('Optical mesh positions must be an array or typed array.');
    }
    if (positions.length === 0 || positions.length % 3 !== 0) {
        throw new RangeError('Optical mesh positions must contain complete XYZ vertices.');
    }
    if (!ArrayBuffer.isView(indices) && !Array.isArray(indices)) {
        throw new TypeError('Optical mesh indices must be an array or typed array.');
    }
    if (indices.length === 0 || indices.length % 3 !== 0) {
        throw new RangeError('Optical mesh indices must contain complete triangles.');
    }
    const positionCopy = Array.from(positions, Number);
    const indexCopy = Array.from(indices, Number);
    if (!positionCopy.every(Number.isFinite)) throw new TypeError('Optical mesh positions must be finite.');
    const vertexCount = positionCopy.length / 3;
    for (const index of indexCopy) {
        if (!Number.isSafeInteger(index) || index < 0 || index >= vertexCount) {
            throw new RangeError('Optical mesh index is outside the vertex buffer.');
        }
    }
    return { positions: positionCopy, indices: indexCopy };
}

function buildTriangles(positions, indices) {
    const triangles = [];
    for (let index = 0; index < indices.length; index += 3) {
        const a = readVertex(positions, indices[index]);
        const b = readVertex(positions, indices[index + 1]);
        const c = readVertex(positions, indices[index + 2]);
        const normal = cross3(subtract3(b, a), subtract3(c, a));
        const normalLength = length3(normal);
        if (normalLength <= EPSILON) continue;
        const min = componentBounds([a, b, c], 'min');
        const max = componentBounds([a, b, c], 'max');
        triangles.push({
            primitiveIndex: index / 3,
            a, b, c,
            min, max,
            centroid: scale3(add3(add3(a, b), c), 1 / 3),
            normal: scale3(normal, 1 / normalLength)
        });
    }
    return triangles;
}

function buildBvh(triangles, leafSize) {
    const bounds = componentBounds(triangles.flatMap((triangle) => [triangle.min, triangle.max]), 'bounds');
    if (triangles.length <= leafSize) return { bounds, triangles };
    const centroidBounds = componentBounds(triangles.map((triangle) => triangle.centroid), 'bounds');
    const extents = AXES.map((axis) => centroidBounds.max[axis] - centroidBounds.min[axis]);
    const axisIndex = extents.indexOf(Math.max(...extents));
    const axis = AXES[axisIndex];
    if (extents[axisIndex] <= EPSILON) return { bounds, triangles };
    triangles.sort((left, right) =>
        left.centroid[axis] - right.centroid[axis] || left.primitiveIndex - right.primitiveIndex
    );
    const middle = Math.floor(triangles.length / 2);
    return {
        bounds,
        left: buildBvh(triangles.slice(0, middle), leafSize),
        right: buildBvh(triangles.slice(middle), leafSize)
    };
}

function intersectBvh(node, segment, minParameter, maximumParameter, counters) {
    if (!intersectAabbSegment(node.bounds, segment, maximumParameter)) return null;
    counters.bvhNodesVisited++;
    if (node.triangles) {
        let best = null;
        for (const triangle of node.triangles) {
            if (!intersectAabbSegment({ min: triangle.min, max: triangle.max }, segment, best?.parameter ?? maximumParameter)) continue;
            counters.trianglesTested++;
            const parameter = intersectTriangle(segment.from, segment.direction, triangle, minParameter, best?.parameter ?? maximumParameter);
            if (parameter === null) continue;
            if (!best || parameter < best.parameter) {
                best = {
                    parameter,
                    primitiveIndex: triangle.primitiveIndex,
                    point: interpolate3(segment.from, segment.to, parameter),
                    normal: triangle.normal,
                    barycentric: triangleBarycentric(segment.from, segment.direction, triangle, parameter)
                };
            }
        }
        return best;
    }
    const left = intersectBvh(node.left, segment, minParameter, maximumParameter, counters);
    const rightLimit = left ? Math.min(maximumParameter, left.parameter) : maximumParameter;
    const right = intersectBvh(node.right, segment, minParameter, rightLimit, counters);
    if (!left) return right;
    if (!right) return left;
    return left.parameter <= right.parameter ? left : right;
}

function intersectTriangle(origin, direction, triangle, minimumParameter, maximumParameter) {
    const edge1 = subtract3(triangle.b, triangle.a);
    const edge2 = subtract3(triangle.c, triangle.a);
    const p = cross3(direction, edge2);
    const determinant = dot3(edge1, p);
    if (Math.abs(determinant) <= EPSILON) return null;
    const inverse = 1 / determinant;
    const tVector = subtract3(origin, triangle.a);
    const u = dot3(tVector, p) * inverse;
    if (u < -EPSILON || u > 1 + EPSILON) return null;
    const q = cross3(tVector, edge1);
    const v = dot3(direction, q) * inverse;
    if (v < -EPSILON || u + v > 1 + EPSILON) return null;
    const parameter = dot3(edge2, q) * inverse;
    if (parameter <= minimumParameter || parameter > maximumParameter + EPSILON) return null;
    return parameter;
}

function triangleBarycentric(origin, direction, triangle, parameter) {
    const point = interpolate3(origin, add3(origin, direction), parameter);
    const v0 = subtract3(triangle.b, triangle.a);
    const v1 = subtract3(triangle.c, triangle.a);
    const v2 = subtract3(point, triangle.a);
    const d00 = dot3(v0, v0);
    const d01 = dot3(v0, v1);
    const d11 = dot3(v1, v1);
    const d20 = dot3(v2, v0);
    const d21 = dot3(v2, v1);
    const denominator = d00 * d11 - d01 * d01;
    if (Math.abs(denominator) <= EPSILON) return { u: 1, v: 0, w: 0 };
    const v = (d11 * d20 - d01 * d21) / denominator;
    const w = (d00 * d21 - d01 * d20) / denominator;
    return { u: 1 - v - w, v, w };
}

function intersectHeightfield(entry, segment, minimumParameter, maximumParameter, counters) {
    const range = intersectXZBounds(segment, entry.bounds, minimumParameter, maximumParameter);
    if (!range) return null;
    const span = range.max - range.min;
    if (span <= EPSILON) return null;
    const horizontalDistance = Math.hypot(segment.direction.x, segment.direction.z) * span;
    // Terrain dimensions cover the complete source heightmap, not this chunk.
    // Use the source raster's world-space sample spacing so chunk size does not
    // accidentally change the intersection sampling density.
    const cellSpacing = terrainCellSpacing(entry);
    const cell = Math.max(Math.min(cellSpacing.x, cellSpacing.z), 1e-6);
    const sampleCount = Math.min(4096, Math.max(8, Math.ceil(horizontalDistance / cell * 2)));
    let previousT = range.min;
    let previousValue = heightDifference(entry, segment, previousT, counters);
    for (let index = 1; index <= sampleCount; index++) {
        const currentT = range.min + span * index / sampleCount;
        const currentValue = heightDifference(entry, segment, currentT, counters);
        if (previousValue === null) {
            previousT = currentT;
            previousValue = currentValue;
            continue;
        }
        if (currentValue !== null && crossesSurface(previousValue, currentValue)) {
            const parameter = refineHeightfieldRoot(entry, segment, previousT, currentT, counters);
            if (parameter !== null && parameter > minimumParameter && parameter <= maximumParameter + EPSILON) {
                const point = interpolate3(segment.from, segment.to, parameter);
                return {
                    parameter,
                    primitiveIndex: null,
                    point: { x: point.x, y: entry.terrain.sampleHeight(point.x, point.z), z: point.z },
                    normal: heightfieldNormal(entry, point.x, point.z),
                    barycentric: null
                };
            }
        }
        previousT = currentT;
        previousValue = currentValue;
    }
    return null;
}

function heightDifference(entry, segment, parameter, counters) {
    const point = interpolate3(segment.from, segment.to, parameter);
    if (!insideHeightfieldBounds(entry.bounds, point.x, point.z)) return null;
    const height = entry.terrain.sampleHeight(point.x, point.z);
    counters.heightfieldSamples++;
    if (!Number.isFinite(height)) return null;
    return point.y - height;
}

function refineHeightfieldRoot(entry, segment, left, right, counters) {
    let leftValue = heightDifference(entry, segment, left, counters);
    let rightValue = heightDifference(entry, segment, right, counters);
    if (leftValue === null || rightValue === null) return null;
    if (Math.abs(leftValue) <= 1e-10) return left;
    if (Math.abs(rightValue) <= 1e-10) return right;
    if (!crossesSurface(leftValue, rightValue)) return null;
    for (let iteration = 0; iteration < 36; iteration++) {
        const middle = (left + right) / 2;
        const middleValue = heightDifference(entry, segment, middle, counters);
        if (middleValue === null) return null;
        if (Math.abs(middleValue) <= 1e-10) return middle;
        if ((leftValue < 0) === (middleValue < 0)) {
            left = middle;
            leftValue = middleValue;
        } else {
            right = middle;
            rightValue = middleValue;
        }
    }
    return (left + right) / 2;
}

function heightfieldNormal(entry, x, z) {
    const cellSpacing = terrainCellSpacing(entry);
    const epsilonX = Math.max(cellSpacing.x * 0.5, 1e-4);
    const epsilonZ = Math.max(cellSpacing.z * 0.5, 1e-4);
    const x0 = Math.max(entry.bounds.minX, x - epsilonX);
    const x1 = Math.min(entry.bounds.maxX, x + epsilonX);
    const z0 = Math.max(entry.bounds.minZ, z - epsilonZ);
    const z1 = Math.min(entry.bounds.maxZ, z + epsilonZ);
    const centerHeight = entry.terrain.sampleHeight(x, z);
    const hx0 = entry.terrain.sampleHeight(x0, z);
    const hx1 = entry.terrain.sampleHeight(x1, z);
    const hz0 = entry.terrain.sampleHeight(x, z0);
    const hz1 = entry.terrain.sampleHeight(x, z1);
    const gradientX = estimateHeightGradient(hx0, centerHeight, hx1, x0, x, x1);
    const gradientZ = estimateHeightGradient(hz0, centerHeight, hz1, z0, z, z1);
    return normalize3({ x: -gradientX, y: 1, z: -gradientZ });
}

function terrainCellSpacing(entry) {
    const terrain = entry.terrain;
    const worldWidth = terrainAxisSpan(terrain, 'x', entry.bounds.maxX - entry.bounds.minX);
    const worldDepth = terrainAxisSpan(terrain, 'z', entry.bounds.maxZ - entry.bounds.minZ);
    return {
        x: worldWidth / Math.max(1, terrain.width - 1),
        z: worldDepth / Math.max(1, terrain.height - 1)
    };
}

function terrainAxisSpan(terrain, axis, fallback) {
    const minimum = terrain.min?.[axis];
    const maximum = terrain.max?.[axis];
    if (Number.isFinite(minimum) && Number.isFinite(maximum) && maximum > minimum) {
        return maximum - minimum;
    }
    const explicitSize = axis === 'x' ? terrain.worldSizeX : terrain.worldSizeZ;
    if (Number.isFinite(explicitSize) && explicitSize > 0) return explicitSize;
    if (Number.isFinite(terrain.worldSize) && terrain.worldSize > 0) return terrain.worldSize;
    return fallback;
}

function estimateHeightGradient(low, center, high, lowCoordinate, centerCoordinate, highCoordinate) {
    if (Number.isFinite(low) && Number.isFinite(high) && highCoordinate - lowCoordinate > EPSILON) {
        return (high - low) / (highCoordinate - lowCoordinate);
    }
    if (Number.isFinite(center) && Number.isFinite(high) && highCoordinate - centerCoordinate > EPSILON) {
        return (high - center) / (highCoordinate - centerCoordinate);
    }
    if (Number.isFinite(center) && Number.isFinite(low) && centerCoordinate - lowCoordinate > EPSILON) {
        return (center - low) / (centerCoordinate - lowCoordinate);
    }
    return 0;
}

function intersectXZBounds(segment, bounds, minimum, maximum) {
    let lower = minimum;
    let upper = maximum;
    for (const axis of ['x', 'z']) {
        const min = axis === 'x' ? bounds.minX : bounds.minZ;
        const max = axis === 'x' ? bounds.maxX : bounds.maxZ;
        const origin = segment.from[axis];
        const direction = segment.direction[axis];
        if (Math.abs(direction) <= EPSILON) {
            if (origin < min || origin > max) return null;
            continue;
        }
        let first = (min - origin) / direction;
        let second = (max - origin) / direction;
        if (first > second) [first, second] = [second, first];
        lower = Math.max(lower, first);
        upper = Math.min(upper, second);
        if (lower > upper) return null;
    }
    return { min: lower, max: upper };
}

function intersectAabbSegment(bounds, segment, maximumParameter = 1) {
    let lower = 0;
    let upper = maximumParameter;
    for (const axis of AXES) {
        const origin = segment.from[axis];
        const direction = segment.direction[axis];
        if (Math.abs(direction) <= EPSILON) {
            if (origin < bounds.min[axis] - EPSILON || origin > bounds.max[axis] + EPSILON) return false;
            continue;
        }
        let first = (bounds.min[axis] - origin) / direction;
        let second = (bounds.max[axis] - origin) / direction;
        if (first > second) [first, second] = [second, first];
        lower = Math.max(lower, first);
        upper = Math.min(upper, second);
        if (lower > upper) return false;
    }
    return upper >= 0 && lower <= maximumParameter;
}

function makeHit(entry, candidate, segment, primitiveType) {

        objectId: entry.id,
        sliceId: entry.sliceId,

        parameter: candidate.parameter,
        point: { ...candidate.point },
        normal: { ...candidate.normal },
        barycentric: candidate.barycentric ? { ...candidate.barycentric } : null,
        segmentDistance: segment.length * candidate.parameter,
        material: { ...entry.material }
    };
}

function createSegment(from, to) {

}

function readVertex(positions, index) {
    const offset = index * 3;
    return { x: positions[offset], y: positions[offset + 1], z: positions[offset + 2] };
}

function componentBounds(points, mode) {
    const min = { x: Infinity, y: Infinity, z: Infinity };
    const max = { x: -Infinity, y: -Infinity, z: -Infinity };
    for (const point of points) {
        for (const axis of AXES) {
            min[axis] = Math.min(min[axis], point[axis]);
            max[axis] = Math.max(max[axis], point[axis]);
        }
    }
    return mode === 'min' ? min : mode === 'max' ? max : { min, max };
}

function isAllowedChunk(chunkKey, allowedChunks) {
    // Non-chunked static surfaces are slice-global and remain queryable while
    // the caller narrows streamed terrain to a set of resident chunk addresses.
    return !allowedChunks || chunkKey === null || allowedChunks.has(chunkKey);
}

function isBetterHit(candidate, current, objectId) {
    if (!current) return true;
    if (candidate.parameter < current.parameter - EPSILON) return true;
    if (Math.abs(candidate.parameter - current.parameter) <= EPSILON) {
        return `${objectId}` < `${current.objectId}`;
    }
    return false;
}

function crossesSurface(left, right) {
    return left === 0 || right === 0 || (left < 0) !== (right < 0);
}

function insideHeightfieldBounds(bounds, x, z) {
    return x >= bounds.minX - EPSILON && x <= bounds.maxX + EPSILON &&
        z >= bounds.minZ - EPSILON && z <= bounds.maxZ + EPSILON;
}

function countTriangles(meshes) {
    let count = 0;
    for (const entry of meshes.values()) count += entry.triangleCount;
    return count;
}

function describeEntry(entry) {
    return {
        id: entry.id,
        sliceId: entry.sliceId,

        triangleCount: entry.triangleCount ?? 0
    };
}

function subtract3(a, b) { return { x: a.x - b.x, y: a.y - b.y, z: a.z - b.z }; }
function add3(a, b) { return { x: a.x + b.x, y: a.y + b.y, z: a.z + b.z }; }
function scale3(value, scale) { return { x: value.x * scale, y: value.y * scale, z: value.z * scale }; }
function cross3(a, b) { return { x: a.y * b.z - a.z * b.y, y: a.z * b.x - a.x * b.z, z: a.x * b.y - a.y * b.x }; }
function dot3(a, b) { return a.x * b.x + a.y * b.y + a.z * b.z; }
function length3(value) { return Math.hypot(value.x, value.y, value.z); }
function normalize3(value) {
    const length = length3(value);
    if (!Number.isFinite(length) || length <= EPSILON) throw new RangeError('Cannot normalize a zero-length surface normal.');
    return scale3(value, 1 / length);
}
function interpolate3(a, b, t) {
    return { x: a.x + (b.x - a.x) * t, y: a.y + (b.y - a.y) * t, z: a.z + (b.z - a.z) * t };
}

/** Slice-native ray tracing: Hamiltonian integration plus authoritative surface events. */

import { ambientToSlice, assertVector4, dot4, sliceToAmbient } from './spatial-4d.js?v=somnium-20261108-phase16c-anomaly-optics-2';
import { createOpticalMetricField, spatialTransformMetricProvider } from './optical-metric.js?v=somnium-20261108-phase16c-anomaly-optics-2';
import { traceOpticalRay } from './optical-ray-integrator.js?v=somnium-20261108-phase16c-anomaly-optics-2';

const DEFAULT_SLICE_TOLERANCE = 1e-7;

function createSliceOpticalField(slice, spatialTransform, options = {}) {
    if (!slice?.id || !slice.frame) throw new TypeError('A registered WorldSlice is required for slice optics.');
    if (!spatialTransform || typeof spatialTransform.metricAtWorld !== 'function') {
        throw new TypeError('A slice SpatialTransform with metricAtWorld() is required.');
    }
    if (options.wCoupling !== undefined && options.wCoupling !== 0) {
        throw new RangeError('Slice-native optical fields must constrain the slice-normal direction (wCoupling = 0).');
    }
    return createOpticalMetricField({
        sliceFrame: slice.frame,
        sliceMetricAt: spatialTransformMetricProvider(spatialTransform),
        refractiveIndexAt: options.refractiveIndexAt ?? options.refractiveIndex ?? 1,
        wCoupling: 0,
        gradientStep: options.gradientStep,
        containsPosition(position) {
            const local = ambientToSlice(slice.frame, position).position;
            return typeof spatialTransform.containsMetricDomainPosition !== 'function' ||
                spatialTransform.containsMetricDomainPosition(local);
        }
    });
}

/**
 * Trace within one 3D slice and stop at the first registered triangle or
 * analytic heightfield hit. Every segment is queried in slice-local physical
 * coordinates; render-space mesh projections are never consulted.
 */
function traceOpticalRayInSlice(field, sceneRegistry, sliceId, initialRay, options = {}) {
    if (!field?.frame || typeof field.sample !== 'function') {
        throw new TypeError('A slice OpticalMetricField is required.');
    }
    if (!sceneRegistry || typeof sceneRegistry.intersectSegment !== 'function') {
        throw new TypeError('An OpticalSceneRegistry is required.');
    }
    if (typeof sliceId !== 'string' || !sliceId.trim()) throw new TypeError('A sliceId is required.');

    const tolerance = options.sliceTolerance ?? DEFAULT_SLICE_TOLERANCE;
    if (!Number.isFinite(tolerance) || tolerance <= 0) throw new RangeError('Slice tolerance must be positive and finite.');
    assertVector4(initialRay?.position);
    assertVector4(initialRay?.momentum);
    const initialLocal = ambientToSlice(field.frame, initialRay.position);
    if (Math.abs(initialLocal.normalOffset) > tolerance) {
        throw new RangeError('Slice-native optical ray must begin on its registered 3D slice.');
    }
    const momentumMagnitude = Math.hypot(
        initialRay.momentum.x, initialRay.momentum.y,
        initialRay.momentum.z, initialRay.momentum.w
    );
    if (Math.abs(dot4(initialRay.momentum, field.frame.normal)) > tolerance * Math.max(1, momentumMagnitude)) {
        throw new RangeError('Slice-native optical ray momentum must be tangent to its registered 3D slice.');
    }
    const callerEvent = options.segmentEvent;
    const integrationOptions = { ...options };
    delete integrationOptions.sliceTolerance;
    delete integrationOptions.allowedChunkAddresses;
    integrationOptions.segmentEvent = (segment) => {
        const start = ambientToSlice(field.frame, segment.startPosition);
        const end = ambientToSlice(field.frame, segment.endPosition);
        if (Math.abs(start.normalOffset) > tolerance || Math.abs(end.normalOffset) > tolerance) {
            throw new RangeError('Slice-native optical ray left its 3D slice.');
        }
        const hit = sceneRegistry.intersectSegment(sliceId, start.position, end.position, {
            allowedChunkAddresses: options.allowedChunkAddresses,
            minimumParameter: options.minimumHitParameter ?? 1e-7
        });
        const surfaceEvent = hit ? { ...hit, kind: 'surface-hit' } : null;
        const topologyEvent = typeof callerEvent === 'function' ? callerEvent(segment) : null;
        if (surfaceEvent && topologyEvent) {
            return surfaceEvent.parameter <= topologyEvent.parameter ? surfaceEvent : topologyEvent;
        }
        return surfaceEvent ?? topologyEvent;
    };

    const result = traceOpticalRay(field, initialRay, integrationOptions);
    if (result.event?.kind === 'surface-hit' && result.event.point) {
        const ambientPosition = sliceToAmbient(field.frame, result.event.point);
        result.finalState.position = ambientPosition;
        result.event.ambientPosition = { ...ambientPosition };
        const finalSample = result.samples[result.samples.length - 1];
        if (finalSample) finalSample.position = { ...ambientPosition };
        result.status = 'surface-hit';
    }
    result.sliceId = sliceId;
    return result;
}

function createSliceOpticalTracer({ slice, spatialTransform, sceneRegistry, debugRenderer = null, metricOptions = {} }) {
    const field = createSliceOpticalField(slice, spatialTransform, metricOptions);
    return Object.freeze({
        sliceId: slice.id,
        field,
        trace(initialRay, options = {}) {
            const result = traceOpticalRayInSlice(field, sceneRegistry, slice.id, initialRay, options);
            if (options.drawDebug && debugRenderer) {
                debugRenderer.renderTrace(result, { slice, spatialTransform, ...options.debugStyle });
            }
            return result;
        }
    });
}

export { createSliceOpticalField, createSliceOpticalTracer, traceOpticalRayInSlice };

class PoincareGeometry extends GeometryModel {

    constructor(curvature = -1) {

        super("poincare");

        if (
            !Number.isFinite(curvature) ||
            curvature >= 0
        ) {
            throw new Error(
                "Poincare curvature must be negative."
            );
        }

        this.curvature = curvature;

        this.k = -curvature;

        this.radius = 1 / Math.sqrt(this.k);

    }

    isInside(position) {

        const radiusSquared =
            position.x * position.x +
            position.y * position.y +
            position.z * position.z;

        return radiusSquared <
            this.radius * this.radius;

    }

    metricAt(position) {

        if (!this.isInside(position)) {

            throw new RangeError(
                "Position lies outside the Poincare ball."
            );

        }

        const r2 =
            position.x * position.x +
            position.y * position.y +
            position.z * position.z;

        const denominator =
            1 - this.k * r2;

        const factor =
            4 / (denominator * denominator);

        return new MetricTensor([

            factor, 0, 0,

            0, factor, 0,

            0, 0, factor

        ]);

    }

    distance(a, b) {

        if (
            !this.isInside(a) ||
            !this.isInside(b)
        ) {

            throw new RangeError(
                "Both positions must lie inside the Poincare ball."
            );

        }

        const dot = 
            a.x * b.x +
            a.y * b.y +
            a.z * b.z;

        const normA =
            a.x * a.x +
            a.y * a.y +
            a.z * a.z;

        const normB =
            b.x * b.x +
            b.y * b.y +
            b.z * b.z;

        const differenceSquared =

            (a.x - b.x) ** 2 +

            (a.y - b.y) ** 2 +

            (a.z - b.z) ** 2;

        const numerator =
            this.k * differenceSquared;

        const denominator =
            (1 - this.k * normA) *
            (1 - this.k * normB);

        const argument =
            1 + 2 * numerator / denominator;

        return Math.acosh(
            Math.max(1, argument)
        ) / Math.sqrt(this.k);

    }

}

/**
 * PoincareMetric
 *
 * Three-dimensional hyperbolic geometry
 * represented using Poincare ball coordinates.
 *
 * Supports:
 * - Metric tensor evaluation.
 * - Exact hyperbolic distances.
 * - Hyperboloid conversion.
 * - Analytic geodesic interpolation.
 * - Geodesic path generation.
 */

import { MetricTensor } from "../metric-tensor.js";

class PoincareMetric {

    constructor(curvature = -1) {

        if (
            !Number.isFinite(curvature) ||
            curvature >= 0
        ) {
            throw new Error(
                "Poincare curvature must be negative."
            );
        }

        this.curvature = curvature;

        this.k = -curvature;

        this.radius = 1 / Math.sqrt(this.k);
    }

    // -----------------------------------------
    // POSITION VALIDATION
    // -----------------------------------------

    validatePosition(position) {

        if (
            !position ||
            !Number.isFinite(position.x) ||
            !Number.isFinite(position.y) ||
            !Number.isFinite(position.z)
        ) {
            throw new TypeError(
                "Invalid Poincare position."
            );
        }

        if (!this.isInside(position)) {
            throw new RangeError(
                "Position lies outside the open Poincare ball."
            );
        }
    }

    isInside(position) {

        if (
            !position ||
            !Number.isFinite(position.x) ||
            !Number.isFinite(position.y) ||
            !Number.isFinite(position.z)
        ) {
            return false;
        }

        const radiusSquared =
            position.x ** 2 +
            position.y ** 2 +
            position.z ** 2;

        return radiusSquared <
            this.radius ** 2;
    }

    // -----------------------------------------
    // METRIC TENSOR
    // -----------------------------------------

    metricAt(position) {

        this.validatePosition(position);

        const radiusSquared =
            position.x ** 2 +
            position.y ** 2 +
            position.z ** 2;

        const denominator =
            this.radius ** 2 - radiusSquared;

        const factor =
            4 * this.radius ** 4 /
            (denominator ** 2);

        return new MetricTensor([
            factor, 0, 0,
            0, factor, 0,
            0, 0, factor
        ]);
    }

    // -----------------------------------------
    // HYPERBOLIC DISTANCE
    // -----------------------------------------

    distance(a, b) {

        this.validatePosition(a);
        this.validatePosition(b);

        const dx = a.x - b.x;
        const dy = a.y - b.y;
        const dz = a.z - b.z;

        const differenceSquared =
            dx * dx + dy * dy + dz * dz;

        const normA =
            a.x ** 2 +
            a.y ** 2 +
            a.z ** 2;

        const normB =
            b.x ** 2 +
            b.y ** 2 +
            b.z ** 2;

        const numerator =
            this.k * differenceSquared;

        const denominator =
            (1 - this.k * normA) *
            (1 - this.k * normB);

        const argument =
            1 + 2 * numerator / denominator;

        return Math.acosh(
            Math.max(1, argument)
        ) / Math.sqrt(this.k);
    }

    // -----------------------------------------
    // POINCARE TO HYPERBOLOID
    // -----------------------------------------

    toHyperboloid(position) {

        this.validatePosition(position);

        const R = this.radius;

        const radiusSquared =
            position.x ** 2 +
            position.y ** 2 +
            position.z ** 2;

        const denominator =
            R * R - radiusSquared;

        const factor =
            2 * R * R / denominator;

        return {
            w:
                R * (R * R + radiusSquared) /
                denominator,

            x: factor * position.x,

            y: factor * position.y,

            z: factor * position.z
        };
    }

    // -----------------------------------------
    // HYPERBOLOID TO POINCARE
    // -----------------------------------------

    fromHyperboloid(position) {

        const R = this.radius;

        if (
            !position ||
            !Number.isFinite(position.w) ||
            !Number.isFinite(position.x) ||
            !Number.isFinite(position.y) ||
            !Number.isFinite(position.z) ||
            position.w <= 0
        ) {
            throw new TypeError(
                "Invalid hyperboloid position."
            );
        }

        const denominator =
            position.w + R;

        const result = {
            x: R * position.x / denominator,

            y: R * position.y / denominator,

            z: R * position.z / denominator
        };

        this.validatePosition(result);

        return result;
    }

    // -----------------------------------------
    // MINKOWSKI INNER PRODUCT
    // -----------------------------------------

    minkowskiProduct(a, b) {

        return (
            -a.w * b.w +
            a.x * b.x +
            a.y * b.y +
            a.z * b.z
        );
    }

    // -----------------------------------------
    // GEODESIC INTERPOLATION
    // -----------------------------------------

    interpolateGeodesic(a, b, t) {

        this.validatePosition(a);
        this.validatePosition(b);

        if (
            !Number.isFinite(t) ||
            t < 0 ||
            t > 1
        ) {
            throw new RangeError(
                "Geodesic parameter must be between 0 and 1."
            );
        }

        if (t === 0) return { ...a };
        if (t === 1) return { ...b };

        const A = this.toHyperboloid(a);
        const B = this.toHyperboloid(b);

        const R = this.radius;

        const coshDistance =
            -this.minkowskiProduct(A, B) /
            (R * R);

        const distance = Math.acosh(
            Math.max(1, coshDistance)
        );

        // Handle nearly identical points.

        if (distance < 1e-10) {

            return {
                x: a.x + (b.x - a.x) * t,
                y: a.y + (b.y - a.y) * t,
                z: a.z + (b.z - a.z) * t
            };
        }

        const denominator =
            Math.sinh(distance);

        const weightA =
            Math.sinh((1 - t) * distance) /
            denominator;

        const weightB =
            Math.sinh(t * distance) /
            denominator;

        const point = {
            w:
                weightA * A.w +
                weightB * B.w,

            x:
                weightA * A.x +
                weightB * B.x,

            y:
                weightA * A.y +
                weightB * B.y,

            z:
                weightA * A.z +
                weightB * B.z
        };

        return this.fromHyperboloid(point);
    }

    // -----------------------------------------
    // GEODESIC PATH
    // -----------------------------------------

    geodesic(a, b, segments = 32) {

        this.validatePosition(a);
        this.validatePosition(b);

        if (
            !Number.isInteger(segments) ||
            segments < 1 ||
            segments > 10000
        ) {
            throw new RangeError(
                "Invalid geodesic segment count."
            );
        }

        const points = [];

        for (let i = 0; i <= segments; i++) {

            const t = i / segments;

            points.push(
                this.interpolateGeodesic(
                    a,
                    b,
                    t
                )
            );
        }

        return {
            points,

            length: this.distance(a, b),

            segments,

            geometry: "poincare-3d"
        };
    }

    // -----------------------------------------
    // INFORMATION
    // -----------------------------------------

    getInfo() {

        return {
            geometry: "poincare-3d",

            curvature: this.curvature,

            radius: this.radius,

            dimensions: 3,

            boundaryReachable: false
        };
    }
}

export { PoincareMetric };

/**
 * PoincareProjection
 *
 * Maps Somnium's finite 3D world into an open
 * three-dimensional Poincare ball.
 *
 * All three spatial dimensions participate
 * in the transformation.
 *
 * The world boundary approaches the ball
 * boundary asymptotically.
 */

class PoincareProjection {

    constructor(options = {}) {

        this.bounds = {
            minX: options.minX ?? -500000,
            maxX: options.maxX ?? 500000,

            minY: options.minY ?? -2000,
            maxY: options.maxY ?? 1000,

            minZ: options.minZ ?? -500000,
            maxZ: options.maxZ ?? 500000
        };

        this.curvature = options.curvature ?? -1;

        if (
            !Number.isFinite(this.curvature) ||
            this.curvature >= 0
        ) {
            throw new Error(
                "Poincare curvature must be negative."
            );
        }

        this.k = -this.curvature;

        this.radius = 1 / Math.sqrt(this.k);

	this.characteristicLength =
options.characteristicLength ?? 100000;

        this.boundaryEpsilon =
            options.boundaryEpsilon ?? 1e-9;

        if (
            !Number.isFinite(this.characteristicLength) ||
            this.characteristicLength <= 0
        ) {
            throw new Error(
                "Invalid characteristic length."
            );
        }

        if (
            !Number.isFinite(this.boundaryEpsilon) ||
            this.boundaryEpsilon <= 0 ||
            this.boundaryEpsilon >= 0.01
        ) {
            throw new Error(
                "Invalid boundary epsilon."
            );
        }

        this.center = {
            x: (this.bounds.minX + this.bounds.maxX) / 2,
            y: (this.bounds.minY + this.bounds.maxY) / 2,
            z: (this.bounds.minZ + this.bounds.maxZ) / 2
        };

        this.validateBounds();
    }

    // -----------------------------------------
    // VALIDATION
    // -----------------------------------------

    validateBounds() {

        const values = Object.values(this.bounds);

        if (
            !values.every(Number.isFinite) ||
            this.bounds.minX >= this.bounds.maxX ||
            this.bounds.minY >= this.bounds.maxY ||
            this.bounds.minZ >= this.bounds.maxZ
        ) {
            throw new Error(
                "Invalid three-dimensional world bounds."
            );
        }
    }

    validatePosition(position) {

        if (
            !position ||
            !Number.isFinite(position.x) ||
            !Number.isFinite(position.y) ||
            !Number.isFinite(position.z)
        ) {
            throw new TypeError(
                "Invalid three-dimensional position."
            );
        }
    }

    // -----------------------------------------
    // WORLD BOUNDARY DISTANCE
    // -----------------------------------------

    getBoundaryDistance(direction) {

        const b = this.bounds;

        const distances = [];

        if (direction.x > 0) {
            distances.push(
                (b.maxX - this.center.x) / direction.x
            );
        } else if (direction.x < 0) {
            distances.push(
                (b.minX - this.center.x) / direction.x
            );
        }

        if (direction.y > 0) {
            distances.push(
                (b.maxY - this.center.y) / direction.y
            );
        } else if (direction.y < 0) {
            distances.push(
                (b.minY - this.center.y) / direction.y
            );
        }

        if (direction.z > 0) {
            distances.push(
                (b.maxZ - this.center.z) / direction.z
            );
        } else if (direction.z < 0) {
            distances.push(
                (b.minZ - this.center.z) / direction.z
            );
        }

        return Math.min(...distances);
    }

    // -----------------------------------------
    // WORLD TO POINCARE BALL
    // -----------------------------------------

    worldToBall(position) {

        this.validatePosition(position);

        const dx = position.x - this.center.x;
        const dy = position.y - this.center.y;
        const dz = position.z - this.center.z;

        const distance = Math.hypot(dx, dy, dz);

        if (distance === 0) {
            return { x: 0, y: 0, z: 0 };
        }

        const direction = {
            x: dx / distance,
            y: dy / distance,
            z: dz / distance
        };

        const boundaryDistance =
            this.getBoundaryDistance(direction);

        const normalizedRadius =
            distance / boundaryDistance;

        if (
            normalizedRadius < 0 ||
            normalizedRadius > 1 + 1e-10
        ) {
            throw new RangeError(
                "Position lies outside world bounds."
            );
        }

        // Clamp exact boundary positions slightly inward.
        // The mathematical boundary remains unreachable.

        const u = Math.min(
            normalizedRadius,
            1 - this.boundaryEpsilon
        );

        // Compactify the finite world.

        const hyperbolicDistance =
            this.characteristicLength *
            u / (1 - u);

        // Hyperbolic radial projection.

const projectedRadius = Math.min(
    this.radius * Math.tanh(
        hyperbolicDistance / (2 * this.radius)
    ),
    this.radius * (1 - this.boundaryEpsilon)
);
	
        return {
            x: direction.x * projectedRadius,
            y: direction.y * projectedRadius,
            z: direction.z * projectedRadius
        };
    }

    // -----------------------------------------
    // POINCARE BALL TO WORLD
    // -----------------------------------------

    ballToWorld(position) {

        this.validatePosition(position);

        const radius = Math.hypot(
            position.x,
            position.y,
            position.z
        );

        if (radius >= this.radius) {
            throw new RangeError(
                "Position is outside the open Poincare ball."
            );
        }

        if (radius === 0) {
            return { ...this.center };
        }

        const direction = {
            x: position.x / radius,
            y: position.y / radius,
            z: position.z / radius
        };

        const ratio = radius / this.radius;

        const hyperbolicDistance =
            2 * this.radius * Math.atanh(ratio);

        const u =
            hyperbolicDistance /
            (
                this.characteristicLength +
                hyperbolicDistance
            );

        const boundaryDistance =
            this.getBoundaryDistance(direction);

        const worldDistance =
            u * boundaryDistance;

        return {
            x: this.center.x +
                direction.x * worldDistance,

            y: this.center.y +
                direction.y * worldDistance,

            z: this.center.z +
                direction.z * worldDistance
        };
    }

    // -----------------------------------------
    // PROJECTION INFORMATION
    // -----------------------------------------

    getInfo() {

        return {
            geometry: "poincare-3d",

            dimensions: 3,

            curvature: this.curvature,

            radius: this.radius,

            center: { ...this.center },

            bounds: { ...this.bounds },

            characteristicLength:
                this.characteristicLength,

            boundaryReachable: false
        };
    }
}

export { PoincareProjection };

/** Bounded rendering exaggeration; never modifies the authoritative Poincare metric. */

const DEFAULT_GAMMA = 0.5;
const DEFAULT_MAX_SCALE = 4;
const DEFAULT_BOUNDARY_EPSILON = 1e-4;

function calculatePoincareMetricFactor(radius, ballRadius) {
    validateRadii(radius, ballRadius);
    if (radius >= ballRadius) {
        throw new RangeError('Poincare metric is undefined at or beyond the ball boundary.');
    }
    return 2 * ballRadius ** 2 / (ballRadius ** 2 - radius ** 2);
}

// This bounded rendering exaggeration is deliberately separate from the true metric.
function calculatePoincareVisualScale(radius, ballRadius, options = {}) {
    validateRadii(radius, ballRadius);
    const gamma = options.gamma ?? DEFAULT_GAMMA;
    const maxScale = options.maxScale ?? DEFAULT_MAX_SCALE;
    const boundaryEpsilon = options.boundaryEpsilon ?? DEFAULT_BOUNDARY_EPSILON;
    validateOptions(gamma, maxScale, boundaryEpsilon);

    const safeRadius = Math.min(radius, ballRadius * (1 - boundaryEpsilon));
    const normalizedRadius = safeRadius / ballRadius;
    const rawScale = (1 - normalizedRadius ** 2) ** -gamma;
    return Math.min(maxScale, Math.max(1, rawScale));
}

function validateRadii(radius, ballRadius) {
    if (!Number.isFinite(radius) || radius < 0) {
        throw new RangeError('Poincare radius must be finite and non-negative.');
    }
    if (!Number.isFinite(ballRadius) || ballRadius <= 0) {
        throw new RangeError('Poincare ball radius must be finite and positive.');
    }
}

function validateOptions(gamma, maxScale, boundaryEpsilon) {
    if (!Number.isFinite(gamma) || gamma < 0) {
        throw new RangeError('Visual expansion gamma must be finite and non-negative.');
    }
    if (!Number.isFinite(maxScale) || maxScale < 1) {
        throw new RangeError('Maximum visual scale must be finite and at least one.');
    }
    if (
        !Number.isFinite(boundaryEpsilon) ||
        boundaryEpsilon <= 0 ||
        boundaryEpsilon >= 1
    ) {
        throw new RangeError('Boundary epsilon must be between zero and one.');
    }
}

    calculatePoincareMetricFactor,
    calculatePoincareVisualScale
};

        target.activeCamera = this.camera;
        target.renderList = [];
        target.refreshRate = 2;
        target.gammaSpace = true;
        target.clearColor = new BABYLON.Color4(0.025, 0.04, 0.09, 1);
        return target;

        const surface = BABYLON.MeshBuilder.CreatePlane(
            'geometryPortalSurface',
            { width, height, sideOrientation: BABYLON.Mesh.DOUBLESIDE },
            this.scene
        );
        surface.position.set(position.x, position.y, position.z);
        return surface;
    }

        material.diffuseTexture = this.texture;
        material.emissiveTexture = this.texture;

        const length = Math.hypot(direction.x, direction.y, direction.z) || 1;
        const normal = { x: direction.x / length, y: direction.y / length, z: direction.z / length };
        const pitch = -Math.asin(Math.max(-1, Math.min(1, normal.y)));
        const yaw = Math.atan2(normal.x, normal.z);

                position.y - normal.y * 0.035,
                position.z - normal.z * 0.035

        }
    }

    }

    setRenderList(meshes) {
        const renderList = meshes.filter(isRenderableMesh);
        if (sameMeshes(renderList, this.ownedMeshes)) return;
        this.ownedMeshes = renderList;
        this.texture.renderList = renderList;
    }

    setVisible(visible) {
        this.visible = Boolean(visible);
        this.surface.isVisible = this.visible;
        const targets = this.scene.customRenderTargets;
        const index = targets.indexOf(this.texture);
        if (this.visible && index < 0) targets.push(this.texture);
        if (!this.visible && index >= 0) targets.splice(index, 1);
    }

    getInfo() {
        return {
            visible: this.visible,
            renderOnly: true,
            renderListSize: this.ownedMeshes.length,
            resolution: this.texture.getSize()
        };
    }

    dispose() {
        this.setVisible(false);
        this.rim?.dispose(false, false);
        this.surface.dispose(false, false);
        this.material.dispose(false, true);
        this.texture.dispose();
        this.camera.dispose();
        this.ownedMeshes = [];
    }
}

function isRenderableMesh(mesh) {
    return Boolean(mesh && !mesh.isDisposed());
}

function capturePose(surface, rim) {
    return {
        surfacePosition: cloneVector(surface.position),
        surfaceRotation: cloneVector(surface.rotation),
        rimPosition: rim ? cloneVector(rim.position) : null,
        rimRotation: rim ? cloneVector(rim.rotation) : null
    };
}

function restorePose(renderer, pose) {
    renderer.surface.position.set(pose.surfacePosition.x, pose.surfacePosition.y, pose.surfacePosition.z);
    renderer.surface.rotation.set(pose.surfaceRotation.x, pose.surfaceRotation.y, pose.surfaceRotation.z);
    if (renderer.rim && pose.rimPosition) {
        renderer.rim.position.set(pose.rimPosition.x, pose.rimPosition.y, pose.rimPosition.z);
        renderer.rim.rotation.set(pose.rimRotation.x, pose.rimRotation.y, pose.rimRotation.z);
    }
}

function cloneVector(value) {
    return { x: value.x, y: value.y, z: value.z };
}

function sameMeshes(left, right) {
    return left.length === right.length && left.every(function sameMesh(mesh, index) {
        return mesh === right[index];
    });
}

        this.camera = requireObject(options.cameraController, 'cameraController');
        this.topology = requireObject(options.topology, 'topology');
        const suppliedAnomaly = requireObject(options.anomaly ?? options.corridor, 'anomaly');
        this.corridor = typeof this.topology.getCorridor === 'function'
            ? this.topology.getCorridor(suppliedAnomaly.id)
            : suppliedAnomaly;
        if (this.corridor.type !== SLICE_TRANSITION_ANOMALY) {
            throw new TypeError('SliceTransitionTraversalController requires a slice-transition anomaly.');
        }
        this.renderers = options.corridorRenderers ?? new Map();
        this.chunkGraph = options.chunkGraph ?? null;
        this.corridorChunk = options.corridorChunk ?? this.chunkGraph?.getCorridorChunk(this.corridor.id) ?? null;
        this.worlds = new Map([
            [options.source.slice.id, options.source],
            [options.destination.slice.id, options.destination]
        ]);
        this.currentSliceId = options.initialSliceId ?? options.source.slice.id;
        this.onSliceChanged = options.onSliceChanged ?? noop;
        this.getInfluenceModifiers = options.getInfluenceModifiers ?? null;
        if (this.getInfluenceModifiers !== null && typeof this.getInfluenceModifiers !== 'function') {
            throw new TypeError('getInfluenceModifiers must be a function when provided.');
        }
        this.options = {
            triggerDistance: this.corridor.entryRadius ?? 3.4,
            headingThreshold: 0.2,
            speed: 8,
            sprintSpeed: 12,
            entitySpeed: 8,
            ...options.traversal
        };
        this.validateOptions();
        this.active = false;
        this.disposed = false;
        this.state = 'idle';
        this.progress = 0;
        this.direction = 0;
        this.startWorld = null;
        this.target = null;
        this.spatialPosition4D = null;
        this.entityTransits = new Map();
        this.entryYaw = 0;
        this.entryPosition = null;
        this.entryLock = null;
        this.lastEntryDiagnostic = { reason: 'NOT_CHECKED', anomalyId: this.corridor.id };
    }

    update(deltaSeconds) {
        if (!this.active || this.disposed) return;
        const delta = clamp(Number(deltaSeconds) || 0, 0, 0.1);
        if (this.state === 'idle') {
            if (!this.tryBeginTransit()) this.advanceAmbientInfluenceMovement(delta);
        } else if (this.state === 'transit') this.advanceTransit(delta);
        this.advanceOtherEntities(delta);
    }

    setActive(active) {
        this.active = Boolean(active) && !this.disposed;
    }

    ownsSlice(sliceId) {
        return this.worlds.has(sliceId);
    }

    get corridorLength() {
        if (!this.startWorld || !this.target) return 0;
        return Math.abs(this.getPlaneCoordinate(this.target.slice.frame.origin, this.startWorld.slice.frame.normal) -
            this.getPlaneCoordinate(this.startWorld.slice.frame.origin, this.startWorld.slice.frame.normal));
    }

    getMovementInput() {
        return this.movement.getMovementInput?.() ?? {
            forward: 0, strafe: 0, vertical: 0, crossSectionW: 0, turn: 0, sprint: false
        };
    }

    tryBeginTransit() {
        const entry = this.findEntry();
        return entry ? this.beginTransit(entry) : false;
    }

    findEntry() {
        const input = this.getMovementInput();
        if (input.forward <= 0.05) {
            this.lastEntryDiagnostic = { reason: 'NO_FORWARD_INPUT', anomalyId: this.corridor.id };
            return null;
        }
        const position = this.player.worldPosition;
        if (this.entryLock?.sliceId === this.currentSliceId) {
            const distanceFromExit = Math.hypot(
                position.x - this.entryLock.position.x,
                position.z - this.entryLock.position.z
            );
            if (distanceFromExit < 1.25) {
                this.lastEntryDiagnostic = { reason: 'EXIT_LOCKED', anomalyId: this.corridor.id, distance: distanceFromExit };
                return null;
            }
            this.entryLock = null;
        }
        const yaw = this.player.yaw;
        const forwardX = Math.sin(yaw);
        const forwardZ = Math.cos(yaw);
        const neighbors = this.topology.getTraversableNeighbors(this.currentSliceId)
            .filter((neighbor) => neighbor.corridorId === this.corridor.id);
        if (!neighbors.length) {
            this.lastEntryDiagnostic = { reason: 'NO_TOPOLOGY_CONNECTION', anomalyId: this.corridor.id };
            return null;
        }
        let diagnostic = { reason: 'TOO_FAR', anomalyId: this.corridor.id, distance: Infinity, entryRadius: this.options.triggerDistance };

        for (const neighbor of neighbors) {
            const fromSource = this.corridor.source.sliceId === this.currentSliceId;
            const endpoint = fromSource ? this.corridor.source : this.corridor.destination;
            const dx = endpoint.position.x - position.x;
            const dz = endpoint.position.z - position.z;
            const distance = Math.hypot(dx, dz);
            diagnostic = { ...diagnostic, distance, entryRadius: this.options.triggerDistance };
            if (distance > this.options.triggerDistance) continue;
            const heading = distance < 1e-6 ? 1 : (forwardX * dx + forwardZ * dz) / distance;
            if (heading < this.options.headingThreshold) {
                diagnostic = { ...diagnostic, reason: 'WRONG_HEADING', heading, headingThreshold: this.options.headingThreshold };
                continue;
            }
            const currentWorld = this.worlds.get(this.currentSliceId);
            const ambientPosition = currentWorld ? this.getAmbientPlayerPosition() : null;
            if (!ambientPosition || !this.isInsideRegion(ambientPosition)) {
                diagnostic = { ...diagnostic, reason: 'OUTSIDE_ANOMALY_REGION', heading };
                continue;
            }
            if (!this.hasChunkConnection(this.currentSliceId, position)) {
                diagnostic = { ...diagnostic, reason: 'NO_CHUNK_CONNECTION', heading };
                continue;
            }
            this.lastEntryDiagnostic = { reason: 'READY', anomalyId: this.corridor.id, distance, heading };
            return { fromSource, destinationSliceId: neighbor.sliceId };
        }
        this.lastEntryDiagnostic = diagnostic;
        return null;
    }

    getEntryDiagnostic() {
        return { ...this.lastEntryDiagnostic };
    }

    getAmbientPlayerPosition() {
        const world = this.worlds.get(this.currentSliceId);
        if (!world) return null;
        const base = sliceToAmbient(world.slice.frame, this.player.worldPosition);
        const prior = this.player.spatialPosition4D;
        const normalOffset = prior && this.player.sliceId === this.currentSliceId
            ? ambientToSlice(world.slice.frame, prior).normalOffset : 0;
        return add4(base, scale4(world.slice.frame.normal, normalOffset));
    }

    getAnomalyInfluence(position4D) {
        const modifiers = this.getInfluenceModifiers?.({
            player: this.player, anomaly: this.corridor, position4D
        }) ?? {};
        return evaluateAnomalyInfluence(this.corridor, position4D, modifiers);
    }

    getInfluenceMovementInfo(position4D, direction, sliceId = this.currentSliceId) {
        if (!Number.isFinite(direction) || Math.abs(direction) < 1e-6 ||
            !this.active || this.disposed || this.state !== 'idle' || !this.corridor.traversable) return null;
        const neighbor = this.topology.getTraversableNeighbors(sliceId)
            .find((entry) => entry.corridorId === this.corridor.id);
        if (!neighbor) return null;
        const sourceWorld = this.worlds.get(sliceId);
        const targetWorld = this.worlds.get(neighbor.sliceId);
        if (!sourceWorld || !targetWorld) return null;
        const normal = sourceWorld.slice.frame.normal;
        const currentPlane = dot4(position4D, normal);
        const targetPlane = dot4(targetWorld.slice.frame.origin, normal);
        const remaining = targetPlane - currentPlane;
        const targetDirection = Math.sign(remaining);
        const influence = this.getAnomalyInfluence(position4D);
        if (influence.freedom <= 0 || influence.wMovementScale <= 0) return null;
        return {
            anomalyId: this.corridor.id,
            sourceWorld,
            targetWorld,
            sourceSliceId: sliceId,
            targetSliceId: neighbor.sliceId,
            direction: targetDirection,
            inputDirection: Math.sign(direction),
            movingTowardTarget: targetDirection !== 0 && Math.sign(direction) === targetDirection,
            remainingWDistance: Math.abs(remaining),
            freedom: influence.freedom,
            wReach: influence.wReach,
            movementScale: influence.wMovementScale,
            influenceDistance: influence.distance
        };
    }

    advanceAmbientInfluenceMovement(deltaSeconds) {
        const input = this.getMovementInput();
        if (!input.crossSectionW) return false;
        const position4D = this.getAmbientPlayerPosition();
        if (!position4D) return false;
        const info = this.getInfluenceMovementInfo(position4D, input.crossSectionW);
        if (!info) return false;
        const speed = input.sprint ? this.options.sprintSpeed : this.options.speed;
        const frame = info.sourceWorld.slice.frame;
        const amount = Math.sign(input.crossSectionW) * speed *
            info.movementScale * Math.min(Math.max(deltaSeconds, 0), 0.1);
        if (Math.abs(amount) <= 1e-10) return false;
        let candidate = add4(position4D, scale4(frame.normal, amount));
        const currentPlane = dot4(position4D, frame.normal);
        const candidatePlane = dot4(candidate, frame.normal);
        const targetPlane = dot4(info.targetWorld.slice.frame.origin, frame.normal);
        const crossed = info.direction > 0
            ? candidatePlane >= targetPlane - 1e-8
            : candidatePlane <= targetPlane + 1e-8;

        if (crossed) {
            candidate = add4(candidate, scale4(frame.normal, targetPlane - candidatePlane));
            const sourceLocal = ambientToSlice(info.sourceWorld.slice.frame, candidate).position;
            const targetLocal = ambientToSlice(info.targetWorld.slice.frame, candidate).position;
            const sourceAddress = this.chunkGraph?.getAddress(info.sourceSliceId, sourceLocal);
            const targetAddress = this.chunkGraph?.getAddress(info.targetSliceId, targetLocal);
            const reachability = sourceAddress && targetAddress
                ? this.chunkGraph.getReachability(sourceAddress, targetAddress, {
                    position4D: candidate, corridorId: this.corridor.id
                })
                : { reachable: true };
            if (!reachability.reachable) {
                candidate = add4(candidate, scale4(frame.normal, -info.direction * 1e-7));
            } else {
                this.startWorld = info.sourceWorld;
                this.target = info.targetWorld;
                this.direction = info.direction;
                this.spatialPosition4D = candidate;
                this.completeTransit(info.targetWorld, candidate);
                return true;
            }
        }

        this.player.spatialPosition4D = candidate;
        const local = ambientToSlice(info.sourceWorld.slice.frame, candidate);
        if (Math.abs(local.normalOffset) <= 1e-7) this.player.spatialPosition4D = null;
        return true;
    }

    beginTransit(entry) {
        this.startWorld = this.worlds.get(this.currentSliceId);
        this.target = this.worlds.get(entry.destinationSliceId);
        if (!this.startWorld || !this.target) {
            throw new Error(`Missing slice-transition world for anomaly ${this.corridor.id}.`);
        }
        this.direction = entry.fromSource ? 1 : -1;
        this.entryYaw = this.player.yaw;
        this.entryPosition = { ...this.player.worldPosition };
        this.spatialPosition4D = this.getAmbientPlayerPosition();
        if (!this.isInsideRegion(this.spatialPosition4D)) {
            throw new RangeError(`Player entered outside anomaly bounds: ${this.corridor.id}.`);
        }
        this.progress = this.calculateProgress(this.spatialPosition4D);
        this.state = 'transit';
        this.movement.setMovementSuspended(true);
        this.player.setSpatialContext?.(
            this.startWorld.transform,
            this.startWorld.slice.id,
            this.startWorld.layerMask
        );
        this.camera.clearRenderAnchor?.();
        this.camera.setSpatialContext(this.startWorld.transform, this.startWorld.layerMask);
        this.corridorChunk?.enter(this.player, {
            direction: this.direction,
            progress: clamp(this.progress, 0, 1),
            position4D: this.spatialPosition4D
        });
        this.onSliceChanged(this.currentSliceId, {
            transitioning: true,
            anomalyId: this.corridor.id,
            corridorId: this.corridor.id,
            type: this.corridor.type,
            renderSliceId: this.currentSliceId,
            direction: this.direction,
            position4D: { ...this.spatialPosition4D }
        });
        return true;
    }

    hasChunkConnection(sliceId, position) {
        if (!this.chunkGraph) return true;
        const address = this.chunkGraph.getAddress(sliceId, position);
        for (let offsetX = -1; offsetX <= 1; offsetX++) {
            for (let offsetZ = -1; offsetZ <= 1; offsetZ++) {
                const nearby = {
                    sliceId,
                    chunkX: address.chunkX + offsetX,
                    chunkZ: address.chunkZ + offsetZ
                };
                const connected = this.chunkGraph.getNeighbors(nearby).some((node) =>
                    node.kind === 'corridor' && node.id === this.corridor.id
                );
                if (connected) return true;
            }
        }
        return false;
    }

    advanceTransit(deltaSeconds) {
        const input = this.getMovementInput();
        const speed = input.sprint ? this.options.sprintSpeed : this.options.speed;
        if (Number.isFinite(input.turn) && input.turn !== 0) {
            this.player.yaw += input.turn * (this.movement.turnSpeed ?? 2.2) * deltaSeconds;
        }
        const yaw = this.player.yaw;
        const forward = clamp(Number(input.forward) || 0, -1, 1);
        const strafe = clamp(Number(input.strafe) || 0, -1, 1);
        const vertical = clamp(Number(input.vertical) || 0, -1, 1);
        const crossSectionW = clamp(Number(input.crossSectionW) || 0, -1, 1);
        const horizontalLength = Math.hypot(forward, strafe);
        const horizontalScale = horizontalLength > 1 ? 1 / horizontalLength : 1;
        const dx = (Math.sin(yaw) * forward + Math.cos(yaw) * strafe) * speed * deltaSeconds * horizontalScale;
        const dz = (Math.cos(yaw) * forward - Math.sin(yaw) * strafe) * speed * deltaSeconds * horizontalScale;
        const dy = vertical * speed * deltaSeconds;
        const frame = this.startWorld.slice.frame;
        let candidate = add4(
            add4(
                add4(this.spatialPosition4D, scale4(frame.axes[0], dx)),
                scale4(frame.axes[1], dy)
            ),
            scale4(frame.axes[2], dz)
        );
        const influence = this.getAnomalyInfluence(this.spatialPosition4D);
        const wSpeedScale = influence.wMovementScale;
        candidate = add4(candidate, scale4(frame.normal, crossSectionW * speed * deltaSeconds * wSpeedScale));

        const startPlane = this.getPlaneCoordinate(frame.origin, frame.normal);
        const destinationPlane = this.getPlaneCoordinate(this.target.slice.frame.origin, frame.normal);
        const targetDirection = Math.sign(destinationPlane - startPlane);
        const currentPlane = this.getPlaneCoordinate(this.spatialPosition4D, frame.normal);
        const nextPlane = this.getPlaneCoordinate(candidate, frame.normal);
        const targetReached = targetDirection > 0
            ? nextPlane >= destinationPlane - 1e-8
            : nextPlane <= destinationPlane + 1e-8;
        const hasEnteredSliceInterval = (currentPlane - startPlane) * targetDirection > 1e-8;
        const returningToSource = hasEnteredSliceInterval &&
            (nextPlane - currentPlane) * targetDirection < -1e-8 &&
            (nextPlane - startPlane) * targetDirection <= 1e-8;

        candidate = clampToRegion(candidate, this.corridor.region);
        candidate = constrainAmbientPositionToWorldChart(this.startWorld, candidate);
        this.spatialPosition4D = candidate;
        this.progress = clamp(this.calculateProgress(candidate), 0, 1);
        this.updatePlayerPosition();
        this.corridorChunk?.updateOccupant(this.player, this.progress, candidate);

        if (targetReached) {
            this.completeTransit(this.target, candidate);
        } else if (returningToSource) {
            this.completeTransit(this.startWorld, candidate);
        }
    }

    updatePlayerPosition() {
        const local = ambientToSlice(this.startWorld.slice.frame, this.spatialPosition4D).position;
        this.player.spatialPosition4D = { ...this.spatialPosition4D };
        this.player.setWorldPosition(local);
    }

    calculateProgress(position4D) {
        if (!this.startWorld || !this.target) return 0;
        const normal = this.startWorld.slice.frame.normal;
        const start = this.getPlaneCoordinate(this.startWorld.slice.frame.origin, normal);
        const end = this.getPlaneCoordinate(this.target.slice.frame.origin, normal);
        const current = this.getPlaneCoordinate(position4D, normal);
        return Math.abs(end - start) <= 1e-9 ? 0 : (current - start) / (end - start);
    }

    getPlaneCoordinate(position, normal) {
        return dot4(position, normal);
    }

    isInsideRegion(position) {
        return AXES.every((axis) =>
            position[axis] >= this.corridor.region.min[axis] - 1e-7 &&
            position[axis] <= this.corridor.region.max[axis] + 1e-7
        );
    }

    completeTransit(world, position4D = this.spatialPosition4D) {
        const arrival = this.createArrivalPosition(world, position4D);
        this.player.setSpatialContext?.(world.transform, world.slice.id, world.layerMask);
        this.player.spatialPosition4D = null;
        this.player.setWorldPosition(arrival);
        this.movement.setTerrain(world.terrain);
        if (world.movementProfile) this.movement.setMovementProfile?.(world.movementProfile);
        this.movement.setMovementSuspended(false);
        this.camera.clearRenderAnchor?.();
        this.camera.setSpatialContext(world.transform, world.layerMask);
        this.currentSliceId = world.slice.id;
        this.entryLock = { sliceId: world.slice.id, position: { ...arrival } };
        this.corridorChunk?.leave(this.player);
        this.state = 'idle';
        this.progress = 0;
        this.direction = 0;
        this.startWorld = null;
        this.target = null;
        this.spatialPosition4D = null;
        this.entryPosition = null;
        this.onSliceChanged(this.currentSliceId, {
            transitioning: false,
            anomalyId: this.corridor.id,
            corridorId: this.corridor.id,
            type: this.corridor.type
        });
        return arrival;
    }

    createArrivalPosition(world, position4D) {
        const local = ambientToSlice(world.slice.frame, position4D).position;
        const fallback = world.slice.id === this.corridor.destination.sliceId
            ? this.corridor.destination.position : this.corridor.source.position;
        const terrain = world.terrain;
        const x = clampToTerrain(local.x, terrain.min.x, terrain.max.x);
        const z = clampToTerrain(local.z, terrain.min.z, terrain.max.z);
        const sample = terrain.sampleAt(x, z);
        let arrival;
        if (!sample || !Number.isFinite(sample.y)) {
            const fallbackX = clampToTerrain(fallback.x, terrain.min.x, terrain.max.x);
            const fallbackZ = clampToTerrain(fallback.z, terrain.min.z, terrain.max.z);
            const fallbackSample = terrain.sampleAt(fallbackX, fallbackZ);
            if (!fallbackSample || !Number.isFinite(fallbackSample.y)) {
                throw new RangeError('Slice-transition arrival lies outside destination terrain.');
            }
            arrival = { x: fallbackX, y: fallbackSample.y, z: fallbackZ };
        } else {
            arrival = { x, y: sample.y, z };
        }
        return world.transform?.constrainWorldPosition?.(arrival) ?? arrival;
    }

    tryBeginEntityTransit(entity, fromSliceId = entity?.sliceId) {
        if (!this.active || this.disposed || !entity?.worldPosition || !Number.isFinite(entity.yaw)) return false;
        const inputForward = { x: Math.sin(entity.yaw), z: Math.cos(entity.yaw) };
        for (const neighbor of this.topology.getTraversableNeighbors(fromSliceId)) {
            if (neighbor.corridorId !== this.corridor.id) continue;
            const fromSource = this.corridor.source.sliceId === fromSliceId;
            const endpoint = fromSource ? this.corridor.source : this.corridor.destination;
            const dx = endpoint.position.x - entity.worldPosition.x;
            const dz = endpoint.position.z - entity.worldPosition.z;
            const distance = Math.hypot(dx, dz);
            if (distance <= 1e-6 || distance > this.options.triggerDistance) continue;
            const heading = (inputForward.x * dx + inputForward.z * dz) / distance;
            if (heading >= this.options.headingThreshold) return this.beginEntityTransit(entity, fromSliceId);
        }
        return false;
    }

    beginEntityTransit(entity, fromSliceId = entity?.sliceId) {
        if (!this.active || this.disposed) throw new Error('Slice-transition anomaly is not active.');
        if (!entity || entity === this.player || this.entityTransits.has(entity)) return false;
        const neighbor = this.topology.getTraversableNeighbors(fromSliceId)
            .find((entry) => entry.corridorId === this.corridor.id);
        if (!neighbor) return false;
        const source = this.worlds.get(fromSliceId);
        const target = this.worlds.get(neighbor.sliceId);
        if (!source || !target || !this.hasChunkConnection(fromSliceId, entity.worldPosition)) return false;
        const position4D = sliceToAmbient(source.slice.frame, entity.worldPosition);
        if (!this.isInsideRegion(position4D)) return false;
        this.entityTransits.set(entity, {
            entity, source, target, fromSliceId,
            direction: fromSliceId === this.corridor.source.sliceId ? 1 : -1,
            speed: entity.anomalySpeed ?? this.options.entitySpeed,
            position4D
        });
        this.corridorChunk?.enter(entity, { direction: fromSliceId === this.corridor.source.sliceId ? 1 : -1,
            progress: clamp(this.calculateEntityProgress(position4D, source, target), 0, 1), position4D });
        return true;
    }

    calculateEntityProgress(position4D, source, target) {
        const normal = source.slice.frame.normal;
        const start = dot4(source.slice.frame.origin, normal);
        const end = dot4(target.slice.frame.origin, normal);
        return Math.abs(end - start) <= 1e-9 ? 0 : (dot4(position4D, normal) - start) / (end - start);
    }

    advanceOtherEntities(deltaSeconds) {
        for (const [entity, state] of this.entityTransits) {
            const normal = state.source.slice.frame.normal;
            const start = dot4(state.source.slice.frame.origin, normal);
            const end = dot4(state.target.slice.frame.origin, normal);
            const targetDirection = Math.sign(end - start);
            const step = scale4(normal, targetDirection * state.speed * deltaSeconds);
            const candidate = clampToRegion(add4(state.position4D, step), this.corridor.region);
            state.position4D = candidate;
            const progress = clamp(this.calculateEntityProgress(candidate, state.source, state.target), 0, 1);
            this.corridorChunk?.updateOccupant(entity, progress, candidate);
            if (progress >= 1 - 1e-8) {
                const arrival = this.createArrivalPosition(state.target, candidate);
                entity.setSpatialContext?.(state.target.transform, state.target.slice.id, state.target.layerMask);
                entity.setWorldPosition?.(arrival);
                if (!entity.setWorldPosition) entity.worldPosition = arrival;
                entity.spatialPosition4D = null;
                this.corridorChunk?.leave(entity);
                this.entityTransits.delete(entity);
            }
        }
    }

    getEntityTransitState(entity) {
        const state = this.entityTransits.get(entity);
        return state ? {
            fromSliceId: state.fromSliceId,
            toSliceId: state.target.slice.id,
            progress: clamp(this.calculateEntityProgress(state.position4D, state.source, state.target), 0, 1),
            direction: state.direction,
            position4D: { ...state.position4D }
        } : null;
    }

    getState() {
        return {
            state: this.state,
            currentSliceId: this.currentSliceId,
            startSliceId: this.startWorld?.slice.id ?? null,
            destinationSliceId: this.target?.slice.id ?? null,
            anomalyId: this.corridor.id,
            anomalyType: this.corridor.type,
            progress: this.progress,
            direction: this.direction,
            position4D: this.spatialPosition4D && { ...this.spatialPosition4D },
            bounds: this.corridor.region,
            influence: this.corridor.influence ? { ...this.corridor.influence, center: { ...this.corridor.influence.center } } : null,
            entryDiagnostic: this.getEntryDiagnostic(),
            corridorOccupants: this.corridorChunk?.occupants.size ?? this.entityTransits.size
        };
    }

    validateOptions() {
        const { triggerDistance, speed, sprintSpeed, entitySpeed, headingThreshold } = this.options;
        if (![triggerDistance, speed, sprintSpeed, entitySpeed].every(Number.isFinite) ||
            triggerDistance <= 0 || speed <= 0 || sprintSpeed < speed || entitySpeed <= 0) {
            throw new RangeError('Slice-transition movement options must be positive and valid.');
        }
        if (!Number.isFinite(headingThreshold) || headingThreshold < -1 || headingThreshold > 1) {
            throw new RangeError('Slice-transition headingThreshold must be between -1 and 1.');
        }
    }

    dispose() {
        if (this.disposed) return;
        this.disposed = true;
        this.active = false;
        if (this.state === 'transit') {
            this.movement.setMovementSuspended(false);
            this.corridorChunk?.leave(this.player);
        }
        for (const entity of this.entityTransits.keys()) this.corridorChunk?.leave(entity);
        this.entityTransits.clear();
        this.state = 'disposed';
        this.target = null;
        this.startWorld = null;
    }
}

function constrainAmbientPositionToWorldChart(world, position4D) {
    if (typeof world?.transform?.constrainWorldPosition !== 'function') return position4D;
    const frame = world.slice.frame;
    const local = ambientToSlice(frame, position4D);
    const constrained = world.transform.constrainWorldPosition(local.position);
    if (AXES.slice(0, 3).every((axis) => Math.abs(constrained[axis] - local.position[axis]) <= 1e-10)) {
        return position4D;
    }
    return add4(
        sliceToAmbient(frame, constrained),
        scale4(frame.normal, local.normalOffset)
    );
}

    return {
        x: clamp(position.x, region.min.x, region.max.x),
        y: clamp(position.y, region.min.y, region.max.y),
        z: clamp(position.z, region.min.z, region.max.z),
        w: clamp(position.w, region.min.w, region.max.w)
    };
}

    return clamp(value, minimum, maximum);
}

function clamp(value, minimum, maximum) {
    return Math.min(maximum, Math.max(minimum, value));
}

    if (!value || typeof value !== 'object') throw new TypeError(`${label} is required.`);

/**
 * Engine-independent primitives for the ambient four-dimensional space.
 * A slice frame contains three tangent axes and one orthogonal normal.
 */

const AXES = ['x', 'y', 'z', 'w'];
const PLANES = ['xy', 'xz', 'xw', 'yz', 'yw', 'zw'];
const FRAME_TOLERANCE = 1e-9;
const ZERO_TOLERANCE = 1e-12;

function createVector4(value) {
    assertVector4(value);
    return { x: value.x, y: value.y, z: value.z, w: value.w };
}

function assertVector4(value) {
    if (!value || !AXES.every(function isFiniteAxis(axis) {
        return Number.isFinite(value[axis]);
    })) {
        throw new TypeError('Expected a finite four-dimensional vector.');
    }
}

function add4(a, b) {
    assertVector4(a);
    assertVector4(b);
    return mapVector4(a, b, function addAxis(left, right) {
        return left + right;
    });
}

function subtract4(a, b) {
    assertVector4(a);
    assertVector4(b);
    return mapVector4(a, b, function subtractAxis(left, right) {
        return left - right;
    });
}

function scale4(vector, factor) {
    assertVector4(vector);
    assertFinite(factor, 'Scale');
    const result = {
        x: vector.x * factor,
        y: vector.y * factor,
        z: vector.z * factor,
        w: vector.w * factor
    };
    assertVector4(result);
    return result;
}

function dot4(a, b) {
    assertVector4(a);
    assertVector4(b);
    const result = a.x * b.x + a.y * b.y + a.z * b.z + a.w * b.w;
    assertFinite(result, 'Dot product');
    return result;
}

function norm4(vector) {
    assertVector4(vector);
    // Math.hypot scales its inputs to avoid premature square overflow.
    const length = Math.hypot(vector.x, vector.y, vector.z, vector.w);
    if (!Number.isFinite(length)) throw new RangeError('4D vector length exceeds numeric range.');
    return length;
}

function normalize4(vector) {
    const length = norm4(vector);
    if (length <= ZERO_TOLERANCE) {
        throw new RangeError('Cannot normalize a zero-length 4D vector.');
    }
    return scale4(vector, 1 / length);
}

function interpolate4(a, b, t) {
    assertUnitParameter(t);
    return add4(a, scale4(subtract4(b, a), t));
}

function rotate4(vector, plane, angle) {
    assertVector4(vector);
    assertFinite(angle, 'Rotation angle');
    if (!PLANES.includes(plane)) {
        throw new RangeError(`Unsupported 4D rotation plane: ${plane}`);
    }

    const [first, second] = plane.split('');
    const result = { ...vector };
    const cosine = Math.cos(angle);
    const sine = Math.sin(angle);
    result[first] = vector[first] * cosine - vector[second] * sine;
    result[second] = vector[first] * sine + vector[second] * cosine;
    assertVector4(result);
    return result;
}

function createCanonicalSliceFrame(w = 0) {
    assertFinite(w, 'Slice w coordinate');
    return createSliceFrame({
        origin: { x: 0, y: 0, z: 0, w },
        axes: [basis4(0), basis4(1), basis4(2)],
        normal: basis4(3)
    });
}

function createSliceFrame(frame) {
    if (!frame || !Array.isArray(frame.axes) || frame.axes.length !== 3) {
        throw new TypeError('A 3D slice frame requires exactly three tangent axes.');
    }

    const normalized = {
        origin: createVector4(frame.origin),
        axes: frame.axes.map(createVector4),
        normal: normalize4(createVector4(frame.normal))
    };
    validateOrthonormalFrame(normalized);
    return normalized;
}

function validateOrthonormalFrame(frame) {
    const basis = [...frame.axes, frame.normal];
    for (let row = 0; row < basis.length; row++) {
        for (let column = row; column < basis.length; column++) {
            const expected = row === column ? 1 : 0;
            if (Math.abs(dot4(basis[row], basis[column]) - expected) > FRAME_TOLERANCE) {
                throw new RangeError('Slice frame axes must be orthonormal in 4D.');
            }
        }
    }
}

function sliceToAmbient(frame, position) {
    const local = createPosition3(position);
    return frame.axes.reduce(function addFrameAxis(point, axis, index) {
        return add4(point, scale4(axis, local[['x', 'y', 'z'][index]]));
    }, { ...frame.origin });
}

function ambientToSlice(frame, position) {
    assertVector4(position);
    const offset = subtract4(position, frame.origin);
    return {
        position: {
            x: dot4(offset, frame.axes[0]),
            y: dot4(offset, frame.axes[1]),
            z: dot4(offset, frame.axes[2])
        },
        normalOffset: dot4(offset, frame.normal)
    };
}

function createPerpendicularFrame(tangent) {
    const direction = normalize4(tangent);
    const candidates = [0, 1, 2, 3]
        .map(basis4)
        .sort(function leastAlignedFirst(a, b) {
            return Math.abs(dot4(a, direction)) - Math.abs(dot4(b, direction));
        });
    return selectPerpendicularAxes(candidates, direction, []);
}

function transportPerpendicularFrame(previousAxes, tangent) {
    if (!Array.isArray(previousAxes) || previousAxes.length !== 3) {
        throw new TypeError('Frame transport requires three previous axes.');
    }
    const direction = normalize4(tangent);
    // Project old axes into the new tangent's normal space before orthogonalizing.
    const projected = previousAxes.map(function projectAxis(axis) {
        return subtract4(axis, scale4(direction, dot4(axis, direction)));
    });
    // Canonical axes fill dimensions lost at near-90-degree tangent changes.
    projected.push(basis4(0), basis4(1), basis4(2), basis4(3));
    return selectPerpendicularAxes(projected, direction, []);
}

function selectPerpendicularAxes(candidates, direction, selected) {
    for (const candidate of candidates) {
        let residual = subtract4(
            candidate,
            scale4(direction, dot4(candidate, direction))
        );
        for (const axis of selected) {
            residual = subtract4(residual, scale4(axis, dot4(residual, axis)));
        }
        const length = norm4(residual);
        if (length > ZERO_TOLERANCE) selected.push(scale4(residual, 1 / length));
        if (selected.length === 3) break;
    }

    if (selected.length !== 3) {
        throw new RangeError('Unable to construct a stable 4D cross-section frame.');
    }
    return selected;
}

function createPosition3(position) {
    if (!position || !['x', 'y', 'z'].every(function isFiniteAxis(axis) {
        return Number.isFinite(position[axis]);
    })) {
        throw new TypeError('Expected a finite three-dimensional position.');
    }
    return { x: position.x, y: position.y, z: position.z };
}

function basis4(index) {
    return { x: +(index === 0), y: +(index === 1), z: +(index === 2), w: +(index === 3) };
}

function mapVector4(a, b, operation) {
    const result = {
        x: operation(a.x, b.x),
        y: operation(a.y, b.y),
        z: operation(a.z, b.z),
        w: operation(a.w, b.w)
    };
    assertVector4(result);
    return result;
}

function assertFinite(value, label) {
    if (!Number.isFinite(value)) throw new TypeError(`${label} must be finite.`);
}

function assertUnitParameter(t) {
    assertFinite(t, 'Interpolation parameter');
    if (t < 0 || t > 1) throw new RangeError('Parameter must be between 0 and 1.');
}

export {
    add4,
    ambientToSlice,
    assertVector4,
    createCanonicalSliceFrame,
    createPerpendicularFrame,
    createPosition3,
    createSliceFrame,
    createVector4,
    dot4,
    interpolate4,
    norm4,
    normalize4,
    rotate4,
    scale4,
    sliceToAmbient,
    subtract4,
    transportPerpendicularFrame
};

/** Serializable definition and deterministic sampling for a cubic 4D corridor. */

import {
    add4,
    createPerpendicularFrame,
    createPosition3,
    createVector4,
    norm4,
    normalize4,
    scale4,
    subtract4,
    transportPerpendicularFrame
} from './spatial-4d.js';
import { createAnomalyOptics } from './optical-contracts.js?v=somnium-20261108-phase16c-anomaly-optics-2';

const SPATIAL_CORRIDOR_SCHEMA_VERSION = 1;
const MAX_CORRIDOR_SAMPLES = 4096;

function createSpatialCorridor(options = {}) {
    const schemaVersion = options.schemaVersion ?? SPATIAL_CORRIDOR_SCHEMA_VERSION;
    if (schemaVersion !== SPATIAL_CORRIDOR_SCHEMA_VERSION) {
        throw new RangeError(`Unsupported SpatialCorridor schema version: ${schemaVersion}`);
    }

    const corridor = {
        schemaVersion,
        id: requireId(options.id, 'SpatialCorridor id'),
        type: 'bezier',
        revision: options.revision ?? 1,
        source: createEndpoint(options.source, 'source'),
        destination: createEndpoint(options.destination, 'destination'),
        centerline: validateCenterline(options.centerline),
        radiusProfile: createRadiusProfile(options.radiusProfile),
        optics: createAnomalyOptics(options.optics),
        // Scales ambient arc length into proper distance inside an exotic corridor.
        intrinsicLengthScale: options.intrinsicLengthScale ?? 1,
        visible: options.visible ?? true,
        traversable: options.traversable ?? false,
        bidirectional: options.bidirectional ?? true
    };

    if (!Number.isSafeInteger(corridor.revision) || corridor.revision < 1) {
        throw new RangeError('Corridor revision must be a positive safe integer.');
    }
    if (!Number.isFinite(corridor.intrinsicLengthScale) ||
        corridor.intrinsicLengthScale <= 0 || corridor.intrinsicLengthScale > 1) {
        throw new RangeError('Corridor intrinsicLengthScale must be in (0, 1].');
    }
    for (const field of ['visible', 'traversable', 'bidirectional']) {
        if (typeof corridor[field] !== 'boolean') {
            throw new TypeError(`Corridor ${field} must be boolean.`);
        }
    }
    if (distance4(corridor.centerline[0], corridor.centerline[3]) <= 1e-12) {
        throw new RangeError('Corridor endpoints must be distinct in 4D space.');
    }
    return corridor;
}

function sampleSpatialCorridor(corridor, t) {
    assertUnitParameter(t);
    const centerline = validateCenterline(corridor.centerline);
    return {
        t,
        position: cubicBezier(centerline, t),
        tangent: cubicBezierTangent(centerline, t),
        radius: sampleRadiusProfile(corridor.radiusProfile, t)
    };
}

function sampleSpatialCorridorFrames(corridor, segments) {
    if (!Number.isSafeInteger(segments) || segments < 1 || segments > MAX_CORRIDOR_SAMPLES) {
        throw new RangeError(`Corridor segments must be between 1 and ${MAX_CORRIDOR_SAMPLES}.`);
    }

    const samples = [];
    let previousAxes = null;
    for (let index = 0; index <= segments; index++) {
        const sample = sampleSpatialCorridor(corridor, index / segments);
        sample.axes = previousAxes
            ? transportPerpendicularFrame(previousAxes, sample.tangent)
            : createPerpendicularFrame(sample.tangent);
        previousAxes = sample.axes;
        samples.push(sample);
    }
    return samples;
}

function cubicBezier(points, t) {
    assertUnitParameter(t);
    const [p0, p1, p2, p3] = validateCenterline(points);
    const inverse = 1 - t;
    const a = inverse ** 3;
    const b = 3 * inverse ** 2 * t;
    const c = 3 * inverse * t ** 2;
    const d = t ** 3;
    return add4(
        add4(scale4(p0, a), scale4(p1, b)),
        add4(scale4(p2, c), scale4(p3, d))
    );
}

function cubicBezierTangent(points, t) {
    assertUnitParameter(t);
    const [p0, p1, p2, p3] = validateCenterline(points);
    const inverse = 1 - t;
    const derivative = add4(
        add4(
            scale4(subtract4(p1, p0), 3 * inverse ** 2),
            scale4(subtract4(p2, p1), 6 * inverse * t)
        ),
        scale4(subtract4(p3, p2), 3 * t ** 2)
    );
    if (norm4(derivative) > 1e-12) return normalize4(derivative);

    // A stationary Bezier endpoint can still have a well-defined one-sided tangent.
    const nearbyT = t < 1 ? Math.min(1, t + 1e-4) : Math.max(0, t - 1e-4);
    const nearby = cubicBezier(points, nearbyT);
    const origin = cubicBezier(points, t);
    const difference = t < 1
        ? subtract4(nearby, origin)
        : subtract4(origin, nearby);
    if (norm4(difference) > 1e-12) return normalize4(difference);

    const chord = subtract4(p3, p0);
    if (norm4(chord) > 1e-12) return normalize4(chord);
    throw new RangeError('Corridor centreline has no stable tangent.');
}

function sampleRadiusProfile(profile, t) {
    const keyframes = createRadiusProfile(profile);
    for (let index = 1; index < keyframes.length; index++) {
        const right = keyframes[index];
        if (t > right.t) continue;
        const left = keyframes[index - 1];
        const blend = (t - left.t) / (right.t - left.t);
        return left.radius + (right.radius - left.radius) * blend;
    }
    return keyframes[keyframes.length - 1].radius;
}

function createRadiusProfile(profile = null) {
    const source = profile ?? [
        { t: 0, radius: 1 },
        { t: 1, radius: 1 }
    ];
    if (!Array.isArray(source) || source.length < 2) {
        throw new TypeError('Corridor radius profile requires at least two keyframes.');
    }

    const result = source.map(function normalizeKeyframe(keyframe) {
        if (!keyframe || !Number.isFinite(keyframe.t) || !Number.isFinite(keyframe.radius)) {
            throw new TypeError('Invalid corridor radius keyframe.');
        }
        if (keyframe.t < 0 || keyframe.t > 1 || keyframe.radius <= 0) {
            throw new RangeError('Corridor keyframes require t in [0, 1] and positive radius.');
        }
        return { t: keyframe.t, radius: keyframe.radius };
    });

    if (result[0].t !== 0 || result[result.length - 1].t !== 1) {
        throw new RangeError('Corridor radius profile must cover t = 0 through t = 1.');
    }
    for (let index = 1; index < result.length; index++) {
        if (result[index].t <= result[index - 1].t) {
            throw new RangeError('Corridor radius profile times must increase strictly.');
        }
    }
    return result;
}

function createEndpoint(endpoint, label) {
    if (!endpoint) throw new TypeError(`Corridor ${label} endpoint is required.`);
    return {
        sliceId: requireId(endpoint.sliceId, `${label} slice id`),
        position: createPosition3(endpoint.position)
    };
}

function validateCenterline(points) {
    if (!Array.isArray(points) || points.length !== 4) {
        throw new TypeError('A cubic 4D corridor requires four centreline control points.');
    }
    return points.map(createVector4);
}

function distance4(a, b) {
    return norm4(subtract4(a, b));
}

function requireId(value, label) {
    if (typeof value !== 'string' || value.trim() === '') {
        throw new TypeError(`${label} must be a non-empty string.`);
    }
    return value.trim();
}

function assertUnitParameter(t) {
    if (!Number.isFinite(t) || t < 0 || t > 1) {
        throw new RangeError('Corridor parameter must be between 0 and 1.');
    }
}

export {
    SPATIAL_CORRIDOR_SCHEMA_VERSION,
    createSpatialCorridor,
    cubicBezier,
    cubicBezierTangent,
    sampleRadiusProfile,
    sampleSpatialCorridor,
    sampleSpatialCorridorFrames
};

/**

import { EuclideanGeometry } from "./euclidean-geometry.js";
import { MetricTensor } from "../metric-tensor.js";

class SpatialTransform {

    constructor(options = {}) {

        this.geometryName =
            options.geometry ?? "poincare";

        this.renderScale =
            options.renderScale ?? 10000;

        this.enabled = options.enabled ?? true;
        this.visualScaleOptions = {
            gamma: options.visualScaleGamma ?? 0.5,
            maxScale: options.maxVisualScale ?? 4,
            boundaryEpsilon: options.visualBoundaryEpsilon ?? 1e-4
        };

        if (
            !Number.isFinite(this.renderScale) ||

        }

    }

    // -----------------------------------------
    // GEOMETRY SELECTION
    // -----------------------------------------

    setGeometry(name) {

        if (!["euclidean", "poincare", "spherical", "inverted-poincare"].includes(name)) {
            throw new Error(
                `Unsupported geometry: ${name}`
            );
        }

        this.geometryName = name;

        console.log(
            "[SpatialTransform] Geometry:",
            name
        );

    }

    // -----------------------------------------
    // WORLD TO GEOMETRY COORDINATES
    // -----------------------------------------

    worldToGeometry(position) {

        if (!this.enabled || this.geometryName === "euclidean") return copyPosition(position);
        if (this.geometryName === "poincare") return this.projection.worldToBall(position);
        if (this.geometryName === "spherical") return this.worldToSphericalChart(position);
        return this.worldToInvertedPoincare(position);
    }

    // -----------------------------------------
    // GEOMETRY TO WORLD COORDINATES
    // -----------------------------------------

    geometryToWorld(position) {

        if (!this.enabled || this.geometryName === "euclidean") return copyPosition(position);
        if (this.geometryName === "poincare") return this.projection.ballToWorld(position);
        if (this.geometryName === "spherical") return this.sphericalChartToWorld(position);
        return this.invertedPoincareToWorld(position);
    }

    // -----------------------------------------
    // WORLD TO RENDER COORDINATES
    // -----------------------------------------

    worldToRender(position) {

        const point =
            this.worldToGeometry(position);

        if (!this.enabled || this.geometryName === "euclidean") return point;
        return scalePosition(point, this.renderScale);
    }

    // -----------------------------------------
    // RENDER TO WORLD COORDINATES
    // -----------------------------------------

    renderToWorld(position) {

        if (
            !this.enabled ||
            this.geometryName === "euclidean"
        ) {
            return { ...position };
        }

        return this.geometryToWorld({
            x: position.x / this.renderScale,
            y: position.y / this.renderScale,
            z: position.z / this.renderScale
        });
    }

    /**
     * Constrain an authoritative world-space position to the finite domain of
     * Poincare-based charts before projecting it. This is useful for camera/player
     * positions that move through an anomaly region extending beyond a slice's
     * local chart bounds. The projection itself remains strict.
     */
    /**
     * Returns whether a world point and the centered finite-difference samples
     * used by metricAtWorld() remain inside this transform's finite chart.
     * Optical integration uses this guard to terminate at chart boundaries
     * instead of allowing an RK stage to throw from worldToBall().
     */
    containsMetricDomainPosition(position, options = {}) {
        if (!position || ![position.x, position.y, position.z].every(Number.isFinite)) {
            return false;
        }
        if (!this.enabled || !["poincare", "inverted-poincare"].includes(this.geometryName)) {
            return true;
        }

        const derivativeStep = options.jacobianStep ?? 0.01;
        const margin = options.boundaryMarginWorld ?? derivativeStep * 1.1;
        if (!Number.isFinite(derivativeStep) || derivativeStep <= 0 ||
            !Number.isFinite(margin) || margin < derivativeStep) {
            throw new RangeError("Invalid metric-domain boundary margin.");
        }
        const bounds = this.projection.bounds;
        for (const axis of ["x", "y", "z"]) {
            if (position[axis] < bounds[`min${axis.toUpperCase()}`] + margin ||
                position[axis] > bounds[`max${axis.toUpperCase()}`] - margin) {
                return false;
            }
        }

        try {
            this.projection.worldToBall(position);
            for (const axis of ["x", "y", "z"]) {
                for (const direction of [-1, 1]) {
                    const sample = { ...position, [axis]: position[axis] + direction * derivativeStep };
                    this.projection.worldToBall(sample);
                }
            }
            return true;
        } catch (error) {
            if (error instanceof RangeError) return false;
            throw error;
        }
    }

    constrainWorldPosition(position, options = {}) {
        validateRenderPosition(position);

        if (!this.enabled || !["poincare", "inverted-poincare"].includes(this.geometryName)) {
            return copyPosition(position);
        }

        const bounds = this.projection.bounds;
        const margin = options.boundaryMargin ?? Math.max(
            this.projection.boundaryEpsilon,
            1e-7
        );
        if (!Number.isFinite(margin) || margin <= 0 || margin >= 0.01) {
            throw new RangeError("World chart boundary margin must be between zero and 0.01.");
        }

        const clampAxis = (value, minimum, maximum) => {
            const inset = (maximum - minimum) * margin;
            return Math.min(maximum - inset, Math.max(minimum + inset, value));
        };

        return {
            x: clampAxis(position.x, bounds.minX, bounds.maxX),
            y: clampAxis(position.y, bounds.minY, bounds.maxY),
            z: clampAxis(position.z, bounds.minZ, bounds.maxZ)
        };
    }

    /**
     * Keep free-camera coordinates inside the active chart before attempting
     * an inverse projection. The Poincare chart is an open ball, so camera
     * movement must never be allowed to step onto or beyond its boundary.
     *
     * This is intentionally a render-space constraint; authoritative world
     * positions and the strict projection validation remain unchanged.
     */
    constrainRenderPosition(position, options = {}) {
        validateRenderPosition(position);

        if (!this.enabled || this.geometryName !== "poincare") {
            return copyPosition(position);
        }

        const margin = options.boundaryMargin ?? Math.max(
            this.projection.boundaryEpsilon,
            this.visualScaleOptions.boundaryEpsilon,
            1e-6
        );
        if (!Number.isFinite(margin) || margin <= 0 || margin >= 1) {
            throw new RangeError("Render chart boundary margin must be between zero and one.");
        }

        const renderRadius = this.projection.radius * this.renderScale;
        const maximumRadius = renderRadius * (1 - margin);
        const distance = Math.hypot(position.x, position.y, position.z);
        if (distance <= maximumRadius || distance === 0) {
            return copyPosition(position);
        }

        const factor = maximumRadius / distance;
        return {
            x: position.x * factor,
            y: position.y * factor,
            z: position.z * factor
        };
    }

    worldToSphericalChart(position) {
        this.projection.validatePosition(position);
        const offset = subtractPosition(position, this.projection.center);
        const distance = lengthPosition(offset);
        if (distance === 0) return { x: 0, y: 0, z: 0 };
        const radius = this.projection.radius;
        const chartDistance = 2 * radius * Math.atan(distance / (2 * radius));
        return scalePosition(offset, chartDistance / distance);
    }

    sphericalChartToWorld(position) {
        this.projection.validatePosition(position);
        const distance = lengthPosition(position);
        const chartLimit = Math.PI * this.projection.radius;
        if (distance >= chartLimit) throw new RangeError("Spherical chart reaches its antipodal limit.");
        if (distance === 0) return copyPosition(this.projection.center);
        const worldDistance = 2 * this.projection.radius * Math.tan(distance / (2 * this.projection.radius));
        return addPosition(this.projection.center, scalePosition(position, worldDistance / distance));
    }

    worldToInvertedPoincare(position) {
        const ball = this.projection.worldToBall(position);
        const distance = lengthPosition(ball);
        const epsilon = this.projection.boundaryEpsilon;
        if (distance <= epsilon * this.projection.radius) {
            throw new RangeError("Inverted-Poincare chart is singular at its centre.");
        }
        const invertedRadius = Math.min(
            this.projection.radius / epsilon,
            this.projection.radius ** 2 / distance
        );
        return scalePosition(ball, invertedRadius / distance);
    }

    invertedPoincareToWorld(position) {
        this.projection.validatePosition(position);
        const distance = lengthPosition(position);
        if (distance <= this.projection.radius) {
            throw new RangeError("Inverted-Poincare coordinates must lie outside the sphere.");
        }
        const ballRadius = this.projection.radius ** 2 / distance;
        const ball = scalePosition(position, ballRadius / distance);
        return this.projection.ballToWorld(ball);
    }

    getVisualScale(position, options = {}) {
        if (!this.enabled || this.geometryName !== "poincare") return 1;
        const ball = this.projection.worldToBall(position);
        return calculatePoincareVisualScale(
            lengthPosition(ball),
            this.projection.radius,
            { ...this.visualScaleOptions, ...options }
        );
    }

    // -----------------------------------------
    // NUMERICAL JACOBIAN
    // -----------------------------------------

    getJacobian(position, options = {}) {

        const h = options.step ?? 0.01;

        if (
            !Number.isFinite(h) ||
            h <= 0
        ) {
            throw new Error(
                "Invalid Jacobian step."
            );
        }

        const origin =
            this.worldToRender(position);

        const columns = [];

        for (const axis of ["x", "y", "z"]) {

            const forward = { ...position };
            const backward = { ...position };

            forward[axis] += h;
            backward[axis] -= h;

            const a = this.worldToRender(forward);
            const b = this.worldToRender(backward);

            columns.push([
                (a.x - b.x) / (2 * h),
                (a.y - b.y) / (2 * h),
                (a.z - b.z) / (2 * h)
            ]);
        }

        return [
            columns[0][0],
            columns[1][0],
            columns[2][0],

            columns[0][1],
            columns[1][1],
            columns[2][1],

            columns[0][2],
            columns[1][2],
            columns[2][2]
        ];
    }

    // -----------------------------------------
    // DIRECTION TRANSFORMATION
    // -----------------------------------------

    transformDirection(position, direction) {

        const J = this.getJacobian(position);

        return {
            x:
                J[0] * direction.x +
                J[1] * direction.y +
                J[2] * direction.z,

            y:
                J[3] * direction.x +
                J[4] * direction.y +
                J[5] * direction.z,

            z:
                J[6] * direction.x +
                J[7] * direction.y +
                J[8] * direction.z
        };
    }

    // -----------------------------------------
    // SPATIAL METRIC
    // -----------------------------------------

	
    metricAtWorld(position) {

        if (
            !this.enabled ||
            this.geometryName === "euclidean"
        ) {
            return this.euclidean.metricAt(position);
        }

        if (this.geometryName === "spherical") {
            return new MetricTensor(this.sphericalMetricAt(position));
        }

        const ballPosition = this.projection.worldToBall(position);
        const geometryMetric = this.metric.metricAt(ballPosition).values;

        // Convert the rendering Jacobian into
        // the mathematical projection Jacobian.

        const J = this.getJacobian(position).map(
            value => value / this.renderScale
        );

        // Pull back G into world coordinates:
        // G_world = transpose(J) * G_ball * J

        const values = new Array(9).fill(0);

        for (let i = 0; i < 3; i++) {
            for (let j = 0; j < 3; j++) {

                for (let k = 0; k < 3; k++) {
                    for (let l = 0; l < 3; l++) {

                        values[i * 3 + j] +=
                            J[k * 3 + i] *
                            geometryMetric[k * 3 + l] *
                            J[l * 3 + j];
                    }
                }
            }
        }

        // J^T G J is symmetric analytically. The two off-diagonal sums can
        // differ by a few ulps because their floating-point accumulation order
        // is reversed; symmetrize before MetricTensor's strict validation.
        for (let row = 0; row < 3; row++) {
            for (let column = row + 1; column < 3; column++) {
                const average = 0.5 * (
                    values[row * 3 + column] + values[column * 3 + row]
                );
                values[row * 3 + column] = average;
                values[column * 3 + row] = average;
            }
        }
        return new MetricTensor(values);
    }



    sphericalMetricAt(position) {
        const chart = this.worldToSphericalChart(position);
        const radius = lengthPosition(chart);
        const sphereRadius = this.projection.radius;
        const ratio = radius === 0 ? 1 : Math.sin(radius / sphereRadius) / (radius / sphereRadius);
        const direction = radius === 0 ? { x: 0, y: 0, z: 0 } : scalePosition(chart, 1 / radius);
        const radialCorrection = 1 - ratio ** 2;
        const chartMetric = [
            ratio ** 2 + radialCorrection * direction.x ** 2,
            radialCorrection * direction.x * direction.y,
            radialCorrection * direction.x * direction.z,
            radialCorrection * direction.y * direction.x,
            ratio ** 2 + radialCorrection * direction.y ** 2,
            radialCorrection * direction.y * direction.z,
            radialCorrection * direction.z * direction.x,
            radialCorrection * direction.z * direction.y,
            ratio ** 2 + radialCorrection * direction.z ** 2
        ];
        return pullbackMetric(chartMetric, this.getJacobian(position).map((value) => value / this.renderScale));
    }

    sphericalDistance(first, second) {
        const a = this.worldToSphericalChart(first);
        const b = this.worldToSphericalChart(second);
        const radius = this.projection.radius;
        const ra = lengthPosition(a);
        const rb = lengthPosition(b);
        const dot = dotPosition(a, b);
        const cosAngle = Math.cos(ra / radius) * Math.cos(rb / radius) +
            (ra === 0 || rb === 0 ? 0 : Math.sin(ra / radius) * Math.sin(rb / radius) * dot / (ra * rb));
        return radius * Math.acos(Math.max(-1, Math.min(1, cosAngle)));
    }

    // -----------------------------------------
    // HYPERBOLIC DISTANCE
    // -----------------------------------------

    distance(a, b) {

        if (this.geometryName === "euclidean") return this.euclidean.distance(a, b);
        if (this.geometryName === "spherical") return this.sphericalDistance(a, b);

        const pointA = this.projection.worldToBall(a);
        const pointB = this.projection.worldToBall(b);

        return this.metric.distance(
            pointA,
            pointB
        );
    }

    // -----------------------------------------
    // DIAGNOSTICS
    // -----------------------------------------

    getInfo() {

        return {
            geometry: this.geometryName,

            enabled: this.enabled,

            renderScale: this.renderScale,

            projection: this.projection.getInfo(),
            metric: this.metric.getInfo(),
            visualScale: this.geometryName === "poincare" ? "bounded-conformal" : "unit"
        };
    }

}

function copyPosition(position) {
    return { x: position.x, y: position.y, z: position.z };
}

function validateRenderPosition(position) {
    if (!position || ![position.x, position.y, position.z].every(Number.isFinite)) {
        throw new TypeError("Render position must contain finite x, y, and z coordinates.");
    }
}

function addPosition(a, b) {
    return { x: a.x + b.x, y: a.y + b.y, z: a.z + b.z };
}

function subtractPosition(a, b) {
    return { x: a.x - b.x, y: a.y - b.y, z: a.z - b.z };
}

function scalePosition(position, factor) {
    return { x: position.x * factor, y: position.y * factor, z: position.z * factor };
}

function dotPosition(a, b) { return a.x * b.x + a.y * b.y + a.z * b.z; }
function lengthPosition(position) { return Math.hypot(position.x, position.y, position.z); }

function pullbackMetric(metric, jacobian) {
    const values = new Array(9).fill(0);
    for (let row = 0; row < 3; row++) {
        for (let column = 0; column < 3; column++) {
            for (let first = 0; first < 3; first++) {
                for (let second = 0; second < 3; second++) {
                    values[row * 3 + column] += jacobian[first * 3 + row] *
                        metric[first * 3 + second] * jacobian[second * 3 + column];
                }
            }
        }
    }
    return values;
}

/** Builds reusable Babylon meshes from a topographic terrain sampler. */

function createTopographicTerrainMesh(scene, terrain, options = {}) {
    const buffers = buildTopographicTerrainBuffers(terrain, options);
    const normals = new Array(buffers.positions.length).fill(0);
    BABYLON.VertexData.ComputeNormals(buffers.positions, buffers.indices, normals);
    const data = new BABYLON.VertexData();
    data.positions = buffers.positions;
    data.normals = normals;
    data.colors = buffers.colors;
    data.indices = buffers.indices;
    const mesh = new BABYLON.Mesh(options.name ?? 'topographicTerrain', scene);
    data.applyToMesh(mesh, true);
    mesh.isPickable = options.isPickable ?? false;
    if (Number.isInteger(options.layerMask)) mesh.layerMask = options.layerMask;
    mesh.metadata = {
        ...(mesh.metadata || {}),
        topographicTerrain: true,
        terrainResolution: { x: buffers.resolutionX, z: buffers.resolutionZ },
        worldSizeX: terrain.worldSizeX,
        worldSizeZ: terrain.worldSizeZ
    };
    return mesh;
}

function buildTopographicTerrainBuffers(terrain, options = {}) {
    validateTerrain(terrain);
    const maxSegments = options.maxSegments ?? 100;
    const resolutionX = resolveResolution(options.resolutionX, terrain.width, maxSegments);
    const resolutionZ = resolveResolution(options.resolutionZ, terrain.height, maxSegments);
    const positions = [];
    const colors = [];
    const indices = [];
    for (let row = 0; row <= resolutionZ; row++) {
        for (let column = 0; column <= resolutionX; column++) {
            const x = terrain.min.x + column / resolutionX * terrain.worldSizeX;
            const z = terrain.min.z + row / resolutionZ * terrain.worldSizeZ;
            const height = terrain.sampleHeight(x, z);
            if (!Number.isFinite(height)) throw new Error(`Invalid topographic height at (${x}, ${z}).`);
            positions.push(x, height, z);
            writeElevationColor(height, terrain.minHeight, terrain.maxHeight, colors);
            if (row < resolutionZ && column < resolutionX) {
                writeCell(row, column, resolutionX, indices);
            }
        }
    }
    return { positions, colors, indices, resolutionX, resolutionZ };
}

function validateTerrain(terrain) {
    if (!terrain || !Number.isFinite(terrain.worldSizeX) || !Number.isFinite(terrain.worldSizeZ) ||
        !terrain.min || !Number.isFinite(terrain.min.x) || !Number.isFinite(terrain.min.z) ||
        !Number.isFinite(terrain.minHeight) || !Number.isFinite(terrain.maxHeight) ||
        terrain.minHeight >= terrain.maxHeight || !Number.isSafeInteger(terrain.width) || terrain.width < 2 ||
        !Number.isSafeInteger(terrain.height) || terrain.height < 2 ||
        typeof terrain.sampleHeight !== 'function') {
        throw new TypeError('A valid topographic terrain sampler is required.');
    }
}

function resolveResolution(requested, sourceSize, maximum) {
    const resolution = requested ?? Math.min(maximum, sourceSize - 1);
    if (!Number.isSafeInteger(resolution) || resolution < 1 || resolution > 2048) {
        throw new RangeError('Terrain mesh resolution must be an integer between 1 and 2048.');
    }
    return Math.min(resolution, sourceSize - 1, maximum);
}

function writeElevationColor(height, minimum, maximum, colors) {
    const value = clamp((height - minimum) / (maximum - minimum), 0, 1);
    const rgb = value < 0.48
        ? [0.12 + value * 0.12, 0.3 + value * 0.45, 0.16 + value * 0.08]
        : value < 0.76
            ? [0.42 + value * 0.2, 0.38 + value * 0.12, 0.2]
            : [0.68 + value * 0.2, 0.7 + value * 0.18, 0.72 + value * 0.16];
    colors.push(rgb[0], rgb[1], rgb[2], 1);
}

function writeCell(row, column, resolutionX, indices) {
    const first = row * (resolutionX + 1) + column;
    const second = first + 1;
    const third = first + resolutionX + 1;
    const fourth = third + 1;
    indices.push(first, third, second, second, third, fourth);
}

function clamp(value, minimum, maximum) {
    return Math.min(maximum, Math.max(minimum, value));
}

export { buildTopographicTerrainBuffers, createTopographicTerrainMesh };

/** Versioned, serializable identity for one ordinary 3D cross-section. */

import {
    createCanonicalSliceFrame,
    createSliceFrame
} from './spatial-4d.js';

const WORLD_SLICE_SCHEMA_VERSION = 1;
const GEOMETRY_TYPES = new Set([
    'euclidean',
    'poincare',
    'spherical',
    'inverted-poincare'
]);

function createWorldSlice(options = {}) {
    const schemaVersion = options.schemaVersion ?? WORLD_SLICE_SCHEMA_VERSION;
    if (schemaVersion !== WORLD_SLICE_SCHEMA_VERSION) {
        throw new RangeError(`Unsupported WorldSlice schema version: ${schemaVersion}`);
    }

    const id = requireId(options.id, 'WorldSlice id');
    const w = options.w ?? 0;
    if (!Number.isFinite(w)) throw new TypeError('WorldSlice w must be finite.');

    const frame = options.frame
        ? createSliceFrame(options.frame)
        : createCanonicalSliceFrame(w);
    if (
        options.frame &&
        options.w !== undefined &&
        Math.abs(frame.origin.w - w) > 1e-9
    ) {
        throw new RangeError('WorldSlice w must match the supplied frame origin.');
    }
    const geometry = options.geometry ?? 'euclidean';
    if (!GEOMETRY_TYPES.has(geometry)) {
        throw new RangeError(`Unsupported slice geometry: ${geometry}`);
    }

    const seed = options.seed ?? 0;
    if (!(typeof seed === 'string' || Number.isSafeInteger(seed))) {
        throw new TypeError('WorldSlice seed must be a string or safe integer.');
    }

    const revision = options.revision ?? 1;
    if (!Number.isSafeInteger(revision) || revision < 1) {
        throw new RangeError('WorldSlice revision must be a positive safe integer.');
    }

    return {
        schemaVersion,
        id,
        w: frame.origin.w,
        frame,
        geometry,
        seed,
        revision
    };
}

function createSliceAddress(sliceId, chunkX, chunkZ) {
    const id = requireId(sliceId, 'Slice address id');
    assertChunkCoordinate(chunkX, 'chunkX');
    assertChunkCoordinate(chunkZ, 'chunkZ');
    return { schemaVersion: 1, sliceId: id, chunkX, chunkZ };
}

function getSliceAddressKey(address) {
    if (!address || (address.schemaVersion !== undefined && address.schemaVersion !== 1)) {
        throw new RangeError('Unsupported slice address schema version.');
    }
    const normalized = createSliceAddress(
        address.sliceId,
        address.chunkX,
        address.chunkZ
    );
    return JSON.stringify([
        normalized.sliceId,
        normalized.chunkX,
        normalized.chunkZ
    ]);
}

function requireId(value, label) {
    if (typeof value !== 'string' || value.trim() === '') {
        throw new TypeError(`${label} must be a non-empty string.`);
    }
    return value.trim();
}

function assertChunkCoordinate(value, label) {
    if (!Number.isSafeInteger(value)) {
        throw new TypeError(`${label} must be a safe integer.`);
    }
}

export {
    WORLD_SLICE_SCHEMA_VERSION,
    createSliceAddress,
    createWorldSlice,
    getSliceAddressKey
};

    }

    removeCorridor(corridorId) {
        if (!this.#corridors.delete(corridorId)) throw new Error(`Unknown corridor: ${corridorId}`);
    }

    getSlice(sliceId) {
        return cloneRecord(this.#requireSlice(sliceId));
    }

    getCorridor(corridorId) {
        const corridor = this.#corridors.get(corridorId);
        if (!corridor) throw new Error(`Unknown corridor: ${corridorId}`);
        return cloneRecord(corridor);
    }

    // Preview visibility never grants simulation or traversal permission.
    getVisibleDestinations(sliceId) {
        this.#requireSlice(sliceId);
        return this.#findCorridors(sliceId)
            .filter(function isVisible(link) {
                return link.corridor.visible;
            })
            .map(function toVisibleDestination(link) {
                return { corridorId: link.corridor.id, sliceId: link.toSliceId, renderOnly: true };
            });
    }

    // One-way links permit travel only from their declared source.
    getTraversableNeighbors(sliceId) {
        this.#requireSlice(sliceId);
        return this.#findCorridors(sliceId)
            .filter(function isTraversable(link) {
                return link.corridor.traversable && (
                    link.fromSource || link.corridor.bidirectional
                );
            })
            .map(function toTraversableNeighbor(link) {
                return { corridorId: link.corridor.id, sliceId: link.toSliceId };
            });
    }

    hasTraversableConnection(fromSliceId, toSliceId) {
        return this.getTraversableNeighbors(fromSliceId).some(function matchesDestination(link) {
            return link.sliceId === toSliceId;
        });
    }

    // Optical topology is deliberately independent from player traversal.
    // An anomaly may transmit light without being physically traversable.
    getOpticallyConnectedNeighbors(sliceId) {
        this.#requireSlice(sliceId);
        return this.#findCorridors(sliceId)
            .filter(function isOpticallyConnected(link) {
                const optics = link.corridor.optics;
                return link.corridor.type === SLICE_TRANSITION_ANOMALY &&
                    optics?.enabled === true && (link.fromSource || optics.bidirectional === true);
            })
            .map(function describeOpticalNeighbor(link) {
                return {
                    corridorId: link.corridor.id,
                    sliceId: link.toSliceId,
                    entrySide: link.fromSource ? 'source' : 'destination',
                    fromSource: link.fromSource
                };
            });
    }

    hasOpticalConnection(fromSliceId, toSliceId) {
        return this.getOpticallyConnectedNeighbors(fromSliceId).some(function matchesOpticalDestination(link) {
            return link.sliceId === toSliceId;
        });
    }

    toJSON() {
        return {
            schemaVersion: WORLD_TOPOLOGY_SCHEMA_VERSION,
            slices: Array.from(this.#slices.values(), cloneRecord),
            corridors: Array.from(this.#corridors.values(), cloneRecord)
        };
    }

    static fromJSON(data) {
        if (!data || data.schemaVersion !== WORLD_TOPOLOGY_SCHEMA_VERSION) {
            throw new RangeError(
                `Unsupported WorldTopology schema version: ${data?.schemaVersion}`
            );
        }
        if (!Array.isArray(data.slices) || !Array.isArray(data.corridors)) {
            throw new TypeError('WorldTopology data requires slice and corridor arrays.');
        }
        const topology = new WorldTopology();
        for (const slice of data.slices) topology.registerSlice(slice);
        for (const corridor of data.corridors) topology.registerCorridor(corridor);
        return topology;
    }

    #requireSlice(sliceId) {
        const slice = this.#slices.get(sliceId);
        if (!slice) throw new Error(`Unknown slice: ${sliceId}`);
        return slice;
    }

    #findCorridors(sliceId) {
        const links = [];
        for (const corridor of this.#corridors.values()) {
            if (corridor.source.sliceId === sliceId) {
                links.push({
                    corridor,
                    fromSource: true,
                    toSliceId: corridor.destination.sliceId
                });
            } else if (corridor.destination.sliceId === sliceId) {
                links.push({
                    corridor,
                    fromSource: false,
                    toSliceId: corridor.source.sliceId
                });
            }
        }
        return links;
    }

    #resolveInfluence(corridor) {
        if (corridor.type !== SLICE_TRANSITION_ANOMALY) return corridor;
        const source = this.#requireSlice(corridor.source.sliceId);
        const destination = this.#requireSlice(corridor.destination.sliceId);
        const sourceAmbient = sliceToAmbient(source.frame, corridor.source.position);
        const destinationAmbient = sliceToAmbient(destination.frame, corridor.destination.position);
        return resolveAnomalyInfluence(corridor, sourceAmbient, destinationAmbient, source.frame.normal);
    }

    #validateEndpoints(corridor) {
        const source = this.#requireSlice(corridor.source.sliceId);
        const destination = this.#requireSlice(corridor.destination.sliceId);
        const sourcePoint = sliceToAmbient(source.frame, corridor.source.position);
        const destinationPoint = sliceToAmbient(destination.frame, corridor.destination.position);
        if (corridor.type === BEZIER_CORRIDOR_ANOMALY) {
            const sourceError = norm4(subtract4(corridor.centerline[0], sourcePoint));
            const destinationError = norm4(subtract4(corridor.centerline[3], destinationPoint));
            if (sourceError > this.#endpointTolerance || destinationError > this.#endpointTolerance) {
                throw new RangeError('Corridor centreline endpoints must match their slice endpoints.');
            }
            return;
        }

        if (corridor.type !== SLICE_TRANSITION_ANOMALY) {
            throw new RangeError(`Unsupported topology anomaly type: ${corridor.type}`);
        }
        if (!containsAmbientPoint(corridor.region, sourcePoint, this.#endpointTolerance) ||
            !containsAmbientPoint(corridor.region, destinationPoint, this.#endpointTolerance)) {
            throw new RangeError('Slice-transition anomaly bounds must contain both slice endpoints.');
        }
        const normalAlignment = Math.abs(
            source.frame.normal.x * destination.frame.normal.x +
            source.frame.normal.y * destination.frame.normal.y +
            source.frame.normal.z * destination.frame.normal.z +
            source.frame.normal.w * destination.frame.normal.w
        );
        if (normalAlignment < 1 - 1e-7) {
            throw new RangeError('Slice-transition anomalies require parallel or anti-parallel slice normals.');
        }
        const separation = Math.abs(dot4(
            subtract4(destination.frame.origin, source.frame.origin),
            source.frame.normal
        ));
        if (separation <= this.#endpointTolerance) {
            throw new RangeError('Slice-transition anomalies must connect distinct 3D slice planes.');
        }
    }
}

function cloneRecord(record) {
    return JSON.parse(JSON.stringify(record));
}

export { WORLD_TOPOLOGY_SCHEMA_VERSION, WorldTopology };
