import { WorldState } from "./world-state.js";

const DATABASE_NAME = "somnium-worlds";
const DATABASE_VERSION = 1;
const STORE_NAME = "worlds";

class WorldPersistence {
    constructor() {
        this.database = null;
        this.opening = null;
    }

    async open() {
        if (this.database) {
            return this.database;
        }

        if (this.opening) {
            return this.opening;
        }

        if (!globalThis.indexedDB) {
            throw new Error(
                "IndexedDB is unavailable in this browser."
            );
        }

        this.opening = new Promise((resolve, reject) => {
            const request = indexedDB.open(
                DATABASE_NAME,
                DATABASE_VERSION
            );

            request.onupgradeneeded = () => {
                const database = request.result;

                if (!database.objectStoreNames.contains(STORE_NAME)) {
                    database.createObjectStore(STORE_NAME, {
                        keyPath: "metadata.id"
                    });
                }
            };

            request.onsuccess = () => {
                this.database = request.result;

                this.database.onversionchange = () => {
                    this.database.close();
                    this.database = null;
                };

                resolve(this.database);
            };

            request.onerror = () => {
                reject(request.error);
            };

            request.onblocked = () => {
                console.warn(
                    "[WorldPersistence] Database upgrade blocked."
                );
            };
        });

        try {
            return await this.opening;
        } finally {
            this.opening = null;
        }
    }

    async transaction(mode, operation) {
        const database = await this.open();

        return new Promise((resolve, reject) => {
            const tx = database.transaction(STORE_NAME, mode);
            const store = tx.objectStore(STORE_NAME);

            let result;

            try {
                result = operation(store, tx);
            } catch (error) {
                tx.abort();
                reject(error);
                return;
            }

            tx.oncomplete = () => resolve(result);
            tx.onerror = () => reject(tx.error);
            tx.onabort = () => reject(
                tx.error ?? new Error("Transaction aborted.")
            );
        });
    }

    async createWorld(name, options = {}) {
        const cleanName = String(name ?? "").trim();

        if (!cleanName) {
            throw new Error("Please provide a world name.");
        }

        const world = new WorldState({
            ...options,
            name: cleanName
        });

        world.validate();

        await this.transaction("readwrite", store => {
            store.add(world.serialize());
        });

        console.log(
            "[WorldPersistence] Created world:",
            world.metadata.name
        );

        return world;
    }

    async listWorlds() {
        const records = await new Promise(async (resolve, reject) => {
            try {
                const database = await this.open();
                const tx = database.transaction(STORE_NAME, "readonly");
                const request = tx.objectStore(STORE_NAME).getAll();

                request.onsuccess = () => resolve(request.result);
                request.onerror = () => reject(request.error);
            } catch (error) {
                reject(error);
            }
        });

        return records
            .map(record => ({
                id: record.metadata.id,
                name: record.metadata.name,
                createdAt: record.metadata.createdAt,
                updatedAt: record.metadata.updatedAt,
                revision: record.metadata.revision,
                worldSeed: record.generation.worldSeed
            }))
            .sort((a, b) =>
                b.updatedAt.localeCompare(a.updatedAt)
            );
    }

    async loadWorld(id) {
        const record = await new Promise(async (resolve, reject) => {
            try {
                const database = await this.open();
                const tx = database.transaction(STORE_NAME, "readonly");
                const request = tx.objectStore(STORE_NAME).get(id);

                request.onsuccess = () => resolve(request.result);
                request.onerror = () => reject(request.error);
            } catch (error) {
                reject(error);
            }
        });

        if (!record) {
            throw new Error(`World not found: ${id}`);
        }

        return WorldState.deserialize(record);
    }

    async saveWorld(world, options = {}) {
        if (!(world instanceof WorldState)) {
            throw new TypeError("Expected a WorldState instance.");
        }

        world.validate();

        const serialized = world.serialize();

        await this.transaction("readwrite", (store, tx) => {
            const request = store.get(world.metadata.id);

            request.onsuccess = () => {
                const existing = request.result;

                if (!existing) {
                    if (!options.allowCreate) {
                        tx.abort();
                        return;
                    }

                    world.touch();
                    store.add(world.serialize());
                    return;
                }

                // Prevent overwriting changes from another tab.
                if (
                    existing.metadata.revision !==
                    world.metadata.revision
                ) {
                    tx.abort();
                    return;
                }

                world.touch();
                store.put(world.serialize());
            };
        });

        return world;
    }

    async renameWorld(id, newName) {
        const world = await this.loadWorld(id);
        const cleanName = String(newName ?? "").trim();

        if (!cleanName) {
            throw new Error("World name cannot be empty.");
        }

        world.metadata.name = cleanName;

        return this.saveWorld(world);
    }

    async deleteWorld(id) {
        await this.transaction("readwrite", store => {
            store.delete(id);
        });

        console.log("[WorldPersistence] Deleted world:", id);
    }

    close() {
        this.database?.close();
        this.database = null;
    }
}

export { WorldPersistence };

import { WorldState } from "./world-state.js";

class WorldSession {
    constructor(ecology) {
        if (!ecology?.populationManager) {
            throw new Error(
                "WorldSession requires ChunkEcologySystem."
            );
        }

        this.ecology = ecology;
    }

    capture(world, elapsedTime = 0) {
        if (!(world instanceof WorldState)) {
            throw new TypeError("Expected WorldState.");
        }

        const manager = this.ecology.populationManager;

        const chunkRecords = [];

        for (const record of manager.records.values()) {
            const exported = manager.exportRecord(
                record.x,
                record.z
            );

            if (exported) {
                chunkRecords.push(exported);
            }
        }

        world.generation.worldSeed = manager.worldSeed;

        world.simulation.elapsedTime = elapsedTime;

        world.ecology.chunkRecords = chunkRecords;


        return world;
    }

    restore(world) {
        if (!(world instanceof WorldState)) {
            throw new TypeError("Expected WorldState.");
        }

        const ecology = this.ecology;
        const manager = ecology.populationManager;

        // Deactivate currently rendered populations.
        for (const record of manager.records.values()) {
            if (record.active) {
                manager.deactivateRecord(record);
            }
        }

        manager.records.clear();
        manager.pending.length = 0;
        manager.activeKeys.clear();
        manager.simulationCursor = 0;

        ecology.worldSeed = world.generation.worldSeed;
        manager.worldSeed = world.generation.worldSeed;

        // Restore all previously generated populations.
        for (const record of world.ecology.chunkRecords) {
            manager.importRecord(record);
        }

        ecology.renderDirty = true;

        return {
            elapsedTime: world.simulation.elapsedTime,
            worldSeed: world.generation.worldSeed,
            restoredChunks: world.ecology.chunkRecords.length
        };
    }

    getStats() {
        return {
            worldSeed: this.ecology.populationManager.worldSeed,
            generatedChunks:
                this.ecology.populationManager.records.size
        };
    }
}

export { WorldSession };

const WORLD_SCHEMA_VERSION = 1;

function createId() {
    if (globalThis.crypto?.randomUUID) {
        return globalThis.crypto.randomUUID();
    }

    return `world-${Date.now()}-${Math.random()
        .toString(36).slice(2)}`;
}

function cloneData(value) {
    return JSON.parse(JSON.stringify(value));
}

class WorldState {
    constructor(options = {}) {
        const now = new Date().toISOString();

        this.schemaVersion = WORLD_SCHEMA_VERSION;

        this.metadata = {
            id: options.id ?? createId(),
            name: options.name ?? "Untitled World",
            createdAt: options.createdAt ?? now,
            updatedAt: now,
            revision: options.revision ?? 0
        };

        this.generation = {
            worldSeed: options.worldSeed ?? 91827,
            generatorVersion: options.generatorVersion ?? 1,
            configuration: options.configuration ?? {}
        };

        this.simulation = {
            elapsedTime: options.elapsedTime ?? 0,
            simulationRate: options.simulationRate ?? 1,
            paused: options.paused ?? false
        };

        this.ecology = {
            chunkRecords: options.chunkRecords ?? []
        };

        this.entities = {
            manual: options.manualEntities ?? [],
            overrides: options.entityOverrides ?? [],
            removedProceduralIds: options.removedProceduralIds ?? []
        };

        this.environment = {
            state: options.environmentState ?? {}
        };
    }

    validate() {
        if (!this.metadata.id || !this.metadata.name.trim()) {
            throw new Error("World requires an ID and name.");
        }

        if (!Number.isInteger(this.generation.worldSeed)) {
            throw new Error("Invalid world seed.");
        }

        if (
            !Number.isFinite(this.simulation.elapsedTime) ||
            this.simulation.elapsedTime < 0
        ) {
            throw new Error("Invalid simulation time.");
        }

        if (!Array.isArray(this.ecology.chunkRecords)) {
            throw new Error("Invalid ecology records.");
        }

        if (
            !Array.isArray(this.entities.manual) ||
            !Array.isArray(this.entities.overrides) ||
            !Array.isArray(this.entities.removedProceduralIds)
        ) {
            throw new Error("Invalid entity collections.");
        }

        return true;
    }

    touch() {
        this.metadata.updatedAt = new Date().toISOString();
        this.metadata.revision++;
    }

    serialize() {
        this.validate();

        return cloneData(this);
    }

    static deserialize(data) {
        if (!data || typeof data !== "object") {
            throw new Error("Invalid world data.");
        }

        if (data.schemaVersion !== WORLD_SCHEMA_VERSION) {
            throw new Error(
                `Unsupported world schema: ${data.schemaVersion}`
            );
        }

        const state = new WorldState({
            id: data.metadata?.id,
            name: data.metadata?.name,
            createdAt: data.metadata?.createdAt,
            revision: data.metadata?.revision,
            worldSeed: data.generation?.worldSeed,
            generatorVersion: data.generation?.generatorVersion,
            configuration: data.generation?.configuration,
            elapsedTime: data.simulation?.elapsedTime,
            simulationRate: data.simulation?.simulationRate,
            paused: data.simulation?.paused,
            chunkRecords: data.ecology?.chunkRecords,
            manualEntities: data.entities?.manual,
            entityOverrides: data.entities?.overrides,
            removedProceduralIds:
                data.entities?.removedProceduralIds,
            environmentState: data.environment?.state
        });

        state.metadata.updatedAt =
            data.metadata.updatedAt ?? state.metadata.updatedAt;

        state.validate();

        return state;
    }
}

export {
    WorldState,
    WORLD_SCHEMA_VERSION
};
