class BuildingGenerator {
    constructor(options = {}) {
        this.defaultWallHeight = options.wallHeight ?? 3.2;
        this.defaultWallThickness = options.wallThickness ?? 0.18;
    }

    createBox(output, kind, position, dimensions, rotation = 0, material = "wall") {
        if (
            dimensions.x <= 0 ||
            dimensions.y <= 0 ||
            dimensions.z <= 0
        ) return;

        output.push({
            kind,
            position,
            dimensions,
            rotation,
            material
        });
    }

    getRoomEdges(room) {
        const b = room.bounds;

        return [
            {
                side: "north",
                x1: b.minX, z1: b.minZ,
                x2: b.maxX, z2: b.minZ
            },
            {
                side: "south",
                x1: b.minX, z1: b.maxZ,
                x2: b.maxX, z2: b.maxZ
            },
            {
                side: "west",
                x1: b.minX, z1: b.minZ,
                x2: b.minX, z2: b.maxZ
            },
            {
                side: "east",
                x1: b.maxX, z1: b.minZ,
                x2: b.maxX, z2: b.maxZ
            }
        ];
    }

    edgeKey(edge, floor) {
        const a = `${edge.x1},${edge.z1}`;
        const b = `${edge.x2},${edge.z2}`;

        return [
            floor,
            ...[a, b].sort()
        ].join(":");
    }

    generateWall(output, edge, openings, floorY, height, thickness) {
        const horizontal = edge.side === "north" ||
            edge.side === "south";

        const length = horizontal
            ? Math.abs(edge.x2 - edge.x1)
            : Math.abs(edge.z2 - edge.z1);

        const centerZ = (edge.z1 + edge.z2) * 0.5;

        const validOpenings = openings
            .filter(o => o.side === edge.side)
            .map(o => ({
                ...o,
                center: o.offset * length
            }))
            .filter(o =>
                o.center - o.width * 0.5 >= 0 &&
                o.center + o.width * 0.5 <= length
            )
            .sort((a, b) => a.center - b.center);

        let cursor = 0;

            if (end - start <= 0.01) return;

            const middle = (start + end) * 0.5;
            const along = middle - length * 0.5;

            this.createBox(
                output,
                "walls",
                {
                    x: centerX + (horizontal ? along : 0),
                    y: centerY,
                    z: centerZ + (horizontal ? 0 : along)

                    x: horizontal ? end - start : thickness,
                    y: height,
                    z: horizontal ? thickness : end - start

                0,
                "wall"

                "ceiling"

            for (const edge of this.getRoomEdges(room)) {

                if (!edges.has(key)) {
                    edges.set(key, {
                        edge,
                        openings: [],
                        floorY,
                        height: room.wallHeight,
                        thickness: room.wallThickness
                    });
                }

		edges.get(key).openings.push(
    ...(room.openings ?? [])
        .filter(o => o.side === edge.side)
        .map(o => ({
            ...o,
            side: edges.get(key).edge.side
        }))
);

            }
        }

        for (const item of edges.values()) {
            this.generateWall(
                output,
                item.edge,
                item.openings,
                item.floorY,
                item.height,
                item.thickness
            );

            const kind = component.kind === "furniture"

                    ? "fixtures"
                    : component.kind;

            this.createBox(

                kind,
                component.position,
                component.dimensions,
                component.rotation,
                component.material

        }

    }

function createId(prefix = "building") {
    return `${prefix}-${globalThis.crypto?.randomUUID?.() ??
        `${Date.now()}-${Math.random().toString(36).slice(2)}`}`;
}

function clone(value) {
    return JSON.parse(JSON.stringify(value));
}

class BuildingModel {
    constructor(options = {}) {
        this.id = options.id ?? createId();

        this.name = options.name ?? "New Building";
        this.type = options.type ?? "residential";

        this.position = {
            x: options.position?.x ?? 0,
            y: options.position?.y ?? 0,
            z: options.position?.z ?? 0
        };

        this.rotation = options.rotation ?? 0;

        this.rooms = clone(options.rooms ?? []);
        this.furniture = clone(options.furniture ?? []);
        this.fixtures = clone(options.fixtures ?? []);

        // Independently placed architectural elements.
        this.components = clone(options.components ?? []);

        this.metadata = {
            version: 1,
            createdAt: options.metadata?.createdAt ??
                new Date().toISOString()
        };
    }

    addRoom(options = {}) {
        const room = {
            id: createId("room"),
            name: options.name ?? "Room",
            floor: options.floor ?? 0,

            bounds: {
                minX: options.minX ?? 0,
                maxX: options.maxX ?? 8,
                minZ: options.minZ ?? 0,
                maxZ: options.maxZ ?? 8
            },

            wallHeight: options.wallHeight ?? 3.2,
            wallThickness: options.wallThickness ?? 0.18,

            floorThickness: options.floorThickness ?? 0.15,

            openings: clone(options.openings ?? [])
        };

        this.rooms.push(room);
        return room;
    }

    addOpening(roomId, options = {}) {
        const room = this.rooms.find(r => r.id === roomId);

        if (!room) {
            throw new Error("Room not found.");
        }

        const opening = {
            id: createId("opening"),
            type: options.type ?? "door",
            side: options.side ?? "north",
            offset: options.offset ?? 0.5,
            width: options.width ?? 0.9,
            height: options.height ?? 2.1,
            sill: options.sill ?? 0
        };

        room.openings.push(opening);
        return opening;
    }

    addComponent(options = {}) {
        const component = {
            id: createId("component"),
            kind: options.kind ?? "furniture",
            type: options.type ?? "custom",

            position: {
                x: options.position?.x ?? 0,
                y: options.position?.y ?? 0,
                z: options.position?.z ?? 0
            },

            dimensions: {
                x: options.dimensions?.x ?? 1,
                y: options.dimensions?.y ?? 1,
                z: options.dimensions?.z ?? 1
            },

            rotation: options.rotation ?? 0,
            material: options.material ?? "wood"
        };

        this.components.push(component);
        return component;
    }

    addFurniture(options = {}) {
        return this.addComponent({
            ...options,
            kind: "furniture"
        });
    }

    validate() {
        if (!this.id || !Array.isArray(this.rooms)) {
            throw new Error("Invalid building.");
        }

        for (const room of this.rooms) {
            const b = room.bounds;

            if (
                !b ||
                ![b.minX, b.maxX, b.minZ, b.maxZ].every(Number.isFinite) ||
                b.maxX <= b.minX ||
                b.maxZ <= b.minZ ||
                !Number.isFinite(room.floor) ||
                !Number.isFinite(room.wallHeight) ||
                room.wallHeight <= 0
            ) {
                throw new Error(`Invalid room: ${room.id}`);
            }

            for (const opening of room.openings ?? []) {
                if (
                    !["north", "south", "east", "west"].includes(opening.side) ||
                    !["door", "window"].includes(opening.type) ||
                    !Number.isFinite(opening.offset) ||
                    opening.offset < 0 ||
                    opening.offset > 1 ||
                    !Number.isFinite(opening.width) ||
                    opening.width <= 0 ||
                    !Number.isFinite(opening.height) ||
                    opening.height <= 0 ||
                    !Number.isFinite(opening.sill) ||
                    opening.sill < 0 ||
                    opening.sill + opening.height > room.wallHeight
                ) {
                    throw new Error(`Invalid opening in room ${room.id}`);
                }
            }
        }

        return true;
    }

    serialize() {
        this.validate();

        return {
            id: this.id,
            type: "building",
            version: 1,
            name: this.name,
            buildingType: this.type,
            position: clone(this.position),
            rotation: this.rotation,
            rooms: clone(this.rooms),
            furniture: clone(this.furniture),
            fixtures: clone(this.fixtures),
            components: clone(this.components),
            metadata: clone(this.metadata)
        };
    }

    static deserialize(data) {
        if (!data || data.type !== "building" ||
            ![1].includes(data.version)) {
            throw new Error("Unsupported building record.");
        }

        const building = new BuildingModel({
            id: data.id,
            name: data.name,
            type: data.buildingType,
            position: data.position,
            rotation: data.rotation,
            rooms: data.rooms,
            furniture: data.furniture,
            fixtures: data.fixtures,
            components: data.components,
            metadata: data.metadata
        });

        building.validate();
        return building;
    }
}

import {
    ThinInstanceRenderer
} from "../rendering/thin-instance-renderer.js";

const MATERIALS = {
    wall: [0.72, 0.69, 0.62],
    floor: [0.48, 0.34, 0.22],
    ceiling: [0.83, 0.81, 0.75],
    glass: [0.35, 0.70, 0.86],
    wood: [0.34, 0.20, 0.11],
    furniture: [0.48, 0.31, 0.20]
};

const BUCKETS = [
    "walls",
    "floors",
    "ceilings",
    "glazing",
    "trim",
    "doors",
    "furniture",
    "fixtures"
];

/**
 * BuildingRenderer
 *
 * Converts authoritative architectural components into
 * thin-instance rendering batches.
 *
 * IMPORTANT:
 * This class never applies SpatialTransform directly.
 * ThinInstanceRenderer delegates projection to the
 * existing GeometryRenderer.
 */
class BuildingRenderer {
    constructor(scene, spatialTransform = null) {
        if (!scene) {
            throw new Error(
                "BuildingRenderer requires a scene."
            );
        }

        this.scene = scene;

        // Retained for compatibility with existing integration.
        // Projection ownership belongs to GeometryRenderer.
        this.transform = spatialTransform;

        this.buckets = new Map();
        this.disposed = false;

        this.statistics = {
            submittedComponents: 0,
            rejectedComponents: 0,
            renderedInstances: 0
        };

        for (const kind of BUCKETS) {
            this.createBucket(kind);
        }
    }

    createBucket(kind) {
        const mesh = BABYLON.MeshBuilder.CreateBox(
            `building-prototype-${kind}`,
            { size: 1 },
            this.scene
        );

        const material = new BABYLON.StandardMaterial(
            `building-material-${kind}`,
            this.scene
        );

        const colorName =
            kind === "walls" ? "wall" :
            kind === "floors" ? "floor" :
            kind === "ceilings" ? "ceiling" :
            kind === "glazing" ? "glass" :
            kind === "trim" || kind === "doors" ? "wood" :
            "furniture";

        material.diffuseColor = new BABYLON.Color3(
            ...MATERIALS[colorName]
        );

        if (kind === "glazing") {
            material.alpha = 0.42;

            material.transparencyMode =
                BABYLON.Material.MATERIAL_ALPHABLEND;

            material.backFaceCulling = false;

            material.specularColor =
                new BABYLON.Color3(0.8, 0.9, 1);
        }

        mesh.material = material;

        const renderer = new ThinInstanceRenderer(
            this.scene,
            `building-${kind}`,
            mesh
        );

        this.buckets.set(kind, {
            mesh,
            material,
            renderer
        });
    }

    clear() {
        for (const bucket of this.buckets.values()) {
            bucket.renderer.clear();
        }

        this.statistics.submittedComponents = 0;
        this.statistics.rejectedComponents = 0;
        this.statistics.renderedInstances = 0;
    }

    addComponent(component, building) {
        if (
            !component ||
            !building ||
            !component.position ||
            !component.dimensions
        ) {
            this.statistics.rejectedComponents++;
            return false;
        }

        const bucket = this.buckets.get(component.kind);

        if (!bucket) {
            this.statistics.rejectedComponents++;
            return false;
        }

        const values = [
            component.position.x,
            component.position.y,
            component.position.z,
            component.dimensions.x,
            component.dimensions.y,
            component.dimensions.z,
            building.position.x,
            building.position.y,
            building.position.z
        ];

        if (
            !values.every(Number.isFinite) ||
            values.slice(3, 6).some(value => value <= 0)
        ) {
            this.statistics.rejectedComponents++;
            return false;
        }

        const angle = building.rotation ?? 0;

        if (!Number.isFinite(angle)) {
            this.statistics.rejectedComponents++;
            return false;
        }

        const cos = Math.cos(angle);
        const sin = Math.sin(angle);

        const local = component.position;

        const rotated = {
            x: local.x * cos - local.z * sin,
            y: local.y,
            z: local.x * sin + local.z * cos
        };

        // Authoritative world coordinates.
        // Do not project this position here.
        const world = {
            x: building.position.x + rotated.x,
            y: building.position.y + rotated.y,
            z: building.position.z + rotated.z
        };

        const rotation = BABYLON.Quaternion.RotationAxis(
            BABYLON.Axis.Y,
            angle + (component.rotation ?? 0)
        );

        // Dimensions remain in authoritative units.
        // GeometryRenderer applies the spatial projection.
        const scale = new BABYLON.Vector3(
            component.dimensions.x,
            component.dimensions.y,
            component.dimensions.z
        );

        const matrix = BABYLON.Matrix.Compose(
            scale,
            rotation,
            new BABYLON.Vector3(
                world.x,
                world.y,
                world.z
            )
        );

        bucket.renderer.addMatrix(matrix);

        this.statistics.submittedComponents++;

        return true;
    }

    update(buildings) {
        if (this.disposed) return;

        this.clear();

        for (const building of buildings) {
            if (!Array.isArray(building.generatedGeometry)) {
                continue;
            }

            for (const component of building.generatedGeometry) {
                this.addComponent(component, building);
            }
        }

        for (const bucket of this.buckets.values()) {
            bucket.renderer.update();

            this.statistics.renderedInstances +=
                bucket.renderer.matrices.length;
        }
    }

    getStats() {
        const result = {};

        for (const [kind, bucket] of this.buckets) {
            result[kind] = {
                instances: bucket.renderer.matrices.length
            };
        }

        return {
            buckets: result,
            submittedComponents:
                this.statistics.submittedComponents,
            rejectedComponents:
                this.statistics.rejectedComponents,
            renderedInstances:
                this.statistics.renderedInstances
        };
    }

    dispose() {
        if (this.disposed) return;

        this.disposed = true;

        for (const bucket of this.buckets.values()) {
            bucket.renderer.dispose?.();
            bucket.mesh.dispose();
            bucket.material.dispose();
        }

        this.buckets.clear();
    }
}

export { BuildingRenderer };

import { BuildingModel } from "./building-model.js";
import { BuildingGenerator } from "./building-generator.js";
import { BuildingRenderer } from "./building-renderer.js";

class BuildingSystem {
    constructor(scene, geometryRenderer, options = {}) {
        this.scene = scene;
        this.geometryRenderer = geometryRenderer;

        this.generator = new BuildingGenerator();

        this.renderer = new BuildingRenderer(
            scene,
            options.spatialTransform ?? null
        );

        this.buildings = new Map();
        this.disposed = false;
        this.terrain = options.terrain ?? null;
    }

    createBuilding(options = {}) {
        const building = new BuildingModel(options);

        building.validate();

        this.buildings.set(building.id, building);
        this.rebuild(building);

        return building;
    }

    addBuilding(building) {
        if (!(building instanceof BuildingModel)) {
            building = BuildingModel.deserialize(building);
        }

        building.validate();
        this.buildings.set(building.id, building);
        this.rebuild(building);

        return building;
    }

    removeBuilding(id) {
        const removed = this.buildings.delete(id);

        if (removed) this.refresh();

        return removed;
    }

    getBuilding(id) {
        return this.buildings.get(id) ?? null;
    }

    rebuild(building) {
        building.validate();

        building.generatedGeometry =
            this.generator.generate(building);

        this.refresh();
    }

    refresh() {
        if (this.disposed) return;

        this.renderer.update(
            Array.from(this.buildings.values())
        );
    }

    createRoom(buildingId, options = {}) {
        const building = this.getBuilding(buildingId);

        if (!building) {
            throw new Error("Building not found.");
        }

        const room = building.addRoom(options);

        this.rebuild(building);

        return room;
    }

    addOpening(buildingId, roomId, options = {}) {
        const building = this.getBuilding(buildingId);

        if (!building) {
            throw new Error("Building not found.");
        }

        const opening = building.addOpening(
            roomId,
            options
        );

        this.rebuild(building);

        return opening;
    }

    addFurniture(buildingId, options = {}) {
        const building = this.getBuilding(buildingId);

        if (!building) {
            throw new Error("Building not found.");
        }

        const furniture = building.addFurniture(options);

        this.rebuild(building);

        return furniture;
    }

    captureWorld(world) {
        if (!world?.entities) {
            throw new Error("Invalid world state.");
        }

        const other = world.entities.manual.filter(
            entity => entity.type !== "building"
        );

        const committed = Array.from(
            this.buildings.values()
        ).filter(
            building =>
                building.metadata?.editorPreview !== true
        );

        world.entities.manual = [
            ...other,
            ...committed.map(
                building => building.serialize()
            )
        ];

        return world;
    }

    restoreWorld(world) {
        if (
            !world?.entities ||
            !Array.isArray(world.entities.manual)
        ) {
            throw new Error(
                "Invalid world entity collection."
            );
        }

        this.buildings.clear();

        for (const entity of world.entities.manual) {
            if (entity.type === "building") {
                this.addBuilding(entity);
            }
        }

        this.refresh();
    }

    getStats() {
        const buildings = Array.from(
            this.buildings.values()
        );

        return {
            buildings: buildings.length,
            provisional: buildings.filter(
                building =>
                    building.metadata?.editorPreview === true
            ).length,
            committed: buildings.filter(
                building =>
                    building.metadata?.editorPreview !== true
            ).length,
            rooms: buildings.reduce(
                (sum, building) =>
                    sum + building.rooms.length,
                0
            ),
            instances: this.renderer.getStats()
        };
    }

    dispose() {
        if (this.disposed) return;

        this.disposed = true;
        this.buildings.clear();
        this.renderer.dispose();
    }
}

export { BuildingSystem };
