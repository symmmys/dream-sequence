class Environment {

    constructor(lightDirection) {

        this.sunDirection = lightDirection
            .negate()
            .normalize();

        // Environmental lighting configuration
        this.baseLight = 1.0;

        this.minimumLight = 0.12;

        // How strongly surrounding trees reduce light
        this.shadingStrength = 0.85;

        // Spatial grid for efficient canopy sampling
        this.canopyCellSize = 10;

        this.canopyGrid = new Map();
    }


    getCellKey(x, z) {

        const cx = Math.floor(x / this.canopyCellSize);
        const cz = Math.floor(z / this.canopyCellSize);

        return `${cx},${cz}`;
    }


    rebuildCanopyGrid(trees) {

        this.canopyGrid.clear();

        for (const tree of trees) {

            if (!tree.alive) continue;

            const age = tree.ageFactor;

            if (age < 0.1) continue;

            const height =
                tree.species.trunkLength *
                age *
                tree.genome.maxHeight;

            const radius =
                height *
                tree.species.canopy.width *
                tree.genome.canopyWidth;

            const key = this.getCellKey(
                tree.position.x,
                tree.position.z
            );

            if (!this.canopyGrid.has(key)) {
                this.canopyGrid.set(key, []);
            }

            this.canopyGrid.get(key).push({
                x: tree.position.x,
                z: tree.position.z,
                radius,
                height
            });

        }

    }

sampleLight(position) {

    const key = this.getCellKey(
        position.x,
        position.z
    );

    const cell = this.canopyGrid.get(key) || [];

    let shade = 0;

    for (const canopy of cell) {

        // Ignore canopies that are below the sample
        if (canopy.height <= position.y) continue;

        const dx = position.x - canopy.x;
        const dz = position.z - canopy.z;

        const distanceSquared =
            dx * dx + dz * dz;

        const radiusSquared =
            canopy.radius * canopy.radius;

        if (distanceSquared < radiusSquared) {

            const influence =
                1 - Math.sqrt(
                    distanceSquared / radiusSquared
                );

            shade += influence *
                this.shadingStrength *
                Math.min(1, canopy.height / 5);

        }

    }

    // Start with generous available light
    const intensity = BABYLON.Scalar.Clamp(
        1.0 - shade * 0.35,
        0.15,
        1.0
    );

    return {
        direction: this.sunDirection,
        intensity
    };

}

}

/** Compatibility terrain adapter for Babylon gameplay and chunk systems. */
class HeightmapTerrain {
    constructor(options = {}) {
        this.url = options.url ?? './assets/0_somnium_map_topographic.png';
        this.fallbackUrl = options.fallbackUrl ?? null;
        this.loadedSource = null;
        this.worldSize = options.worldSize ?? 1000000;
        this.minHeight = options.minHeight ?? -2000;
        this.maxHeight = options.maxHeight ?? 1000;
        this.heightChannel = options.heightChannel ?? 'average';
        this.min = { x: -this.worldSize / 2, y: this.minHeight, z: -this.worldSize / 2 };
        this.max = { x: this.worldSize / 2, y: this.maxHeight, z: this.worldSize / 2 };
        this.width = 0;
        this.height = 0;
        this.pixels = null;
        this.mapData = null;
        this.ready = false;
        this.diagnostics = null;
        this.loader = options.loader ?? new BrowserHeightmapLoader();
    }

    async load() {
        let mapData;
        this.loadedSource = this.url;
        try {
            const raster = await this.loader.load(this.url);
            assertCompleteRaster(raster, Boolean(this.fallbackUrl));
            mapData = this.createMapData(raster);
        } catch (error) {
            if (!this.fallbackUrl || this.fallbackUrl === this.url) throw error;
            console.warn(`[HeightmapTerrain] Primary map failed validation; loading fallback ${this.fallbackUrl}.`, error);
            mapData = this.createMapData(await this.loader.load(this.fallbackUrl));
            this.loadedSource = this.fallbackUrl;
        }
        this.mapData = mapData;
        this.width = this.mapData.width;
        this.height = this.mapData.height;
        this.pixels = this.mapData.pixels;
        this.ready = true;
        this.diagnostics = this.getDiagnostics(32);
        console.log('[HeightmapTerrain] Loaded:', this.diagnostics);
        return this;
    }

    createMapData(raster) {
        return createHeightmapData({
            ...raster,
            channel: this.heightChannel,
            minHeight: this.minHeight,
            maxHeight: this.maxHeight
        });
    }

    getNormalizedCoordinates(x, z) {
        return { u: (x - this.min.x) / this.worldSize, v: (z - this.min.z) / this.worldSize };
    }

    contains(x, z) {
        return x >= this.min.x && x <= this.max.x && z >= this.min.z && z <= this.max.z;
    }

    getPixelValue(x, z) {
        this.assertReady();
        const column = clamp(Math.floor(x), 0, this.width - 1);
        const row = clamp(Math.floor(z), 0, this.height - 1);
        return decodeHeightmapPixel(this.mapData, column, row);
    }

    getHeightPixel(x, z) { return this.getPixelValue(x, z); }

    sampleHeight(x, z) {
        this.assertReady();
        if (!this.contains(x, z)) return null;
        const coordinates = this.getNormalizedCoordinates(x, z);
        return sampleHeightmapData(this.mapData, coordinates.u, coordinates.v);
    }

    sampleMapValue(x, z) {
        this.assertReady();
        if (!this.contains(x, z)) return null;
        const coordinates = this.getNormalizedCoordinates(x, z);
        return sampleNormalizedHeightmapData(this.mapData, coordinates.u, coordinates.v);
    }

    sampleAt(x, z) {
        const height = this.sampleHeight(x, z);
        return height === null ? null : new BABYLON.Vector3(x, height, z);
    }

    sample() {
        return this.sampleAt(
            BABYLON.Scalar.RandomRange(this.min.x, this.max.x),
            BABYLON.Scalar.RandomRange(this.min.z, this.max.z)
        );
    }

    sampleNear(center, radius) {
        const angle = Math.random() * Math.PI * 2;
        const distance = radius * Math.sqrt(Math.random());
        return this.sampleAt(
            center.x + Math.cos(angle) * distance,
            center.z + Math.sin(angle) * distance
        );
    }

    sampleSlopeAt(x, z, spacing = 4) {
        const left = this.sampleHeight(x - spacing, z);
        const right = this.sampleHeight(x + spacing, z);
        const back = this.sampleHeight(x, z - spacing);
        const front = this.sampleHeight(x, z + spacing);
        if ([left, right, back, front].some(isNull)) return null;
        const dx = (right - left) / (2 * spacing);
        const dz = (front - back) / (2 * spacing);
        const magnitude = Math.hypot(dx, dz);
        const angle = Math.atan(magnitude);
        return {
            gradientX: dx,
            gradientZ: dz,
            magnitude,
            angle,
            degrees: angle * 180 / Math.PI,
            normal: new BABYLON.Vector3(-dx, 1, -dz).normalize()
        };
    }

    sampleProfileAt(x, z, spacing = 4) {
        const position = this.sampleAt(x, z);
        const slope = this.sampleSlopeAt(x, z, spacing);
        if (!position || !slope) return null;
        return {
            position,
            elevation: clamp((position.y - this.minHeight) / (this.maxHeight - this.minHeight), 0, 1),
            slope: slope.magnitude,
            slopeDegrees: slope.degrees,
            normal: slope.normal
        };
    }

    getDiagnostics(gridSize = 32) {
        if (!this.ready) return { ready: false, message: 'Heightmap has not finished loading.' };
        const samples = Math.max(1, Math.floor(gridSize));
        let minimum = Infinity;
        let maximum = -Infinity;
        let total = 0;
        let count = 0;
        const extremaSamples = [];
        for (let row = 0; row <= samples; row++) {
            for (let column = 0; column <= samples; column++) {
                const x = this.min.x + column / samples * this.worldSize;
                const z = this.min.z + row / samples * this.worldSize;
                const height = this.sampleHeight(x, z);
                minimum = Math.min(minimum, height);
                maximum = Math.max(maximum, height);
                total += height;
                count++;
                if (isDiagnosticPoint(column, row, samples)) extremaSamples.push({ x, z, height });
            }
        }
        return {
            ready: true,
            image: { width: this.width, height: this.height },
            source: this.loadedSource,
            usingFallback: this.loadedSource !== this.url,
            worldSize: this.worldSize,
            heightChannel: this.heightChannel,
            configuredHeightRange: { min: this.minHeight, max: this.maxHeight },
            sampledHeightRange: { min: minimum, max: maximum, variation: maximum - minimum },
            averageHeight: total / count,
            sampleCount: count,
            samples: extremaSamples
        };
    }

    assertReady() {
        if (!this.ready || !this.mapData) throw new Error('Heightmap not loaded.');
    }

    dispose() {
        this.pixels = null;
        this.mapData = null;
        this.ready = false;
        this.diagnostics = null;
        this.loadedSource = null;
    }

function assertCompleteRaster(raster, checkTransparentTail) {
    if (!checkTransparentTail || !raster?.pixels || raster.height < 2) return;
    const threshold = Math.max(8, Math.ceil(raster.height * 0.01));
    let transparentRows = 0;
    for (let row = raster.height - 1; row >= 0; row--) {
        let transparent = true;
        for (let column = 0; column < raster.width; column++) {
            if (raster.pixels[(row * raster.width + column) * 4 + 3] !== 0) {
                transparent = false;
                break;
            }
        }
        if (!transparent) break;
        transparentRows++;
    }
    if (transparentRows >= threshold) {
        throw new Error(`Topographic raster has ${transparentRows} fully transparent trailing rows.`);
    }
}

function isNull(value) { return value === null; }
function clamp(value, minimum, maximum) { return Math.min(maximum, Math.max(minimum, value)); }
function isDiagnosticPoint(column, row, size) {
    const midpoint = Math.floor(size / 2);
    return [0, midpoint, size].includes(column) && [0, midpoint, size].includes(row);
}

class MoistureField {
    constructor(terrain, options = {}) {
        this.terrain = terrain;

        this.seaLevel = options.seaLevel ?? -100;

        this.baseMoisture = options.baseMoisture ?? 0.08;
        this.oceanMoisture = options.oceanMoisture ?? 0.82;

        this.coastalRadius = options.coastalRadius ?? 24000;
        this.atmosphericScale = options.atmosphericScale ?? 7000;

        this.cloudContribution = options.cloudContribution ?? 0.25;
    }

    clamp(value) {
        return Math.max(0, Math.min(1, value));
    }

    getSurfaceHeight(x, z) {
        const height = this.terrain.sampleHeight(x, z);

        if (height === null || !Number.isFinite(height)) {
            return null;
        }

        return height;
    }

    getOceanInfluence(x, z) {
        const centerHeight = this.getSurfaceHeight(x, z);

        if (centerHeight === null) {
            return 0;
        }

        // Directly over the ocean.
        if (centerHeight <= this.seaLevel) {
            return 1;
        }

        // Sample the surrounding terrain to estimate
        // proximity to ocean water.
        const directions = [
            [1, 0],
            [-1, 0],
            [0, 1],
            [0, -1],
            [0.7071, 0.7071],
            [-0.7071, 0.7071],
            [0.7071, -0.7071],
            [-0.7071, -0.7071]
        ];

        let influence = 0;

        const distances = [
            this.coastalRadius * 0.4,
            this.coastalRadius
        ];

        for (let distanceIndex = 0;
             distanceIndex < distances.length;
             distanceIndex++) {

            const distance = distances[distanceIndex];

            const weight = distanceIndex === 0 ? 0.65 : 0.35;

            let oceanSamples = 0;

            for (const direction of directions) {
                const height = this.getSurfaceHeight(
                    x + direction[0] * distance,
                    z + direction[1] * distance
                );

                if (height !== null && height <= this.seaLevel) {
                    oceanSamples++;
                }
            }

            influence += (
                oceanSamples / directions.length
            ) * weight;
        }

        return this.clamp(influence);
    }

    sample(position, cloudMoisture = 0) {
        if (!position) {
            return this.baseMoisture;
        }

        const surfaceHeight = this.getSurfaceHeight(
            position.x,
            position.z
        );

        if (surfaceHeight === null) {
            return this.baseMoisture;
        }

        const referenceHeight = Math.max(
            surfaceHeight,
            this.seaLevel
        );

        const altitude = Math.max(
            0,
            position.y - referenceHeight
        );

        // Moisture availability decreases with altitude.
        const altitudeFactor = Math.exp(
            -altitude / this.atmosphericScale
        );

        const oceanInfluence = this.getOceanInfluence(
            position.x,
            position.z
        );

        const oceanContribution =
            oceanInfluence *
            this.oceanMoisture *
            altitudeFactor;

        const cloudContribution =
            this.clamp(cloudMoisture) *
            this.cloudContribution;

        return this.clamp(
            this.baseMoisture +
            oceanContribution +
            cloudContribution
        );
    }

    sampleSurface(x, z) {
        const position = {
            x,
            y: this.seaLevel,
            z
        };

        return this.sample(position);
    }

    getStats() {
        return {
            seaLevel: this.seaLevel,
            baseMoisture: this.baseMoisture,
            oceanMoisture: this.oceanMoisture,
            coastalRadius: this.coastalRadius,
            atmosphericScale: this.atmosphericScale
        };
    }
}

export { MoistureField };

class SpatialGrid {

    constructor(cellSize = 2.5) {

        this.cellSize = cellSize;
        this.cells = new Map();

    }

    getKey(x, z) {

        const cx = Math.floor(x / this.cellSize);
        const cz = Math.floor(z / this.cellSize);

        return `${cx},${cz}`;
    }

    insert(tree) {

        const key = this.getKey(
            tree.position.x,
            tree.position.z
        );

        if (!this.cells.has(key)) {
            this.cells.set(key, []);
        }

        this.cells.get(key).push(tree);

    }

    isAvailable(position, minSpacing) {

        const cx = Math.floor(
            position.x / this.cellSize
        );

        const cz = Math.floor(
            position.z / this.cellSize
        );

        const searchRadius = Math.ceil(
            minSpacing / this.cellSize
        );

        const minSpacingSquared =
            minSpacing * minSpacing;

        for (
            let x = cx - searchRadius;
            x <= cx + searchRadius;
            x++
        ) {

            for (
                let z = cz - searchRadius;
                z <= cz + searchRadius;
                z++
            ) {

                const key = `${x},${z}`;
                const cell = this.cells.get(key);

                if (!cell) continue;

                for (const tree of cell) {

                    if (!tree.alive) continue;

                    const dx =
                        position.x - tree.position.x;

                    const dz =
                        position.z - tree.position.z;

                    if (
                        dx * dx + dz * dz <
                        minSpacingSquared
                    ) {
                        return false;
                    }

                }

            }

        }

        return true;

    }

}
export { SpatialGrid }

class TerrainSampler {

    constructor(scene, ground) {

        this.scene = scene;
        this.ground = ground;

        ground.isPickable = true;
        ground.computeWorldMatrix(true);

        const bounds = ground.getBoundingInfo().boundingBox;

        this.min = bounds.minimumWorld.clone();
        this.max = bounds.maximumWorld.clone();
    }


    sample() {

        const x = BABYLON.Scalar.RandomRange(
            this.min.x,
            this.max.x
        );

        const z = BABYLON.Scalar.RandomRange(
            this.min.z,
            this.max.z
        );

        const origin = new BABYLON.Vector3(
            x,
            this.max.y + 100,
            z
        );

        const ray = new BABYLON.Ray(
            origin,
            new BABYLON.Vector3(0, -1, 0),
            this.max.y - this.min.y + 200
        );

        const hit = this.scene.pickWithRay(
            ray,
            mesh => mesh === this.ground,
            true
        );

        if (!hit || !hit.hit || !hit.pickedPoint) {
            return null;
        }

        return hit.pickedPoint.clone();
    }


	
sampleNear(center, radius) {

    const angle = Math.random() * Math.PI * 2;

    // Uniform distribution over a circular area
    const distance = radius * Math.sqrt(Math.random());

    const x = center.x + Math.cos(angle) * distance;
    const z = center.z + Math.sin(angle) * distance;

    return this.sampleAt(x, z);
}


sampleAt(x, z) {

    if (
        x < this.min.x || x > this.max.x ||
        z < this.min.z || z > this.max.z
    ) {
        return null;
    }

    const origin = new BABYLON.Vector3(
        x,
        this.max.y + 100,
        z
    );

    const ray = new BABYLON.Ray(
        origin,
        new BABYLON.Vector3(0, -1, 0),
        this.max.y - this.min.y + 200
    );

    const hit = this.scene.pickWithRay(
        ray,
        mesh => mesh === this.ground,
        true
    );

    if (!hit || !hit.hit || !hit.pickedPoint) {
        return null;
    }

    return hit.pickedPoint.clone();
}


sampleSlopeAt(x, z, spacing = 4) {

    const left = this.sampleAt(
        x - spacing,
        z
    );

    const right = this.sampleAt(
        x + spacing,
        z
    );

    const back = this.sampleAt(
        x,
        z - spacing
    );

    const front = this.sampleAt(
        x,
        z + spacing
    );

    if (!left || !right || !back || !front) {
        return null;
    }

    const dx = (
        right.y - left.y
    ) / (spacing * 2);

    const dz = (
        front.y - back.y
    ) / (spacing * 2);

    const magnitude = Math.sqrt(
        dx * dx + dz * dz
    );

    const angle = Math.atan(magnitude);

    return {

        gradientX: dx,

        gradientZ: dz,

        magnitude,

        angle,

        degrees: angle * 180 / Math.PI,

        normal: new BABYLON.Vector3(
            -dx,
            1,
            -dz
        ).normalize()

    };

}


sampleProfileAt(x, z, spacing = 4) {

    const position = this.sampleAt(x, z);

    if (!position) return null;

    const slope = this.sampleSlopeAt(
        x,
        z,
        spacing
    );

    if (!slope) return null;

    const elevation = BABYLON.Scalar.Clamp(

        (position.y - this.min.y) /
        Math.max(1, this.max.y - this.min.y),

        0,
        1

    );

    return {

        position,

        elevation,

        slope: slope.magnitude,

        slopeDegrees: slope.degrees,

        normal: slope.normal

    };

}



}

/** Browser-only image decoding adapter. It returns plain RGBA raster data. */

class BrowserHeightmapLoader {
    async load(source, options = {}) {
        if (isImageData(source)) return copyImageData(source);
        if (typeof createImageBitmap === 'function') {
            return this.loadImageBitmap(source, options);
        }
        return this.loadHtmlImage(source, options);
    }

    async loadImageBitmap(source, options) {
        let bitmap;
        try {
            bitmap = await createImageBitmap(await resolveImageSource(source));
            return this.readBitmap(bitmap, options);
        } catch (error) {
            throw createLoadError(source, error);
        } finally {
            bitmap?.close?.();
        }
    }

    readBitmap(bitmap, options) {
        const canvas = document.createElement('canvas');
        canvas.width = bitmap.width;
        canvas.height = bitmap.height;
        const context = canvas.getContext('2d', { willReadFrequently: true });
        if (!context) throw new Error('Unable to create a heightmap canvas context.');
        context.drawImage(bitmap, 0, 0);
        return copyImageData(context.getImageData(0, 0, bitmap.width, bitmap.height), options);
    }

    async loadHtmlImage(source, options) {
        if (typeof source !== 'string') {
            throw new TypeError('This browser requires an image URL when createImageBitmap is unavailable.');
        }
        const image = new Image();
        image.crossOrigin = options.crossOrigin ?? 'anonymous';
        const loaded = new Promise(resolveImageLoad.bind(null, image));
        image.src = source;
        try {
            await loaded;
            const canvas = document.createElement('canvas');
            canvas.width = image.naturalWidth;
            canvas.height = image.naturalHeight;
            const context = canvas.getContext('2d', { willReadFrequently: true });
            if (!context) throw new Error('Unable to create a heightmap canvas context.');
            context.drawImage(image, 0, 0);
            return copyImageData(context.getImageData(0, 0, canvas.width, canvas.height), options);
        } catch (error) {
            throw createLoadError(source, error);
        }
    }
}

async function resolveImageSource(source) {
    if (typeof source === 'string') {
        const response = await fetch(source, { mode: 'cors' });
        if (!response.ok) throw new Error(`HTTP ${response.status} while loading map.`);
        return response.blob();
    }
    if (source instanceof Blob) return source;
    throw new TypeError('Heightmap source must be a URL, Blob, File, or ImageData.');
}

function isImageData(value) {
    return Boolean(value && Number.isSafeInteger(value.width) &&
        Number.isSafeInteger(value.height) && value.data);
}

function copyImageData(imageData) {
    return {
        width: imageData.width,
        height: imageData.height,
        pixels: new Uint8Array(imageData.data)
    };
}

function resolveImageLoad(image, resolve, reject) {
    image.onload = resolve;
    image.onerror = reject;
}

function createLoadError(source, error) {
    const label = typeof source === 'string' ? source : 'provided image data';
    return new Error(`Unable to decode topographic map ${label}: ${error.message}`, { cause: error });
}

export { BrowserHeightmapLoader };

/** Pure raster validation, elevation decoding, and bilinear sampling. */

const CHANNELS = new Set([
    'red', 'green', 'blue', 'alpha', 'average', 'luminance'
]);

function createHeightmapData(options = {}) {
    const width = options.width;
    const height = options.height;
    const pixels = options.pixels ?? options.data;
    const channel = options.channel ?? 'luminance';
    const minHeight = options.minHeight ?? 0;
    const maxHeight = options.maxHeight ?? 1;
    const colorStops = normalizeColorStops(options.colorStops);

    validateRaster(width, height, pixels);
    validateHeightRange(minHeight, maxHeight);
    validateChannel(channel, colorStops);

    return {
        width,
        height,
        pixels: new Uint8Array(pixels),
        channel,
        minHeight,
        maxHeight,
        colorStops
    };
}

function validateRaster(width, height, pixels) {
    if (!Number.isSafeInteger(width) || width < 2) {
        throw new RangeError('Heightmap width must be an integer of at least two.');
    }
    if (!Number.isSafeInteger(height) || height < 2) {
        throw new RangeError('Heightmap height must be an integer of at least two.');
    }
    if (!pixels || !Number.isSafeInteger(pixels.length)) {
        throw new TypeError('Heightmap pixels must be a typed or array-like RGBA buffer.');
    }
    if (pixels.length !== width * height * 4) {
        throw new RangeError('Heightmap RGBA buffer length does not match its dimensions.');
    }
    for (let index = 0; index < pixels.length; index++) {
        const value = pixels[index];
        if (!Number.isFinite(value) || value < 0 || value > 255) {
            throw new RangeError(`Heightmap pixel ${index} is outside the byte range.`);
        }
    }
}

function validateHeightRange(minHeight, maxHeight) {
    if (!Number.isFinite(minHeight) || !Number.isFinite(maxHeight) || minHeight >= maxHeight) {
        throw new RangeError('Heightmap elevation bounds must be finite and increasing.');
    }
}

function validateChannel(channel, colorStops) {
    if (colorStops.length) return;
    if (!CHANNELS.has(channel)) {
        throw new RangeError(`Unsupported heightmap channel: ${channel}`);
    }
}

function normalizeColorStops(stops = []) {
    if (!Array.isArray(stops)) {
        throw new TypeError('Heightmap colorStops must be an array.');
    }
    const normalized = stops.map(normalizeColorStop);
    if (normalized.length && normalized.length < 2) {
        throw new RangeError('A color elevation palette requires at least two stops.');
    }
    return normalized;
}

function normalizeColorStop(stop) {
    if (!stop || !Array.isArray(stop.color) || stop.color.length < 3) {
        throw new TypeError('Each color stop requires an RGB color and normalized value.');
    }
    const color = stop.color.slice(0, 3);
    if (!color.every(isByte)) throw new RangeError('Palette RGB channels must be bytes.');
    if (!Number.isFinite(stop.value) || stop.value < 0 || stop.value > 1) {
        throw new RangeError('Palette values must be normalized to [0, 1].');
    }
    return { color, value: stop.value };
}

function isByte(value) {
    return Number.isInteger(value) && value >= 0 && value <= 255;
}

function sampleHeightmapData(data, u, v) {
    return data.minHeight + sampleNormalizedHeightmapData(data, u, v) * (data.maxHeight - data.minHeight);
}

function sampleNormalizedHeightmapData(data, u, v) {
    validateSampleCoordinates(u, v);
    const x = clamp(u, 0, 1) * (data.width - 1);
    const z = clamp(v, 0, 1) * (data.height - 1);
    const x0 = Math.floor(x);
    const z0 = Math.floor(z);
    const x1 = Math.min(x0 + 1, data.width - 1);
    const z1 = Math.min(z0 + 1, data.height - 1);
    const tx = x - x0;
    const tz = z - z0;
    const top = lerp(
        decodeHeightmapPixel(data, x0, z0),
        decodeHeightmapPixel(data, x1, z0),
        tx
    );
    const bottom = lerp(
        decodeHeightmapPixel(data, x0, z1),
        decodeHeightmapPixel(data, x1, z1),
        tx
    );
    return lerp(top, bottom, tz);
}

function decodeHeightmapPixel(data, x, z) {
    const offset = (z * data.width + x) * 4;
    const pixels = data.pixels;
    if (data.colorStops.length) return decodePalette(pixels, offset, data.colorStops);

    const red = pixels[offset];
    const green = pixels[offset + 1];
    const blue = pixels[offset + 2];
    const alpha = pixels[offset + 3];
    switch (data.channel) {
        case 'red': return red / 255;
        case 'green': return green / 255;
        case 'blue': return blue / 255;
        case 'alpha': return alpha / 255;
        case 'average': return (red + green + blue) / 765;
        default: return (0.2126 * red + 0.7152 * green + 0.0722 * blue) / 255;
    }
}

function decodePalette(pixels, offset, stops) {
    let nearest = stops[0];
    let nearestDistance = Infinity;
    for (const stop of stops) {
        const red = pixels[offset] - stop.color[0];
        const green = pixels[offset + 1] - stop.color[1];
        const blue = pixels[offset + 2] - stop.color[2];
        const distance = red * red + green * green + blue * blue;
        if (distance < nearestDistance) {
            nearest = stop;
            nearestDistance = distance;
        }
    }
    return nearest.value;
}

function validateSampleCoordinates(u, v) {
    if (!Number.isFinite(u) || !Number.isFinite(v)) {
        throw new TypeError('Heightmap sample coordinates must be finite.');
    }
}

function clamp(value, minimum, maximum) {
    return Math.min(maximum, Math.max(minimum, value));
}

function lerp(start, end, amount) {
    return start + (end - start) * amount;
}

function getHeightmapStatistics(data, samples = 24) {
    if (!Number.isSafeInteger(samples) || samples < 1) {
        throw new RangeError('Heightmap statistic sample count must be positive.');
    }
    let minimum = Infinity;
    let maximum = -Infinity;
    let total = 0;
    let count = 0;
    for (let row = 0; row <= samples; row++) {
        for (let column = 0; column <= samples; column++) {
            const value = sampleHeightmapData(data, column / samples, row / samples);
            minimum = Math.min(minimum, value);
            maximum = Math.max(maximum, value);
            total += value;
            count++;
        }
    }
    return { minimum, maximum, average: total / count, count };
}

export {
    createHeightmapData,
    decodeHeightmapPixel,
    getHeightmapStatistics,
    sampleHeightmapData,
    sampleNormalizedHeightmapData
};
