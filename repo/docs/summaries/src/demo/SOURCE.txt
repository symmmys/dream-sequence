 * Three-dimensional presentation of travel between the two ends of a 4D
 * slice-transition anomaly. All positions and scales in this module are
 * display-space only; canonical anomaly coordinates are never modified.
 */

import { PortalPreviewRenderer } from '../world/fields/geometry/portal-preview-renderer.js';

const PASSAGE_RENDER_LAYER = 0x40000000;
const DEFAULTS = Object.freeze({
    mouthSeparation: 80,
    cameraInset: 8,
    mouthSeparationAxis: null,
    mouthSeparationCurve: null,
    entryAngularScaleStart: 1,
    entryAngularScaleEnd: 1,
    exitAngularScaleStart: 1,
    exitAngularScaleEnd: 1,
    resolution: 512
});

class AnomalyPassagePresentation {
    constructor(scene, pair, options = {}) {
        if (!scene || !pair?.corridor || !pair.sourcePortal || !pair.destinationPortal) {
            throw new TypeError('Anomaly passage presentation requires a scene and complete portal pair.');
        }
        const radii = pair.corridor.view?.interiorBoundaryRadii;
        if (!radii || !Number.isFinite(radii.source) || radii.source <= 0 ||
            !Number.isFinite(radii.destination) || radii.destination <= 0) {
            throw new TypeError(`Anomaly ${pair.corridor.id} requires positive source and destination mouth radii.`);
        }
        this.scene = scene;
        this.pair = pair;
        this.radii = { source: radii.source, destination: radii.destination };
        this.config = normalizeConfig({ ...pair.corridor.view?.passagePresentation, ...options });
        this.active = false;
        this.frame = null;
        this.previousClearColor = null;
        this.lastState = null;
        this.entry = this.createMouthRenderer('entry', Math.max(radii.source, radii.destination));
        this.exit = this.createMouthRenderer('exit', Math.max(radii.source, radii.destination));
    }

    createMouthRenderer(name, radius) {
        const renderer = new PortalPreviewRenderer(this.scene, {
            namePrefix: `anomalyPassage_${this.pair.corridor.id}_${name}`,
            position: { x: 0, y: 0, z: 0 },
            width: radius * 2,
            height: radius * 2,
            resolution: this.config.resolution,
            apertureShape: 'circle',
            previewSurface: 'spherical',
            maxApertureScale: 4,
            camera: { layerMask: 0 }
        });
        renderer.surface.layerMask = PASSAGE_RENDER_LAYER;
        renderer.surface.isPickable = false;
        renderer.setHomePose();
        renderer.surface.metadata = {
            ...(renderer.surface.metadata || {}),
            anomalyPassageMouth: name,
            anomalyId: this.pair.corridor.id,
            displaySpaceOnly: true,
            renderOnly: true
        };
        renderer.setVisible(false);
        return renderer;

            exitCameraPose = null, presentationDirection = null
        } = options;
        if (!camera || !startWorld || !targetWorld || !Number.isFinite(progress)) {
            throw new TypeError('Anomaly passage update requires a camera, endpoint worlds, and finite progress.');
        }

            entryAngularScaleStart: this.config.entryAngularScaleStart,
            entryAngularScaleEnd: this.config.entryAngularScaleEnd,
            exitAngularScaleStart: this.config.exitAngularScaleStart,
            exitAngularScaleEnd: this.config.exitAngularScaleEnd

        const exitRadius = startIsSource ? this.radii.destination : this.radii.source;
        const entryMeshes = startIsSource
            ? this.pair.destinationPortal.ownedMeshes
            : this.pair.sourcePortal.ownedMeshes;
        const exitMeshes = startIsSource
            ? this.pair.sourcePortal.ownedMeshes
            : this.pair.destinationPortal.ownedMeshes;

        this.exit.setRenderList(exitMeshes ?? []);

            baseCameraPosition,
            add3(baseCameraPosition, scale3(entryForward, 100)),

        this.entry.setVisible(true);
        this.exit.setVisible(true);

        this.lastState = {
            active: true,
            anomalyId: this.pair.corridor.id,
            progress: t,
            easedProgress: eased,
            direction: startIsSource ? 1 : -1,
            startSliceId: startWorld.slice.id,
            destinationSliceId: targetWorld.slice.id,
            separation: layout.separation,
            axis: { ...this.frame.axis },

            exit: {
                position: { ...layout.exit.position },
                physicalRadius: exitRadius,
                scale: layout.exit.scale * exitRadius / maxRadius,
                apparentAngularRadius: layout.exit.apparentAngularRadius,
                cameraDistance: layout.exit.cameraDistance,
            },
            renderLayer: PASSAGE_RENDER_LAYER
        };
        return this.lastState;
    }

    getBaseCameraPosition(camera) {
        const current = vector(camera.position);
        if (this.active && this.lastCameraPose &&
            isSamePoint(current, this.lastCameraPose.appliedPosition)) {
            return { ...this.lastCameraPose.basePosition };
        }
        return current;
    }

    restoreCameraPoseForControllerUpdate(camera) {
        if (!camera || !this.active || !this.lastCameraPose) return false;
        if (!isSamePoint(camera.position, this.lastCameraPose.appliedPosition)) {
            // An external controller has already supplied a new base pose.
            // Do not overwrite its position; simply discard the old mapping.
            this.lastCameraPose = null;
            return false;
        }
        const base = this.lastCameraPose.basePosition;
        const forward = cameraForward(camera);
        const target = add3(base, scale3(forward, 100));
        camera.position.set(base.x, base.y, base.z);
        camera.setTarget?.(new BABYLON.Vector3(target.x, target.y, target.z));
        this.lastCameraPose = null;
        return true;
    }

    activate(camera, baseCameraPosition = vector(camera.position), passageDirection = cameraForward(camera)) {
        const overrideAxis = this.config.mouthSeparationAxis;
        const axis = overrideAxis ? normalize3(overrideAxis) : normalize3(passageDirection);
        this.frame = {
            anchor: { ...baseCameraPosition },
            axis
        };
        this.previousClearColor = cloneColor(this.scene.clearColor);
        this.active = true;
    }

    deactivate(camera = null, restoreLayerMask = null) {
        if (!this.active && !this.entry.visible && !this.exit.visible) return;
        this.entry.setVisible(false);
        this.exit.setVisible(false);
        if (this.previousClearColor) this.scene.clearColor = cloneColor(this.previousClearColor);
        if (camera && this.lastCameraPose && isSamePoint(camera.position, this.lastCameraPose.appliedPosition)) {
            const base = this.lastCameraPose.basePosition;
            // Preserve a camera orbit/rotation performed after the last render
            // update rather than restoring a stale target from that update.
            const forward = cameraForward(camera);
            const target = add3(base, scale3(forward, 100));
            camera.position.set(base.x, base.y, base.z);
            camera.setTarget?.(new BABYLON.Vector3(target.x, target.y, target.z));
        }
        if (camera && Number.isSafeInteger(restoreLayerMask)) camera.layerMask = restoreLayerMask;
        this.active = false;
        this.frame = null;
        this.previousClearColor = null;
        this.lastState = null;
        this.lastCameraPose = null;
    }

    getInfo() {
        return {
            ...(this.lastState ?? { active: false, anomalyId: this.pair.corridor.id }),
            entryRenderer: this.entry.getInfo(),
            exitRenderer: this.exit.getInfo(),
            configuration: { ...this.config }
        };
    }

    dispose(camera = null, restoreLayerMask = null) {
        this.deactivate(camera, restoreLayerMask);
        this.entry.dispose();
        this.exit.dispose();
    }
}

function calculateAnomalyPassageLayout(options = {}) {
    const progress = clampFinite(options.progress ?? 0, 0, 1, 'progress');
    const anchor = requireVector(options.anchor, 'anchor');
    const cameraPosition = requireVector(options.cameraPosition ?? anchor, 'cameraPosition');
    const axis = normalize3(requireVector(options.axis, 'axis'));
    const fixedSeparation = positiveFinite(options.mouthSeparation ?? DEFAULTS.mouthSeparation, 'mouthSeparation');
    const inset = positiveFinite(options.cameraInset ?? DEFAULTS.cameraInset, 'cameraInset');
    const curve = options.mouthSeparationCurve ?? null;
    let separation = fixedSeparation;
    if (curve !== null) {
        if (!Array.isArray(curve) || curve.length !== 2 ||
            !curve.every((value) => Number.isFinite(value) && value > inset * 2)) {
            throw new RangeError('mouthSeparationCurve must contain two distances greater than twice cameraInset.');
        }
        separation = lerp(curve[0], curve[1], smootherstep(progress));
    }
    if (separation <= inset * 2) {
        throw new RangeError('mouthSeparation must exceed twice cameraInset.');
    }
    const entryStart = scaleValue(options.entryAngularScaleStart ?? DEFAULTS.entryAngularScaleStart, 'entryAngularScaleStart');
    const entryEnd = scaleValue(options.entryAngularScaleEnd ?? DEFAULTS.entryAngularScaleEnd, 'entryAngularScaleEnd');
    const exitStart = scaleValue(options.exitAngularScaleStart ?? DEFAULTS.exitAngularScaleStart, 'exitAngularScaleStart');
    const exitEnd = scaleValue(options.exitAngularScaleEnd ?? DEFAULTS.exitAngularScaleEnd, 'exitAngularScaleEnd');
    const entryRadius = positiveFinite(options.entryRadius ?? 1, 'entryRadius');
    const exitRadius = positiveFinite(options.exitRadius ?? 1, 'exitRadius');
    const eased = smootherstep(progress);
    // Keep the entrance behind the player and the exit ahead. The camera
    // starts one inset beyond the entrance and advances toward the exit,
    // finishing one inset before it. The axis is a display-space proxy for W.
    const entryPosition = subtract3(anchor, scale3(axis, inset));
    const exitPosition = add3(anchor, scale3(axis, separation - inset));
    const cameraOffset = scale3(axis, progress * (separation - inset * 2));
    const displayCameraPosition = add3(cameraPosition, cameraOffset);
    const entryDistance = distance3(displayCameraPosition, entryPosition);
    const exitDistance = distance3(displayCameraPosition, exitPosition);
    const entryScale = lerp(entryStart, entryEnd, eased);
    const exitScale = lerp(exitStart, exitEnd, eased);
    return {
        progress,
        easedProgress: eased,
        separation,
        cameraOffset,
        cameraPosition: displayCameraPosition,
        entry: {
            position: entryPosition,
            scale: entryScale,
            cameraDistance: entryDistance,
            apparentAngularRadius: apparentAngularRadius(entryRadius * entryScale, entryDistance)
        },
        exit: {
            position: exitPosition,
            scale: exitScale,
            cameraDistance: exitDistance,
            apparentAngularRadius: apparentAngularRadius(exitRadius * exitScale, exitDistance)
        }
    };
}

function normalizeConfig(options = {}) {
    const config = { ...DEFAULTS, ...options };
    positiveFinite(config.mouthSeparation, 'mouthSeparation');
    positiveFinite(config.cameraInset, 'cameraInset');
    if (config.mouthSeparation <= config.cameraInset * 2) {
        throw new RangeError('mouthSeparation must exceed twice cameraInset.');
    }
    if (config.resolution !== 256 && config.resolution !== 512 && config.resolution !== 1024 &&
        config.resolution !== 2048) {
        throw new RangeError('Passage render resolution must be 256, 512, 1024, or 2048.');
    }
    if (config.mouthSeparationAxis !== null) {
        config.mouthSeparationAxis = normalize3(requireVector(config.mouthSeparationAxis, 'mouthSeparationAxis'));
    }
    // Validate optional curve and normalized scale endpoints eagerly.
    calculateAnomalyPassageLayout({
        anchor: { x: 0, y: 0, z: 0 }, axis: { x: 0, y: 0, z: 1 }, progress: 0,
        mouthSeparation: config.mouthSeparation,
        cameraInset: config.cameraInset,
        mouthSeparationCurve: config.mouthSeparationCurve,
        entryAngularScaleStart: config.entryAngularScaleStart,
        entryAngularScaleEnd: config.entryAngularScaleEnd,
        exitAngularScaleStart: config.exitAngularScaleStart,
        exitAngularScaleEnd: config.exitAngularScaleEnd
    });
    return config;
}

function smootherstep(value) {
    return value * value * value * (value * (value * 6 - 15) + 10);
}
function clamp(value, min, max) { return Math.max(min, Math.min(max, value)); }
function clampFinite(value, min, max, name) {
    if (!Number.isFinite(value)) throw new TypeError(`${name} must be finite.`);
    return clamp(value, min, max);
}
function positiveFinite(value, name) {
    if (!Number.isFinite(value) || value <= 0) throw new RangeError(`${name} must be positive and finite.`);
    return value;
}
function scaleValue(value, name) {
    if (!Number.isFinite(value) || value < 0 || value > 4) throw new RangeError(`${name} must be between 0 and 4.`);
    return value;
}
function requireVector(value, name) {
    const vector = Array.isArray(value) ? { x: value[0], y: value[1], z: value[2] } : value;
    if (!vector || ![vector.x, vector.y, vector.z].every(Number.isFinite)) {
        throw new TypeError(`${name} must contain three finite components.`);
    }
    return { x: vector.x, y: vector.y, z: vector.z };
}
function normalize3(value) {
    const length = Math.hypot(value.x, value.y, value.z);
    if (length <= 1e-9) throw new RangeError('Passage direction must have non-zero length.');
    return { x: value.x / length, y: value.y / length, z: value.z / length };
}
function cameraForward(camera) {
    const direction = camera.getForwardRay?.().direction ?? camera.forward ?? { x: 0, y: 0, z: 1 };
    return normalize3(vector(direction));
}
function vector(value) { return { x: value.x, y: value.y, z: value.z }; }
function add3(a, b) { return { x: a.x + b.x, y: a.y + b.y, z: a.z + b.z }; }
function subtract3(a, b) { return { x: a.x - b.x, y: a.y - b.y, z: a.z - b.z }; }
function scale3(value, scalar) { return { x: value.x * scalar, y: value.y * scalar, z: value.z * scalar }; }
function lerp(a, b, t) { return a + (b - a) * t; }
function distance3(a, b) { return Math.hypot(a.x - b.x, a.y - b.y, a.z - b.z); }
function apparentAngularRadius(radius, distance) {
    if (distance <= 1e-9) return Math.PI / 2;
    return Math.asin(clamp(radius / distance, 0, 1));
}
function isSamePoint(point, expected) {
    return Boolean(point && expected && distance3(point, expected) <= 1e-4);
}
function cloneColor(color) {
    if (!color) return new BABYLON.Color4(0, 0, 0, 1);
    return new BABYLON.Color4(color.r, color.g, color.b, color.a ?? 1);
}

/** Coordinates several single-corridor controllers as one connected world graph. */

class CorridorTraversalNetwork {
    constructor(options = {}) {
        this.topology = requireObject(options.topology, 'topology');
        this.worlds = new Map(options.worlds ?? []);
        this.controllers = [...(options.controllers ?? [])];
        if (!this.controllers.length) throw new TypeError('Traversal network requires at least one controller.');
        this.currentSliceId = options.initialSliceId ?? this.controllers[0].currentSliceId;
        this.onSliceChanged = options.onSliceChanged ?? noop;
        this.active = false;
        this.disposed = false;
        this.activeController = null;
        this.bindControllers();
    }

    bindControllers() {
        for (const controller of this.controllers) {
            const notify = controller.onSliceChanged;
            controller.currentSliceId = this.currentSliceId;
            controller.onSliceChanged = this.handleSliceChanged.bind(this, controller, notify);
        }
    }

    handleSliceChanged(controller, notify, sliceId, state) {
        this.currentSliceId = sliceId;
        this.syncSlice(sliceId);
        if (typeof notify === 'function') notify(sliceId, state);
        this.onSliceChanged(sliceId, state);
    }

    update(deltaSeconds) {
        if (!this.active || this.disposed) return;
        const delta = Math.max(0, Math.min(deltaSeconds, 0.1));
        const transit = this.getActiveController();
        if (transit) {
            transit.update(delta);
            for (const controller of this.controllers) {
                if (controller !== transit) controller.advanceOtherEntities(delta);
            }
            if (transit.state !== 'transit') this.activeController = null;
            this.currentSliceId = transit.currentSliceId;
            this.syncSlice(this.currentSliceId);
            return;
        }

        this.activeController = null;
        this.syncSlice(this.currentSliceId);
        for (const controller of this.controllers) {
            if (controller.tryBeginTransit()) {
                this.activeController = controller;
                break;
            }
        }
        if (!this.activeController) this.advanceAmbientInfluenceMovement(delta);
        this.currentSliceId = this.controllers[0]?.currentSliceId ?? this.currentSliceId;
        this.syncSlice(this.currentSliceId);
        for (const controller of this.controllers) controller.advanceOtherEntities(delta);
    }

    advanceAmbientInfluenceMovement(deltaSeconds) {
        const input = this.controllers[0]?.getMovementInput?.() ?? {};
        const direction = Number(input.crossSectionW) || 0;
        if (!direction) return false;

        // Controllers are synchronized to the active slice, but not every
        // controller owns a WorldSlice binding for that slice. Resolve one
        // authoritative ambient position first, then evaluate only routes
        // that actually connect from the current slice.
        const position4D = this.getAmbientPlayerPosition();
        if (!position4D) return false;
        const candidates = this.controllers.filter((controller) =>
            controller.ownsSlice?.(this.currentSliceId) ??
                (controller.worlds instanceof Map
                    ? controller.worlds.has(this.currentSliceId)
                    : controller.currentSliceId === this.currentSliceId)
        ).map((controller) => {
            const influence = controller.getInfluenceMovementInfo?.(
                position4D, direction, this.currentSliceId
            );
            return influence ? { controller, influence } : null;
        }).filter(Boolean).sort((left, right) =>
            right.influence.movementScale - left.influence.movementScale
        );
        if (!candidates.length) return false;
        return candidates[0].controller.advanceAmbientInfluenceMovement(deltaSeconds, position4D);
    }

    getAmbientPlayerPosition() {
        const active = this.getActiveController();
        if (active?.spatialPosition4D) return { ...active.spatialPosition4D };

        // A network may contain routes that do not touch the current slice.
        // Do not stop at the first synchronized controller: find the first
        // controller that owns a binding for this actual WorldSlice and can
        // resolve a finite ambient position.
        let fallback = null;
        // Multiple anomalies can have separate presentation mouths in one
        // slice. Prefer the controller whose visible portal approach contains
        // the player; otherwise an unrelated route can rebase coordinates
        // against its own canonical endpoint and keep the correct W field off.
        for (const controller of this.controllers) {
            const ownsSlice = typeof controller.ownsSlice === 'function'
                ? controller.ownsSlice(this.currentSliceId)
                : controller.worlds instanceof Map
                    ? controller.worlds.has(this.currentSliceId)
                    : controller.currentSliceId === this.currentSliceId;
            if (!ownsSlice || typeof controller.getAmbientPlayerPosition !== 'function') continue;
            const position = controller.getAmbientPlayerPosition();
            if (!isFinitePosition4(position)) continue;
            if (controller.isNearPresentationPortal?.(this.currentSliceId)) return { ...position };
            if (!fallback) fallback = { ...position };
        }
        return fallback;
    }

    getEntryDiagnostics() {
        return this.controllers.map((controller) => ({
            anomalyId: controller.corridor.id,
            ...controller.getEntryDiagnostic?.()
        }));
    }

    syncSlice(sliceId) {
        for (const controller of this.controllers) controller.currentSliceId = sliceId;
    }

    setActive(active) {
        this.active = Boolean(active) && !this.disposed;
        for (const controller of this.controllers) controller.setActive(this.active);
    }

    getActiveController() {
        return this.activeController?.state === 'transit'
            ? this.activeController
            : this.controllers.find(function inTransit(controller) {
                return controller.state === 'transit';
            }) ?? null;
    }

    get state() { return this.getActiveController()?.state ?? 'idle'; }
    get corridor() { return this.getActiveController()?.corridor ?? null; }
    get corridorLength() { return this.getActiveController()?.corridorLength ?? 0; }
    get startWorld() { return this.getActiveController()?.startWorld ?? null; }
    get target() { return this.getActiveController()?.target ?? null; }
    get direction() { return this.getActiveController()?.direction ?? 0; }
    get progress() { return this.getActiveController()?.progress ?? 0; }
    get offsetU() { return this.getActiveController()?.offsetU ?? 0; }
    get offsetV() { return this.getActiveController()?.offsetV ?? 0; }
    get offsetW() { return this.getActiveController()?.offsetW ?? 0; }
    get activeRenderer() { return this.getActiveController()?.activeRenderer ?? null; }

    getRoutesForSlice(sliceId) {
        return this.topology.getTraversableNeighbors(sliceId)
            .map(this.resolveRoute.bind(this, sliceId))
            .filter(Boolean);
    }

    resolveRoute(sliceId, link) {
        const controller = this.controllers.find(function matchingController(entry) {
            return entry.corridor.id === link.corridorId;
        });
        if (!controller) return null;
        return {
            ...link, controller, corridor: controller.corridor,
            fromSource: controller.corridor.source.sliceId === sliceId,
            renderer: controller.renderers.get(sliceId) ?? controller.renderer,
            corridorLength: controller.corridorLength
        };
    }

    getRouteForSlice(sliceId, corridorId) {
        return this.getRoutesForSlice(sliceId).find(function matchRoute(route) {
            return route.corridorId === corridorId;
        }) ?? null;
    }

    beginEntityTransit(entity, fromSliceId = entity?.sliceId) {
        const routes = this.getRoutesForSlice(fromSliceId);
        const preferred = routes.find(function preferredRoute(route) {
            return route.corridorId === entity?.corridorId;
        });
        for (const route of preferred ? [preferred] : routes) {
            if (route.controller.beginEntityTransit(entity, fromSliceId)) return true;
        }
        return false;
    }

    tryBeginEntityTransit(entity, fromSliceId = entity?.sliceId) {
        for (const route of this.getRoutesForSlice(fromSliceId)) {
            if (route.controller.tryBeginEntityTransit(entity, fromSliceId)) return true;
        }
        return false;
    }

    getEntityTransitState(entity) {
        for (const controller of this.controllers) {
            const state = controller.getEntityTransitState(entity);
            if (state) return state;
        }
        return null;
    }

    getState() {
        return {
            state: this.state,
            currentSliceId: this.currentSliceId,
            activeCorridorId: this.corridor?.id ?? null,
            routes: this.controllers.map(function summarize(controller) {
                return controller.getState();
            })
        };
    }

    dispose() {
        if (this.disposed) return;
        this.setActive(false);
        this.disposed = true;
        for (const controller of this.controllers) controller.dispose();
        this.controllers.length = 0;
        this.activeController = null;
    }
}

        Number.isFinite(position.z) && Number.isFinite(position.w);
}

    updateFreeWorld(delta, input) {
        const speed = this.freeSpeed * (input.sprint ? 2 : 1);
        const forward = this.getForwardVector();
        const right = { x: Math.cos(this.freeYaw), y: 0, z: -Math.sin(this.freeYaw) };
        const movement = add3(
            add3(scale3(forward, input.forward), scale3(right, input.strafe)),
            { x: 0, y: input.vertical, z: 0 }
        );
        const length = length3(movement);
        if (length > 1e-9) {
            const candidate = add3(
                this.freePosition,
                scale3(movement, speed * delta / Math.max(1, length))
            );
            this.freePosition = this.constrainFreePosition(
                this.traversal?.worlds.get(this.freeSliceId), candidate
            );
        }
        this.advanceFreeInfluenceMovement(delta, input, speed);
        this.applyFreePose();
    }

    advanceFreeInfluenceMovement(delta, input, speed) {
        if (!input.crossSectionW || !this.traversal || !this.freeSliceId) return false;
        const world = this.traversal.worlds.get(this.freeSliceId);
        const worldPosition = this.renderFreePositionToWorld(world);
        if (!world || !worldPosition) return false;
        const routes = this.traversal.getRoutesForSlice?.(this.freeSliceId) ?? this.getLegacyRoute();
        const candidates = routes.map((route) => {
            const physicalPosition = route.controller?.getPhysicalEndpointLocalPosition?.(world, worldPosition) ?? worldPosition;
            const base = sliceToAmbient(world.slice.frame, physicalPosition);
            const current = add4(base, scale4(world.slice.frame.normal, this.freeW));
            const info = route.controller?.getInfluenceMovementInfo?.(current, input.crossSectionW, this.freeSliceId);
            return info ? { route, info, current } : null;
        }).filter(Boolean).sort((left, right) =>
            right.info.movementScale - left.info.movementScale
        );
        if (!candidates.length) return false;

        const { route, info, current } = candidates[0];
        const amount = Math.sign(input.crossSectionW) * speed * delta * info.movementScale;
        if (Math.abs(amount) <= 1e-10) return false;
        const normal = world.slice.frame.normal;
        let candidate = add4(current, scale4(normal, amount));
        const nextPlane = dot4(candidate, normal);
        const targetPlane = dot4(info.targetWorld.slice.frame.origin, normal);
        const crossed = info.direction > 0
            ? nextPlane >= targetPlane - 1e-8
            : nextPlane <= targetPlane + 1e-8;

        if (crossed) {
            candidate = add4(candidate, scale4(normal, targetPlane - nextPlane));
            const sourceLocal = ambientToSlice(world.slice.frame, candidate).position;
            const targetLocal = ambientToSlice(info.targetWorld.slice.frame, candidate).position;
            const graph = route.controller?.chunkGraph;
            const sourceAddress = graph?.getAddress(this.freeSliceId, sourceLocal);
            const targetAddress = graph?.getAddress(info.targetSliceId, targetLocal);
            const reachability = sourceAddress && targetAddress
                ? graph.getReachability(sourceAddress, targetAddress, {
                    position4D: candidate, corridorId: info.anomalyId
                }) : { reachable: true };
            if (reachability.reachable) {
                this.completeFreeInfluenceHandoff(info.targetWorld, candidate, route, world);
                return true;
            }
            candidate = add4(candidate, scale4(normal, -info.direction * 1e-7));
        }

        this.freeW = ambientToSlice(world.slice.frame, candidate).normalOffset;
        return true;
    }

    completeFreeInfluenceHandoff(world, position4D, route = null, fromWorld = null) {
        const local = route?.controller?.createArrivalPosition
            ? route.controller.createArrivalPosition(world, position4D, fromWorld)
            : ambientToSlice(world.slice.frame, position4D).position;
        const arrival = world.transform?.constrainWorldPosition?.(local) ?? local;
        this.freeSliceId = world.slice.id;
        this.freeW = 0;
        this.freeTransit = null;
        this.freeEntryLock = { sliceId: world.slice.id, position: { ...arrival } };
        this.setSpatialContext(world.transform, world.layerMask);
        this.freePosition = world.transform.worldToRender(arrival);
        this.applyFreePose();
    }

    getFreeAmbientPosition() {
        if (this.mode !== 'free' || !this.freePosition || !this.freeSliceId) return null;
        if (this.freeTransit?.position4D) return { ...this.freeTransit.position4D };
        const world = this.traversal?.worlds.get(this.freeSliceId);
        const position = this.renderFreePositionToWorld(world);
        if (!world || !position) return null;
        const routes = this.traversal.getRoutesForSlice?.(this.freeSliceId) ?? this.getLegacyRoute();
        const nearby = routes.find((route) =>
            route.controller?.isNearPresentationPortal?.(this.freeSliceId, position)
        );
        const physicalPosition = nearby?.controller?.getPhysicalEndpointLocalPosition?.(world, position) ?? position;
        return add4(
            sliceToAmbient(world.slice.frame, physicalPosition),
            scale4(world.slice.frame.normal, this.freeW)
        );
    }

    tryEnterCorridor(forwardInput) {
        if (forwardInput <= 0 || !this.traversal || !this.freeSliceId) return false;
        if (this.freeEntryLock?.sliceId === this.freeSliceId) {
            const world = this.traversal.worlds.get(this.freeSliceId);
            const position = this.renderFreePositionToWorld(world);
            if (position && Math.hypot(
                position.x - this.freeEntryLock.position.x,
                position.z - this.freeEntryLock.position.z
            ) < 1.25) return false;
            this.freeEntryLock = null;
        }
        const routes = this.traversal.getRoutesForSlice?.(this.freeSliceId) ?? this.getLegacyRoute();
        for (const route of routes) {
            const corridor = route.corridor;
            const fromSource = route.fromSource ?? (corridor.source.sliceId === this.freeSliceId);
            const endpoint = fromSource ? corridor.source : corridor.destination;
            const world = this.traversal.worlds.get(this.freeSliceId);
            const renderer = route.renderer ?? route.controller?.renderers?.get(this.freeSliceId);
            const presentationEndpoint = route.controller?.getPresentationEndpoint?.(this.freeSliceId);
            const mouthLocal = presentationEndpoint ?? endpoint.position;
            const mouth = renderer?.getDisplaySample(fromSource ? 0 : 1).position ??
                world?.transform?.worldToRender(mouthLocal);
            if (!mouth) continue;
            const delta = subtract3(mouth, this.freePosition);
            const distance = length3(delta);
            const entryDistance = corridor.type === SLICE_TRANSITION_ANOMALY
                ? (route.controller?.getEntryTriggerDistance?.(this.freeSliceId) ?? corridor.entryRadius ?? this.freeEntryDistance)
                : this.freeEntryDistance;
            if (distance > entryDistance) continue;
            const facing = distance < 1e-6 ? 1 :
                dot3(this.getForwardVector(), scale3(delta, 1 / distance));
            if (facing < 0.25) continue;
            const startWorld = world;
            const targetWorld = this.traversal.worlds.get(route.sliceId);
            if (!startWorld || !targetWorld) continue;
            this.lookYawOffset = 0;
            this.lookPitchOffset = 0;
            if (corridor.type === SLICE_TRANSITION_ANOMALY) {
                const position3 = this.renderFreePositionToWorld(startWorld);
                if (!position3) continue;
                const physicalPosition3 = route.controller?.getPhysicalEndpointLocalPosition?.(startWorld, position3) ?? position3;
                const basePosition4D = sliceToAmbient(startWorld.slice.frame, physicalPosition3);
                const position4D = add4(basePosition4D, scale4(startWorld.slice.frame.normal, this.freeW));
                const resolvedAnomaly = corridor.influence?.center ? corridor : resolveAnomalyInfluence(
                    corridor,
                    sliceToAmbient(startWorld.slice.frame, fromSource ? corridor.source.position : corridor.destination.position),
                    sliceToAmbient(targetWorld.slice.frame, fromSource ? corridor.destination.position : corridor.source.position),
                    startWorld.slice.frame.normal
                );
                if (!isInsideBounds(position4D, resolvedAnomaly.region)) continue;
                this.freeTransit = {
                    type: SLICE_TRANSITION_ANOMALY,
                    controller: route.controller ?? this.traversal,
                    anomaly: resolvedAnomaly,
                    corridor: resolvedAnomaly,
                    startWorld,
                    targetWorld,
                    direction: fromSource ? 1 : -1,
                    position4D,
                    bounds: corridor.region,
                    progress: 0
                };
                this.freeW = ambientToSlice(startWorld.slice.frame, position4D).normalOffset;
                this.updateFreeAnomalyCamera();
            } else {
                if (!renderer) continue;
                this.freeTransit = {
                    type: 'bezier',
                    controller: route.controller ?? this.traversal,
                    corridor,
                    corridorId: corridor.id,
                    corridorLength: route.corridorLength ?? route.controller?.corridorLength ?? this.traversal.corridorLength,
                    startWorld, targetWorld,
                    direction: fromSource ? 1 : -1,
                    progress: 0, offsetU: 0, offsetV: 0, offsetW: 0, renderer
                };
                this.updateFreeTransitCamera();
            }
            return true;
        }
        return false;
    }

    getLegacyRoute() {
        const corridor = this.traversal?.corridor;
        if (!corridor) return [];
        const link = this.traversal.topology?.getTraversableNeighbors(this.freeSliceId)
            .find(function findLink(neighbor) { return neighbor.corridorId === corridor.id; });
        return link ? [{
            ...link, corridor,
            fromSource: corridor.source.sliceId === this.freeSliceId,
            renderer: this.traversal.renderers?.get(this.freeSliceId) ?? this.traversal.renderer,
            controller: this.traversal,
            corridorLength: this.traversal.corridorLength
        }] : [];
    }

    updateFreeAnomaly(delta, input) {
        const transit = this.freeTransit;
        const frame = transit.startWorld.slice.frame;
        const speed = this.freeSpeed * (input.sprint ? 2 : 1);
        const forward = this.getForwardVector();
        const right = { x: Math.cos(this.freeYaw), y: 0, z: -Math.sin(this.freeYaw) };
        const movement = add3(
            add3(scale3(forward, input.forward), scale3(right, input.strafe)),
            { x: 0, y: input.vertical, z: 0 }
        );
        const length = length3(movement);
        const distance = length > 1e-9 ? speed * delta / Math.max(1, length) : 0;
        const raw = { ...transit.position4D };
        for (const [index, key] of ['x', 'y', 'z'].entries()) {
            const component = key === 'x' ? movement.x : key === 'y' ? movement.y : movement.z;
            const axis = frame.axes[index];
            raw.x += axis.x * component * distance;
            raw.y += axis.y * component * distance;
            raw.z += axis.z * component * distance;
            raw.w += axis.w * component * distance;
        }
        const influence = evaluateAnomalyInfluence(transit.anomaly ?? transit.corridor, transit.position4D);
        const wStep = input.crossSectionW * speed * delta * influence.wMovementScale;
        raw.x += frame.normal.x * wStep;
        raw.y += frame.normal.y * wStep;
        raw.z += frame.normal.z * wStep;
        raw.w += frame.normal.w * wStep;

        const startPlane = dot4(frame.origin, frame.normal);
        const targetPlane = dot4(transit.targetWorld.slice.frame.origin, frame.normal);
        const currentPlane = dot4(transit.position4D, frame.normal);
        const nextPlane = dot4(raw, frame.normal);
        const targetDirection = Math.sign(targetPlane - startPlane);
        const reachedTarget = targetDirection > 0
            ? nextPlane >= targetPlane - 1e-8
            : nextPlane <= targetPlane + 1e-8;
        const hasEnteredSliceInterval = (currentPlane - startPlane) * targetDirection > 1e-8;
        const returnedToSource = hasEnteredSliceInterval &&
            (nextPlane - currentPlane) * targetDirection < -1e-8 &&
            (nextPlane - startPlane) * targetDirection <= 1e-8;

        transit.position4D = clampAmbient(raw, transit.bounds);
        transit.position4D = constrainAmbientPositionToWorldChart(transit.startWorld, transit.position4D);
        transit.progress = clamp((dot4(transit.position4D, frame.normal) - startPlane) /
            (targetPlane - startPlane), 0, 1);
        if (reachedTarget) {
            transit.position4D = clampAmbient(raw, transit.bounds);
            this.completeFreeAnomaly(transit.targetWorld, transit);
        } else if (returnedToSource) {
            this.completeFreeAnomaly(transit.startWorld, transit);
        } else {
            this.updateFreeAnomalyCamera();
        }
    }

    updateFreeAnomalyCamera() {
        const transit = this.freeTransit;
        if (!transit) return;
        transit.position4D = constrainAmbientPositionToWorldChart(transit.startWorld, transit.position4D);
        const local = ambientToSlice(transit.startWorld.slice.frame, transit.position4D).position;
        this.freePosition = transit.startWorld.transform.worldToRender(local);
        this.freeW = ambientToSlice(transit.startWorld.slice.frame, transit.position4D).normalOffset;
        this.applyFreePose();
    }

    completeFreeAnomaly(world, transit) {
        const ambient = transit.position4D;
        const controller = transit.controller;
        const baseArrival = controller?.createArrivalPosition
            ? controller.createArrivalPosition(world, ambient, transit.startWorld)
            : ambientToSlice(world.slice.frame, ambient).position;
        const arrival = { ...baseArrival, y: baseArrival.y + 4 };
        this.freeSliceId = world.slice.id;
        this.freeEntryLock = { sliceId: world.slice.id, position: { x: arrival.x, y: arrival.y, z: arrival.z } };
        const chartSafeArrival = world.transform.constrainWorldPosition?.(arrival) ?? arrival;
        this.freeW = 0;
        this.setSpatialContext(world.transform, world.layerMask);
        this.freePosition = world.transform.worldToRender(chartSafeArrival);
        this.freeTransit = null;
        this.applyFreePose();
    }

    updateFreeTransit(delta, input) {
        const transit = this.freeTransit;
        const speed = this.freeSpeed * (input.sprint ? 2 : 1);
        transit.progress = clamp(
            transit.progress + input.forward * speed * delta / (transit.corridorLength ?? this.traversal.corridorLength),
            0,
            1
        );
        transit.offsetU += input.strafe * this.freeTransverseSpeed * delta;
        transit.offsetV += input.vertical * this.freeTransverseSpeed * delta;
        transit.offsetW += input.crossSectionW * this.freeTransverseSpeed * delta;
        this.constrainFreeOffset(transit);
        if (transit.progress >= 1 || (transit.progress <= 0 && input.forward < 0)) {
            this.updateFreeTransitCamera();
            const destination = transit.progress >= 1 ? transit.targetWorld : transit.startWorld;
            this.completeFreeTransit(destination, transit);
            return;
        }
        this.updateFreeTransitCamera();
    }

    constrainFreeOffset(transit) {
        const sample = this.getFreeCorridorSample(transit);
        const scale = transit.renderer.options?.displayScale ?? 1;
        const limit = sample.radius / Math.max(scale, 1e-9) * 0.62;
        const length = Math.hypot(transit.offsetU, transit.offsetV, transit.offsetW);
        if (length <= limit || length <= 1e-9) return;
        const factor = limit / length;
        transit.offsetU *= factor;
        transit.offsetV *= factor;
        transit.offsetW *= factor;
    }

    getFreeCorridorSample(transit) {
        const t = transit.direction > 0 ? transit.progress : 1 - transit.progress;
        return transit.renderer.getDisplaySample(t);
    }

    updateFreeTransitCamera() {
        const transit = this.freeTransit;
        const sample = this.getFreeCorridorSample(transit);
        const scale = transit.renderer.options?.displayScale ?? 1;
        const axes = getDisplayAxes(sample);
        const position = add3(
            add3(
                add3(sample.position, scale3(axes[0], transit.offsetU * scale)),
                scale3(axes[1], transit.offsetV * scale)
            ),
            scale3(axes[2], transit.offsetW * scale)
        );
        this.freePosition = position;
        const tangent = normalize3(sample.tangent);
        const tangentYaw = Math.atan2(tangent.x * transit.direction, tangent.z * transit.direction);
        const tangentPitch = Math.asin(clamp(tangent.y * transit.direction, -1, 1));
        this.freeYaw = tangentYaw + this.lookYawOffset;
        this.freePitch = clamp(tangentPitch + this.lookPitchOffset, -1.48, 1.48);
        this.applyFreePose();
        transit.spatialPosition4D = applyAmbientOffsets(sample, transit);
    }

    completeFreeTransit(world, transit) {
        const ambientPosition = transit.spatialPosition4D ?? this.getAmbientPosition(transit);
        const arrival = (transit.controller ?? this.traversal).createArrivalPosition(world, ambientPosition);
        const rawCameraWorld = { ...arrival, y: arrival.y + 4 };
        const cameraWorld = world.transform?.constrainWorldPosition?.(rawCameraWorld) ?? rawCameraWorld;
        this.freeSliceId = world.slice.id;
        this.freeW = world.fourDimensionalRegion ? ambientPosition.w - world.slice.frame.origin.w : 0;
        this.freeW = clamp(this.freeW, world.wBounds?.min ?? -12, world.wBounds?.max ?? 12);
        this.setSpatialContext(world.transform, world.layerMask);
        this.freePosition = world.transform.worldToRender(cameraWorld);
        const sample = transit.renderer.getDisplaySample(transit.direction > 0 ? 1 : 0);
        const tangent = normalize3(sample.tangent);
        const tangentYaw = Math.atan2(tangent.x * transit.direction, tangent.z * transit.direction);
        const tangentPitch = Math.asin(clamp(tangent.y * transit.direction, -1, 1));
        this.freeYaw = tangentYaw + this.lookYawOffset;
        this.freePitch = clamp(tangentPitch + this.lookPitchOffset, -1.48, 1.48);
        this.freeTransit = null;
        this.applyFreePose();
    }

    getAmbientPosition(transit) {
        return applyAmbientOffsets(this.getFreeCorridorSample(transit), transit);
    }

    getForwardVector() {
        const horizontal = Math.cos(this.freePitch);
        return {
            x: Math.sin(this.freeYaw) * horizontal,
            y: Math.sin(this.freePitch),
            z: Math.cos(this.freeYaw) * horizontal
        };
    }

    applyFreePose() {
        if (!this.freePosition) return;
        const forward = this.getForwardVector();
        this.camera.position.set(this.freePosition.x, this.freePosition.y, this.freePosition.z);
        this.camera.setTarget(new BABYLON.Vector3(
            this.freePosition.x + forward.x,
            this.freePosition.y + forward.y,
            this.freePosition.z + forward.z
        ));
    }

    getModeInfo() {
        return {
            mode: this.mode,
            freeSliceId: this.freeSliceId,
            inCorridor: Boolean(this.freeTransit),
            anomalyType: this.freeTransit?.type ?? null,
            corridorProgress: this.freeTransit?.progress ?? null,
            corridorPosition4D: this.freeTransit?.position4D ?? this.freeTransit?.spatialPosition4D ?? null,
            fourthCoordinate: this.freeW,
            speed: this.freeSpeed
        };
    }

    getFreeTransitState() {
        return this.freeTransit;
    }

    getViewSliceId() {
        if (this.mode === 'free') {
            return this.freeTransit?.startWorld.slice.id ??
                this.freeSliceId ?? this.traversal?.currentSliceId ?? this.character.sliceId;
        }
        if (this.traversal?.state === 'transit') return this.traversal.startWorld.slice.id;
        return this.traversal?.currentSliceId ?? this.character.sliceId;
    }

    getFreeW() { return this.mode === 'free' ? this.freeW : 0; }

    getFreeWorldPosition() {
        if (this.mode !== 'free' || this.freeTransit || !this.freePosition) return null;
        const world = this.traversal?.worlds.get(this.freeSliceId);
        if (typeof world?.transform?.renderToWorld !== 'function') return null;
        const position = this.renderFreePositionToWorld(world);
        return isFinitePosition(position) ? position : null;
    }

    constrainFreePosition(world, position = this.freePosition) {
        if (!position) return position;
        const constrained = world?.transform?.constrainRenderPosition?.(position) ?? position;
        this.freePosition = copyVector(constrained);
        return this.freePosition;
    }

    renderFreePositionToWorld(world) {
        if (typeof world?.transform?.renderToWorld !== 'function' || !this.freePosition) return null;
        this.constrainFreePosition(world);
        try {
            return world.transform.renderToWorld(this.freePosition);
        } catch (error) {
            // A chart-boundary miss should prevent corridor entry for this
            // frame, not terminate the entire render/update loop.
            if (error instanceof RangeError) return null;
            throw error;
        }
    }

    dispose() {
        if (this.disposed) return;
        window.removeEventListener('keydown', this.onFreeKeyDown);
        window.removeEventListener('keyup', this.onFreeKeyUp);
        window.removeEventListener('blur', this.onFreeBlur);
        this.canvas?.removeEventListener('pointerdown', this.onFreePointerDown);
        window.removeEventListener('pointermove', this.onFreePointerMove);
        window.removeEventListener('pointerup', this.onFreePointerUp);
        this.playerController?.setInputEnabled(true);
        this.freeKeys.clear();
        this.freeTransit = null;
        super.dispose();
    }
}

function constrainAmbientPositionToWorldChart(world, position4D) {
    if (typeof world?.transform?.constrainWorldPosition !== 'function') return position4D;
    const frame = world.slice.frame;
    const local = ambientToSlice(frame, position4D);
    const constrained = world.transform.constrainWorldPosition(local.position);
    if (['x', 'y', 'z'].every((axis) => Math.abs(constrained[axis] - local.position[axis]) <= 1e-10)) {
        return position4D;
    }
    return add4(
        sliceToAmbient(frame, constrained),
        scale4(frame.normal, local.normalOffset)
    );
}

function applyAmbientOffsets(sample, transit) {
    const position = { ...sample.spatialPosition4D };
    for (const [index, key] of ['offsetU', 'offsetV', 'offsetW'].entries()) {
        const axis = sample.axes4D[index];
        const amount = transit[key];
        position.x += axis.x * amount;
        position.y += axis.y * amount;
        position.z += axis.z * amount;
        position.w += axis.w * amount;
    }
    return position;
}

function getDisplayAxes(sample) {
    const first = normalize3(sample.axes?.[0] ?? sample.crossSectionAxes?.[0]);
    const secondRaw = normalize3(sample.axes?.[1] ?? sample.crossSectionAxes?.[1]);
    const second = normalize3(subtract3(secondRaw, scale3(first, dot3(secondRaw, first))));
    const third = normalize3(sample.crossSectionAxes?.[2] ?? cross3(sample.tangent, first));
    return [first, second, length3(third) > 1e-9 ? third : normalize3(cross3(first, second))];
}

function isInsideBounds(position, bounds) {
    return ['x', 'y', 'z', 'w'].every((axis) =>
        position[axis] >= bounds.min[axis] - 1e-7 && position[axis] <= bounds.max[axis] + 1e-7
    );
}

function clampAmbient(position, bounds) {
    return {
        x: clamp(position.x, bounds.min.x, bounds.max.x),
        y: clamp(position.y, bounds.min.y, bounds.max.y),
        z: clamp(position.z, bounds.min.z, bounds.max.z),
        w: clamp(position.w, bounds.min.w, bounds.max.w)
    };
}

function keyAxis(keys, positive, negative) {
    return Number(keys.has(positive)) - Number(keys.has(negative));
}

function copyVector(value) { return { x: value.x, y: value.y, z: value.z }; }
function add3(left, right) { return { x: left.x + right.x, y: left.y + right.y, z: left.z + right.z }; }
function subtract3(left, right) { return { x: left.x - right.x, y: left.y - right.y, z: left.z - right.z }; }
function scale3(value, amount) { return { x: value.x * amount, y: value.y * amount, z: value.z * amount }; }
function dot3(left, right) { return left.x * right.x + left.y * right.y + left.z * right.z; }
function cross3(left, right) {
    return {
        x: left.y * right.z - left.z * right.y,
        y: left.z * right.x - left.x * right.z,
        z: left.x * right.y - left.y * right.x
    };
}
function length3(value) { return Math.hypot(value.x, value.y, value.z); }
function normalize3(value) {
    const length = length3(value);
    return length > 1e-9 ? scale3(value, 1 / length) : { x: 0, y: 0, z: 1 };
}
function clamp(value, minimum, maximum) { return Math.min(maximum, Math.max(minimum, value)); }

function isFinitePosition(position) {
    return Boolean(position && ['x', 'y', 'z'].every(function finiteAxis(axis) {
        return Number.isFinite(position[axis]);
    }));
}

function createTransitionRegion(sourcePoint, destinationPoint, padding = {}) {
    const horizontalPadding = padding.horizontalPadding ?? 6;
    const verticalPadding = padding.verticalPadding ?? 6;
    const wPadding = padding.wPadding ?? 6;
    const min = {
        x: Math.min(sourcePoint.x, destinationPoint.x) - horizontalPadding,
        y: Math.min(sourcePoint.y, destinationPoint.y) - verticalPadding,
        z: Math.min(sourcePoint.z, destinationPoint.z) - horizontalPadding,
        w: Math.min(sourcePoint.w, destinationPoint.w) - wPadding
    };
    const max = {
        x: Math.max(sourcePoint.x, destinationPoint.x) + horizontalPadding,
        y: Math.max(sourcePoint.y, destinationPoint.y) + verticalPadding,
        z: Math.max(sourcePoint.z, destinationPoint.z) + horizontalPadding,

function createAnomalyBoundaryRenderers(scene, portalPairs, registry, layerMasks) {
    const entries = [];
    for (const pair of portalPairs) {
        if (!['open-4d-hypercube', 'open-4d-cylinder'].includes(pair.corridor.view?.interiorBoundary)) continue;
        for (const sliceId of [pair.sourceSliceId, pair.destinationSliceId]) {
            const binding = registry.require(sliceId);
            const endpoint = pair.corridor.source.sliceId === sliceId
                ? pair.corridor.source : pair.corridor.destination;
            const endpointAmbient = sliceToAmbient(binding.slice.frame, endpoint.position);
            const sourceBinding = registry.require(pair.corridor.source.sliceId);
            const destinationBinding = registry.require(pair.corridor.destination.sliceId);
            const sourceAmbient = sliceToAmbient(sourceBinding.slice.frame, pair.corridor.source.position);
            const destinationAmbient = sliceToAmbient(
                destinationBinding.slice.frame, pair.corridor.destination.position
            );
            const renderer = new AnomalyBoundaryRenderer(
                scene,
                pair.corridor,
                binding.slice,
                binding.spatialTransform,
                {
                    layerMask: layerMasks.get(sliceId),
                    presentationEndpoint: pair.presentationEndpointsBySlice?.[sliceId] ?? null,
                    sourceAmbient,
                    destinationAmbient,
                    gridSegments: 5,
                    opacity: 0.11,
                    edgeAlpha: 0.72
                }
            );
            entries.push({
                anomaly: pair.corridor,
                corridorId: pair.corridor.id,
                sliceId,
                endpointW: endpointAmbient.w,
                renderer
            });
        }
    }
    return entries;
}

function createAnomalyTraversalController(options) {
    const { anomaly, corridorRenderers, ...rest } = options;
    if (anomaly.type === SLICE_TRANSITION_ANOMALY) {
        return new SliceTransitionTraversalController({
            ...rest,
            anomaly,
            corridor: anomaly,
            corridorRenderers
        });
    }
    return new CorridorTraversalController({
        ...rest,
        corridor: anomaly,
        corridorRenderer: corridorRenderers?.get(options.source.slice.id),
        corridorRenderers
    });
}

function createDemoWorldBinding(slice, transform, terrain, layerMask, address, options = {}) {
    return {
        slice, transform, terrain, layerMask, address,
        fourDimensionalRegion: Boolean(options.fourDimensionalRegion),
        wBounds: options.wBounds ?? null,
        movementProfile: options.movementProfile ? { ...options.movementProfile } : null,
        spatialNeighbors: [], adjacentWorldChunks: [],
        setSpatialNeighbors: function setSpatialNeighbors(neighbors, worldChunks) {
            this.spatialNeighbors = neighbors;
            this.adjacentWorldChunks = worldChunks;
        }
    };
}

function createMainPreviewCamera(transform, anchor) {
    const bounds = transform.projection?.bounds;
    const cameraWorld = {
        x: anchor.x,
        y: Math.min(anchor.y + 90, (bounds?.maxY ?? anchor.y + 90) - 1),
        z: Math.max(anchor.z - 260, (bounds?.minZ ?? anchor.z - 260) + 1)
    };
    const targetWorld = {
        x: anchor.x,
        y: clamp(anchor.y + 18, (bounds?.minY ?? anchor.y - 18) + 1, (bounds?.maxY ?? anchor.y + 18) - 1),
        z: clamp(anchor.z + 520, (bounds?.minZ ?? anchor.z + 520) + 1, (bounds?.maxZ ?? anchor.z + 520) - 1)
    };
    const position = transform.worldToRender(cameraWorld);
    const target = transform.worldToRender(targetWorld);
    return {
        x: position.x, y: position.y, z: position.z,
        targetX: target.x, targetY: target.y, targetZ: target.z,
        fov: 0.88, maxZ: 100000
    };
}

class GeometryDemoWorld {
    constructor(resources) {
        Object.assign(this, resources);
        this.active = false;
        this.disposed = false;
        this.corridorRenderer = this.mainSourceRenderer;
        this.destinationCorridorRenderer = this.mainDestinationRenderer;
        this.didStart = false;
        this.gridEnabled = false;
        this.currentSliceId = this.hypercubeSlice.id;
        this.grids = new Map([
            [this.sourceSlice.id, this.sourceGrid],
            [this.hypercubeSlice.id, this.hypercubeGrid],
            [this.destination.slice.id, this.mainWorldGrid]
        ]);
        this.onGridKeyDown = this.handleGridKeyDown.bind(this);
        if (typeof window !== 'undefined') window.addEventListener('keydown', this.onGridKeyDown);
        this.initializeOpticalSpotlight();
        this.corridorRenderers = [
            [this.sourceSlice.id, this.mainSourceRenderer],
            [this.destination.slice.id, this.mainDestinationRenderer],
            [this.hypercubeSlice.id, this.hypercubeSourceRenderer],
            [this.sourceSlice.id, this.hypercubeDestinationRenderer]
        ].filter(([, renderer]) => Boolean(renderer))
            .map(([sliceId, renderer]) => ({ sliceId, renderer }));
        this.setActive(false);
    }

    setActive(active) {
        if (this.disposed) return;
        this.active = Boolean(active);
        for (const mesh of [...this.sourceMeshes, ...this.hypercubeMeshes]) mesh.isVisible = this.active;
        this.player.root.setEnabled(this.active && this.cameraController.mode !== 'free');
        this.traversalController?.setActive(this.active);
        for (const entry of this.corridorRenderers) entry.renderer.mesh.isVisible = this.active;
        for (const entry of this.anomalyBoundaryRenderers ?? []) {
            entry.renderer.setVisible(this.active && entry.anomaly.visible);
        }
        for (const pair of this.portalPairs) {
            pair.sourcePortal.setVisible(false);
            pair.destinationPortal.setVisible(false);
            pair.passagePresentation?.deactivate(this.camera, this.getWorld(this.cameraController.getViewSliceId()).layerMask);
        }
        if (!this.active) {
            for (const grid of this.grids.values()) grid.setActive(false);
            this.spotlightSourceLight?.setEnabled(false);
            this.spotlightMarker && (this.spotlightMarker.isVisible = false);
            this.spotlightAimLine && (this.spotlightAimLine.isVisible = false);
            this.destinationSpotlight?.setEnabled(false);
            if (this.destinationSpotlightRayLine) this.destinationSpotlightRayLine.isVisible = false;
            this.clearOpticalSpotlightShadowCasters();
            return;
        }
        if (this.cameraController.mode !== 'follow') this.cameraController.setMode('follow');
        this.spotlightSourceLight?.setEnabled(Boolean(this.spotlightTestRayEnabled));
        this.spotlightMarker && (this.spotlightMarker.isVisible = true);
        this.spotlightAimLine && (this.spotlightAimLine.isVisible = true);
        this.spotlightNeedsTrace = true;
        this.didStart = true;
        const transitioning = this.traversalController?.state === 'transit';
        this.activateSlice(this.currentSliceId, {
            transitioning,
            renderSliceId: this.traversalController?.startWorld?.slice.id ?? this.currentSliceId
        });
        this.updateCameraPresentation();
        this.updatePortalLists();
    }

    activateSlice(sliceId, state = {}) {
        this.currentSliceId = sliceId;
        const transitioning = Boolean(state.transitioning);
        const renderWorld = this.getWorld(transitioning ? (state.renderSliceId ?? sliceId) : sliceId);
        this.camera.layerMask = renderWorld.layerMask;
        this.cameraController.setSpatialContext(renderWorld.transform, renderWorld.layerMask);
        this.player.setLayerMask(renderWorld.layerMask);
        this.updateCameraPresentation();
    }

    getWorld(sliceId) {
        const world = [this.hypercubeWorld, this.sourceWorld, this.destinationWorld]
            .find(function matchWorld(entry) { return entry.slice.id === sliceId; });
        if (!world) throw new Error(`Unknown geometry demo slice: ${sliceId}`);
        return world;
    }

    update(deltaSeconds) {
        if (!this.active || this.disposed) return;
        // A follow camera smooths from its previous position. Remove the
        // passage's temporary display-space translation before letting that
        // controller calculate its next authoritative camera pose; otherwise
        // the follow-camera lerp would absorb the visual offset and make it
        // accumulate or drift across frames.
        for (const pair of this.portalPairs ?? []) {
            pair.passagePresentation?.restoreCameraPoseForControllerUpdate(this.camera);
        }
        this.playerController.update(deltaSeconds);
        this.traversalController.update(deltaSeconds);
        this.cameraController.update(deltaSeconds);
        const viewSliceId = this.cameraController.getViewSliceId();
        const ambientPosition4D = this.cameraController.mode === 'free'
            ? this.cameraController.getFreeAmbientPosition()
            : this.traversalController.getAmbientPlayerPosition?.();
        this.updateAnomalyBoundaryRenderers(viewSliceId, ambientPosition4D);
        let viewerW = 0;
        if (viewSliceId === this.hypercubeSlice.id) {
            if (this.cameraController.mode === 'free') {
                viewerW = this.cameraController.getFreeW();
            } else if (ambientPosition4D) {
                viewerW = ambientToSlice(this.hypercubeSlice.frame, ambientPosition4D).normalOffset;
            }
        }
        this.hypercubeRenderer.update(deltaSeconds, viewerW);
        this.updateHypercubeSliceHud();
        this.updateCameraPresentation();
        this.updatePortalPreview();
        this.updatePortalLists();
        this.updateOpticalSpotlight(deltaSeconds);
        const focus = this.getDestinationFocus();
        const focusDirection = this.destinationPreviewDirection ?? { x: 0, z: 1 };
        const reachabilityContext = ambientPosition4D ? {
            sourceSliceId: this.cameraController.mode === 'free'
                ? this.cameraController.freeSliceId : this.traversalController.currentSliceId,
            position4D: ambientPosition4D,
            influenceOptions: this.playerController.getAnomalyInfluenceModifiers?.() ??
                this.playerController.anomalyInfluenceModifiers ?? {}
        } : null;
        this.destinationWorldManager?.update(focus, focusDirection, reachabilityContext);
        const destinationMeshes = this.destinationWorldManager?.getMeshes() ?? [];
        const destinationPreviewMeshes = getDestinationPreviewRenderList(
            [...destinationMeshes, ...this.getAnomalyBoundaryMeshes(this.destination.slice.id)],
            this.destinationSpotlightRayLine
        );
        this.mainSourcePortal.setRenderList(destinationPreviewMeshes);
        this.hypercubeMainPortal.setRenderList(destinationPreviewMeshes);
    }

    updateHypercubeSliceHud() {
        const viewSliceId = this.cameraController.getViewSliceId();
        if (viewSliceId !== this.hypercubeSlice.id) {
            const position4D = this.cameraController.mode === 'free'
                ? this.cameraController.getFreeAmbientPosition?.()
                : this.traversalController.getAmbientPlayerPosition?.();
            const routes = this.traversalController.getRoutesForSlice?.(viewSliceId) ?? [];
            const availableFields = position4D ? routes.map((route) => ({
                route,
                influence: route.controller?.getAnomalyInfluence?.(position4D)
            })).filter((entry) => Number(entry.influence?.wMovementScale) > 0) : [];
            availableFields.sort((left, right) =>
                right.influence.wMovementScale - left.influence.wMovementScale
            );
            const best = availableFields[0] ?? null;
            const freedomPercent = best
                ? Math.round(best.influence.wMovementScale * 100) : 0;
            const label = best
                ? `W FREEDOM ACTIVE · ${freedomPercent}%`
                : 'W SLICE INACTIVE';
            if (label !== this.lastHypercubeSliceLabel) {
                this.lastHypercubeSliceLabel = label;
                this.onSliceInfoChanged?.(label, best ? {
                    anomalyId: best.route.corridorId,
                    freedom: best.influence.freedom,
                    wMovementScale: best.influence.wMovementScale,
                    wReach: best.influence.wReach,
                    position4D: { ...position4D }
                } : null);
            }
            return;
        }
        const info = this.hypercubeRenderer.getInfo();
        const active = this.cameraController.mode === 'free'
            ? this.cameraController.getFreeTransitState()
            : this.traversalController.getActiveController?.();
        const inAnomaly = (active?.state === 'transit' || active?.type === SLICE_TRANSITION_ANOMALY) &&
            active.startWorld?.slice.id === this.hypercubeSlice.id &&
            (active.spatialPosition4D || active.position4D);
        const label = inAnomaly
            ? `ANOMALY W ${info.viewerW.toFixed(2)} · ${info.sliceVertexCount} VERTICES · ${info.sliceFaceCount} FACES`
            : `W ${info.viewerW.toFixed(2)} · ${info.sliceVertexCount} VERTICES · ${info.sliceFaceCount} FACES`;
        if (label === this.lastHypercubeSliceLabel) return;
        this.lastHypercubeSliceLabel = label;
        this.onSliceInfoChanged?.(label, info);
    }

    updateCameraPresentation() {
        if (!this.corridorRenderers) return updateLegacyCameraPresentation(this);
        const viewSliceId = this.cameraController.getViewSliceId();
        const world = this.getWorld(viewSliceId);
        this.camera.layerMask = world.layerMask;
        for (const entry of this.corridorRenderers) {
            entry.renderer.mesh.isVisible = this.active && entry.sliceId === viewSliceId;
        }
        for (const [sliceId, grid] of this.grids) {
            grid.setActive(this.active && sliceId === viewSliceId);
        }
        for (const pair of this.portalPairs) {
            const sourceActive = pair.sourceSliceId === viewSliceId;
            const destinationActive = pair.destinationSliceId === viewSliceId;
            const viewEnabled = pair.corridor.view?.enabled !== false;
            pair.sourcePortal.setVisible(this.active && viewEnabled && sourceActive);
            pair.destinationPortal.setVisible(this.active && viewEnabled && destinationActive);
        }
        this.onViewChanged?.(viewSliceId);
    }

    updatePortalLists() {
        const poincareMeshes = [
            ...this.sourceMeshes,
            ...this.getAnomalyBoundaryMeshes(this.sourceSlice.id),
            this.mainSourceRenderer?.mesh,
            this.hypercubeDestinationRenderer?.mesh
        ].filter(Boolean);
        const hypercubeMeshes = [
            ...this.hypercubeMeshes,
            ...this.getAnomalyBoundaryMeshes(this.hypercubeSlice.id),
            this.hypercubeSourceRenderer?.mesh
        ].filter(Boolean);
        this.mainDestinationPortal.setRenderList(poincareMeshes);
        this.hypercubeSourcePortal.setRenderList(poincareMeshes);
        this.hypercubeDestinationPortal.setRenderList(hypercubeMeshes);
        this.hypercubeMainDestinationPortal.setRenderList(hypercubeMeshes);
    }

    getAnomalyBoundaryMeshes(sliceId) {
        return (this.anomalyBoundaryRenderers ?? [])
            .filter((entry) => entry.sliceId === sliceId)
            .flatMap((entry) => [entry.renderer.mesh, entry.renderer.edgeMesh])
            .filter(Boolean);
    }

    updateAnomalyBoundaryRenderers(viewSliceId, ambientPosition4D) {
        for (const entry of this.anomalyBoundaryRenderers ?? []) {
            const region = entry.anomaly.region;
            const viewerInsideWSpan = ambientPosition4D &&
                ambientPosition4D.w >= region.min.w - 1e-7 &&
                ambientPosition4D.w <= region.max.w + 1e-7;
            const viewerW = entry.sliceId === viewSliceId && viewerInsideWSpan
                ? ambientPosition4D.w : entry.endpointW;
            entry.renderer.update(viewerW);
            entry.renderer.setVisible(this.active && entry.anomaly.visible);
        }
    }

    toggleSpotlightEditor() {
        this.setSpotlightEditorEnabled(!this.spotlightEditorActive);
        return this.spotlightEditorActive;
    }

    setSpotlightEditorEnabled(enabled) {
        this.spotlightEditorActive = Boolean(enabled);
        if (!this.spotlightEditorActive) this.handleSpotlightPointerUp();
        this.onSpotlightEditorStateChanged?.(this.spotlightEditorActive);
        return this.spotlightEditorActive;
    }

    toggleSpotlightTestRay() {
        return this.setSpotlightTestRayEnabled(!this.spotlightTestRayEnabled);
    }

    getDestinationFocus() {
        if (this.currentSliceId !== this.destination.slice.id && this.destinationPreviewFocus) {
            return this.destinationPreviewFocus;
        }
        if (this.cameraController.mode === 'free' &&
            this.cameraController.freeSliceId === this.destination.slice.id) {
            return this.cameraController.getFreeWorldPosition() ?? this.mainAnchor;
        }
        return this.currentSliceId === this.destination.slice.id
            ? this.player.worldPosition : this.mainAnchor;
    }

    beginEntityTransit(entity, fromSliceId = entity?.sliceId) {
        return this.traversalController.beginEntityTransit(entity, fromSliceId);
    }

    tryBeginEntityTransit(entity, fromSliceId = entity?.sliceId) {

    }

    toggleGrid() {
        const sliceId = this.cameraController.getViewSliceId();
        if (!this.grids.has(sliceId)) return false;
        this.gridEnabled = !this.gridEnabled;
        for (const grid of this.grids.values()) grid.setVisible(this.gridEnabled);
        this.updateCameraPresentation();
        return this.gridEnabled;
    }

    initializeOpticalSpotlight() {
        const mouth = this.hypercubeMainCorridor.source.position;
        this.spotlightSource = aimOpticalSpotlightSource(createOpticalSpotlightSource({
            id: 'hypercube-main-world-spotlight',
            sliceId: this.hypercubeSlice.id,
            // Aim slightly upward through the mouth so the transported ray
            // emerges just above the Main World terrain instead of underground.
            position: { x: mouth.x - 10, y: mouth.y - 0.75, z: mouth.z },
            direction: { x: 1, y: -0.25, z: 0 },
            color: [1, 0.78, 0.48],
            intensity: 1.8,
            range: 64,
            coneAngle: 0.52,
            exponent: 3,
            enabled: true,
            castsShadows: true
        }), mouth);
        this.spotlightEditorActive = false;
        this.spotlightTestRayEnabled = true;
        this.spotlightPointerDragging = false;
        this.spotlightPreviousPointer = { x: 0, y: 0 };
        this.spotlightPointerSensitivity = 0.0035;
        this.spotlightNeedsTrace = true;
        this.spotlightTraceState = { transmitted: false, reason: 'not-traced-yet' };
        this.spotlightShadowCasters = new Set();
        this.spotlightOriginalReceiveShadows = new Map();
        this.spotlightShadowRefreshElapsed = Infinity;
        this.spotlightShadowCasterLimit = 18;
        this.spotlightVirtualSourceDistance = 0.5;
        this.spotlightDebugRayLength = 400;

        const anomalyId = this.hypercubeMainCorridor.id;
        const restrictedTopology = {
            getCorridor: (id) => this.topology.getCorridor(id),
            getOpticallyConnectedNeighbors: (sliceId) =>
                this.topology.getOpticallyConnectedNeighbors(sliceId)
                    .filter((neighbor) => neighbor.corridorId === anomalyId),
            hasOpticalConnection: (fromSliceId, toSliceId) =>
                this.topology.getOpticallyConnectedNeighbors(fromSliceId)
                    .some((neighbor) => neighbor.corridorId === anomalyId && neighbor.sliceId === toSliceId)
        };
        this.spotlightTracer = this.opticalScene
            ? createAnomalyOpticalTracer({
                topology: restrictedTopology,
                sliceRegistry: this.registry,
                sceneRegistry: this.opticalScene,
                chunkGraph: this.chunkGraph
            })
            : null;

        if (typeof BABYLON === 'undefined' || !this.scene) return;

        const sourceTransform = this.hypercubeWorld.transform;
        const sourcePosition = sourceTransform.worldToRender(this.spotlightSource.position);
        const sourceTarget = sourceTransform.worldToRender(add3(
            this.spotlightSource.position,
            scale3(this.spotlightSource.direction, 12)
        ));
        const sourceDirection = normalize3(subtract3(sourceTarget, sourcePosition));
        this.spotlightSourceLight = new BABYLON.SpotLight(
            'geometryDemoOpticalSpotlightSource',
            toBabylonVector3(sourcePosition),
            toBabylonVector3(sourceDirection),
            this.spotlightSource.coneAngle,
            this.spotlightSource.exponent,
            this.scene
        );
        this.spotlightSourceLight.diffuse = BABYLON.Color3.FromArray(this.spotlightSource.color);
        this.spotlightSourceLight.specular = BABYLON.Color3.FromArray(this.spotlightSource.color);
        this.spotlightSourceLight.intensity = this.spotlightSource.intensity;
        this.spotlightSourceLight.range = distance3(sourcePosition, sourceTarget) + 8;
        this.spotlightSourceLight.includeOnlyWithLayerMask = HYPERCUBE_LAYER;
        this.spotlightSourceLight.setEnabled(false);

        this.spotlightMarkerMaterial = new BABYLON.StandardMaterial(
            'geometryDemoOpticalSpotlightMarkerMaterial', this.scene
        );
        this.spotlightMarkerMaterial.disableLighting = true;
        this.spotlightMarkerMaterial.diffuseColor = BABYLON.Color3.FromArray(this.spotlightSource.color);
        this.spotlightMarkerMaterial.emissiveColor = BABYLON.Color3.FromArray(this.spotlightSource.color).scale(0.9);
        this.spotlightMarkerMaterial.specularColor = BABYLON.Color3.Black();
        this.spotlightMarker = BABYLON.MeshBuilder.CreateSphere(
            'geometryDemoOpticalSpotlightMarker', { diameter: 0.72, segments: 16 }, this.scene
        );
        this.spotlightMarker.position.copyFrom(toBabylonVector3(sourcePosition));
        this.spotlightMarker.material = this.spotlightMarkerMaterial;
        this.spotlightMarker.layerMask = HYPERCUBE_LAYER;
        this.spotlightMarker.isPickable = false;
        this.spotlightMarker.isVisible = false;

        this.spotlightAimLine = BABYLON.MeshBuilder.CreateLines(
            'geometryDemoOpticalSpotlightAim', {
                points: [toBabylonVector3(sourcePosition), toBabylonVector3(sourceTarget)]
            }, this.scene
        );
        this.spotlightAimLine.color = BABYLON.Color3.FromArray(this.spotlightSource.color);
        this.spotlightAimLine.layerMask = HYPERCUBE_LAYER;
        this.spotlightAimLine.isPickable = false;
        this.spotlightAimLine.isVisible = false;

        this.destinationSpotlight = new BABYLON.SpotLight(
            'geometryDemoTransmittedSpotlight', BABYLON.Vector3.Zero(),
            new BABYLON.Vector3(1, 0, 0), this.spotlightSource.coneAngle,
            this.spotlightSource.exponent, this.scene
        );
        this.destinationSpotlight.diffuse = BABYLON.Color3.FromArray(this.spotlightSource.color);
        this.destinationSpotlight.specular = BABYLON.Color3.FromArray(this.spotlightSource.color);
        this.destinationSpotlight.intensity = 0;
        this.destinationSpotlight.includeOnlyWithLayerMask = MAIN_WORLD_LAYER;
        this.destinationSpotlight.includedOnlyMeshes = [];
        this.destinationSpotlight.setEnabled(false);

        this.destinationSpotlightRayLine = BABYLON.MeshBuilder.CreateLines(
            'geometryDemoTransmittedSpotlightRay', {
                points: [BABYLON.Vector3.Zero(), new BABYLON.Vector3(1, 0, 0)]
            }, this.scene
        );
        this.destinationSpotlightRayLine.color = BABYLON.Color3.FromArray(this.spotlightSource.color);
        this.destinationSpotlightRayLine.alpha = 0.96;
        this.destinationSpotlightRayLine.layerMask = MAIN_WORLD_LAYER;
        this.destinationSpotlightRayLine.isPickable = false;
        this.destinationSpotlightRayLine.alwaysSelectAsActiveMesh = true;
        this.destinationSpotlightRayLine.metadata = {
            ...(this.destinationSpotlightRayLine.metadata || {}),
            opticalDebugRay: true,
            anomalyId: this.hypercubeMainCorridor.id,
            sourceSliceId: this.hypercubeSlice.id,
            destinationSliceId: this.destination.slice.id
        };
        this.destinationSpotlightRayLine.isVisible = false;

        if (this.spotlightSource.castsShadows && BABYLON.ShadowGenerator) {
            try {
                this.spotlightShadowGenerator = new BABYLON.ShadowGenerator(1024, this.destinationSpotlight);
                this.spotlightShadowGenerator.bias = 0.0005;
                this.spotlightShadowGenerator.normalBias = 0.02;
                this.spotlightShadowGenerator.usePercentageCloserFiltering = true;
                if (BABYLON.ShadowGenerator.QUALITY_MEDIUM !== undefined) {
                    this.spotlightShadowGenerator.filteringQuality = BABYLON.ShadowGenerator.QUALITY_MEDIUM;
                }
            } catch (error) {
                console.warn('[GeometryDemo] Transmitted spotlight shadow map unavailable:', error);
                this.spotlightShadowGenerator = null;
            }
        }

        this.onSpotlightEditorKeyDown = this.handleSpotlightEditorKeyDown.bind(this);
        this.onSpotlightPointerDown = this.handleSpotlightPointerDown.bind(this);
        this.onSpotlightPointerMove = this.handleSpotlightPointerMove.bind(this);
        this.onSpotlightPointerUp = this.handleSpotlightPointerUp.bind(this);
        this.onSpotlightPointerBlur = this.handleSpotlightPointerUp.bind(this);
        if (typeof window !== 'undefined') {
            window.addEventListener('keydown', this.onSpotlightEditorKeyDown, true);
            window.addEventListener('pointerdown', this.onSpotlightPointerDown, true);
            window.addEventListener('pointermove', this.onSpotlightPointerMove, true);
            window.addEventListener('pointerup', this.onSpotlightPointerUp, true);
            window.addEventListener('blur', this.onSpotlightPointerBlur);
        }
        this.syncSpotlightSourcePresentation();
    }

    syncSpotlightSourcePresentation() {
        if (!this.spotlightSource || !this.spotlightSourceLight || typeof BABYLON === 'undefined') return;
        const transform = this.hypercubeWorld.transform;
        const sourcePosition = transform.worldToRender(this.spotlightSource.position);
        const sourceTarget = transform.worldToRender(add3(
            this.spotlightSource.position,
            scale3(this.spotlightSource.direction, 12)
        ));
        const direction = normalize3(subtract3(sourceTarget, sourcePosition));
        this.spotlightSourceLight.position.copyFrom(toBabylonVector3(sourcePosition));
        this.spotlightSourceLight.direction.copyFrom(toBabylonVector3(direction));
        this.spotlightSourceLight.angle = this.spotlightSource.coneAngle;
        this.spotlightSourceLight.exponent = this.spotlightSource.exponent;
        this.spotlightSourceLight.intensity = this.spotlightSource.intensity;
        this.spotlightSourceLight.range = distance3(sourcePosition, sourceTarget) + 8;
        this.spotlightMarker?.position.copyFrom(toBabylonVector3(sourcePosition));
        if (this.spotlightAimLine) {
            this.spotlightAimLine = BABYLON.MeshBuilder.CreateLines(
                'geometryDemoOpticalSpotlightAim', {
                    points: [toBabylonVector3(sourcePosition), toBabylonVector3(sourceTarget)],
                    instance: this.spotlightAimLine
                }, this.scene
            );
        }
    }

    handleSpotlightEditorKeyDown(event) {
        if (!this.active || this.disposed || event.ctrlKey || event.metaKey || event.altKey) return;
        const tagName = String(event.target?.tagName ?? '').toUpperCase();
        if (['INPUT', 'TEXTAREA', 'SELECT'].includes(tagName) || event.target?.isContentEditable) return;
        if (event.code === 'KeyP' && !event.repeat) {
            event.preventDefault?.();
            event.stopImmediatePropagation?.();
            this.setSpotlightTestRayEnabled(!this.spotlightTestRayEnabled);
            console.info(`[OpticalSpotlight] Test light/ray ${this.spotlightTestRayEnabled ? 'ON' : 'OFF'}.`);
            return;
        }
        if (event.code === 'KeyL' && !event.repeat) {
            event.preventDefault?.();
            event.stopImmediatePropagation?.();
            this.setSpotlightEditorEnabled(!this.spotlightEditorActive);
            console.info(this.spotlightEditorActive
                ? '[OpticalSpotlight] Edit mode ON: WASD/arrows move X/Z, R/F move Y, U/O yaw, I/K pitch; drag mouse to aim; Esc exits.'
                : '[OpticalSpotlight] Edit mode OFF.');
            return;
        }
        if (!this.spotlightEditorActive) return;
        if (event.code === 'Escape') {
            event.preventDefault?.();
            event.stopImmediatePropagation?.();
            this.setSpotlightEditorEnabled(false);
            console.info('[OpticalSpotlight] Edit mode OFF.');
            return;
        }
        const step = event.shiftKey ? 2 : 0.5;
        const angle = event.shiftKey ? Math.PI / 12 : Math.PI / 36;
        const movement = {
            ArrowLeft: { x: -step, y: 0, z: 0 }, KeyA: { x: -step, y: 0, z: 0 },
            ArrowRight: { x: step, y: 0, z: 0 }, KeyD: { x: step, y: 0, z: 0 },
            ArrowUp: { x: 0, y: 0, z: step }, KeyW: { x: 0, y: 0, z: step },
            ArrowDown: { x: 0, y: 0, z: -step }, KeyS: { x: 0, y: 0, z: -step },
            KeyR: { x: 0, y: step, z: 0 }, KeyF: { x: 0, y: -step, z: 0 }
        }[event.code];
        const rotations = {
            KeyU: [-angle, 0], KeyO: [angle, 0],
            KeyI: [0, angle], KeyK: [0, -angle]
        }[event.code];
        if (!movement && !rotations) return;
        event.preventDefault?.();
        event.stopImmediatePropagation?.();
        this.spotlightSource = movement
            ? moveOpticalSpotlightSource(this.spotlightSource, movement)
            : rotateOpticalSpotlightSource(this.spotlightSource, rotations[0], rotations[1]);
        this.syncSpotlightSourcePresentation();
        this.spotlightNeedsTrace = true;
    }

    handleSpotlightPointerDown(event) {
        if (!this.active || this.disposed || !this.spotlightEditorActive || event.button !== 0) return;
        const canvas = this.scene?.getEngine?.()?.getRenderingCanvas?.();
        if (canvas && event.target !== canvas && !canvas.contains?.(event.target)) return;
        this.spotlightPointerDragging = true;
        this.spotlightPreviousPointer = { x: event.clientX, y: event.clientY };
        event.preventDefault?.();
        event.stopImmediatePropagation?.();
    }

    handleSpotlightPointerMove(event) {
        if (!this.spotlightPointerDragging || !this.active || !this.spotlightEditorActive || this.disposed) return;
        const yaw = (event.clientX - this.spotlightPreviousPointer.x) * this.spotlightPointerSensitivity;
        const pitch = -(event.clientY - this.spotlightPreviousPointer.y) * this.spotlightPointerSensitivity;
        this.spotlightPreviousPointer = { x: event.clientX, y: event.clientY };
        if (Math.abs(yaw) + Math.abs(pitch) <= 1e-9) return;
        this.spotlightSource = rotateOpticalSpotlightSource(this.spotlightSource, yaw, pitch);
        this.syncSpotlightSourcePresentation();
        this.spotlightNeedsTrace = true;
        event.preventDefault?.();
        event.stopImmediatePropagation?.();
    }

    handleSpotlightPointerUp() {
        this.spotlightPointerDragging = false;
    }

    setSpotlightTestRayEnabled(enabled) {
        this.spotlightTestRayEnabled = Boolean(enabled);
        if (this.spotlightSource) this.spotlightSource.enabled = this.spotlightTestRayEnabled;
        this.onSpotlightTestRayStateChanged?.(this.spotlightTestRayEnabled);
        this.spotlightSourceLight?.setEnabled(this.active && this.spotlightTestRayEnabled);
        if (!this.spotlightTestRayEnabled) {
            this.spotlightTraceState = { transmitted: false, reason: 'test-ray-disabled' };
            this.destinationSpotlight?.setEnabled(false);
            if (this.destinationSpotlight) this.destinationSpotlight.intensity = 0;
            if (this.destinationSpotlightRayLine) this.destinationSpotlightRayLine.isVisible = false;
            this.clearOpticalSpotlightShadowCasters();
        } else {
            this.spotlightNeedsTrace = true;
        }
        return this.spotlightTestRayEnabled;
    }

    updateOpticalSpotlight(deltaSeconds = 0) {
        if (!this.active || !this.spotlightSource) return;
        if (this.spotlightNeedsTrace) {
            this.spotlightNeedsTrace = false;
            this.refreshOpticalSpotlightTrace();
            this.spotlightShadowRefreshElapsed = Infinity;
        }
        if (this.destinationSpotlight?.isEnabled?.()) {
            this.spotlightShadowRefreshElapsed += Math.max(0, deltaSeconds || 0);
            if (this.spotlightShadowRefreshElapsed >= 0.45) {
                this.spotlightShadowRefreshElapsed = 0;
                this.updateSpotlightShadowCasters();
            }
        }
    }

    refreshOpticalSpotlightTrace() {
        if (!this.spotlightTracer || !this.destinationSpotlight) {
            this.spotlightTraceState = { transmitted: false, reason: 'optical-renderer-unavailable' };
            this.destinationSpotlight?.setEnabled(false);
            if (this.destinationSpotlightRayLine) this.destinationSpotlightRayLine.isVisible = false;
            this.clearOpticalSpotlightShadowCasters();
            return;
        }
        if (!this.spotlightTestRayEnabled || !this.spotlightSource.enabled || !this.hypercubeMainCorridor.optics?.enabled) {
            this.spotlightTraceState = { transmitted: false, reason: 'source-or-anomaly-disabled' };
            this.destinationSpotlight.setEnabled(false);
            if (this.destinationSpotlightRayLine) this.destinationSpotlightRayLine.isVisible = false;
            this.clearOpticalSpotlightShadowCasters();
            return;
        }
        const sourceBinding = this.registry.require(this.spotlightSource.sliceId);
        const destinationBinding = this.registry.require(this.destination.slice.id);
        const result = traceOpticalSpotlightSource({
            source: this.spotlightSource,
            sourceBinding,
            destinationBinding,
            tracer: this.spotlightTracer,
            anomalyId: this.hypercubeMainCorridor.id,
            virtualSourceDistance: this.spotlightVirtualSourceDistance
        });
        if (!result.transmitted) {
            this.spotlightTraceState = result;
            this.destinationSpotlight.setEnabled(false);
            this.destinationSpotlight.intensity = 0;
            if (this.destinationSpotlightRayLine) this.destinationSpotlightRayLine.isVisible = false;
            this.clearOpticalSpotlightShadowCasters();
            return;
        }

        const presentationResult = this.mapOpticalSpotlightResultToPresentation(result);
        this.spotlightTraceState = {
            ...result,
            presentationOffset: presentationResult.presentationOffset ?? { x: 0, z: 0 },
            presentationVirtualPosition: { ...presentationResult.virtualPosition },
            presentationExitPosition: { ...presentationResult.exitPosition }
        };
        const transform = destinationBinding.spatialTransform;
        const lightPosition = transform.worldToRender(presentationResult.virtualPosition);
        const directionSample = transform.worldToRender(add3(
            presentationResult.virtualPosition, scale3(result.direction, 2)
        ));
        const lightDirection = normalize3(subtract3(directionSample, lightPosition));
        const rangeSample = transform.worldToRender(add3(
            presentationResult.virtualPosition, scale3(result.direction, this.spotlightSource.range)
        ));
        this.destinationSpotlight.position.copyFrom(toBabylonVector3(lightPosition));
        this.destinationSpotlight.direction.copyFrom(toBabylonVector3(lightDirection));
        this.destinationSpotlight.angle = this.spotlightSource.coneAngle;
        this.destinationSpotlight.exponent = this.spotlightSource.exponent;
        this.destinationSpotlight.range = Math.max(1, distance3(lightPosition, rangeSample));
        this.destinationSpotlight.intensity = this.spotlightSource.intensity * result.energy;
        this.destinationSpotlight.diffuse = BABYLON.Color3.FromArray(this.spotlightSource.color);
        this.destinationSpotlight.specular = BABYLON.Color3.FromArray(this.spotlightSource.color);
        this.destinationSpotlight.setEnabled(this.active);
        this.updateDestinationSpotlightRay(presentationResult, transform);
        this.updateSpotlightShadowCasters(true);
    }

    mapOpticalSpotlightResultToPresentation(result) {
        const pair = this.portalPairs?.find((entry) => entry.corridor.id === result.anomalyId);
        const presentation = pair?.presentationEndpointsBySlice?.[result.destinationSliceId];
        const endpoint = endpointForSlice(pair?.corridor, result.destinationSliceId);
        if (!presentation || !endpoint) return result;
        const offset = {
            x: presentation.x - endpoint.position.x,
            z: presentation.z - endpoint.position.z
        };
        const translate = (position) => position ? {
            ...position, x: position.x + offset.x, z: position.z + offset.z
        } : position;
        return {
            ...result,
            exitPosition: translate(result.exitPosition),
            virtualPosition: translate(result.virtualPosition),
            presentationOffset: offset
        };
    }

    updateDestinationSpotlightRay(result, transform) {
        if (!this.destinationSpotlightRayLine || !this.active) return false;
        let projected;
        try {
            projected = projectOpticalSpotlightExitRay(
                result, transform, this.spotlightDebugRayLength ?? 400
            );
        } catch (error) {
            // A debug segment may extend beyond a finite destination chart even
            // when the aperture exit itself is valid. Hide only that diagnostic.
            if (error instanceof RangeError) {
                this.destinationSpotlightRayLine.isVisible = false;
                return false;
            }
            throw error;
        }
        if (!projected) {
            this.destinationSpotlightRayLine.isVisible = false;
            return false;
        }
        this.destinationSpotlightRayLine = BABYLON.MeshBuilder.CreateLines(
            'geometryDemoTransmittedSpotlightRay', {
                points: [
                    toBabylonVector3(projected.startRender),
                    toBabylonVector3(projected.endRender)
                ],
                instance: this.destinationSpotlightRayLine
            }, this.scene
        );
        this.destinationSpotlightRayLine.layerMask = MAIN_WORLD_LAYER;
        this.destinationSpotlightRayLine.alpha = 0.96;
        this.destinationSpotlightRayLine.isPickable = false;
        this.destinationSpotlightRayLine.isVisible = true;
        this.destinationSpotlightRayLine.metadata = {
            ...(this.destinationSpotlightRayLine.metadata || {}),
            opticalDebugRay: true,
            anomalyId: result.anomalyId,
            sourceSliceId: this.hypercubeSlice.id,
            destinationSliceId: result.destinationSliceId,
            worldStart: projected.startWorld,
            worldEnd: projected.endWorld
        };
        return true;
    }

    updateSpotlightShadowCasters(force = false) {
        if (!this.destinationSpotlight || !this.destinationWorldManager) return;
        const lightPosition = this.destinationSpotlight.position;
        const sceneMeshes = this.scene?.meshes ?? [];
        const worldMeshes = new Set([
            ...(this.destinationWorldManager.getMeshes?.() ?? []),
            ...sceneMeshes
        ]);
        const excludedName = /portal|corridor|preview|sky|cloud|water|(?:^|[-_])rain(?:[-_]|$)|debug|grid|hud|camera|geometrydemo/i;
        const maxDistance = Math.max(1, Number(this.destinationSpotlight.range) || Infinity);
        const candidates = [...worldMeshes]
            .filter((mesh) => mesh && !mesh.isDisposed?.() &&
                mesh.isEnabled?.() !== false && mesh.isVisible !== false &&
                ((mesh.layerMask ?? MAIN_WORLD_LAYER) & MAIN_WORLD_LAYER) !== 0 &&
                !excludedName.test(String(mesh.name ?? '')))
            .map((mesh) => {
                let center = mesh.position;
                try {
                    mesh.computeWorldMatrix?.(true);
                    center = mesh.getBoundingInfo?.().boundingSphere?.centerWorld ?? center;
                } catch { /* Use the mesh origin when bounds are not available yet. */ }
                return { mesh, distance: center ? distance3(center, lightPosition) : Infinity };
            })
            .filter((entry) => entry.distance <= maxDistance * 1.15)
            .sort((a, b) => a.distance - b.distance)
            .slice(0, this.spotlightShadowCasterLimit)
            .map((entry) => entry.mesh);
        if (!this.spotlightShadowGenerator) {
            this.destinationSpotlight.includedOnlyMeshes = candidates;
            return;
        }
        const next = new Set(candidates);
        if (!force && next.size === this.spotlightShadowCasters.size &&
            [...next].every((mesh) => this.spotlightShadowCasters.has(mesh))) return;

        for (const mesh of this.spotlightShadowCasters) {
            if (next.has(mesh)) continue;
            this.spotlightShadowGenerator?.removeShadowCaster?.(mesh, false);
            if (this.spotlightOriginalReceiveShadows.has(mesh)) {
                mesh.receiveShadows = this.spotlightOriginalReceiveShadows.get(mesh);
                this.spotlightOriginalReceiveShadows.delete(mesh);
            }
        }
        for (const mesh of next) {
            if (!this.spotlightShadowCasters.has(mesh)) {
                this.spotlightOriginalReceiveShadows.set(mesh, Boolean(mesh.receiveShadows));
                this.spotlightShadowGenerator?.addShadowCaster?.(mesh, false);
            }
            mesh.receiveShadows = true;
        }
        this.spotlightShadowCasters = next;
        this.destinationSpotlight.includedOnlyMeshes = candidates;
    }

    clearOpticalSpotlightShadowCasters() {
        for (const mesh of this.spotlightShadowCasters ?? []) {
            this.spotlightShadowGenerator?.removeShadowCaster?.(mesh, false);
        }
        for (const [mesh, receiveShadows] of this.spotlightOriginalReceiveShadows ?? []) {
            if (!mesh.isDisposed?.()) mesh.receiveShadows = receiveShadows;
        }
        this.spotlightShadowCasters?.clear();
        this.spotlightOriginalReceiveShadows?.clear();
        if (this.destinationSpotlight) this.destinationSpotlight.includedOnlyMeshes = [];
    }

    getOpticalSpotlightDiagnostics() {
        return {
            source: this.spotlightSource ? {
                ...this.spotlightSource,
                position: { ...this.spotlightSource.position },
                direction: { ...this.spotlightSource.direction },
                color: [...this.spotlightSource.color]
            } : null,
            editing: Boolean(this.spotlightEditorActive),
            testRayEnabled: Boolean(this.spotlightTestRayEnabled && this.spotlightSource?.enabled),
            trace: this.spotlightTraceState ? {
                transmitted: Boolean(this.spotlightTraceState.transmitted),
                reason: this.spotlightTraceState.reason ?? null,
                anomalyId: this.spotlightTraceState.anomalyId ?? null,
                energy: this.spotlightTraceState.energy ?? 0,
                virtualPosition: this.spotlightTraceState.presentationVirtualPosition ??
                    this.spotlightTraceState.virtualPosition ?? null,
                physicalVirtualPosition: this.spotlightTraceState.virtualPosition ?? null,
                presentationOffset: this.spotlightTraceState.presentationOffset ?? null,
                direction: this.spotlightTraceState.direction ?? null,
                transitions: this.spotlightTraceState.trace?.transitions?.length ?? 0,
                pathLength: this.spotlightTraceState.diagnostics?.pathLength ?? null
            } : null,
            receiverEnabled: Boolean(this.destinationSpotlight?.isEnabled?.()),
            destinationRayVisible: Boolean(this.destinationSpotlightRayLine?.isVisible),
            destinationRayLength: this.spotlightDebugRayLength ?? 0,
            shadowGeneratorAvailable: Boolean(this.spotlightShadowGenerator),
            shadowCasterCount: this.spotlightShadowCasters?.size ?? 0,
            shadowCasterLimit: this.spotlightShadowCasterLimit,
            controls: 'L toggle edit; P test light/ray on/off; WASD/arrows move X/Z; R/F move Y; U/O yaw; I/K pitch; drag mouse to aim; Shift=faster; Escape exits'
        };
    }

    disposeOpticalSpotlight() {
        this.clearOpticalSpotlightShadowCasters();
        this.spotlightShadowGenerator?.dispose?.();
        this.destinationSpotlight?.dispose?.();
        this.destinationSpotlightRayLine?.dispose?.();
        this.spotlightSourceLight?.dispose?.();
        this.spotlightMarker?.dispose?.();
        this.spotlightAimLine?.dispose?.();
        this.spotlightMarkerMaterial?.dispose?.();
        this.spotlightTracer = null;
    }

    handleGridKeyDown(event) {
        if (!this.active || this.disposed || event.code !== 'KeyG' || event.repeat ||
            event.ctrlKey || event.metaKey || event.altKey) return;
        const tagName = String(event.target?.tagName ?? '').toUpperCase();
        if (['INPUT', 'TEXTAREA', 'SELECT'].includes(tagName) || event.target?.isContentEditable) return;
        event.preventDefault?.();
        this.toggleGrid();
    }

    toggleCameraMode() { return this.cameraController.toggleMode(); }

    updatePortalPreview() {
        this.destinationPreviewFocus = null;
        this.destinationPreviewDirection = null;
        if (!this.active) return;
        if (!Array.isArray(this.portalPairs)) return updateLegacyPortalPreview(this);

        const viewSliceId = this.cameraController.getViewSliceId();
        const viewerWorld = this.getWorld(viewSliceId);
        const playerController = this.traversalController.getActiveController?.();
        const playerTransit = playerController?.state === 'transit' ? playerController : null;
        const cameraTransit = this.cameraController.getFreeTransitState();
        const transit = this.cameraController.mode === 'free' ? cameraTransit : playerTransit;
        const transitAnomaly = transit?.type === SLICE_TRANSITION_ANOMALY ||
            transit?.corridor?.type === SLICE_TRANSITION_ANOMALY ||
            transit?.anomaly?.type === SLICE_TRANSITION_ANOMALY;
        const transitCorridorId = transit?.corridorId ?? transit?.corridor?.id ??
            transit?.anomaly?.id ?? playerController?.corridor?.id ?? null;
        const transitTargetSliceId = transit?.targetWorld?.slice?.id ??
            transit?.target?.slice?.id ?? transit?.toSliceId ?? null;
        this.destinationApproachState = null;

        const passagePair = transitAnomaly
            ? this.portalPairs.find((entry) => entry.corridor.id === transitCorridorId)
            : null;
        for (const pair of this.portalPairs) {
            if (pair !== passagePair) {
                pair.passagePresentation?.deactivate(this.camera, viewerWorld.layerMask);
            }
        }

        if (transitAnomaly) {
            const presentation = passagePair?.passagePresentation;
            const startWorld = transit?.startWorld ?? this.getWorld(viewSliceId);
            const targetWorld = transit?.targetWorld ?? transit?.target ??
                (transitTargetSliceId ? this.getWorld(transitTargetSliceId) : null);
            if (presentation && startWorld && targetWorld) {
                const progress = clamp(transit.progress ?? 0, 0, 1);
                const viewerPresentationPosition = passagePair.presentationEndpointsBySlice?.[startWorld.slice.id];
                const exitPose = mapAnomalyCameraPose({
                    camera: this.camera, viewerWorld: startWorld, remoteWorld: targetWorld,
                    anomaly: passagePair.corridor, approachProgress: progress,
                    viewerEndpoint: viewerPresentationPosition
                        ? { sliceId: startWorld.slice.id, position: viewerPresentationPosition }
                        : null
                });
                for (const pair of this.portalPairs) {
                    pair.sourcePortal.setVisible(false);
                    pair.destinationPortal.setVisible(false);
                }
                const presentationDirection = this.cameraController.mode === 'follow' &&
                    Number.isFinite(this.player?.yaw)
                    ? { x: Math.sin(this.player.yaw), y: 0, z: Math.cos(this.player.yaw) }
                    : null;
                const passageState = presentation.update({
                    camera: this.camera, startWorld, targetWorld, progress,
                    exitCameraPose: exitPose, presentationDirection
                });
                this.destinationApproachState = {
                    anomalyId: passagePair.corridor.id, progress,
                    entryScale: passageState.entry.scale, exitScale: passageState.exit.scale,
                    mouthSeparation: passageState.separation,
                    cameraConvergence: exitPose?.approachEase ?? null,
                    remoteEyeLocalPosition: exitPose?.remoteLocalPosition ?? null
                };
                if (targetWorld.slice.id === this.destination.slice.id && exitPose?.remoteLocalPosition) {
                    this.destinationPreviewFocus = { ...exitPose.remoteLocalPosition };
                    const horizontalLength = Math.hypot(exitPose.direction.x, exitPose.direction.z);
                    this.destinationPreviewDirection = horizontalLength > 1e-6
                        ? { x: exitPose.direction.x / horizontalLength, z: exitPose.direction.z / horizontalLength }
                        : { x: 0, z: 1 };
                }
                return;
            }
        }

        for (const pair of this.portalPairs) {
            const viewingFromSource = viewSliceId === pair.sourceSliceId;
            const viewingFromDestination = viewSliceId === pair.destinationSliceId;
            if (!viewingFromSource && !viewingFromDestination) continue;

            const portal = viewingFromSource ? pair.sourcePortal : pair.destinationPortal;
            const remoteSliceId = viewingFromSource ? pair.destinationSliceId : pair.sourceSliceId;
            const remoteWorld = this.getWorld(remoteSliceId);
            const approachProgress = transitAnomaly && transitCorridorId === pair.corridor.id &&
                transitTargetSliceId === remoteSliceId
                ? clamp(transit.progress ?? 0, 0, 1) : null;
            const viewConfig = pair.corridor.view ?? {
                enabled: true, mapping: 'endpoint-frame', apertureShape: 'circle',
                crossSection: 'constant', centerNormalOffset: 0
            };
            if (viewConfig.enabled === false) {
                portal.setVisible?.(false);
                continue;
            }

            if (viewConfig.mapping === 'endpoint-frame') {
                const viewerPresentationPosition = pair.presentationEndpointsBySlice?.[viewSliceId];
                const pose = mapAnomalyCameraPose({
                    camera: this.camera,
                    viewerWorld,
                    remoteWorld,
                    anomaly: pair.corridor,
                    approachProgress,
                    viewerEndpoint: viewerPresentationPosition
                        ? { sliceId: viewSliceId, position: viewerPresentationPosition }
                        : null
                });
                if (!pose) {
                    portal.setVisible?.(false);
                    continue;
                }
                portal.setCameraPose?.(pose.position, pose.target, pose.up, this.camera.fov);
                if (approachProgress !== null) {
                    this.destinationApproachState = {
                        anomalyId: pair.corridor.id,
                        progress: approachProgress,
                        cameraConvergence: pose.approachEase,
                        remoteEyeLocalPosition: { ...pose.remoteLocalPosition }
                    };
                }
                if (remoteSliceId === this.destination.slice.id) {
                    this.destinationPreviewFocus = { ...pose.remoteLocalPosition };
                    const horizontalLength = Math.hypot(pose.direction.x, pose.direction.z);
                    this.destinationPreviewDirection = horizontalLength > 1e-6
                        ? { x: pose.direction.x / horizontalLength, z: pose.direction.z / horizontalLength }
                        : { x: 0, z: 1 };
                }
            }

            const apertureScale = calculateAnomalyApertureScale({
                anomaly: pair.corridor,
                viewerWorld,
                viewConfig,
                approachProgress,
                viewingFromSource,
                viewerNormalOffset: this.getViewerNormalOffset(viewerWorld)
            });
            portal.setApertureScale?.(apertureScale);
            if (approachProgress !== null && this.destinationApproachState?.anomalyId === pair.corridor.id) {
                this.destinationApproachState.apertureScale = apertureScale;
            }
        }

        // Cubic Bezier corridors retain their established moving look-ahead
        // preview. Slice-transition anomalies instead expose a fixed 4D mouth
        // whose remote camera and visible aperture now converge smoothly as
        // the observer advances toward the far mouth.
        if (transitAnomaly) return;

        if (!transit) {
            for (const pair of this.portalPairs) {
                pair.passagePresentation?.deactivate(
                    this.camera, this.getWorld(viewSliceId).layerMask
                );
            }
            for (const pair of this.portalPairs) {
                if (pair.corridor.type !== SLICE_TRANSITION_ANOMALY) {
                    pair.sourcePortal.resetPose();
                    pair.destinationPortal.resetPose();
                }
            }
            return;
        }

        const corridorId = transit.corridorId ?? transit.corridor?.id ?? playerController?.corridor.id;
        const pair = this.portalPairs.find(function findPair(entry) { return entry.corridor.id === corridorId; });
        if (!pair) return;
        const length = transit.corridorLength ?? playerController?.corridorLength ?? 1;
        const lead = clamp(14 / Math.max(1, length), 0.08, 0.22);
        const progress = clamp(transit.progress ?? 0, 0, 1);
        const previewProgress = Math.min(0.995, progress + lead);
        const t = transit.direction > 0 ? previewProgress : 1 - previewProgress;
        const renderer = transit.activeRenderer ?? transit.renderer;
        const sample = renderer?.getDisplaySample?.(t);
        if (!sample) return;
        const portal = transit.direction > 0 ? pair.sourcePortal : pair.destinationPortal;
        portal.setPose(sample.position, scaleVector(sample.tangent, transit.direction));
        portal.setVisible(true);
        for (const other of this.portalPairs) {
            if (other === pair || other.corridor.type === SLICE_TRANSITION_ANOMALY) continue;
            other.sourcePortal.setVisible(false);
            other.destinationPortal.setVisible(false);
        }
    }

    getViewerNormalOffset(viewerWorld) {
        let position4D = null;
        if (this.cameraController.mode === 'free') {
            position4D = this.cameraController.getFreeAmbientPosition?.();
        } else {
            position4D = this.traversalController.getAmbientPlayerPosition?.();
        }
        if (!position4D) return this.cameraController.mode === 'free'
            ? (this.cameraController.getFreeW?.() ?? 0) : 0;
        return ambientToSlice(viewerWorld.slice.frame, position4D).normalOffset;
    }

    getDiagnostics() {
        return {
            defaultSlice: this.hypercubeSlice.id,
            sourceSlice: this.sourceSlice.id,
            destinationSlice: this.destination.slice.id,
            corridors: this.portalPairs.map(function describePair(pair) {
                return { id: pair.corridor.id, type: pair.corridor.type, source: pair.sourceSliceId, destination: pair.destinationSliceId };
            }),
            hypercube: this.hypercubeRenderer.getInfo(),
            chunkGraph: this.chunkGraph.getStats(),
            destinationChunking: this.destinationWorldManager?.getStats() ?? null,
            traversal: this.traversalController.getState(),
            camera: this.cameraController.getModeInfo(),
            viewSliceId: this.cameraController.getViewSliceId(),
            visibleDestinations: this.topology.getVisibleDestinations(this.cameraController.getViewSliceId()),
            traversableNeighbors: this.topology.getTraversableNeighbors(this.cameraController.getViewSliceId()),
            opticalSpotlight: this.getOpticalSpotlightDiagnostics(),
            destinationApproach: this.destinationApproachState ?? null,
            portalViews: this.portalPairs.map((pair) => ({
                anomalyId: pair.corridor.id,
                view: pair.corridor.view ?? null,
                source: pair.sourcePortal.getInfo?.() ?? null,
                destination: pair.destinationPortal.getInfo?.() ?? null,
                passage: pair.passagePresentation?.getInfo?.() ?? null
            })),
            anomalyBoundaries: (this.anomalyBoundaryRenderers ?? []).map((entry) => entry.renderer.getInfo()),
            anomalyEntryDiagnostics: this.traversalController.getEntryDiagnostics?.() ?? [],
            ambientPlayerPosition4D: this.cameraController.mode === 'free'
                ? this.cameraController.getFreeAmbientPosition()
                : this.traversalController.getAmbientPlayerPosition?.() ?? null,
            sourceGrid: this.sourceGrid.getInfo(),
            hypercubeGrid: this.hypercubeGrid.getInfo(),
            mainWorldGrid: this.mainWorldGrid.getInfo(),
            metricGridEnabled: this.gridEnabled,
            geometry: this.adapter.getStats()
        };
    }

    setDestinationWorldManager(manager) {
        if (!manager || manager.sliceId !== this.destination.slice.id) {
            throw new TypeError('Geometry demo requires the main WorldManager for its destination slice.');
        }
        if (manager.chunkGraph !== this.chunkGraph || manager.spatialAdapter !== this.adapter ||
            manager.terrain !== this.destination.terrain) {
            throw new Error('Main WorldManager must share the main terrain, geometry adapter, and chunk graph.');
        }
        this.destinationWorldManager = manager;
        this.spotlightShadowRefreshElapsed = Infinity;
        this.spotlightNeedsTrace = true;
        return manager;
    }

    dispose() {
        if (this.disposed) return;
        this.setActive(false);
        this.disposed = true;
        if (typeof window !== 'undefined') {
            window.removeEventListener('keydown', this.onGridKeyDown);
            window.removeEventListener('keydown', this.onSpotlightEditorKeyDown, true);
            window.removeEventListener('pointerdown', this.onSpotlightPointerDown, true);
            window.removeEventListener('pointermove', this.onSpotlightPointerMove, true);
            window.removeEventListener('pointerup', this.onSpotlightPointerUp, true);
            window.removeEventListener('blur', this.onSpotlightPointerBlur);
        }
        this.disposeOpticalSpotlight();
        this.traversalController?.dispose();
        this.chunkGraph.dispose();
        this.cameraController.dispose();
        this.camera.dispose();
        this.playerController.dispose();
        this.player.dispose();
        for (const grid of this.grids.values()) grid.dispose();
        for (const pair of this.portalPairs) {
            pair.passagePresentation?.dispose();
            pair.sourcePortal.dispose();
            pair.destinationPortal.dispose();
        }
        for (const entry of this.corridorRenderers) entry.renderer.dispose();
        for (const entry of this.anomalyBoundaryRenderers ?? []) entry.renderer.dispose();
        this.hypercubeRenderer.dispose();
        disposeMeshes(this.sourceMeshes);
        disposeMeshes(this.hypercubeAuxiliaryMeshes ?? []);
        this.adapter.dispose();
        this.registry.dispose();
    }
}

function updateLegacyCameraPresentation(world) {
    const viewSliceId = world.cameraController.getViewSliceId();
    const isSource = viewSliceId === world.sourceSlice.id;
    const renderWorld = isSource ? world.sourceWorld : world.destinationWorld;
    world.camera.layerMask = renderWorld.layerMask;
    world.corridorRenderer.mesh.isVisible = world.active && isSource;
    world.destinationCorridorRenderer.mesh.isVisible = world.active && !isSource;
    world.debugGrid?.setActive(world.active && isSource && !world.cameraController.getFreeTransitState?.());
}

function updateLegacyPortalPreview(world) {
    const controller = world.traversalController;
    const playerTransit = controller.state === 'transit' ? controller : null;
    const cameraTransit = world.cameraController.getFreeTransitState?.();
    const transit = world.cameraController.mode === 'free' ? cameraTransit : playerTransit;
    const sourceActive = world.cameraController.getViewSliceId?.() === world.sourceSlice.id;
    if (!transit) {
        world.portal.resetPose();
        world.destinationPortal.resetPose();
        world.portal.setVisible(sourceActive);
        world.destinationPortal.setVisible(!sourceActive);
        world.portal.rim.isVisible = sourceActive;
        world.destinationPortal.rim.isVisible = !sourceActive;
        return;
    }
    const length = controller.corridorLength || 1;
    const lead = clamp(14 / Math.max(1, length), 0.08, 0.22);
    const progress = clamp(transit.progress ?? 0, 0, 1);
    const t = transit.direction > 0 ? Math.min(0.995, progress + lead) : 1 - Math.min(0.995, progress + lead);
    const sample = (transit.activeRenderer ?? transit.renderer ?? controller.renderer).getDisplaySample(t);
    const target = transit.direction > 0 ? world.portal : world.destinationPortal;
    const other = transit.direction > 0 ? world.destinationPortal : world.portal;
    target.setPose(sample.position, scaleVector(sample.tangent, transit.direction));
    target.setVisible(true);
    target.rim.isVisible = true;
    other.resetPose();
    other.setVisible(false);
    other.rim.isVisible = false;
}

function createHypercubeTransform() {
    return new SpatialTransform({ geometry: 'euclidean', renderScale: 1, enabled: true });
}

function createHypercubeMovementSurface() {
    return {
        min: { x: -34, y: -10, z: -30 },
        max: { x: 34, y: 20, z: 34 },
        sampleAt: function sampleHypercubeGround(x, z) {
            if (x < -34 || x > 34 || z < -30 || z > 34) return null;
            return { x, y: 0, z };
        }
    };
}

function createSourceTransform() {
    return new SpatialTransform({
        geometry: 'poincare',
        renderScale: 24,
        visualScaleGamma: 0.72,
        maxVisualScale: 3.5,
        bounds: {
            minX: -46, maxX: 46,
            minY: -5, maxY: 18,
            minZ: -48, maxZ: 46,
            characteristicLength: 0.7,
            boundaryEpsilon: 1e-5
        }
    });
}

function createRoomPalette(scene) {
    return {
        amber: makeMaterial(scene, 'demoAmber', '#e59c42', '#3b210d'),

/** Renders the actual 3D cross-section of a rotating 4D tesseract. */

import {
    createHypercubeEdges, createHypercubeSlice, createHypercubeVertices
} from '../world/fields/geometry/hypercube-4d.js?v=somnium-20261109-phase17i-bidirectional-spherical-passage-1';

const MAX_SECTION_VERTICES = 32;
const MAX_SECTION_EDGES = 64;

class Hypercube4DRenderer {
    constructor(scene, transform, options = {}) {
        if (!scene || !transform?.worldToRender) {
            throw new TypeError('Hypercube renderer requires a scene and spatial transform.');
        }
        this.scene = scene;
        this.transform = transform;
        this.center = { ...(options.center ?? { x: 0, y: 4, z: 0 }) };
        this.angularSpeeds = { xw: 0.68, yw: 0.49, zw: 0.33, ...options.angularSpeeds };
        this.angles = { xw: 0.42, yw: 0.36, zw: 0.32, ...options.initialAngles };
        this.angularSpeed = this.angularSpeeds.xw;
        this.edgeRadius = options.edgeRadius ?? 0.095;
        this.radialSegments = options.radialSegments ?? 7;
        this.vertexDiameter = options.vertexDiameter ?? 0.13;
        this.layerMask = options.layerMask ?? 0x0FFFFFFF;
        this.opticalScene = options.opticalScene ?? null;
        this.opticalSliceId = options.sliceId ?? null;
        this.opticalChunkAddress = options.chunkAddress ?? null;
        this.opticalRevision = 0;
        this.opticalRegistrationId = this.opticalSliceId ? `hypercube-surface:${this.opticalSliceId}` : null;
        this.viewerW = options.viewerW ?? 0;
        this.disposed = false;
        this.vertices = createHypercubeVertices(options.halfExtent ?? 4.2);
        this.edges = createHypercubeEdges(this.vertices);
        this.surfaceMaterial = this.createMaterial('geometryDemoTesseractSliceSurface', 0.2, 0.16);
        this.edgeMaterial = this.createMaterial('geometryDemoTesseractSliceEdges', 1, 1);
        this.surfaceMesh = this.createSurfaceMesh();
        this.edgeMesh = this.createEdgeMesh();
        this.mesh = this.edgeMesh;
        this.vertexMeshes = this.createVertexMeshes();
        this.meshes = [this.surfaceMesh, this.edgeMesh, ...this.vertexMeshes];
        this.slice = createHypercubeSlice(this.vertices, this.angles, this.viewerW);
        this.update(0, this.viewerW);
    }

    createMaterial(name, alpha, emissive) {
        const material = new BABYLON.StandardMaterial(name, this.scene);
        material.emissiveColor = new BABYLON.Color3(0.28 * emissive, 0.88 * emissive, emissive);
        material.diffuseColor = new BABYLON.Color3(0.12, 0.56, 0.82);
        material.specularColor = BABYLON.Color3.White();
        material.disableLighting = true;
        material.backFaceCulling = false;
        material.alpha = alpha;
        return material;
    }

    createSurfaceMesh() {
        const mesh = new BABYLON.Mesh('geometryDemoTesseractSliceSurface', this.scene);
        mesh.setVerticesData(BABYLON.VertexBuffer.PositionKind, new Float32Array(MAX_SECTION_VERTICES * 3), true, 3);
        mesh.setVerticesData(BABYLON.VertexBuffer.NormalKind, new Float32Array(MAX_SECTION_VERTICES * 3), true, 3);
        mesh.material = this.surfaceMaterial;
        this.configureMesh(mesh, { section: 'surface' });
        return mesh;
    }

    createEdgeMesh() {
        const mesh = new BABYLON.Mesh('geometryDemoTesseractSliceEdges', this.scene);
        mesh.setVerticesData(BABYLON.VertexBuffer.PositionKind,
            new Float32Array(MAX_SECTION_EDGES * this.radialSegments * 6), true, 3);
        mesh.setVerticesData(BABYLON.VertexBuffer.NormalKind,
            new Float32Array(MAX_SECTION_EDGES * this.radialSegments * 6), true, 3);
        mesh.material = this.edgeMaterial;
        this.configureMesh(mesh, { section: 'edges' });
        return mesh;
    }

    configureMesh(mesh, metadata) {
        mesh.layerMask = this.layerMask;
        mesh.isVisible = true;
        mesh.isPickable = false;
        mesh.alwaysSelectAsActiveMesh = true;
        mesh.metadata = { geometryDemo: true, hypercube4D: true, ...metadata };
    }

    createVertexMeshes() {
        return Array.from({ length: MAX_SECTION_VERTICES }, function createMarker(unused, index) {
            const mesh = BABYLON.MeshBuilder.CreateSphere(
                `geometryDemoTesseractSliceVertex${index}`,
                { diameter: this.vertexDiameter, segments: 8 },
                this.scene
            );
            mesh.material = this.edgeMaterial;
            mesh.layerMask = this.layerMask;
            mesh.isPickable = false;
            mesh.isVisible = false;
            mesh.alwaysSelectAsActiveMesh = true;
            mesh.metadata = { geometryDemo: true, hypercube4DSectionVertex: index };
            return mesh;
        }, this);
    }

    toRenderPoint(point) {
        return this.transform.worldToRender({
            x: this.center.x + point.x,
            y: this.center.y + point.y,
            z: this.center.z + point.z
        });
    }

    createSurfaceBuffers(section) {
        if (section.vertices.length > MAX_SECTION_VERTICES) {
            throw new RangeError('Tesseract section exceeds the renderer vertex capacity.');
        }
        const positions = new Float32Array(MAX_SECTION_VERTICES * 3);
        const normals = new Float32Array(MAX_SECTION_VERTICES * 3);
        const points = section.vertices.map(this.toRenderPoint.bind(this));
        for (let index = 0; index < points.length; index++) writeVector3(positions, index * 3, points[index]);
        const indices = [];
        for (const face of section.faces) {
            for (let index = 1; index < face.length - 1; index++) {
                const a = face[0];
                const b = face[index];
                const c = face[index + 1];
                indices.push(a, b, c);
                const normal = normalize3(cross3(subtract3(points[b], points[a]), subtract3(points[c], points[a])));
                for (const vertex of [a, b, c]) {
                    normals[vertex * 3] += normal.x;
                    normals[vertex * 3 + 1] += normal.y;
                    normals[vertex * 3 + 2] += normal.z;
                }
            }
        }
        for (let index = 0; index < section.vertices.length; index++) {
            const offset = index * 3;
            const normal = normalize3({ x: normals[offset], y: normals[offset + 1], z: normals[offset + 2] });
            writeVector3(normals, offset, normal);
        }
        return { positions, normals, indices, points };
    }

    createEdgeBuffers(section, points) {
        if (section.edges.length > MAX_SECTION_EDGES) {
            throw new RangeError('Tesseract section exceeds the renderer edge capacity.');
        }
        const vertexCount = MAX_SECTION_EDGES * this.radialSegments * 2;
        const positions = new Float32Array(vertexCount * 3);
        const normals = new Float32Array(vertexCount * 3);
        const indices = [];
        let offset = 0;
        for (const edge of section.edges) {
            const start = points[edge[0]];
            const end = points[edge[1]];
            offset = appendTube(start, end, this.edgeRadius, this.radialSegments, offset, positions, normals, indices);
        }
        return { positions, normals, indices };
    }

    update(deltaSeconds, viewerW = 0) {
        if (this.disposed) return;
        const delta = Math.max(0, Math.min(Number(deltaSeconds) || 0, 0.1));
        for (const plane of ['xw', 'yw', 'zw']) {
            this.angles[plane] = (this.angles[plane] + delta * this.angularSpeeds[plane]) % (Math.PI * 2);
        }
        this.angle = this.angles.xw;
        this.viewerW = Number.isFinite(viewerW) ? viewerW : 0;
        this.slice = createHypercubeSlice(this.vertices, this.angles, this.viewerW);
        const surface = this.createSurfaceBuffers(this.slice);
        const edges = this.createEdgeBuffers(this.slice, surface.points);
        this.updateMesh(this.surfaceMesh, surface);
        this.updateMesh(this.edgeMesh, edges);
        this.updateOpticalSurface(this.slice);
        const visible = this.slice.vertices.length > 0;
        this.surfaceMesh.isVisible = visible && this.slice.faces.length > 0;
        this.edgeMesh.isVisible = visible && this.slice.edges.length > 0;
        for (let index = 0; index < this.vertexMeshes.length; index++) {
            const marker = this.vertexMeshes[index];
            const point = surface.points[index];
            marker.isVisible = Boolean(point);
            if (point) marker.position.set(point.x, point.y, point.z);
        }
    }

    updateOpticalSurface(section) {
        if (!this.opticalScene || !this.opticalSliceId || !this.opticalRegistrationId) return;
        const positions = new Float64Array(section.vertices.length * 3);
        section.vertices.forEach((point, index) => {
            const offset = index * 3;
            positions[offset] = this.center.x + point.x;
            positions[offset + 1] = this.center.y + point.y;
            positions[offset + 2] = this.center.z + point.z;
        });
        const indices = [];
        for (const face of section.faces) {
            for (let index = 1; index < face.length - 1; index++) {
                indices.push(face[0], face[index], face[index + 1]);
            }
        }
        if (indices.length === 0) {
            this.opticalScene.unregister(this.opticalRegistrationId);
            return;
        }
        this.opticalRevision++;
        this.opticalScene.registerTriangleMesh({
            id: this.opticalRegistrationId,
            sliceId: this.opticalSliceId,
            chunkAddress: this.opticalChunkAddress,
            positions,
            indices,
            material: { id: 'material:hypercube-cross-section', refractiveIndex: 1, transmission: 1 },
            revision: this.opticalRevision
        });
    }

    updateMesh(mesh, buffers) {
        mesh.updateVerticesData(BABYLON.VertexBuffer.PositionKind, buffers.positions, false, false);
        mesh.updateVerticesData(BABYLON.VertexBuffer.NormalKind, buffers.normals, false, false);
        mesh.setIndices(buffers.indices);
        mesh.refreshBoundingInfo();
    }

    getInfo() {
        return {
            vertices: this.vertices.length,
            edges: this.edges.length,
            rotationPlane: 'xw+yw+zw',
            rotationAngles: { ...this.angles },
            angle: this.angle,
            viewerW: this.viewerW,
            sliceVertexCount: this.slice.vertices.length,
            sliceFaceCount: this.slice.faces.length,
            sliceEdgeCount: this.slice.edges.length,
            visibleMeshCount: Number(this.surfaceMesh.isVisible) + Number(this.edgeMesh.isVisible) +
                this.vertexMeshes.filter(function visibleMarker(marker) { return marker.isVisible; }).length,
            mesh: this.edgeMesh
        };
    }

    dispose() {
        if (this.disposed) return;
        this.disposed = true;
        if (this.opticalRegistrationId) this.opticalScene?.unregister(this.opticalRegistrationId);
        for (const mesh of this.meshes) mesh.dispose(false, false);
        this.surfaceMaterial.dispose(false, true);
        this.edgeMaterial.dispose(false, true);
    }
}

function appendTube(start, end, radius, radialSegments, offset, positions, normals, indices) {
    const direction = normalize3(subtract3(end, start));
    const reference = Math.abs(direction.x) < 0.8 ? { x: 1, y: 0, z: 0 } : { x: 0, y: 1, z: 0 };
    const first = normalize3(cross3(direction, reference));
    const second = cross3(direction, first);
    const base = offset / 3;
    for (const point of [start, end]) {
        for (let index = 0; index < radialSegments; index++) {
            const angle = index / radialSegments * Math.PI * 2;
            const normal = add3(scale3(first, Math.cos(angle)), scale3(second, Math.sin(angle)));
            const target = offset;
            positions[target] = point.x + normal.x * radius;
            positions[target + 1] = point.y + normal.y * radius;
            positions[target + 2] = point.z + normal.z * radius;
            normals[target] = normal.x;
            normals[target + 1] = normal.y;
            normals[target + 2] = normal.z;
            offset += 3;
        }
    }
    for (let index = 0; index < radialSegments; index++) {
        const next = (index + 1) % radialSegments;
        const a = base + index;
        const b = base + radialSegments + index;
        const c = base + next;
        const d = base + radialSegments + next;
        indices.push(a, c, b, c, d, b);
    }
    return offset;
}

function writeVector3(target, offset, value) {
    target[offset] = value.x;
    target[offset + 1] = value.y;
    target[offset + 2] = value.z;
}
function subtract3(a, b) { return { x: a.x - b.x, y: a.y - b.y, z: a.z - b.z }; }
function add3(a, b) { return { x: a.x + b.x, y: a.y + b.y, z: a.z + b.z }; }
function scale3(value, amount) { return { x: value.x * amount, y: value.y * amount, z: value.z * amount }; }
function cross3(a, b) {
    return { x: a.y * b.z - a.z * b.y, y: a.z * b.x - a.x * b.z, z: a.x * b.y - a.y * b.x };
}
function length3(value) { return Math.hypot(value.x, value.y, value.z); }
function normalize3(value) {
    const length = length3(value);
    return length > 1e-9 ? scale3(value, 1 / length) : { x: 0, y: 1, z: 0 };
}

/** Applies visibility across current and legacy world-manager contracts. */

export function setWorldManagerVisibility(manager, visible) {
    if (!manager) return;

    const nextVisible = Boolean(visible);
    manager.visible = nextVisible;

    if (applyManagerVisibility(manager, nextVisible)) return;
    applyResidentVisibility(manager, nextVisible);
}

function applyManagerVisibility(manager, visible) {
    if (typeof manager.setVisible !== 'function') return false;

    try {
        manager.setVisible(visible);
        return true;
    } catch (error) {
        console.warn('[GeometryDemo] WorldManager.setVisible failed; using compatibility fallback.', error);
        return false;
    }
}

function applyResidentVisibility(manager, visible) {
    const chunks = manager.chunks;
    if (chunks && typeof chunks.values === 'function') {
        for (const chunk of chunks.values()) setChunkVisibility(chunk, visible);
        return;
    }
    applyMeshVisibility(manager, visible);
}

function setChunkVisibility(chunk, visible) {
    if (!chunk) return;
    chunk.visible = visible;
    if (chunk.mesh) chunk.mesh.isVisible = visible;
}

function applyMeshVisibility(manager, visible) {
    if (typeof manager.getMeshes !== 'function') return;
    for (const mesh of manager.getMeshes() || []) {
        if (mesh) mesh.isVisible = visible;
    }
}
