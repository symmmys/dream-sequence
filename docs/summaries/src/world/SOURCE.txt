sampleWind(position, time) {
    const x = position.x;
    const z = position.z;

    // Large-scale atmospheric circulation.
    const broad = Math.sin(
        x * 0.000035 + z * 0.000018 + time * 0.018
    );

    const secondary = Math.cos(
        z * 0.000045 - x * 0.000012 + time * 0.012
    );

    const direction = new BABYLON.Vector3(
        0.75 + broad * 0.20,
        0,
        0.30 + secondary * 0.25
    ).normalize();

    const speed = 4 + 9 * (
        0.5 + 0.5 * Math.sin(
            x * 0.00002 + z * 0.000025 + time * 0.01
        )
    );

    return direction.scale(speed);
}

updateCloudEnvironment(cloud, deltaSeconds) {
    // Keep the existing moisture and precipitation simulation.
    // These values should be sampled periodically, not every frame.

    cloud.environmentSampleTimer =
        (cloud.environmentSampleTimer || 0) + deltaSeconds;

    const sampleInterval = 2.0;

    if (cloud.environmentSampleTimer >= sampleInterval) {
        cloud.environmentSampleTimer = 0;

        const position = cloud.position;

        // Use the existing environmental sampling functions
        // from your original updateCloud implementation.
        if (this.moistureField) {
            cloud.localMoisture =
                this.moistureField.sample(
                    position.x,
                    position.z
                );
        }
    }

    // Preserve cloud water capacity.
    cloud.waterContent = BABYLON.Scalar.Clamp(
        cloud.waterContent,
        0,
        cloud.maxWaterCapacity
    );

    // Existing precipitation lifecycle.
    if (cloud.isRaining) {
        cloud.rainRemaining -= deltaSeconds;

        if (cloud.rainRemaining <= 0) {
            cloud.isRaining = false;
            cloud.rainIntensity = 0;
            cloud.rainRemaining = 0;
        }
    }

        if (minX >= maxX || minZ >= maxZ) {
            return { trees };
        }

        const attemptsLimit = this.treesPerChunk * 12;

        let attempts = 0;

        while (
            trees.length < this.treesPerChunk &&
            attempts < attemptsLimit
        ) {
            attempts++;

            const x = minX + random() * (maxX - minX);
            const z = minZ + random() * (maxZ - minZ);

            const profile = this.terrain.sampleProfileAt(
                x, z, 4
            );

            if (!profile) continue;

            if (profile.position.y <= this.seaLevel) {
                continue;
            }

            if (profile.slopeDegrees > 28) {
                continue;
            }

	const soil = this.geologicalField
    ? this.geologicalField.sample(x, z)
    : null;

const speciesName = this.chooseSpeciesForSoil(
    soil,
    random
);

            const genome = Genetics.createGenome();

            // Reproducible genetic variation.
            for (const gene of Object.keys(genome)) {
                genome[gene] *= 0.85 + random() * 0.30;
            }

            const initialAge = random() * 20;

            const tree = new Tree(
                speciesName,
                profile.position,
                genome,
                null,
                initialAge
            );

            tree.soilMoisture =
                0.25 + random() * 0.40;

            trees.push(tree);
        }

        return {
            trees,
            seed: context.seed,
            generatedCount: trees.length
        };
    }

simulatePopulation(record, deltaYears) {
    for (const tree of record.population.trees) {
        if (!tree.alive) continue;

        tree.update(

            this.environment,
            this.hydrology

}

                new BABYLON.Vector3(

                    item.position.y,
                    item.position.z
                ),

                item.age
            );

            tree.soilMoisture = item.soilMoisture ?? 0.35;
            tree.alive = item.alive ?? true;

            tree.droughtYears = item.droughtYears ?? 0;
            tree.waterloggingYears =
                item.waterloggingYears ?? 0;
            tree.lowLightYears = item.lowLightYears ?? 0;

            trees.push(tree);
        }

        return { trees };
    }

    rebuildRendering() {
        if (this.disposed) return;

        for (const renderer of Object.values(
            this.trunkRenderers
        )) {
            renderer.clear();
        }

        for (const renderer of Object.values(
            this.foliageRenderers
        )) {
            renderer.clear();
        }

        const allTrees = [];

        for (const record of this.activeRecords) {
            if (!record.active) continue;

            for (const tree of record.population.trees) {
                if (tree.alive) allTrees.push(tree);
            }
        }

        this.environment.rebuildCanopyGrid(allTrees);

        for (const tree of allTrees) {
            const species = tree.species;
            const age = tree.ageFactor;

            const growth =
                age * age * (3 - 2 * age);

            const height =
                species.trunkLength *
                growth *
                tree.genome.maxHeight;

            const radius =
                species.trunkRadius *
                Math.sqrt(Math.max(age, 0.001)) *
                tree.genome.trunkThickness;

            if (height <= 0.05) continue;

            const trunkMatrix = BABYLON.Matrix.Compose(
                new BABYLON.Vector3(
                    radius,
                    height,
                    radius
                ),
                BABYLON.Quaternion.Identity(),
                new BABYLON.Vector3(
                    tree.position.x,
                    tree.position.y + height * 0.5,
                    tree.position.z
                )
            );

            this.trunkRenderers[
                tree.speciesName
            ].addMatrix(trunkMatrix);

            const canopy = species.canopy;

            const canopyHeight = height * canopy.height;

            const canopyWidth =
                height *
                canopy.width *
                tree.genome.canopyWidth;

            const canopyCenter = new BABYLON.Vector3(
                tree.position.x,
                tree.position.y +
                    height * (
                        canopy.startHeight +
                        canopy.height * 0.5
                    ),
                tree.position.z
            );

            const foliageScale = Math.max(
                0.08,
                canopyWidth * 0.42
            );

            const foliageMatrix = BABYLON.Matrix.Compose(
                new BABYLON.Vector3(
                    foliageScale * 1.6,
                    Math.max(0.1, canopyHeight * 0.45),
                    foliageScale * 1.6
                ),
                BABYLON.Quaternion.Identity(),
                canopyCenter
            );

            this.foliageRenderers[
                tree.speciesName
            ].addMatrix(foliageMatrix);
        }

        for (const renderer of Object.values(
            this.trunkRenderers
        )) {
            renderer.update();
        }

        for (const renderer of Object.values(
            this.foliageRenderers
        )) {
            renderer.update();
        }

        this.renderDirty = false;
    }


update(cameraPosition, deltaSeconds = 0) {
    if (this.disposed) return;

    // Streaming and simulation continue normally.
    this.populationManager.updateVisibleChunks(
        cameraPosition,
        this.renderRadius
    );

    const deltaYears = deltaSeconds * 0.05;

    if (deltaYears > 0) {
        this.populationManager.updateSimulation(
            deltaYears
        );
    }

    // Batch expensive rendering work.
    this.renderAccumulator += Math.max(
        0,
        Math.min(deltaSeconds, 0.25)
    );

    if (
        this.renderDirty &&
        (
            this.renderAccumulator >= this.renderInterval
        )
    ) {
        this.rebuildRendering();
        this.renderAccumulator = 0;
    }
}


    getStats() {
        return {
            ...this.populationManager.getStats(),

            activeTrees: Array.from(
                this.activeRecords
            ).reduce(
                (total, record) =>
                    total + record.population.trees.filter(
                        tree => tree.alive
                    ).length,
                0
            )
        };
    }

    dispose() {
        if (this.disposed) return;

        this.disposed = true;

        this.populationManager.dispose();

        for (const renderer of Object.values(
            this.trunkRenderers
        )) {
            renderer.mesh.dispose();
        }

        for (const renderer of Object.values(
            this.foliageRenderers
        )) {
            renderer.mesh.dispose();
        }

        this.activeRecords.clear();
    }


chooseSpeciesForSoil(soil, random) {
    const fallback = random() < 0.55 ? "oak" : "pine";

    if (!soil) {
        return fallback;
    }

    const calculateScore = (speciesName) => {
        const species = Species[speciesName];

        // Some species definitions may not yet have
        // geological affinity data.
        if (!species || !species.soilAffinity) {
            return 0.5;
        }

        const affinity = species.soilAffinity;

        const preferred = Array.isArray(affinity.preferred)
            ? affinity.preferred
            : [];

        const texture = preferred.includes(soil.surfaceSoil)
            ? 1.0
            : (affinity.tolerance ?? 0.5);

        const fertility = 1 - Math.abs(
            (soil.fertility ?? 0.5) -
            (affinity.fertility ?? 0.5)
        );

        const drainage = 1 - Math.abs(
            (soil.drainage ?? 0.5) -
            (affinity.drainage ?? 0.5)
        );

        const moisture = 1 - Math.abs(
            (soil.moisture ?? 0.5) -
            (affinity.moisture ?? 0.5)
        );

        return (
            texture * 0.45 +
            fertility * 0.20 +
            drainage * 0.20 +
            moisture * 0.15
        );
    };

    const oakScore = calculateScore("oak");
    const pineScore = calculateScore("pine");

    const total = oakScore + pineScore;

    if (total <= 0) {
        return fallback;
    }

    return random() < oakScore / total
        ? "oak"
        : "pine";
}

}

import { GeologicalField } from "../geology/geological-field.js";
import { RockSpecies } from "../geology/rock-species.js";
import { RockGenerator } from "../geology/rock-generator.js";
import { CreateSeededRandom } from "../core/random.js";
import { ThinInstanceRenderer } from "../rendering/thin-instance-renderer.js";

export class ChunkGeologySystem {
    constructor(scene, terrain, options = {}) {
        this.scene = scene;
        this.terrain = terrain;

        this.chunkSize = options.chunkSize ?? 2000;
        this.renderRadius = options.renderRadius ?? 3;
        this.rocksPerChunk = options.rocksPerChunk ?? 10;
        this.seaLevel = options.seaLevel ?? -100;
        this.worldSeed = options.worldSeed ?? 91827;

        this.field = new GeologicalField(
            terrain,
            this.worldSeed
        );

        // Persistent geological records.
        this.records = new Map();

        // Currently rendered chunks.
        this.activeChunks = new Set();

        this.renderers = {};
        this.disposed = false;
        this.renderDirty = true;

        this.createRenderers();

        console.log(
            "[Geology] Chunk geology initialized."
        );
    }

    createRenderers() {
        for (const speciesName of Object.keys(RockSpecies)) {
            const config = RockSpecies[speciesName];

            // Surface materials are handled separately.
            if (
                speciesName === "sand" ||
                speciesName === "clay" ||
                speciesName === "fertileSoil"
            ) {
                continue;
            }

		const mesh = BABYLON.MeshBuilder.CreateBox(
    "chunkRock_" + speciesName,
    {
        size: 1,
        faceUV: [
            new BABYLON.Vector4(0, 0, 1, 1),
            new BABYLON.Vector4(0, 0, 1, 1),
            new BABYLON.Vector4(0, 0, 1, 1),
            new BABYLON.Vector4(0, 0, 1, 1),
            new BABYLON.Vector4(0, 0, 1, 1),
            new BABYLON.Vector4(0, 0, 1, 1)
        ]
    },
    this.scene
);

            const material = new BABYLON.StandardMaterial(
                "chunkRockMaterial_" + speciesName,
                this.scene
            );

            material.diffuseColor = config.color.clone();

            material.specularColor =
                new BABYLON.Color3(0.06, 0.06, 0.06);

            if (speciesName === "anomaly") {
                material.emissiveColor =
                    config.color.scale(0.35);
            }

            mesh.material = material;

            this.renderers[speciesName] =
                new ThinInstanceRenderer(
                    this.scene,
                    "chunkRock_" + speciesName,
                    mesh,
                    null
                );
        }
    }

    getChunkKey(x, z) {
        return `${x},${z}`;
    }

    getChunkSeed(chunkX, chunkZ) {
        return (
            this.worldSeed ^
            Math.imul(chunkX, 73856093) ^
            Math.imul(chunkZ, 19349663)
        ) >>> 0;
    }

    generateChunk(chunkX, chunkZ) {
        const key = this.getChunkKey(chunkX, chunkZ);

        if (this.records.has(key)) {
            return this.records.get(key);
        }

        const seed = this.getChunkSeed(
            chunkX,
            chunkZ
        );

        const random = CreateSeededRandom(seed);

        const minX = Math.max(
            chunkX * this.chunkSize,
            this.terrain.min.x
        );

        const maxX = Math.min(
            (chunkX + 1) * this.chunkSize,
            this.terrain.max.x
        );

        const minZ = Math.max(
            chunkZ * this.chunkSize,
            this.terrain.min.z
        );

        const maxZ = Math.min(
            (chunkZ + 1) * this.chunkSize,
            this.terrain.max.z
        );

        const formations = [];

        if (minX >= maxX || minZ >= maxZ) {
            const emptyRecord = {
                key,
                chunkX,
                chunkZ,
                seed,
                formations
            };

            this.records.set(key, emptyRecord);
            return emptyRecord;
        }

        const attemptsLimit = this.rocksPerChunk * 12;

        let attempts = 0;

        while (
            formations.length < this.rocksPerChunk &&
            attempts < attemptsLimit
        ) {
            attempts++;

            const x = minX + random() * (maxX - minX);
            const z = minZ + random() * (maxZ - minZ);

            const sample = this.field.sample(x, z);

            if (!sample) continue;

            if (sample.position.y <= this.seaLevel) {
                continue;
            }

		// Formations come from underlying bedrock,
// not the current surface-soil classification.
if (sample.exposure < 0.45) {
    continue;
}

const speciesName = sample.bedrockType;
const config = RockSpecies[speciesName];

            // Only generate actual geological formations.
            if (!config) continue;

            if (
                speciesName === "sand" ||
                speciesName === "clay" ||
                speciesName === "fertileSoil"
            ) {
                continue;
            }

            const variation = 0.85 + random() * 0.65;

            const formationSeed = (
                seed ^
                Math.imul(formations.length + 1, 2654435761)
            ) >>> 0;

            formations.push({
                id: `${key}:${formations.length}`,
                speciesName,
                position: sample.position.clone(),
                scale: config.scale * variation * 2.0,
                seed: formationSeed,
                age: 0,
                integrity: 1,
                erosionProgress: 0,
                alive: true
            });
        }

        const record = {
            key,
            chunkX,
            chunkZ,
            seed,
            formations
        };

        this.records.set(key, record);

        console.log(
            `[Geology] Generated ${formations.length} formations in ${key}`
        );

        return record;
    }

    update(cameraPosition) {
        if (this.disposed) return;

        const centerX = Math.floor(
            cameraPosition.x / this.chunkSize
        );

        const centerZ = Math.floor(
            cameraPosition.z / this.chunkSize
        );

        const desiredChunks = new Set();

        for (
            let x = centerX - this.renderRadius;
            x <= centerX + this.renderRadius;
            x++
        ) {
            for (
                let z = centerZ - this.renderRadius;
                z <= centerZ + this.renderRadius;
                z++
            ) {
                const minX = x * this.chunkSize;
                const minZ = z * this.chunkSize;

                const maxX = minX + this.chunkSize;
                const maxZ = minZ + this.chunkSize;

                if (
                    maxX <= this.terrain.min.x ||
                    minX >= this.terrain.max.x ||
                    maxZ <= this.terrain.min.z ||
                    minZ >= this.terrain.max.z
                ) {
                    continue;
                }

                const key = this.getChunkKey(x, z);

                desiredChunks.add(key);

                // Generate once, preserve thereafter.
                this.generateChunk(x, z);
            }
        }

        let changed = false;

        for (const key of this.activeChunks) {
            if (!desiredChunks.has(key)) {
                this.activeChunks.delete(key);
                changed = true;
            }
        }

        for (const key of desiredChunks) {
            if (!this.activeChunks.has(key)) {
                this.activeChunks.add(key);
                changed = true;
            }
        }

        if (changed || this.renderDirty) {
            this.rebuildRendering();
        }
    }

    rebuildRendering() {
        if (this.disposed) return;

        for (const renderer of Object.values(this.renderers)) {
            renderer.clear();
        }

        let visibleFormationCount = 0;

        for (const key of this.activeChunks) {
            const record = this.records.get(key);

            if (!record) continue;

            for (const formation of record.formations) {
                if (!formation.alive) continue;

                const config =
                    RockSpecies[formation.speciesName];

                const renderer =
                    this.renderers[formation.speciesName];

                if (!config || !renderer) continue;

                const generator = new RockGenerator(
                    config,
                    formation.seed
                );

                // Conservative initial detail level.
                // Higher recursive detail can be introduced
                // after performance validation.
                const depth = Math.min(
                    1,
                    config.recursionDepth
                );

                const matrices = generator.generate(
                    formation.position,
                    formation.scale,
                    depth
                );

                renderer.addMatrices(matrices);

                visibleFormationCount++;
            }
        }

        for (const renderer of Object.values(this.renderers)) {
            renderer.update();
        }

        this.visibleFormationCount =
            visibleFormationCount;

        this.renderDirty = false;
    }

    getStats() {
        let totalFormations = 0;

        for (const record of this.records.values()) {
            totalFormations += record.formations.filter(
                formation => formation.alive
            ).length;
        }

        return {
            generatedChunks: this.records.size,
            activeChunks: this.activeChunks.size,
            totalFormations,
            visibleFormations: this.visibleFormationCount ?? 0
        };
    }

    dispose() {
        if (this.disposed) return;

        this.disposed = true;

        for (const renderer of Object.values(this.renderers)) {
            renderer.mesh.dispose();
        }

        this.renderers = {};
        this.activeChunks.clear();
        this.records.clear();
    }
}

import { CreateSeededRandom } from "../core/random.js";

export class ChunkPopulationManager {
    constructor(options = {}) {
        this.worldSeed = options.worldSeed ?? 91827;
        this.chunkSize = options.chunkSize ?? 2000;

        this.maxGenerationPerUpdate =
            options.maxGenerationPerUpdate ?? 2;

        this.maxSimulationPerUpdate =
            options.maxSimulationPerUpdate ?? 8;

        this.adapter = options.adapter ?? {};

        this.records = new Map();
        this.pending = [];
        this.activeKeys = new Set();

        this.simulationCursor = 0;

        this.statistics = {
            generated: 0,
            activated: 0,
            deactivated: 0,
            simulated: 0
        };
    }

    getKey(x, z) {
        return `${x},${z}`;
    }

    getSeed(x, z) {
        let seed = this.worldSeed >>> 0;

        seed ^= Math.imul(x | 0, 73856093);
        seed ^= Math.imul(z | 0, 19349663);

        seed = Math.imul(
            seed ^ (seed >>> 16),
            2246822519
        );

        seed = Math.imul(
            seed ^ (seed >>> 13),
            3266489917
        );

        return (seed ^ (seed >>> 16)) >>> 0;
    }

    getChunkCoordinates(position) {
        return {
            x: Math.floor(position.x / this.chunkSize),
            z: Math.floor(position.z / this.chunkSize)
        };
    }

    getRecord(x, z) {
        return this.records.get(this.getKey(x, z)) ?? null;
    }

    requestChunk(x, z, priority = 0) {
        const key = this.getKey(x, z);

        const existing = this.records.get(key);

        if (existing) {
            existing.lastRequested = Date.now();

            this.activateRecord(existing);

            return existing;
        }

        const pending = this.pending.find(
            item => item.key === key
        );

        if (pending) {
            pending.priority = Math.max(
                pending.priority,
                priority
            );

            return null;
        }

        this.pending.push({
            x, z, key, priority
        });

        this.pending.sort(
            (a, b) => b.priority - a.priority
        );

        return null;
    }

    generateRecord(x, z) {
        const key = this.getKey(x, z);

        if (this.records.has(key)) {
            return this.records.get(key);
        }

        const seed = this.getSeed(x, z);

        const random = CreateSeededRandom(seed);

        const bounds = {
            minX: x * this.chunkSize,
            maxX: (x + 1) * this.chunkSize,
            minZ: z * this.chunkSize,
            maxZ: (z + 1) * this.chunkSize
        };

        const context = {
            x, z, key, seed, random, bounds
        };

        const population =
            this.adapter.generate?.(context) ?? {};

        const record = {
            x,
            z,
            key,
            seed,
            bounds,
            population,
            active: false,
            generated: true,
            simulationTime: 0,
            lastRequested: Date.now(),
            metadata: { version: 1 }
        };

        this.records.set(key, record);

        this.statistics.generated++;

        this.activateRecord(record);

        return record;
    }

    activateRecord(record) {
        if (record.active) return;

        this.adapter.activate?.(record);

        record.active = true;

        this.activeKeys.add(record.key);

        this.statistics.activated++;
    }

    deactivateRecord(record) {
        if (!record.active) return;

        this.adapter.deactivate?.(record);

        record.active = false;

        this.activeKeys.delete(record.key);

        this.statistics.deactivated++;
    }

    updateGeneration() {
        let generated = 0;

        while (
            this.pending.length > 0 &&
            generated < this.maxGenerationPerUpdate
        ) {
            const item = this.pending.shift();

            if (this.records.has(item.key)) continue;

            this.generateRecord(item.x, item.z);

            generated++;
        }

        return generated;
    }

    updateSimulation(deltaYears) {
        if (
            !Number.isFinite(deltaYears) ||
            deltaYears < 0
        ) {
            throw new RangeError(
                "deltaYears must be non-negative."
            );
        }

        const active = Array.from(this.activeKeys);

        if (active.length === 0) {
            this.simulationCursor = 0;
            return;
        }

        const count = Math.min(
            active.length,
            this.maxSimulationPerUpdate
        );

        const start =
            this.simulationCursor % active.length;

        for (let i = 0; i < count; i++) {
            const index = (start + i) % active.length;
            const record = this.records.get(active[index]);

            if (!record || !record.active) continue;

            this.adapter.simulate?.(
                record,
                deltaYears
            );

            record.simulationTime += deltaYears;

            this.statistics.simulated++;
        }

        this.simulationCursor =
            (start + count) % active.length;
    }

    updateVisibleChunks(cameraPosition, radius = 3) {
        const center = this.getChunkCoordinates(
            cameraPosition
        );

        const required = new Set();

        for (
            let x = center.x - radius;
            x <= center.x + radius;
            x++
        ) {
            for (
                let z = center.z - radius;
                z <= center.z + radius;
                z++
            ) {
                const dx = x - center.x;
                const dz = z - center.z;

                if (
                    dx * dx + dz * dz >
                    radius * radius
                ) continue;

                const key = this.getKey(x, z);

                required.add(key);

                const distance = Math.sqrt(
                    dx * dx + dz * dz
                );

                this.requestChunk(
                    x, z, 1000 - distance
                );
            }
        }

        // Remove obsolete requests before generating them.
        this.pending = this.pending.filter(
            item => required.has(item.key)
        );

        for (const key of Array.from(this.activeKeys)) {
            if (required.has(key)) continue;

            const record = this.records.get(key);

            if (record) {
                this.deactivateRecord(record);
            }
        }

        this.updateGeneration();
    }

    getPopulation(x, z) {
        return this.getRecord(x, z)?.population ?? null;
    }

    getStats() {
        return {
            ...this.statistics,
            totalRecords: this.records.size,
            activeRecords: this.activeKeys.size,
            pendingRecords: this.pending.length
        };
    }

    exportRecord(x, z) {
        const record = this.getRecord(x, z);

        if (!record) return null;

        return {
            version: 1,
            coordinates: { x, z },
            seed: record.seed,
            bounds: { ...record.bounds },
            simulationTime: record.simulationTime,
            population: this.adapter.serialize
                ? this.adapter.serialize(record)
                : structuredClone(record.population)
        };
    }

    importRecord(data) {
        if (!data || data.version !== 1) {
            throw new Error(
                "Unsupported population record version."
            );
        }

        const { x, z } = data.coordinates;
        const key = this.getKey(x, z);

        const previous = this.records.get(key);

        if (previous?.active) {
            this.deactivateRecord(previous);
        }

        const record = {
            x,
            z,
            key,
            seed: data.seed,
            bounds: { ...data.bounds },

            population: this.adapter.deserialize
                ? this.adapter.deserialize(data.population)
                : structuredClone(data.population),

            active: false,
            generated: true,
            simulationTime: data.simulationTime ?? 0,
            lastRequested: Date.now(),
            metadata: { version: 1 }
        };

        this.records.set(key, record);

        return record;
    }

    dispose() {
        for (const record of this.records.values()) {
            if (record.active) {
                this.deactivateRecord(record);
            }

            this.adapter.dispose?.(record);
        }

        this.records.clear();
        this.pending.length = 0;
        this.activeKeys.clear();

        this.simulationCursor = 0;
    }

class GeodesicSolver {

    constructor(geometry, options = {}) {

        this.geometry = geometry;

        this.iterations =
            options.iterations ?? 80;

        this.relaxation =
            options.relaxation ?? 0.15;

        this.segments =
            options.segments ?? 24;

    }

    solve(start, end) {

        const points = [];

        for (
            let i = 0;
            i <= this.segments;
            i++
        ) {

            const t = i / this.segments;

            points.push({

                x: start.x + (end.x - start.x) * t,

                y: start.y + (end.y - start.y) * t,

                z: start.z + (end.z - start.z) * t

            });

        }

        const step = this.relaxation;

        for (
            let iteration = 0;
            iteration < this.iterations;
            iteration++
        ) {

            for (
                let i = 1;
                i < points.length - 1;
                i++
            ) {

                const previous = points[i - 1];

                const current = points[i];

                const next = points[i + 1];

                const midpoint = {

                    x: (previous.x + next.x) * 0.5,

                    y: (previous.y + next.y) * 0.5,

                    z: (previous.z + next.z) * 0.5

                };

                const candidate = {

                    x: current.x +
                        (midpoint.x - current.x) * step,

                    y: current.y +
                        (midpoint.y - current.y) * step,

                    z: current.z +
                        (midpoint.z - current.z) * step

                };

                // Reject candidates outside bounded geometries.
                if (
                    typeof this.geometry.isInside === "function" &&
                    !this.geometry.isInside(candidate)
                ) {
                    continue;
                }

                points[i] = candidate;

            }

        }

        return {

            points,

            length: this.calculateLength(points)

        };

    }

    calculateLength(points) {

        let length = 0;

        for (
            let i = 0;
            i < points.length - 1;
            i++
        ) {

            length += this.geometry.distance(
                points[i],
                points[i + 1]
            );

        }

        return length;

    }

}

export { GeodesicSolver };

/**
 * GeometryRenderer
 *
 * Applies spatial transformations to Babylon.js
 * mesh geometry.
 *
 * Initial target:
 * - Streamed terrain meshes.
 *
 * Preserves:
 * - Original vertex positions.
 * - Original vertex normals.
 * - Mesh materials.
 * - Vertex colors.
 * - Mesh indices.
 *
 * Simulation coordinates remain unchanged.
 */

class GeometryRenderer {

    constructor(scene, spatialTransform, options = {}) {

        if (!scene || !spatialTransform) {
            throw new Error(
                "GeometryRenderer requires a scene and SpatialTransform."
            );
        }

        this.scene = scene;
        this.transform = spatialTransform;

        this.enabled = options.enabled ?? true;

        this.meshCache = new WeakMap();

        this.statistics = {
            transformedMeshes: 0,
            transformedVertices: 0,
            restoredMeshes: 0
        };

        console.log(
            "[GeometryRenderer] Initialized:",
            this.transform.getInfo()
        );
    }

    // -----------------------------------------
    // MESH TRANSFORMATION
    // -----------------------------------------

    applyMesh(mesh) {

        if (!this.enabled || !mesh) {
            return false;
        }

        if (this.transform.geometryName === "euclidean") {
            return false;
        }

        const cached = this.meshCache.get(mesh);

        if (cached) {
            return false;
        }

        const positions =
            mesh.getVerticesData(
                BABYLON.VertexBuffer.PositionKind
            );

        const indices = mesh.getIndices();

        if (
            !positions ||
            !indices ||
            positions.length === 0
        ) {
            return false;
        }

        // Preserve authoritative mesh geometry.

        const originalPositions =
            new Float32Array(positions);

        const originalNormals =
            mesh.getVerticesData(
                BABYLON.VertexBuffer.NormalKind
            );

        const cache = {
            positions: originalPositions,

            normals: originalNormals
                ? new Float32Array(originalNormals)
                : null
        };

        this.meshCache.set(mesh, cache);

        // Transform every vertex.

        const transformedPositions =
            new Float32Array(positions.length);

        for (
            let i = 0;
            i < positions.length;
            i += 3
        ) {

            const projected =
                this.transform.worldToRender({
                    x: positions[i],
                    y: positions[i + 1],
                    z: positions[i + 2]
                });

            transformedPositions[i] = projected.x;
            transformedPositions[i + 1] = projected.y;
            transformedPositions[i + 2] = projected.z;
        }

        // Recalculate normals in the new geometry.

        const transformedNormals =
            new Float32Array(positions.length);

        BABYLON.VertexData.ComputeNormals(
            transformedPositions,
            indices,
            transformedNormals
        );

        mesh.setVerticesData(
            BABYLON.VertexBuffer.PositionKind,
            transformedPositions,
            true,
            3
        );

        mesh.setVerticesData(
            BABYLON.VertexBuffer.NormalKind,
            transformedNormals,
            true,
            3
        );

        mesh.refreshBoundingInfo();

        mesh.computeWorldMatrix(true);

        mesh.metadata = {
            ...(mesh.metadata || {}),
            spatialGeometry: this.transform.geometryName,
            spatiallyTransformed: true
        };

        this.statistics.transformedMeshes++;

        this.statistics.transformedVertices +=
            positions.length / 3;

        return true;
    }

    // -----------------------------------------
    // RESTORE ORIGINAL GEOMETRY
    // -----------------------------------------

    restoreMesh(mesh) {

        if (!mesh) return false;

        const cache = this.meshCache.get(mesh);

        if (!cache) return false;

        mesh.setVerticesData(
            BABYLON.VertexBuffer.PositionKind,
            cache.positions,
            true,
            3
        );

        if (cache.normals) {

            mesh.setVerticesData(
                BABYLON.VertexBuffer.NormalKind,
                cache.normals,
                true,
                3
            );
        }

        mesh.refreshBoundingInfo();

        mesh.computeWorldMatrix(true);

        if (mesh.metadata) {

            mesh.metadata.spatiallyTransformed = false;

            delete mesh.metadata.spatialGeometry;
        }

        this.meshCache.delete(mesh);

        this.statistics.restoredMeshes++;

        return true;
    }

    // -----------------------------------------
    // RESTORE ALL KNOWN CHUNKS
    // -----------------------------------------

    restoreWorld(worldManager) {

        if (!worldManager) return;

        for (const chunk of worldManager.chunks.values()) {

            if (chunk.mesh) {
                this.restoreMesh(chunk.mesh);
            }
        }
    }

    // -----------------------------------------
    // REPROJECT ALL KNOWN CHUNKS
    // -----------------------------------------

    reprojectWorld(worldManager) {

        if (!worldManager) return;

        for (const chunk of worldManager.chunks.values()) {

            if (!chunk.mesh) continue;

            this.restoreMesh(chunk.mesh);

            this.applyMesh(chunk.mesh);
        }
    }

    // -----------------------------------------
    // CONFIGURATION
    // -----------------------------------------

    setEnabled(enabled, worldManager = null) {

        this.enabled = !!enabled;

        if (!worldManager) return;

        if (this.enabled) {
            this.reprojectWorld(worldManager);
        } else {
            this.restoreWorld(worldManager);
        }
    }

    // -----------------------------------------
    // DIAGNOSTICS
    // -----------------------------------------

    getStats() {

        return {
            enabled: this.enabled,

            geometry:
                this.transform.geometryName,

            ...this.statistics
        };
    }

    // -----------------------------------------
    // DISPOSAL
    // -----------------------------------------

    dispose(worldManager = null) {

        if (worldManager) {
            this.restoreWorld(worldManager);
        }

        this.enabled = false;

        console.log(
            "[GeometryRenderer] Disposed:",
            this.getStats()
        );
    }
	

    // -----------------------------------------
    // THIN INSTANCE TRANSFORMATION
    // -----------------------------------------

    transformInstanceMatrix(matrix) {

        if (
            !this.enabled ||
            this.transform.geometryName === "euclidean"
        ) {
            return matrix;
        }

        if (
            !matrix ||
            !matrix.m ||
            !Array.from(matrix.m).every(Number.isFinite)
        ) {
            return null;
        }

        // Extract translation directly from the matrix.

        const translation = matrix.getTranslation();

        if (
            !translation ||
            !Number.isFinite(translation.x) ||
            !Number.isFinite(translation.y) ||
            !Number.isFinite(translation.z)
        ) {
            return null;
        }

        // Reject positions outside the authoritative
        // projection domain.

        let projected;

        try {

            projected =
                this.transform.worldToRender({
                    x: translation.x,
                    y: translation.y,
                    z: translation.z
                });

        } catch (error) {

            return null;
        }

        const scaling = new BABYLON.Vector3();
        const rotation = new BABYLON.Quaternion();
        const decomposedTranslation =
            new BABYLON.Vector3();

        if (
            !matrix.decompose(
                scaling,
                rotation,
                decomposedTranslation
            )
        ) {
            return null;
        }

        // Validate the local transformation Jacobian.

        let J;

        try {

            J = this.transform.getJacobian({
                x: translation.x,
                y: translation.y,
                z: translation.z
            });

        } catch (error) {

            return null;
        }

        if (!J.every(Number.isFinite)) {
            return null;
        }

        // Estimate volume scaling.

        const determinant =
            J[0] * (J[4] * J[8] - J[5] * J[7]) -
            J[1] * (J[3] * J[8] - J[5] * J[6]) +
            J[2] * (J[3] * J[7] - J[4] * J[6]);

        if (!Number.isFinite(determinant)) {
            return null;
        }

        const scaleFactor = Math.cbrt(
            Math.max(Math.abs(determinant), 1e-18)
        );

        const adjustedScale =
            scaling.scale(scaleFactor);

        return BABYLON.Matrix.Compose(
            adjustedScale,
            rotation,
            new BABYLON.Vector3(
                projected.x,
                projected.y,
                projected.z
            )
        );
    }

	
    // -----------------------------------------
    // CAMERA BOUNDARY PROTECTION
    // -----------------------------------------

    constrainCamera(camera) {

        if (
            !camera ||
            this.transform.geometryName !== "poincare"
        ) {
            return;
        }

        const scale = this.transform.renderScale;
        const radius = this.transform.projection.radius;

        const x = camera.position.x / scale;
        const y = camera.position.y / scale;
        const z = camera.position.z / scale;

        const distance = Math.hypot(x, y, z);

        const maximumRadius =
            radius * (1 - 1e-9);

        if (distance >= maximumRadius) {

            const factor =
                maximumRadius / Math.max(distance, 1e-12);

            camera.position.scaleInPlace(factor);
        }
    }

    getWorldCameraPosition(camera) {

        this.constrainCamera(camera);

        return this.transform.renderToWorld({
            x: camera.position.x,
            y: camera.position.y,
            z: camera.position.z
        });
    }




}

export { GeometryRenderer };

/**
 * GeometryShaderManager
 *
 * GPU-side spatial projection for Babylon.js.
 *
 * Initial implementation:
 * - Three-dimensional Poincare projection.
 * - Configurable curvature.
 * - Configurable world bounds.
 * - Basic diffuse lighting.
 * - Independent shader material management.
 *
 * Initial limitation:
 * - Ordinary meshes only.
 * - Thin instances require a separate shader path.
 */

class GeometryShaderManager {

    constructor(scene, spatialTransform, options = {}) {

        if (!scene || !spatialTransform) {
            throw new Error(
                "GeometryShaderManager requires a scene and SpatialTransform."
            );
        }

        this.scene = scene;
        this.transform = spatialTransform;

        this.enabled = options.enabled ?? true;

        this.materials = new Set();
        this.originalMaterials = new WeakMap();

        this.renderScale =
            options.renderScale ??
            spatialTransform.renderScale;

        this.initializeShaders();

        console.log(
            "[GeometryShaderManager] Initialized."
        );
    }

    // -----------------------------------------
    // SHADER REGISTRATION
    // -----------------------------------------

    initializeShaders() {

        BABYLON.Effect.ShadersStore[
            "somniumPoincareVertexShader"
        ] = this.getVertexShader();

        BABYLON.Effect.ShadersStore[
            "somniumPoincareFragmentShader"
        ] = this.getFragmentShader();
    }

    // -----------------------------------------
    // VERTEX SHADER
    // -----------------------------------------

    getVertexShader() {

        return `
            precision highp float;

            attribute vec3 position;
            attribute vec3 normal;

            uniform mat4 world;
	    uniform mat4 viewProjection;

            uniform vec3 worldCenter;
            uniform vec3 worldHalfBounds;

            uniform float ballRadius;
            uniform float characteristicLength;
            uniform float renderScale;
            uniform float boundaryEpsilon;

            varying vec3 vNormal;
            varying vec3 vWorldPosition;

            float safeTanh(float value) {

                float x = clamp(value, 0.0, 20.0);

                float e = exp(2.0 * x);

                return (e - 1.0) / (e + 1.0);
            }

            vec3 poincareProject(vec3 worldPosition) {

                vec3 delta =
                    worldPosition - worldCenter;

                float distanceFromCenter =
                    length(delta);

                if (distanceFromCenter < 0.000001) {
                    return vec3(0.0);
                }

                vec3 direction =
                    delta / distanceFromCenter;

                vec3 safeDirection =
                    max(abs(direction), vec3(0.000001));

                vec3 boundaryDistances =
                    worldHalfBounds / safeDirection;

                float boundaryDistance =
                    min(
                        boundaryDistances.x,
                        min(
                            boundaryDistances.y,
                            boundaryDistances.z
                        )
                    );

                float u =
                    distanceFromCenter /
                    boundaryDistance;

                u = clamp(
                    u,
                    0.0,
                    1.0 - boundaryEpsilon
                );

                float hyperbolicDistance =
                    characteristicLength *
                    u / (1.0 - u);

                float projectedRadius =
                    ballRadius *
                    safeTanh(
                        hyperbolicDistance /
                        (2.0 * ballRadius)
                    );

                return direction * projectedRadius;
            }

	void main(void) {

    vec4 worldPosition =
        world * vec4(position, 1.0);

    vec3 projectedPosition =
        poincareProject(worldPosition.xyz);

    vWorldPosition =
        projectedPosition * renderScale;

    vNormal =
        normalize(mat3(world) * normal);

    gl_Position =
        viewProjection *
        vec4(vWorldPosition, 1.0);
}

        `;
    }

    // -----------------------------------------
    // FRAGMENT SHADER
    // -----------------------------------------

    getFragmentShader() {

        return `
            precision highp float;

            uniform vec4 baseColor;
            uniform vec3 lightDirection;
            uniform float ambientStrength;

            varying vec3 vNormal;
            varying vec3 vWorldPosition;

            void main(void) {

                vec3 normal =
                    normalize(vNormal);

                float diffuse =
                    max(
                        dot(normal, normalize(-lightDirection)),
                        0.0
                    );

                float illumination =
                    ambientStrength +
                    (1.0 - ambientStrength) * diffuse;

                vec3 color =
                    baseColor.rgb * illumination;

                gl_FragColor =
                    vec4(color, baseColor.a);
            }
        `;
    }

    // -----------------------------------------
    // MATERIAL CREATION
    // -----------------------------------------

    createMaterial(name, options = {}) {

        const material =
            new BABYLON.ShaderMaterial(
                name,
                this.scene,
                {
                    vertex: "somniumPoincare",
                    fragment: "somniumPoincare"
                },
                {
                    attributes: [
                        "position",
                        "normal"
                    ],

                    uniforms: [
                        "world",
                        "worldViewProjection",
                        "worldCenter",
                        "worldHalfBounds",
                        "ballRadius",
                        "characteristicLength",
                        "renderScale",
                        "boundaryEpsilon",
                        "baseColor",
                        "lightDirection",
                        "ambientStrength"
                    ]
                }
            );

        const projection =
            this.transform.projection;

        const bounds = projection.bounds;

        material.setVector3(
            "worldCenter",
            new BABYLON.Vector3(
                projection.center.x,
                projection.center.y,
                projection.center.z
            )
        );

        material.setVector3(
            "worldHalfBounds",
            new BABYLON.Vector3(
                (bounds.maxX - bounds.minX) / 2,
                (bounds.maxY - bounds.minY) / 2,
                (bounds.maxZ - bounds.minZ) / 2
            )
        );

        material.setFloat(
            "ballRadius",
            projection.radius
        );

        material.setFloat(
            "characteristicLength",
            projection.characteristicLength
        );

        material.setFloat(
            "renderScale",
            this.renderScale
        );

        material.setFloat(
            "boundaryEpsilon",
            projection.boundaryEpsilon
        );

        material.setColor4(
            "baseColor",
            options.color ??
                new BABYLON.Color3(0.35, 0.55, 0.3),
            options.alpha ?? 1
        );

        material.setVector3(
            "lightDirection",
            options.lightDirection ??
                new BABYLON.Vector3(-0.5, -1, -0.3).normalize()
        );

        material.setFloat(
            "ambientStrength",
            options.ambientStrength ?? 0.3
        );

        material.backFaceCulling =
            options.backFaceCulling ?? true;

material.onBindObservable.add(mesh => {

    const camera = this.scene.activeCamera;

    if (!camera) return;

    const world = mesh.getWorldMatrix();

    const viewProjection =
        camera.getViewMatrix().multiply(
            camera.getProjectionMatrix()
        );

    material.setMatrix("world", world);

    material.setMatrix(
        "viewProjection",
        viewProjection
    );
});
        this.materials.add(material);

        return material;
    }

    // -----------------------------------------
    // APPLY TO MESH
    // -----------------------------------------

    apply(mesh, options = {}) {

        if (!this.enabled || !mesh) {
            return false;
        }

        if (
            mesh.metadata?.spatiallyTransformed
        ) {
            throw new Error(
                "Cannot apply GPU projection to a CPU-projected mesh."
            );
        }

        if (
            mesh.metadata?.geometryShaderApplied
        ) {
            return false;
        }

        if (!this.originalMaterials.has(mesh)) {

            this.originalMaterials.set(
                mesh,
                mesh.material
            );
        }

        const material =
            this.createMaterial(
                options.name ??
                    `poincare-${mesh.name}`,
                options
            );

        mesh.material = material;

        mesh.metadata = {
            ...(mesh.metadata || {}),
            geometryShaderApplied: true
        };

        return true;
    }

    // -----------------------------------------
    // RESTORE ORIGINAL MATERIAL
    // -----------------------------------------

    restore(mesh) {

        if (!mesh) return false;

        if (!this.originalMaterials.has(mesh)) {
            return false;
        }

        mesh.material =
            this.originalMaterials.get(mesh);

        this.originalMaterials.delete(mesh);

        if (mesh.metadata) {
            delete mesh.metadata.geometryShaderApplied;
        }

        return true;
    }

    // -----------------------------------------
    // CONFIGURATION
    // -----------------------------------------

    setEnabled(enabled) {

        this.enabled = !!enabled;
    }

    // -----------------------------------------
    // DIAGNOSTICS
    // -----------------------------------------

    getStats() {

        return {
            enabled: this.enabled,
            geometry: this.transform.geometryName,
            activeMaterials: this.materials.size,
            renderScale: this.renderScale
        };
    }

    // -----------------------------------------
    // DISPOSAL
    // -----------------------------------------

    dispose() {

        for (const material of this.materials) {
            material.dispose();
        }

        this.materials.clear();

        console.log(
            "[GeometryShaderManager] Disposed."
        );
    }
}

export { GeometryShaderManager };

/**
 * HyperbolicCameraController
 *
 * Converts ordinary Babylon.js camera input into
 * hyperbolically distance-limited geodesic movement.
 *
 * Authoritative world coordinates remain unchanged.
 */

class HyperbolicCameraController {
    constructor(camera, spatialTransform, options = {}) {
        if (!camera || !spatialTransform) {
            throw new Error(
                "HyperbolicCameraController requires a camera and SpatialTransform."
            );
        }

        this.camera = camera;
        this.transform = spatialTransform;
        this.projection = spatialTransform.projection;
        this.metric = spatialTransform.metric;

        this.enabled = options.enabled ?? true;

        this.maxHyperbolicSpeed =
            options.maxHyperbolicSpeed ?? 2.5;

        this.boundaryMargin =
            options.boundaryMargin ?? 1e-7;

        this.maxDeltaSeconds =
            options.maxDeltaSeconds ?? 0.05;

        this.statistics = {
            updates: 0,
            geodesicMoves: 0,
            boundaryCorrections: 0,
            limitedMoves: 0
        };

        this.previousPosition = this.getBallPosition();

        this.constrainCamera();

        this.previousPosition = this.getBallPosition();

        console.log(
            "[HyperbolicCameraController] Initialized:",
            this.getInfo()
        );
    }

    // -----------------------------------------
    // POSITION CONVERSION
    // -----------------------------------------

    getBallPosition() {
        const scale = this.transform.renderScale;

        return {
            x: this.camera.position.x / scale,
            y: this.camera.position.y / scale,
            z: this.camera.position.z / scale
        };
    }

    setBallPosition(position) {
        const scale = this.transform.renderScale;

        this.camera.position.set(
            position.x * scale,
            position.y * scale,
            position.z * scale
        );
    }

    // -----------------------------------------
    // OPEN BALL PROTECTION
    // -----------------------------------------

    constrainPosition(position) {
        const radius = this.projection.radius;

        const maximumRadius =
            radius * (1 - this.boundaryMargin);

        const distance = Math.hypot(
            position.x,
            position.y,
            position.z
        );

        if (distance < maximumRadius) {
            return {
                x: position.x,
                y: position.y,
                z: position.z
            };
        }

        const factor =
            maximumRadius / Math.max(distance, 1e-15);

        this.statistics.boundaryCorrections++;

        return {
            x: position.x * factor,
            y: position.y * factor,
            z: position.z * factor
        };
    }

    constrainCamera() {
        const position = this.constrainPosition(
            this.getBallPosition()
        );

        this.setBallPosition(position);

        return position;
    }

    // -----------------------------------------
    // GEODESIC CAMERA MOVEMENT
    // -----------------------------------------

    update(deltaSeconds) {
        if (!this.enabled) {
            return;
        }

        if (!Number.isFinite(deltaSeconds) ||
            deltaSeconds <= 0) {
            return;
        }

        const elapsed = Math.min(
            deltaSeconds,
            this.maxDeltaSeconds
        );

        const start = this.previousPosition;

        // Babylon.js has already processed ordinary
        // keyboard and mouse camera input.
        const target = this.constrainPosition(
            this.getBallPosition()
        );

        let distance;

        try {
            distance = this.metric.distance(
                start,
                target
            );
        } catch (error) {
            this.setBallPosition(start);
            return;
        }

        if (!Number.isFinite(distance)) {
            this.setBallPosition(start);
            return;
        }

        if (distance < 1e-10) {
            this.setBallPosition(target);
            this.previousPosition = target;
            return;
        }

        // Limit movement by hyperbolic distance.
        const maximumStep =
            this.maxHyperbolicSpeed * elapsed;

        const fraction = Math.min(
            1,
            maximumStep / distance
        );

        if (fraction < 1) {
            this.statistics.limitedMoves++;
        }

        // Analytic Poincare geodesic interpolation.
        const nextPosition =
            this.metric.interpolateGeodesic(
                start,
                target,
                fraction
            );

        const safePosition =
            this.constrainPosition(nextPosition);

        this.setBallPosition(safePosition);

        this.previousPosition = safePosition;

        this.statistics.updates++;
        this.statistics.geodesicMoves++;
    }

    // -----------------------------------------
    // AUTHORITATIVE WORLD POSITION
    // -----------------------------------------

    getWorldPosition() {
        this.constrainCamera();

        return this.transform.renderToWorld({
            x: this.camera.position.x,
            y: this.camera.position.y,
            z: this.camera.position.z
        });
    }

    // -----------------------------------------
    // TELEPORT / POSITION SYNCHRONIZATION
    // -----------------------------------------

    synchronize() {
        this.constrainCamera();
        this.previousPosition = this.getBallPosition();
    }

    // -----------------------------------------
    // CONFIGURATION
    // -----------------------------------------

    setEnabled(enabled) {
        this.enabled = !!enabled;

        this.synchronize();
    }

    getInfo() {
        return {
            enabled: this.enabled,
            geometry: "poincare",
            movement: "hyperbolic-geodesic",
            maxHyperbolicSpeed: this.maxHyperbolicSpeed,
            boundaryMargin: this.boundaryMargin,
            boundaryReachable: false,
            statistics: { ...this.statistics }
        };
    }

    getStats() {
        return this.getInfo();
    }

    // -----------------------------------------
    // DISPOSAL
    // -----------------------------------------

    dispose() {
        this.enabled = false;
        this.camera = null;
        this.previousPosition = null;
    }
}

export { HyperbolicCameraController };

import { GeometryModel } from "./geometry-model.js";
import { MetricTensor } from "../metric-tensor.js";

class PoincareGeometry extends GeometryModel {

    constructor(curvature = -1) {

        super("poincare");

        if (
            !Number.isFinite(curvature) ||
            curvature >= 0
        ) {
            throw new Error(
                "Poincare curvature must be negative."
            );
        }

        this.curvature = curvature;

        this.k = -curvature;

        this.radius = 1 / Math.sqrt(this.k);

    }

    isInside(position) {

        const radiusSquared =
            position.x * position.x +
            position.y * position.y +
            position.z * position.z;

        return radiusSquared <
            this.radius * this.radius;

    }

    metricAt(position) {

        if (!this.isInside(position)) {

            throw new RangeError(
                "Position lies outside the Poincare ball."
            );

        }

        const r2 =
            position.x * position.x +
            position.y * position.y +
            position.z * position.z;

        const denominator =
            1 - this.k * r2;

        const factor =
            4 / (denominator * denominator);

        return new MetricTensor([

            factor, 0, 0,

            0, factor, 0,

            0, 0, factor

        ]);

    }

    distance(a, b) {

        if (
            !this.isInside(a) ||
            !this.isInside(b)
        ) {

            throw new RangeError(
                "Both positions must lie inside the Poincare ball."
            );

        }

        const dot = 
            a.x * b.x +
            a.y * b.y +
            a.z * b.z;

        const normA =
            a.x * a.x +
            a.y * a.y +
            a.z * a.z;

        const normB =
            b.x * b.x +
            b.y * b.y +
            b.z * b.z;

        const differenceSquared =

            (a.x - b.x) ** 2 +

            (a.y - b.y) ** 2 +

            (a.z - b.z) ** 2;

        const numerator =
            this.k * differenceSquared;

        const denominator =
            (1 - this.k * normA) *
            (1 - this.k * normB);

        const argument =
            1 + 2 * numerator / denominator;

        return Math.acosh(
            Math.max(1, argument)
        ) / Math.sqrt(this.k);

    }

}

/**
 * PoincareMetric
 *
 * Three-dimensional hyperbolic geometry
 * represented using Poincare ball coordinates.
 *
 * Supports:
 * - Metric tensor evaluation.
 * - Exact hyperbolic distances.
 * - Hyperboloid conversion.
 * - Analytic geodesic interpolation.
 * - Geodesic path generation.
 */

import { MetricTensor } from "../metric-tensor.js";

class PoincareMetric {

    constructor(curvature = -1) {

        if (
            !Number.isFinite(curvature) ||
            curvature >= 0
        ) {
            throw new Error(
                "Poincare curvature must be negative."
            );
        }

        this.curvature = curvature;

        this.k = -curvature;

        this.radius = 1 / Math.sqrt(this.k);
    }

    // -----------------------------------------
    // POSITION VALIDATION
    // -----------------------------------------

    validatePosition(position) {

        if (
            !position ||
            !Number.isFinite(position.x) ||
            !Number.isFinite(position.y) ||
            !Number.isFinite(position.z)
        ) {
            throw new TypeError(
                "Invalid Poincare position."
            );
        }

        if (!this.isInside(position)) {
            throw new RangeError(
                "Position lies outside the open Poincare ball."
            );
        }
    }

    isInside(position) {

        if (
            !position ||
            !Number.isFinite(position.x) ||
            !Number.isFinite(position.y) ||
            !Number.isFinite(position.z)
        ) {
            return false;
        }

        const radiusSquared =
            position.x ** 2 +
            position.y ** 2 +
            position.z ** 2;

        return radiusSquared <
            this.radius ** 2;
    }

    // -----------------------------------------
    // METRIC TENSOR
    // -----------------------------------------

    metricAt(position) {

        this.validatePosition(position);

        const radiusSquared =
            position.x ** 2 +
            position.y ** 2 +
            position.z ** 2;

        const denominator =
            this.radius ** 2 - radiusSquared;

        const factor =
            4 * this.radius ** 4 /
            (denominator ** 2);

        return new MetricTensor([
            factor, 0, 0,
            0, factor, 0,
            0, 0, factor
        ]);
    }

    // -----------------------------------------
    // HYPERBOLIC DISTANCE
    // -----------------------------------------

    distance(a, b) {

        this.validatePosition(a);
        this.validatePosition(b);

        const dx = a.x - b.x;
        const dy = a.y - b.y;
        const dz = a.z - b.z;

        const differenceSquared =
            dx * dx + dy * dy + dz * dz;

        const normA =
            a.x ** 2 +
            a.y ** 2 +
            a.z ** 2;

        const normB =
            b.x ** 2 +
            b.y ** 2 +
            b.z ** 2;

        const numerator =
            this.k * differenceSquared;

        const denominator =
            (1 - this.k * normA) *
            (1 - this.k * normB);

        const argument =
            1 + 2 * numerator / denominator;

        return Math.acosh(
            Math.max(1, argument)
        ) / Math.sqrt(this.k);
    }

    // -----------------------------------------
    // POINCARE TO HYPERBOLOID
    // -----------------------------------------

    toHyperboloid(position) {

        this.validatePosition(position);

        const R = this.radius;

        const radiusSquared =
            position.x ** 2 +
            position.y ** 2 +
            position.z ** 2;

        const denominator =
            R * R - radiusSquared;

        const factor =
            2 * R * R / denominator;

        return {
            w:
                R * (R * R + radiusSquared) /
                denominator,

            x: factor * position.x,

            y: factor * position.y,

            z: factor * position.z
        };
    }

    // -----------------------------------------
    // HYPERBOLOID TO POINCARE
    // -----------------------------------------

    fromHyperboloid(position) {

        const R = this.radius;

        if (
            !position ||
            !Number.isFinite(position.w) ||
            !Number.isFinite(position.x) ||
            !Number.isFinite(position.y) ||
            !Number.isFinite(position.z) ||
            position.w <= 0
        ) {
            throw new TypeError(
                "Invalid hyperboloid position."
            );
        }

        const denominator =
            position.w + R;

        const result = {
            x: R * position.x / denominator,

            y: R * position.y / denominator,

            z: R * position.z / denominator
        };

        this.validatePosition(result);

        return result;
    }

    // -----------------------------------------
    // MINKOWSKI INNER PRODUCT
    // -----------------------------------------

    minkowskiProduct(a, b) {

        return (
            -a.w * b.w +
            a.x * b.x +
            a.y * b.y +
            a.z * b.z
        );
    }

    // -----------------------------------------
    // GEODESIC INTERPOLATION
    // -----------------------------------------

    interpolateGeodesic(a, b, t) {

        this.validatePosition(a);
        this.validatePosition(b);

        if (
            !Number.isFinite(t) ||
            t < 0 ||
            t > 1
        ) {
            throw new RangeError(
                "Geodesic parameter must be between 0 and 1."
            );
        }

        if (t === 0) return { ...a };
        if (t === 1) return { ...b };

        const A = this.toHyperboloid(a);
        const B = this.toHyperboloid(b);

        const R = this.radius;

        const coshDistance =
            -this.minkowskiProduct(A, B) /
            (R * R);

        const distance = Math.acosh(
            Math.max(1, coshDistance)
        );

        // Handle nearly identical points.

        if (distance < 1e-10) {

            return {
                x: a.x + (b.x - a.x) * t,
                y: a.y + (b.y - a.y) * t,
                z: a.z + (b.z - a.z) * t
            };
        }

        const denominator =
            Math.sinh(distance);

        const weightA =
            Math.sinh((1 - t) * distance) /
            denominator;

        const weightB =
            Math.sinh(t * distance) /
            denominator;

        const point = {
            w:
                weightA * A.w +
                weightB * B.w,

            x:
                weightA * A.x +
                weightB * B.x,

            y:
                weightA * A.y +
                weightB * B.y,

            z:
                weightA * A.z +
                weightB * B.z
        };

        return this.fromHyperboloid(point);
    }

    // -----------------------------------------
    // GEODESIC PATH
    // -----------------------------------------

    geodesic(a, b, segments = 32) {

        this.validatePosition(a);
        this.validatePosition(b);

        if (
            !Number.isInteger(segments) ||
            segments < 1 ||
            segments > 10000
        ) {
            throw new RangeError(
                "Invalid geodesic segment count."
            );
        }

        const points = [];

        for (let i = 0; i <= segments; i++) {

            const t = i / segments;

            points.push(
                this.interpolateGeodesic(
                    a,
                    b,
                    t
                )
            );
        }

        return {
            points,

            length: this.distance(a, b),

            segments,

            geometry: "poincare-3d"
        };
    }

    // -----------------------------------------
    // INFORMATION
    // -----------------------------------------

    getInfo() {

        return {
            geometry: "poincare-3d",

            curvature: this.curvature,

            radius: this.radius,

            dimensions: 3,

            boundaryReachable: false
        };
    }
}

export { PoincareMetric };

/**
 * PoincareProjection
 *
 * Maps Somnium's finite 3D world into an open
 * three-dimensional Poincare ball.
 *
 * All three spatial dimensions participate
 * in the transformation.
 *
 * The world boundary approaches the ball
 * boundary asymptotically.
 */

class PoincareProjection {

    constructor(options = {}) {

        this.bounds = {
            minX: options.minX ?? -500000,
            maxX: options.maxX ?? 500000,

            minY: options.minY ?? -2000,
            maxY: options.maxY ?? 1000,

            minZ: options.minZ ?? -500000,
            maxZ: options.maxZ ?? 500000
        };

        this.curvature = options.curvature ?? -1;

        if (
            !Number.isFinite(this.curvature) ||
            this.curvature >= 0
        ) {
            throw new Error(
                "Poincare curvature must be negative."
            );
        }

        this.k = -this.curvature;

        this.radius = 1 / Math.sqrt(this.k);

	this.characteristicLength =
options.characteristicLength ?? 100000;

        this.boundaryEpsilon =
            options.boundaryEpsilon ?? 1e-9;

        if (
            !Number.isFinite(this.characteristicLength) ||
            this.characteristicLength <= 0
        ) {
            throw new Error(
                "Invalid characteristic length."
            );
        }

        if (
            !Number.isFinite(this.boundaryEpsilon) ||
            this.boundaryEpsilon <= 0 ||
            this.boundaryEpsilon >= 0.01
        ) {
            throw new Error(
                "Invalid boundary epsilon."
            );
        }

        this.center = {
            x: (this.bounds.minX + this.bounds.maxX) / 2,
            y: (this.bounds.minY + this.bounds.maxY) / 2,
            z: (this.bounds.minZ + this.bounds.maxZ) / 2
        };

        this.validateBounds();
    }

    // -----------------------------------------
    // VALIDATION
    // -----------------------------------------

    validateBounds() {

        const values = Object.values(this.bounds);

        if (
            !values.every(Number.isFinite) ||
            this.bounds.minX >= this.bounds.maxX ||
            this.bounds.minY >= this.bounds.maxY ||
            this.bounds.minZ >= this.bounds.maxZ
        ) {
            throw new Error(
                "Invalid three-dimensional world bounds."
            );
        }
    }

    validatePosition(position) {

        if (
            !position ||
            !Number.isFinite(position.x) ||
            !Number.isFinite(position.y) ||
            !Number.isFinite(position.z)
        ) {
            throw new TypeError(
                "Invalid three-dimensional position."
            );
        }
    }

    // -----------------------------------------
    // WORLD BOUNDARY DISTANCE
    // -----------------------------------------

    getBoundaryDistance(direction) {

        const b = this.bounds;

        const distances = [];

        if (direction.x > 0) {
            distances.push(
                (b.maxX - this.center.x) / direction.x
            );
        } else if (direction.x < 0) {
            distances.push(
                (b.minX - this.center.x) / direction.x
            );
        }

        if (direction.y > 0) {
            distances.push(
                (b.maxY - this.center.y) / direction.y
            );
        } else if (direction.y < 0) {
            distances.push(
                (b.minY - this.center.y) / direction.y
            );
        }

        if (direction.z > 0) {
            distances.push(
                (b.maxZ - this.center.z) / direction.z
            );
        } else if (direction.z < 0) {
            distances.push(
                (b.minZ - this.center.z) / direction.z
            );
        }

        return Math.min(...distances);
    }

    // -----------------------------------------
    // WORLD TO POINCARE BALL
    // -----------------------------------------

    worldToBall(position) {

        this.validatePosition(position);

        const dx = position.x - this.center.x;
        const dy = position.y - this.center.y;
        const dz = position.z - this.center.z;

        const distance = Math.hypot(dx, dy, dz);

        if (distance === 0) {
            return { x: 0, y: 0, z: 0 };
        }

        const direction = {
            x: dx / distance,
            y: dy / distance,
            z: dz / distance
        };

        const boundaryDistance =
            this.getBoundaryDistance(direction);

        const normalizedRadius =
            distance / boundaryDistance;

        if (
            normalizedRadius < 0 ||
            normalizedRadius > 1 + 1e-10
        ) {
            throw new RangeError(
                "Position lies outside world bounds."
            );
        }

        // Clamp exact boundary positions slightly inward.
        // The mathematical boundary remains unreachable.

        const u = Math.min(
            normalizedRadius,
            1 - this.boundaryEpsilon
        );

        // Compactify the finite world.

        const hyperbolicDistance =
            this.characteristicLength *
            u / (1 - u);

        // Hyperbolic radial projection.

const projectedRadius = Math.min(
    this.radius * Math.tanh(
        hyperbolicDistance / (2 * this.radius)
    ),
    this.radius * (1 - this.boundaryEpsilon)
);
	
        return {
            x: direction.x * projectedRadius,
            y: direction.y * projectedRadius,
            z: direction.z * projectedRadius
        };
    }

    // -----------------------------------------
    // POINCARE BALL TO WORLD
    // -----------------------------------------

    ballToWorld(position) {

        this.validatePosition(position);

        const radius = Math.hypot(
            position.x,
            position.y,
            position.z
        );

        if (radius >= this.radius) {
            throw new RangeError(
                "Position is outside the open Poincare ball."
            );
        }

        if (radius === 0) {
            return { ...this.center };
        }

        const direction = {
            x: position.x / radius,
            y: position.y / radius,
            z: position.z / radius
        };

        const ratio = radius / this.radius;

        const hyperbolicDistance =
            2 * this.radius * Math.atanh(ratio);

        const u =
            hyperbolicDistance /
            (
                this.characteristicLength +
                hyperbolicDistance
            );

        const boundaryDistance =
            this.getBoundaryDistance(direction);

        const worldDistance =
            u * boundaryDistance;

        return {
            x: this.center.x +
                direction.x * worldDistance,

            y: this.center.y +
                direction.y * worldDistance,

            z: this.center.z +
                direction.z * worldDistance
        };
    }

    // -----------------------------------------
    // PROJECTION INFORMATION
    // -----------------------------------------

    getInfo() {

        return {
            geometry: "poincare-3d",

            dimensions: 3,

            curvature: this.curvature,

            radius: this.radius,

            center: { ...this.center },

            bounds: { ...this.bounds },

            characteristicLength:
                this.characteristicLength,

            boundaryReachable: false
        };
    }
}

export { PoincareProjection };

/**
 * SpatialTransform
 *
 * Coordinates world geometry transformations.
 *
 * Separates:
 * - Authoritative world coordinates.
 * - Mathematical geometry coordinates.
 * - Rendering coordinates.
 *
 * Supports Euclidean and Poincare geometry.
 */

import { PoincareProjection } from "./poincare-projection.js";
import { PoincareMetric } from "./poincare-metric.js";

import { EuclideanGeometry } from "./euclidean-geometry.js";
import { MetricTensor } from "../metric-tensor.js";

class SpatialTransform {

    constructor(options = {}) {

        this.geometryName =
            options.geometry ?? "poincare";

        this.renderScale =
            options.renderScale ?? 10000;

        this.enabled =
            options.enabled ?? true;

        if (
            !Number.isFinite(this.renderScale) ||
            this.renderScale <= 0
        ) {
            throw new Error(
                "Invalid geometry render scale."
            );
        }

        this.euclidean =
            new EuclideanGeometry();

        this.projection =
            options.projection ??
            new PoincareProjection(
                options.bounds ?? {}
            );

        this.metric =
            options.metric ??
            new PoincareMetric(
                this.projection.curvature
            );

        this.setGeometry(this.geometryName);
    }

    // -----------------------------------------
    // GEOMETRY SELECTION
    // -----------------------------------------

    setGeometry(name) {

        if (
            name !== "euclidean" &&
            name !== "poincare"
        ) {
            throw new Error(
                `Unsupported geometry: ${name}`
            );
        }

        this.geometryName = name;

        console.log(
            "[SpatialTransform] Geometry:",
            name
        );

    }

    // -----------------------------------------
    // WORLD TO GEOMETRY COORDINATES
    // -----------------------------------------

    worldToGeometry(position) {

        if (
            !this.enabled ||
            this.geometryName === "euclidean"
        ) {
            return {
                x: position.x,
                y: position.y,
                z: position.z
            };
        }

        return this.projection.worldToBall(position);
    }

    // -----------------------------------------
    // GEOMETRY TO WORLD COORDINATES
    // -----------------------------------------

    geometryToWorld(position) {

        if (
            !this.enabled ||
            this.geometryName === "euclidean"
        ) {
            return { ...position };
        }

        return this.projection.ballToWorld(position);
    }

    // -----------------------------------------
    // WORLD TO RENDER COORDINATES
    // -----------------------------------------

    worldToRender(position) {

        const point =
            this.worldToGeometry(position);

        if (
            !this.enabled ||
            this.geometryName === "euclidean"
        ) {
            return point;
        }

        return {
            x: point.x * this.renderScale,
            y: point.y * this.renderScale,
            z: point.z * this.renderScale
        };
    }

    // -----------------------------------------
    // RENDER TO WORLD COORDINATES
    // -----------------------------------------

    renderToWorld(position) {

        if (
            !this.enabled ||
            this.geometryName === "euclidean"
        ) {
            return { ...position };
        }

        return this.geometryToWorld({
            x: position.x / this.renderScale,
            y: position.y / this.renderScale,
            z: position.z / this.renderScale
        });
    }

    // -----------------------------------------
    // NUMERICAL JACOBIAN
    // -----------------------------------------

    getJacobian(position, options = {}) {

        const h = options.step ?? 0.01;

        if (
            !Number.isFinite(h) ||
            h <= 0
        ) {
            throw new Error(
                "Invalid Jacobian step."
            );
        }

        const origin =
            this.worldToRender(position);

        const columns = [];

        for (const axis of ["x", "y", "z"]) {

            const forward = { ...position };
            const backward = { ...position };

            forward[axis] += h;
            backward[axis] -= h;

            const a = this.worldToRender(forward);
            const b = this.worldToRender(backward);

            columns.push([
                (a.x - b.x) / (2 * h),
                (a.y - b.y) / (2 * h),
                (a.z - b.z) / (2 * h)
            ]);
        }

        return [
            columns[0][0],
            columns[1][0],
            columns[2][0],

            columns[0][1],
            columns[1][1],
            columns[2][1],

            columns[0][2],
            columns[1][2],
            columns[2][2]
        ];
    }

    // -----------------------------------------
    // DIRECTION TRANSFORMATION
    // -----------------------------------------

    transformDirection(position, direction) {

        const J = this.getJacobian(position);

        return {
            x:
                J[0] * direction.x +
                J[1] * direction.y +
                J[2] * direction.z,

            y:
                J[3] * direction.x +
                J[4] * direction.y +
                J[5] * direction.z,

            z:
                J[6] * direction.x +
                J[7] * direction.y +
                J[8] * direction.z
        };
    }

    // -----------------------------------------
    // SPATIAL METRIC
    // -----------------------------------------

	
    metricAtWorld(position) {

        if (
            !this.enabled ||
            this.geometryName === "euclidean"
        ) {
            return this.euclidean.metricAt(position);
        }

        const ballPosition =
            this.projection.worldToBall(position);

        const geometryMetric =
            this.metric.metricAt(ballPosition).values;

        // Convert the rendering Jacobian into
        // the mathematical projection Jacobian.

        const J = this.getJacobian(position).map(
            value => value / this.renderScale
        );

        // Pull back G into world coordinates:
        // G_world = transpose(J) * G_ball * J

        const values = new Array(9).fill(0);

        for (let i = 0; i < 3; i++) {
            for (let j = 0; j < 3; j++) {

                for (let k = 0; k < 3; k++) {
                    for (let l = 0; l < 3; l++) {

                        values[i * 3 + j] +=
                            J[k * 3 + i] *
                            geometryMetric[k * 3 + l] *
                            J[l * 3 + j];
                    }
                }
            }
        }

        return new MetricTensor(values);
    }



    // -----------------------------------------
    // HYPERBOLIC DISTANCE
    // -----------------------------------------

    distance(a, b) {

        if (this.geometryName === "euclidean") {
            return this.euclidean.distance(a, b);
        }

        const pointA =
            this.projection.worldToBall(a);

        const pointB =
            this.projection.worldToBall(b);

        return this.metric.distance(
            pointA,
            pointB
        );
    }

    // -----------------------------------------
    // DIAGNOSTICS
    // -----------------------------------------

    getInfo() {

        return {
            geometry: this.geometryName,

            enabled: this.enabled,

            renderScale: this.renderScale,

            projection:
                this.projection.getInfo(),

            metric:
                this.metric.getInfo()
        };
    }

}

class MetricTensor {

    constructor(values = null) {

        this.values = values
            ? Array.from(values)
            : [
                1, 0, 0,
                0, 1, 0,
                0, 0, 1
            ];

        this.validate();

    }

    validate() {

        if (this.values.length !== 9) {
            throw new Error("MetricTensor requires 9 values.");
        }

        if (!this.values.every(Number.isFinite)) {
            throw new Error("Metric contains non-finite values.");
        }

        const m = this.values;
        const tolerance = 1e-10;

        if (
            Math.abs(m[1] - m[3]) > tolerance ||
            Math.abs(m[2] - m[6]) > tolerance ||
            Math.abs(m[5] - m[7]) > tolerance
        ) {
            throw new Error("Metric must be symmetric.");
        }

        this.cholesky();

    }

    cholesky() {

        const m = this.values;

        const L = new Array(9).fill(0);

        for (let i = 0; i < 3; i++) {

            for (let j = 0; j <= i; j++) {

                let sum = m[i * 3 + j];

                for (let k = 0; k < j; k++) {

                    sum -=
                        L[i * 3 + k] *
                        L[j * 3 + k];

                }

                if (i === j) {

                    if (sum <= 0 || !Number.isFinite(sum)) {
                        throw new Error(
                            "Metric must be positive definite."
                        );
                    }

                    L[i * 3 + j] = Math.sqrt(sum);

                } else {

                    L[i * 3 + j] =
                        sum / L[j * 3 + j];

                }

            }

        }

        return L;

    }

    determinant() {

        const L = this.cholesky();

        const d =
            L[0] * L[4] * L[8];

        return d * d;

    }

    quadraticForm(vector) {

        const m = this.values;

        const x = [vector.x, vector.y, vector.z];

        let result = 0;

        for (let i = 0; i < 3; i++) {

            for (let j = 0; j < 3; j++) {

                result +=
                    x[i] * m[i * 3 + j] * x[j];

            }

        }

        return result;

    }

    norm(vector) {

        return Math.sqrt(
            Math.max(0, this.quadraticForm(vector))
        );

    }

    static identity() {

        return new MetricTensor();

    }

    static interpolate(a, b, t) {

        const A = a.cholesky();
        const B = b.cholesky();

        const L = new Array(9).fill(0);

        for (let i = 0; i < 9; i++) {

            L[i] =
                A[i] + (B[i] - A[i]) * t;

        }

        const values = new Array(9).fill(0);

        // G = L * transpose(L)

        for (let i = 0; i < 3; i++) {

            for (let j = 0; j < 3; j++) {

                for (let k = 0; k < 3; k++) {

                    values[i * 3 + j] +=
                        L[i * 3 + k] *
                        L[j * 3 + k];

                }

            }

        }

        return new MetricTensor(values);

    }

}

export { MetricTensor };

class SpacetimeField extends WorldField {

    constructor(options = {}) {

        super("spacetime", options);

        this.metricProvider =
            options.metricProvider || null;

        this.lapseProvider =
            options.lapseProvider || null;

        this.defaultMetric =
            options.defaultMetric || MetricTensor.identity();

        this.defaultLapse =
            options.defaultLapse ?? 1.0;

        if (
            !Number.isFinite(this.defaultLapse) ||
            this.defaultLapse <= 0
        ) {
            throw new Error("Lapse must be positive.");
        }

    }

    evaluate(position, time = 0) {

        const metric = this.metricProvider
            ? this.metricProvider(position, time)
            : this.defaultMetric;

        const lapse = this.lapseProvider
            ? this.lapseProvider(position, time)
            : this.defaultLapse;

        if (!(metric instanceof MetricTensor)) {
            throw new TypeError("Invalid spatial metric.");
        }

        if (
            !Number.isFinite(lapse) ||
            lapse <= 0
        ) {
            throw new Error("Invalid temporal lapse.");
        }

        return {

            metric,

            lapse,

            coordinateTime: time,

            properTimeRate: lapse

        };

    }

    getProperTime(position, coordinateDelta, time = 0) {

        if (
            !Number.isFinite(coordinateDelta) ||
            coordinateDelta < 0
        ) {
            throw new RangeError("Invalid time interval.");
        }

        return this.sample(position, time).lapse *
            coordinateDelta;

    }

}

class SpatialLattice {

    constructor({
        origin = { x: 0, y: 0, z: 0 },
        dimensions = { x: 16, y: 16, z: 16 },
        cellSize = 10,
        defaultValue = null,
        interpolate = null
    } = {}) {

        if (
            !Number.isFinite(cellSize) ||
            cellSize <= 0
        ) {
            throw new Error("Invalid lattice cell size.");
        }

        this.origin = { ...origin };

        this.dimensions = { ...dimensions };

        this.cellSize = cellSize;

        this.defaultValue = defaultValue;

        this.interpolate = interpolate ||
            SpatialLattice.interpolateValues;

        this.values = new Map();

    }

    getKey(x, y, z) {

        return `${x},${y},${z}`;

    }

    set(x, y, z, value) {

        this.validateCoordinates(x, y, z);

        this.values.set(
            this.getKey(x, y, z),
            value
        );

    }

    get(x, y, z) {

        return this.values.get(
            this.getKey(x, y, z)
        ) ?? this.defaultValue;

    }

    validateCoordinates(x, y, z) {

        const d = this.dimensions;

        if (
            !Number.isInteger(x) ||
            !Number.isInteger(y) ||
            !Number.isInteger(z) ||
            x < 0 || y < 0 || z < 0 ||
            x >= d.x || y >= d.y || z >= d.z
        ) {

            throw new RangeError(
                "Lattice coordinates out of bounds."
            );

        }

    }

    worldToGrid(position) {

        return {

            x: (position.x - this.origin.x) / this.cellSize,

            y: (position.y - this.origin.y) / this.cellSize,

            z: (position.z - this.origin.z) / this.cellSize

        };

    }

    sample(position) {

        const p = this.worldToGrid(position);

        const x0 = Math.floor(p.x);
        const y0 = Math.floor(p.y);
        const z0 = Math.floor(p.z);

        const tx = p.x - x0;
        const ty = p.y - y0;
        const tz = p.z - z0;

        const d = this.dimensions;

        if (
            p.x < 0 || p.y < 0 || p.z < 0 ||
            p.x > d.x - 1 ||
            p.y > d.y - 1 ||
            p.z > d.z - 1
        ) {
            return this.defaultValue;
        }

        const x1 = Math.min(x0 + 1, d.x - 1);
        const y1 = Math.min(y0 + 1, d.y - 1);
        const z1 = Math.min(z0 + 1, d.z - 1);

        const samples = [

            [this.get(x0, y0, z0), (1-tx)*(1-ty)*(1-tz)],
            [this.get(x1, y0, z0), tx*(1-ty)*(1-tz)],
            [this.get(x0, y1, z0), (1-tx)*ty*(1-tz)],
            [this.get(x1, y1, z0), tx*ty*(1-tz)],

            [this.get(x0, y0, z1), (1-tx)*(1-ty)*tz],
            [this.get(x1, y0, z1), tx*(1-ty)*tz],
            [this.get(x0, y1, z1), (1-tx)*ty*tz],
            [this.get(x1, y1, z1), tx*ty*tz]

        ];

        return this.interpolate(samples);

    }

    static interpolateValues(samples) {

	if (samples.some(([value]) => value == null)) {
		return null;
	}
        const first = samples[0][0];

        if (first === null || first === undefined) {
            return null;
        }

        if (typeof first === "number") {

            return samples.reduce(
                (sum, [value, weight]) =>
                    sum + value * weight,
                0
            );

        }

        if (Array.isArray(first)) {

            return first.map((_, i) =>

                samples.reduce(
                    (sum, [value, weight]) =>
                        sum + value[i] * weight,
                    0
                )

            );

        }

        if (
            typeof first === "object" &&
            "x" in first &&
            "y" in first &&
            "z" in first
        ) {

            const result = {
                x: 0,
                y: 0,
                z: 0
            };

            for (const [value, weight] of samples) {

                if (!value) continue;

                result.x += value.x * weight;
                result.y += value.y * weight;
                result.z += value.z * weight;

            }

            return result;

        }

        return first;

    }

}

                this.temporalVariation(position, time);

            value: temperature,

            time

        };

    }

import { WorldState } from "./world-state.js";

const DATABASE_NAME = "somnium-worlds";
const DATABASE_VERSION = 1;
const STORE_NAME = "worlds";

class WorldPersistence {
    constructor() {
        this.database = null;
        this.opening = null;
    }

    async open() {
        if (this.database) {
            return this.database;
        }

        if (this.opening) {
            return this.opening;
        }

        if (!globalThis.indexedDB) {
            throw new Error(
                "IndexedDB is unavailable in this browser."
            );
        }

        this.opening = new Promise((resolve, reject) => {
            const request = indexedDB.open(
                DATABASE_NAME,
                DATABASE_VERSION
            );

            request.onupgradeneeded = () => {
                const database = request.result;

                if (!database.objectStoreNames.contains(STORE_NAME)) {
                    database.createObjectStore(STORE_NAME, {
                        keyPath: "metadata.id"
                    });
                }
            };

            request.onsuccess = () => {
                this.database = request.result;

                this.database.onversionchange = () => {
                    this.database.close();
                    this.database = null;
                };

                resolve(this.database);
            };

            request.onerror = () => {
                reject(request.error);
            };

            request.onblocked = () => {
                console.warn(
                    "[WorldPersistence] Database upgrade blocked."
                );
            };
        });

        try {
            return await this.opening;
        } finally {
            this.opening = null;
        }
    }

    async transaction(mode, operation) {
        const database = await this.open();

        return new Promise((resolve, reject) => {
            const tx = database.transaction(STORE_NAME, mode);
            const store = tx.objectStore(STORE_NAME);

            let result;

            try {
                result = operation(store, tx);
            } catch (error) {
                tx.abort();
                reject(error);
                return;
            }

            tx.oncomplete = () => resolve(result);
            tx.onerror = () => reject(tx.error);
            tx.onabort = () => reject(
                tx.error ?? new Error("Transaction aborted.")
            );
        });
    }

    async createWorld(name, options = {}) {
        const cleanName = String(name ?? "").trim();

        if (!cleanName) {
            throw new Error("Please provide a world name.");
        }

        const world = new WorldState({
            ...options,
            name: cleanName
        });

        world.validate();

        await this.transaction("readwrite", store => {
            store.add(world.serialize());
        });

        console.log(
            "[WorldPersistence] Created world:",
            world.metadata.name
        );

        return world;
    }

    async listWorlds() {
        const records = await new Promise(async (resolve, reject) => {
            try {
                const database = await this.open();
                const tx = database.transaction(STORE_NAME, "readonly");
                const request = tx.objectStore(STORE_NAME).getAll();

                request.onsuccess = () => resolve(request.result);
                request.onerror = () => reject(request.error);
            } catch (error) {
                reject(error);
            }
        });

        return records
            .map(record => ({
                id: record.metadata.id,
                name: record.metadata.name,
                createdAt: record.metadata.createdAt,
                updatedAt: record.metadata.updatedAt,
                revision: record.metadata.revision,
                worldSeed: record.generation.worldSeed
            }))
            .sort((a, b) =>
                b.updatedAt.localeCompare(a.updatedAt)
            );
    }

    async loadWorld(id) {
        const record = await new Promise(async (resolve, reject) => {
            try {
                const database = await this.open();
                const tx = database.transaction(STORE_NAME, "readonly");
                const request = tx.objectStore(STORE_NAME).get(id);

                request.onsuccess = () => resolve(request.result);
                request.onerror = () => reject(request.error);
            } catch (error) {
                reject(error);
            }
        });

        if (!record) {
            throw new Error(`World not found: ${id}`);
        }

        return WorldState.deserialize(record);
    }

    async saveWorld(world, options = {}) {
        if (!(world instanceof WorldState)) {
            throw new TypeError("Expected a WorldState instance.");
        }

        world.validate();

        const serialized = world.serialize();

        await this.transaction("readwrite", (store, tx) => {
            const request = store.get(world.metadata.id);

            request.onsuccess = () => {
                const existing = request.result;

                if (!existing) {
                    if (!options.allowCreate) {
                        tx.abort();
                        return;
                    }

                    world.touch();
                    store.add(world.serialize());
                    return;
                }

                // Prevent overwriting changes from another tab.
                if (
                    existing.metadata.revision !==
                    world.metadata.revision
                ) {
                    tx.abort();
                    return;
                }

                world.touch();
                store.put(world.serialize());
            };
        });

        return world;
    }

    async renameWorld(id, newName) {
        const world = await this.loadWorld(id);
        const cleanName = String(newName ?? "").trim();

        if (!cleanName) {
            throw new Error("World name cannot be empty.");
        }

        world.metadata.name = cleanName;

        return this.saveWorld(world);
    }

    async deleteWorld(id) {
        await this.transaction("readwrite", store => {
            store.delete(id);
        });

        console.log("[WorldPersistence] Deleted world:", id);
    }

    close() {
        this.database?.close();
        this.database = null;
    }
}

export { WorldPersistence };

import { WorldState } from "./world-state.js";

class WorldSession {
    constructor(ecology) {
        if (!ecology?.populationManager) {
            throw new Error(
                "WorldSession requires ChunkEcologySystem."
            );
        }

        this.ecology = ecology;
    }

    capture(world, elapsedTime = 0) {
        if (!(world instanceof WorldState)) {
            throw new TypeError("Expected WorldState.");
        }

        const manager = this.ecology.populationManager;

        const chunkRecords = [];

        for (const record of manager.records.values()) {
            const exported = manager.exportRecord(
                record.x,
                record.z
            );

            if (exported) {
                chunkRecords.push(exported);
            }
        }

        world.generation.worldSeed = manager.worldSeed;

        world.simulation.elapsedTime = elapsedTime;

        world.ecology.chunkRecords = chunkRecords;


        return world;
    }

    restore(world) {
        if (!(world instanceof WorldState)) {
            throw new TypeError("Expected WorldState.");
        }

        const ecology = this.ecology;
        const manager = ecology.populationManager;

        // Deactivate currently rendered populations.
        for (const record of manager.records.values()) {
            if (record.active) {
                manager.deactivateRecord(record);
            }
        }

        manager.records.clear();
        manager.pending.length = 0;
        manager.activeKeys.clear();
        manager.simulationCursor = 0;

        ecology.worldSeed = world.generation.worldSeed;
        manager.worldSeed = world.generation.worldSeed;

        // Restore all previously generated populations.
        for (const record of world.ecology.chunkRecords) {
            manager.importRecord(record);
        }

        ecology.renderDirty = true;

        return {
            elapsedTime: world.simulation.elapsedTime,
            worldSeed: world.generation.worldSeed,
            restoredChunks: world.ecology.chunkRecords.length
        };
    }

    getStats() {
        return {
            worldSeed: this.ecology.populationManager.worldSeed,
            generatedChunks:
                this.ecology.populationManager.records.size
        };
    }
}

const WORLD_SCHEMA_VERSION = 1;

function createId() {
    if (globalThis.crypto?.randomUUID) {
        return globalThis.crypto.randomUUID();
    }

    return `world-${Date.now()}-${Math.random()
        .toString(36).slice(2)}`;
}

function cloneData(value) {
    return JSON.parse(JSON.stringify(value));
}

class WorldState {
    constructor(options = {}) {
        const now = new Date().toISOString();

        this.schemaVersion = WORLD_SCHEMA_VERSION;

        this.metadata = {
            id: options.id ?? createId(),
            name: options.name ?? "Untitled World",
            createdAt: options.createdAt ?? now,
            updatedAt: now,
            revision: options.revision ?? 0
        };

        this.generation = {
            worldSeed: options.worldSeed ?? 91827,
            generatorVersion: options.generatorVersion ?? 1,
            configuration: options.configuration ?? {}
        };

        this.simulation = {
            elapsedTime: options.elapsedTime ?? 0,
            simulationRate: options.simulationRate ?? 1,
            paused: options.paused ?? false
        };

        this.ecology = {
            chunkRecords: options.chunkRecords ?? []
        };

        this.entities = {
            manual: options.manualEntities ?? [],
            overrides: options.entityOverrides ?? [],
            removedProceduralIds: options.removedProceduralIds ?? []
        };

        this.environment = {
            state: options.environmentState ?? {}
        };
    }

    validate() {
        if (!this.metadata.id || !this.metadata.name.trim()) {
            throw new Error("World requires an ID and name.");
        }

        if (!Number.isInteger(this.generation.worldSeed)) {
            throw new Error("Invalid world seed.");
        }

        if (
            !Number.isFinite(this.simulation.elapsedTime) ||
            this.simulation.elapsedTime < 0
        ) {
            throw new Error("Invalid simulation time.");
        }

        if (!Array.isArray(this.ecology.chunkRecords)) {
            throw new Error("Invalid ecology records.");
        }

        if (
            !Array.isArray(this.entities.manual) ||
            !Array.isArray(this.entities.overrides) ||
            !Array.isArray(this.entities.removedProceduralIds)
        ) {
            throw new Error("Invalid entity collections.");
        }

        return true;
    }

    touch() {
        this.metadata.updatedAt = new Date().toISOString();
        this.metadata.revision++;
    }

    serialize() {
        this.validate();

        return cloneData(this);
    }

    static deserialize(data) {
        if (!data || typeof data !== "object") {
            throw new Error("Invalid world data.");
        }

        if (data.schemaVersion !== WORLD_SCHEMA_VERSION) {
            throw new Error(
                `Unsupported world schema: ${data.schemaVersion}`
            );
        }

        const state = new WorldState({
            id: data.metadata?.id,
            name: data.metadata?.name,
            createdAt: data.metadata?.createdAt,
            revision: data.metadata?.revision,
            worldSeed: data.generation?.worldSeed,
            generatorVersion: data.generation?.generatorVersion,
            configuration: data.generation?.configuration,
            elapsedTime: data.simulation?.elapsedTime,
            simulationRate: data.simulation?.simulationRate,
            paused: data.simulation?.paused,
            chunkRecords: data.ecology?.chunkRecords,
            manualEntities: data.entities?.manual,
            entityOverrides: data.entities?.overrides,
            removedProceduralIds:
                data.entities?.removedProceduralIds,
            environmentState: data.environment?.state
        });

        state.metadata.updatedAt =
            data.metadata.updatedAt ?? state.metadata.updatedAt;

        state.validate();

        return state;
    }
}

let terrain = null;

function clamp(value, min, max) {
    return Math.max(min, Math.min(max, value));
}

function lerp(a, b, t) {
    return a + (b - a) * t;
}


function hashNoise2D(x, z, seed = 1) {
    let n =
        Math.imul(x | 0, 374761393) +
        Math.imul(z | 0, 668265263) +
        Math.imul(seed | 0, 1442695041);

    n = Math.imul(
        n ^ (n >>> 13),
        1274126177
    );

    n ^= n >>> 16;

    return (n >>> 0) / 4294967295;
}

function smoothNoise2D(x, z, seed = 1) {
    const ix = Math.floor(x);
    const iz = Math.floor(z);

    const fx = x - ix;
    const fz = z - iz;

    const u = fx * fx * (3 - 2 * fx);
    const v = fz * fz * (3 - 2 * fz);

    const a = hashNoise2D(ix, iz, seed);
    const b = hashNoise2D(ix + 1, iz, seed);
    const c = hashNoise2D(ix, iz + 1, seed);
    const d = hashNoise2D(ix + 1, iz + 1, seed);

    return lerp(
        lerp(a, b, u),
        lerp(c, d, u),
        v
    );
}

function fractalNoise2D(x, z, seed, octaves = 3) {
    let value = 0;
    let amplitude = 1;
    let frequency = 1;
    let totalAmplitude = 0;

    for (let i = 0; i < octaves; i++) {
        value += smoothNoise2D(
            x * frequency,
            z * frequency,
            seed + i * 1013
        ) * amplitude;

        totalAmplitude += amplitude;
        amplitude *= 0.5;
        frequency *= 2;
    }

    return value / totalAmplitude;
}

function mixColor(a, b, t) {
    return [
        lerp(a[0], b[0], t),
        lerp(a[1], b[1], t),
        lerp(a[2], b[2], t)
    ];
}


function pixelValue(x, z) {
    const t = terrain;

    x = clamp(x, 0, t.width - 1);
    z = clamp(z, 0, t.height - 1);

    const index = (Math.floor(z) * t.width + Math.floor(x)) * 4;
    const p = t.pixels;

    const r = p[index];
    const g = p[index + 1];
    const b = p[index + 2];

    switch (t.heightChannel) {
        case "red": return r / 255;
        case "green": return g / 255;
        case "blue": return b / 255;
        case "luminance":
            return (
                0.2126 * r +
                0.7152 * g +
                0.0722 * b
            ) / 255;
        case "average":
        default:
            return (r + g + b) / 765;
    }
}

function heightAt(x, z) {
    const u = clamp(
        (x - terrain.minX) / terrain.worldSize,
        0, 1
    );

    const v = clamp(
        (z - terrain.minZ) / terrain.worldSize,
        0, 1
    );

    const px = u * (terrain.width - 1);
    const pz = v * (terrain.height - 1);

    const x0 = Math.floor(px);
    const z0 = Math.floor(pz);

    const x1 = Math.min(x0 + 1, terrain.width - 1);
    const z1 = Math.min(z0 + 1, terrain.height - 1);

    const tx = px - x0;
    const tz = pz - z0;

    const a = lerp(
        pixelValue(x0, z0),
        pixelValue(x1, z0),
        tx
    );

    const b = lerp(
        pixelValue(x0, z1),
        pixelValue(x1, z1),
        tx
    );

    return lerp(
        terrain.minHeight,
        terrain.maxHeight,
        lerp(a, b, tz)
    );
}


function terrainColor(height, slope, worldX, worldZ) {
    const t = terrain;

    const elevation = clamp(
        (height - t.minHeight) /
        (t.maxHeight - t.minHeight),
        0,
        1
    );

    const slopeDegrees =
        Math.atan(slope) * 180 / Math.PI;

    const seed = 91827;

    const moisture = fractalNoise2D(
        worldX * 0.0025,
        worldZ * 0.0025,
        seed + 1000,
        3
    ) * 0.7 + fractalNoise2D(
        worldX * 0.009,
        worldZ * 0.009,
        seed + 3000,
        3
    ) * 0.3;

    const geology = fractalNoise2D(
        worldX * 0.00375,
        worldZ * 0.00375,
        seed + 2000,
        3
    );

    const variation = fractalNoise2D(
        worldX * 0.025,
        worldZ * 0.025,
        seed + 5000,
        3
    );

    const exposure = clamp(
        slopeDegrees / 42 * 0.78 +
        elevation * 0.12 +
        (1 - moisture) * 0.10,
        0,
        1
    );

    const soilType = soilTypeAt(worldX, worldZ);

    const soilColors = {
        clay:        [0.48, 0.29, 0.21],
        sand:        [0.92, 0.73, 0.39],
        fertileSoil: [0.43, 0.29, 0.18],
        loam:        [0.35, 0.25, 0.16],
        organic:     [0.19, 0.13, 0.085],
        scree:       [0.43, 0.41, 0.37]
    };

    const soil = soilColors[soilType];
    let bedrock;

    if (geology < 0.28) {
        bedrock = [0.63, 0.60, 0.51];
    } else if (geology < 0.48) {
        bedrock = [0.65, 0.43, 0.29];
    } else if (geology > 0.84) {
        bedrock = [0.24, 0.23, 0.22];
    } else {
        bedrock = [0.39, 0.37, 0.34];
    }

    if (elevation > 0.70 && geology > 0.65) {
        bedrock = [0.39, 0.37, 0.34];
    }

    // Blend surface material into exposed bedrock.
    let color = mixColor(
        soil,
        bedrock,
        clamp((exposure - 0.25) / 0.65)
    );

    // Small-scale geological surface variation.
    const texture = 0.92 + variation * 0.16;

    color = color.map(value =>
        clamp(value * texture, 0, 1)
    );

    // Snow and frost at the highest elevations.
    if (elevation > 0.92) {
        const snow = [0.82, 0.84, 0.82];

        color = mixColor(
            color,
            snow,
            clamp((elevation - 0.92) / 0.08)
        );
    }

    return color;
}





function generateMesh(job) {
    const {
        x,
        z,
        size,
        lod,
        neighborLODs
    } = job;

    const resolution = [32, 16, 8, 4][
        clamp(lod, 0, 3)
    ];

    const stride = size / resolution;

    const count = (resolution + 1) ** 2;

    const positions = new Float32Array(count * 3);
    const normals = new Float32Array(count * 3);
    const colors = new Float32Array(count * 4);

    const indices = new Uint32Array(
        resolution * resolution * 6
    );

    const originX = x * size;
    const originZ = z * size;

    const edgeHeight = (edge, index) => {
        const otherLOD = neighborLODs[edge];

        const worldDistance = index * stride;

        if (otherLOD <= lod) {
            return edge === "north" || edge === "south"
                ? heightAt(
                    originX + worldDistance,
                    originZ + (edge === "south" ? size : 0)
                )
                : heightAt(
                    originX + (edge === "east" ? size : 0),
                    originZ + worldDistance
                );
        }

        const coarseResolution = [32, 16, 8, 4][
            clamp(otherLOD, 0, 3)
        ];

        const coarseStride = size / coarseResolution;

        const lower =
            Math.floor(worldDistance / coarseStride) *
            coarseStride;

        const upper = Math.min(
            lower + coarseStride,
            size
        );

        const t = upper === lower
            ? 0
            : (worldDistance - lower) / (upper - lower);

        let ax, az, bx, bz;

        if (edge === "north" || edge === "south") {
            az = bz = originZ +
                (edge === "south" ? size : 0);

            ax = originX + lower;
            bx = originX + upper;
        } else {
            ax = bx = originX +
                (edge === "east" ? size : 0);

            az = originZ + lower;
            bz = originZ + upper;
        }

        return lerp(
            heightAt(ax, az),
            heightAt(bx, bz),
            t
        );
    };

    for (let iz = 0; iz <= resolution; iz++) {
        for (let ix = 0; ix <= resolution; ix++) {
            const worldX = originX + ix * stride;
            const worldZ = originZ + iz * stride;

            let h = heightAt(worldX, worldZ);

            if (iz === 0) h = edgeHeight("north", ix);
            if (iz === resolution) h = edgeHeight("south", ix);
            if (ix === 0) h = edgeHeight("west", iz);
            if (ix === resolution) h = edgeHeight("east", iz);

            const spacing = Math.max(2, stride * 0.25);

            const dx = (
                heightAt(worldX + spacing, worldZ) -
                heightAt(worldX - spacing, worldZ)
            ) / (2 * spacing);

            const dz = (
                heightAt(worldX, worldZ + spacing) -
                heightAt(worldX, worldZ - spacing)
            ) / (2 * spacing);

            const length = Math.hypot(dx, 1, dz);

            const index = iz * (resolution + 1) + ix;
            const offset = index * 3;

            positions[offset] = worldX;
            positions[offset + 1] = h;
            positions[offset + 2] = worldZ;

            normals[offset] = -dx / length;
            normals[offset + 1] = 1 / length;
            normals[offset + 2] = -dz / length;

		const color = terrainColor(
    h,
    Math.hypot(dx, dz),
    worldX,
    worldZ
);

            const colorOffset = index * 4;

            colors[colorOffset] = color[0];
            colors[colorOffset + 1] = color[1];
            colors[colorOffset + 2] = color[2];
            colors[colorOffset + 3] = 1;
        }
    }

    let cursor = 0;

    for (let iz = 0; iz < resolution; iz++) {
        for (let ix = 0; ix < resolution; ix++) {
            const a = iz * (resolution + 1) + ix;
            const b = a + 1;
            const c = a + resolution + 1;
            const d = c + 1;

            indices[cursor++] = a;
            indices[cursor++] = c;
            indices[cursor++] = b;

            indices[cursor++] = b;
            indices[cursor++] = c;
            indices[cursor++] = d;
        }
    }

    return {
        positions,
        normals,
        colors,
        indices
    };
}

self.onmessage = event => {
    const message = event.data;

    if (message.type === "init") {
        terrain = {
            ...message.terrain,
            pixels: new Uint8ClampedArray(message.pixels)
        };

        self.postMessage({ type: "ready" });
        return;
    }

    if (message.type === "generate" && terrain) {
        try {
            const data = generateMesh(message.job);

            self.postMessage({
                type: "complete",
                id: message.id,
                revision: message.revision,
                data: {
                    positions: data.positions,
                    normals: data.normals,
                    colors: data.colors,
                    indices: data.indices
                }
            }, [
                data.positions.buffer,
                data.normals.buffer,
                data.colors.buffer,
                data.indices.buffer
            ]);
        } catch (error) {
            self.postMessage({
                type: "error",
                id: message.id,
                revision: message.revision,
                message: error.message
            });
        }
    }
};

function soilTypeAt(worldX, worldZ) {
    const t = terrain;

    const u = clamp(
        (worldX - t.minX) / t.worldSize,
        0,
        1
    );

    const v = clamp(
        (worldZ - t.minZ) / t.worldSize,
        0,
        1
    );

    const mapValue = pixelValue(
        u * (t.width - 1),
        v * (t.height - 1)
    );

    if (mapValue < 0.18) return "clay";
    if (mapValue < 0.38) return "sand";
    if (mapValue < 0.58) return "fertileSoil";
    if (mapValue < 0.76) return "loam";
    if (mapValue < 0.90) return "organic";

    return "scree";
}

export class WorldChunk {
    constructor(x, z, size, terrain, scene) {
        this.x = x;
        this.z = z;
        this.size = size;
        this.terrain = terrain;
        this.scene = scene;

        this.mesh = null;
        this.lod = -1;
        this.loaded = false;
        this.disposed = false;

        this.revision = 0;

        this.neighbors = {
            north: null,
            south: null,
            east: null,
            west: null
        };
    }

    get originX() {
        return this.x * this.size;
    }

    get originZ() {
        return this.z * this.size;
    }

    getResolution(lod) {
        return [32, 16, 8, 4][
            BABYLON.Scalar.Clamp(lod, 0, 3)
        ];
    }

    setNeighbors(neighbors) {
        this.neighbors = neighbors;
    }

	
applyMeshData(data, lod, material) {
    if (this.disposed) return;

    const vertexData = new BABYLON.VertexData();

    vertexData.positions = Array.from(data.positions);
    vertexData.normals = Array.from(data.normals);
    vertexData.colors = Array.from(data.colors);
    vertexData.indices = Array.from(data.indices);

    if (this.mesh) {
        this.mesh.dispose();
    }

    this.mesh = new BABYLON.Mesh(
        `terrain_chunk_${this.x}_${this.z}`,
        this.scene
    );

    vertexData.applyToMesh(this.mesh, true);

    this.mesh.material = material;

    material.useVertexColors = true;
    material.diffuseColor.set(1, 1, 1);

    this.mesh.useVertexColors = true;
    this.mesh.hasVertexAlpha = false;

    if (
        !this.mesh.isVerticesDataPresent(
            BABYLON.VertexBuffer.ColorKind
        )
    ) {
        console.error(
            "[WorldChunk] Terrain vertex colors missing:",
            this.x,
            this.z
        );
    }

    // IMPORTANT:
    // Terrain must be pickable for architectural editing.

    this.mesh.isPickable = true;

    this.mesh.metadata = {
        chunkX: this.x,
        chunkZ: this.z,
        lod,

        // Explicit identifier for editor terrain picking.
        terrainChunk: true
    };

    // Apply existing spatial transformation.
    this.scene.metadata?.geometryRenderer?.applyMesh(
        this.mesh
    );

    this.lod = lod;
    this.loaded = true;

    console.log(
        "[WorldChunk] Terrain mesh ready for picking:",
        {
            name: this.mesh.name,
            pickable: this.mesh.isPickable,
            terrainChunk: this.mesh.metadata.terrainChunk,
            vertices: this.mesh.getTotalVertices()
        }
    );
}


    generateMesh(lod, material) {
        if (this.disposed) return;

        const resolution = this.getResolution(lod);
        const stride = this.size / resolution;

        const positions = [];
        const indices = [];
        const normals = [];
        const colors = [];

        const minX = this.terrain.min.x;
        const maxX = this.terrain.max.x;
        const minZ = this.terrain.min.z;
        const maxZ = this.terrain.max.z;

        const heightAt = (x, z) =>
            this.terrain.sampleHeight(
                BABYLON.Scalar.Clamp(x, minX, maxX),
                BABYLON.Scalar.Clamp(z, minZ, maxZ)
            );

        for (let z = 0; z <= resolution; z++) {
            for (let x = 0; x <= resolution; x++) {
                const wx = this.originX + x * stride;
                const wz = this.originZ + z * stride;

                const h = heightAt(wx, wz);

                positions.push(wx, h, wz);

                const slope = this.terrain.sampleSlopeAt(
                    wx, wz, Math.max(2, stride * 0.25)
                );

                const normal = slope?.normal ??
                    new BABYLON.Vector3(0, 1, 0);

                normals.push(
                    normal.x, normal.y, normal.z
                );

                const color = this.getTerrainColor(
                    h,
                    slope?.magnitude ?? 0
                );

                colors.push(
                    color.r, color.g, color.b, 1
                );
            }
        }

        for (let z = 0; z < resolution; z++) {
            for (let x = 0; x < resolution; x++) {
                const a = z * (resolution + 1) + x;
                const b = a + 1;
                const c = a + resolution + 1;
                const d = c + 1;

                indices.push(a, c, b, b, c, d);
            }
        }

        this.applyMeshData({
            positions,
            normals,
            colors,
            indices
        }, lod, material);
    }

    getTerrainColor(height, slope) {
        const elevation = BABYLON.Scalar.Clamp(
            (height - this.terrain.minHeight) /
            (this.terrain.maxHeight - this.terrain.minHeight),
            0, 1
        );

        const low = new BABYLON.Color3(0.24, 0.38, 0.19);
        const grass = new BABYLON.Color3(0.42, 0.48, 0.25);
        const mountain = new BABYLON.Color3(0.48, 0.45, 0.39);
        const snow = new BABYLON.Color3(0.82, 0.84, 0.82);

        let color;

        if (elevation < 0.35) {
            color = BABYLON.Color3.Lerp(
                low, grass, elevation / 0.35
            );
        } else if (elevation < 0.78) {
            color = BABYLON.Color3.Lerp(
                grass, mountain, (elevation - 0.35) / 0.43
            );
        } else {
            color = BABYLON.Color3.Lerp(
                mountain, snow, (elevation - 0.78) / 0.22
            );
        }

        return BABYLON.Color3.Lerp(
            color,
            new BABYLON.Color3(0.32, 0.30, 0.28),
            BABYLON.Scalar.Clamp(slope / 0.9, 0, 0.45)
        );
    }

    dispose() {
        this.revision++;

        if (this.mesh) {
            this.mesh.dispose();
            this.mesh = null;
        }

        this.loaded = false;
        this.disposed = true;
    }
}

            !this.workerReady ||
            this.workerBusy ||
            this.meshQueue.length === 0

            return;

        this.stats.pendingMeshes = this.meshQueue.length;

        const chunk = this.chunks.get(job.key);

        if (
            !chunk ||
            chunk.disposed ||
            chunk.revision !== job.revision

            this.pumpWorker();
            return;

        this.currentTask = job;
        this.workerBusy = true;

        this.worker.postMessage({

            id: job.id,
            revision: job.revision,

        this.meshQueue.length === 0 ||

        if (job) {

            if (
                chunk &&
                !chunk.disposed &&
                chunk.revision === job.revision

                try {
                    chunk.generateMesh(
                        job.lod,
                        this.material

if (distance <= this.chunkSize * 8) return 2;

        return 3;

    getPriority(x, z, cameraPosition, forward) {
        const centerX = (x + 0.5) * this.chunkSize;
        const centerZ = (z + 0.5) * this.chunkSize;

        const dx = centerX - cameraPosition.x;
        const dz = centerZ - cameraPosition.z;

        const distance = Math.hypot(dx, dz);

        const length = Math.max(distance, 1);

        const directionDot = (
            dx / length * forward.x +
            dz / length * forward.z
        );

        // Lower scores are generated first.
        return distance -
            Math.max(0, directionDot) *
            this.chunkSize * 4;

    update(cameraPosition, forward = null) {
        if (!this.enabled || !this.terrain.ready) return;

        const center = this.getCoordinates(cameraPosition);

        const direction = forward ?? {
            x: 0,
            z: 1
        };

        const required = [];

        for (
            let x = Math.max(
                this.minChunkX,
                center.x - this.renderRadius
            );
            x <= Math.min(
                this.maxChunkX,
                center.x + this.renderRadius
            );
            x++
        ) {
            for (
                let z = Math.max(
                    this.minChunkZ,
                    center.z - this.renderRadius
                );
                z <= Math.min(
                    this.maxChunkZ,
                    center.z + this.renderRadius
                );
                z++
            ) {
                const dx = x - center.x;
                const dz = z - center.z;

                if (
                    dx * dx + dz * dz >
                    this.renderRadius * this.renderRadius
                ) continue;

                required.push({
                    x,
                    z,
                    priority: this.getPriority(
                        x,
                        z,
                        cameraPosition,
                        direction
                    ),
                    lod: this.getLOD(
                        x,
                        z,
                        cameraPosition
                    )
                });
            }
        }

        required.sort(
            (a, b) => a.priority - b.priority
        );

        let scheduled = 0;

        for (const item of required) {
            const chunk = this.getChunk(item.x, item.z);

            if (!chunk) {
                if (scheduled >= this.maxChunksPerUpdate) {
                    continue;
                }

                this.createChunk(
                    item.x,
                    item.z,
                    item.lod
                );

                scheduled++;
                continue;
            }

            if (chunk.lod !== item.lod) {
                if (scheduled >= this.maxChunksPerUpdate) {
                    continue;
                }

                this.scheduleMesh(chunk, item.lod);
                scheduled++;
            }
        }

        const unloadRadius = this.renderRadius + 1;

        for (const [key, chunk] of this.chunks) {
            const dx = chunk.x - center.x;
            const dz = chunk.z - center.z;

            if (
                dx * dx + dz * dz >
                unloadRadius * unloadRadius
            ) {
                chunk.dispose();
                this.chunks.delete(key);

                this.meshQueue = this.meshQueue.filter(
                    task => task.key !== key
                );

                this.stats.unloaded++;
                this.stats.loaded--;
            }
        }

        this.stats.pendingMeshes = this.meshQueue.length;

        if (!this.worker) {
            this.processFallbackQueue();
        } else {
            this.pumpWorker();
        }
    }

    getStats() {
        return {
            ...this.stats,
            currentChunks: this.chunks.size,
            chunkSize: this.chunkSize,
            renderRadius: this.renderRadius,
            worldSize: this.terrain.worldSize,
            workerActive: !!this.worker,
            workerReady: this.workerReady
        };
    }

    dispose() {
        this.enabled = false;

        if (this.worker) {
            this.worker.terminate();
            this.worker = null;
        }

        for (const chunk of this.chunks.values()) {
            chunk.dispose();
        }

        this.chunks.clear();
        this.meshQueue.length = 0;

        this.material.dispose();

        this.stats.loaded = 0;
        this.stats.pendingMeshes = 0;
    }

failWorker(reason) {
    console.error(
        "[WorldManager] Worker failure:",
        reason
    );

    this.stats.workerErrors++;

    if (this.workerInitTimer) {
        clearTimeout(this.workerInitTimer);
        this.workerInitTimer = null;
    }

    const failedTask = this.currentTask;
    this.currentTask = null;

    if (this.worker) {
        this.worker.terminate();
        this.worker = null;
    }

    this.workerReady = false;
    this.workerBusy = false;

    // Retry the outstanding task synchronously.
    if (failedTask) {
        const chunk = this.chunks.get(failedTask.key);

        if (
            chunk &&
            !chunk.disposed &&
            chunk.revision === failedTask.revision
        ) {
            this.meshQueue.unshift(failedTask);
        }
    }

    this.stats.pendingMeshes = this.meshQueue.length;
    this.processFallbackQueue();
}
